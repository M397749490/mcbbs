  27. class Telnet {
  28. var $sock = NULL;

  30. function telnet($host,$port) {
  31.   $this->sock = fsockopen($host,$port);
  32.   socket_set_timeout($this->sock,2,0);
  33. }

  35. function close() {
  36.   if ($this->sock)  fclose($this->sock);
  37.   $this->sock = NULL;
  38. }

  40. function write($buffer) {
  41.   $buffer = str_replace(chr(255),chr(255).chr(255),$buffer);
  42.   fwrite($this->sock,$buffer);
  43. }

  45. function getc() {
  46.   return fgetc($this->sock);
  47. }

  49. function read_till($what) {
  50.   $buf = '';
  51.   while (1) {
  52.    $IAC = chr(255);
  53.    
  54.    $DONT = chr(254);
  55.    $DO = chr(253);
  56.    
  57.    $WONT = chr(252);
  58.    $WILL = chr(251);
  59.    
  60.    $theNULL = chr(0);

  62.    $c = $this->getc();
  63.    
  64.    if ($c === false) return $buf;
  65.    if ($c == $theNULL) {
  66.     continue;
  67.    }

  69.    if ($c == "1") {
  70.     continue;
  71.    }

  73.    if ($c != $IAC) {
  74.     $buf .= $c;
  75.   
  76.     if ($what == (substr($buf,strlen($buf)-strlen($what)))) {
  77.      return $buf;
  78.     }
  79.     else {
  80.      continue;
  81.     }
  82.    }

  84.    $c = $this->getc();
  85.    
  86.    if ($c == $IAC) {
  87.    $buf .= $c;
  88.    }
  89.    else if (($c == $DO) || ($c == $DONT)) {
  90.     $opt = $this->getc();
  91.     // echo "we wont ".ord($opt)."\n";
  92.     fwrite($this->sock,$IAC.$WONT.$opt);
  93.    }
  94.    elseif (($c == $WILL) || ($c == $WONT)) {
  95.     $opt = $this->getc();
  96.     // echo "we dont ".ord($opt)."\n";

  98.    }
  99.    else {
  100.     // echo "where are we? c=".ord($c)."\n";

  102.   }
  103. }

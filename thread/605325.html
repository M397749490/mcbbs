PlaceholderAPI 配置

# PlaceholderAPI 版本 2.3.0
# 作者by extended_clip

check_updates: true
#是否提示更新最新版本
boolean:
'true': 'yes'
'false': 'no'
date_format: MM/dd/yy HH:mm:ss
#时间格式 开头需大写 MM指月份 dd指天 yy指年 HH指小时 mm指分钟 ss指秒钟
expansions:
pinger:
offline: '&7[&cOFF&7]'
check_interval: 10
online: '&7[&aON&7]'
vault:
formatting:
thousands: k
trillions: T
quadrillions: Q
billions: B
millions: M
bungee:
check_interval: 30

PlaceholderAPI 指令

/placeholderapi #查看placeholderapi版本
/placeholderapi list #查看挂钩的插件
/placeholderapi reload #重载配置
PlaceholderAPI 权限
placeholderapi.admin #全部指令的权限

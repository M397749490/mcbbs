棱镜 PrisonUtils
用于管理药水效果，伤害减免等功能


功能：
      1.自动熔炼，自动冶炼可熔炼的物品，如铁矿
      2.掉落的物品直接进入库存                  
      3.放置特定原因造成的伤害
      4.永久的药水效果
      5.牢不可破的稿子，防止耐久下降
      6.鹤嘴镐，显示主人和耐久等


用法：
     1.下载插件放到mods文件夹
     2.启动服务器生成配置文件
     3.修改配置文件开启功能


配置文件：
  1. # 玩家登录时的永久效果(药剂效果|效果等级)
  2. "Active Potion Effects" {
  3.     "minecraft:absorption"=0
  4.     "minecraft:blindness"=0
  5.     "minecraft:fire_resistance"=0
  6.     "minecraft:glowing"=0
  7.     "minecraft:haste"=0
  8.     "minecraft:health_boost"=0
  9.     "minecraft:hunger"=0
  10.     "minecraft:invisibility"=0
  11.     "minecraft:jump_boost"=0
  12.     "minecraft:luck"=0
  13.     "minecraft:mining_fatigue"=0
  14.     "minecraft:nausea"=0
  15.     "minecraft:night_vision"=0
  16.     "minecraft:poison"=0
  17.     "minecraft:regeneration"=0
  18.     "minecraft:resistance"=0
  19.     "minecraft:saturation"=0
  20.     "minecraft:slowness"=0
  21.     "minecraft:speed"=0
  22.     "minecraft:strength"=0
  23.     "minecraft:unluck"=0
  24.     "minecraft:water_breathing"=0
  25.     "minecraft:weakness"=0
  26.     "minecraft:wither"=0
  27. }
  28. # 防止玩家在下面原因中收到伤害
  29. "Damage Prevention" {
  30.     Drown=false
  31.     Explosive=false
  32.     Fall=false
  33.     Fire=false
  34.     Hunger=false
  35.     Magic=false
  36.     Magma=false
  37.     Suffocate=false
  38.     Void=false
  39. }
  40. # 改变拖动物品的操作方式
  41. "Dropped Items" {
  42.     Auto-Smelt=true
  43.     # 自动熔炼的黑名单（下列物品不受自动熔炼影响）
  44.     "Auto-Smelt BlackList"=[
  45.         "minecraft:log",
  46.         "minecraft:log2"
  47.     ]
  48.     "Dropped to Inventory"=true
  49. }
  50. # 上述三个功能控制开关
  51. Modules {
  52.     "Damage Prevention"=true
  53.     "Dropped Items"=true
  54.     "Permanent Effects"=true
  55. }
  56. # 鹤嘴镐和无限耐久搞启用开关
  57. "Tool Options" {
  58.     "Pickaxe Lore"=false
  59.     "Unbreakable Pickaxes"=false
  60. }
复制代码

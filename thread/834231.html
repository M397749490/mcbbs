  •   #属性显示名
  •   display: '体质'
  •   #最大等级
  •   max: 999
  •   #图标
  •   #默认为墨囊,"_"以" "代替
  •   icon: 'ink sack'
  •   #图标副ID
  •   #如果为羊毛
  •   #白色羊毛填0，橙色羊毛填1
  •   icon-data: 1
  •   #图标LORE
  •   #简单来说就是属性介绍
  •   icon-lore:
  •   - '&6体质等级:&7(&2{amount}&7)'
  •   - ''
  •   - '&7每升一级获得1％生命值'
  •   - '&7需要1属性点.'
  •   #全局配置
  •   global:
  •     # 条件
  •     condition: {}
  •     # 效果
  •     mechanic: {}
  •     # 目标
  •     target: {}
  •   #统计点数
  •   stats:
  •     #生命值
  •     #a=原数值 v=加点数
  •     health: 'a*0.01+1*v'

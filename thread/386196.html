                Player p=event.getPlayer();                                                           
                ItemStack i=new ItemStack(Material.DIAMOND);                             //设定指定物品类型
                if(p.getLocation().getX()>110.0&&p.getLocation().getX()<120.0&&p.getLocation().getY()>60&&p.getLocation().getY()<65){ //设定结界
                        if(!(p.getInventory().getItemInHand==i)){                            //不含有指定物品
                                int X=(int) (p.getLocation().getX()-3.0);          //定义禁止进入后玩家传送到的坐标
                                int Y=(int) p.getLocation().getY();
                                int Z=(int) p.getLocation().getZ();
                                Block b=p.getWorld().getBlockAt(X,Y,Z);            //获得完整Location

                                p.sendMessage("You don't hava a Diamond");  //发送警告
                        }
                }
        }

标题: Fe经济插件怎么改货币名称？
Fe经济插件怎么改货币名称？配置文件如下：
# Fe Config - meloncraft.com
# holdings - The amount of money that players will start out with
# prefix - The message prefix
# currency - The single and multiple names for the currency
# type - The type of database used (sqlite, mysql, mongo, or item)

holdings: 10.0
maxholdings: -1
prefix: 传说中的斗币
topsize: 5
currency:
  negative: false
  major:
    single: Fe
    multiple: Fe
  minor:
    enabled: true
    single:  斗币
    multiple:  斗币
  prefix: ''
type: sqlite
updatecheck: true
mongo:
  host: localhost
  port: 27017
  user: root
  password: minecraft
  database: Fe
mysql:
  host: localhost
  port: 3306
  user: root
  password: minecraft
  database: Fe
  tables:
    accounts: fe_accounts
item:
  item: gold_ingot
  value: 10

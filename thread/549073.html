你用的是ils插件吧？
打开ils文件的config，找到以下内容把true改为false
#是否开启合成物品添加护甲
enableArmourOnCrafted: false
enableArmourOnDrops: false
armourStatFormat: '&3[statName]: &b[statValue]&3%'
armour:
leather: '&b3.6-7.3'
iron: '&b7.8-15.6'
chainmail: '&b6.2-12.5'
gold: '&b5.7-11.4'
diamond: '&b10.4-20.8'
mobHead: '&b4.0-15.0'
defaultCraftedDamage:
#是否开启合成物品添加伤害
enableDamageOnCrafted: fales
enableDamageOnDrops: false
damageStatFormat: '&3[statName]: +&b[statValue]-[statValue2]'
tool:
bow: '&b3.6-7.3'
wood: '&b7.8-15.6'

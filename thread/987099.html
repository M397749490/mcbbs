# 连锁命令方块依次接在循环命令方块上
  1. # 循环,保持开启
  2. scoreboard players tag @a add foo {Inventory:[{Slot:-106b,id:"minecraft:clock"}]}
  3. # 连锁,保持开启
  4. tp @a[tag=foo] 2 3 3
  5. # 连锁,保持开启
  6. scoreboard players tag @a[tag=foo] remove foo

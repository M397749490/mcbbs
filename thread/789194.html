所以……Mojang 发布新材质了？
于是花了 3 分钟（甚至在性能上没有一点优化）写了个不知道是什么东西用于提取新材质（希望没有违背 EULA）
完整代码：
  1. import java.io.File;
  2. import java.io.FileInputStream;
  3. import java.io.IOException;
  4. import java.io.InputStream;
  5. import java.nio.file.Files;
  6. import java.nio.file.StandardCopyOption;

  8. import javax.json.Json;
  9. import javax.json.JsonObject;

  11. public class AssetsExtractor {
  12.     public static void main(String[] args) {
  13.         File assetsRoot = new File("minecraft/assets"); // Path to `assets`
  14.         String version = "1.13-af"; // Index version
  15.         File root = new File("minecraft/extracted");
  16.         File objects = new File(assetsRoot, "objects");
  17.         File indexes = new File(assetsRoot, "indexes");
  18.         try (InputStream in = new FileInputStream(new File(indexes, version + ".json"))) {
  19.             JsonObject index = Json.createReader(in).readObject().getJsonObject("objects");
  20.             for (String key : index.keySet()) {
  21.                 System.out.println("Copying " + key);
  22.                 String hash = index.getJsonObject(key).getString("hash");
  23.                 File file = new File(root, version + '/' + key);
  24.                 file.getParentFile().mkdirs();
  25.                 Files.copy(new File(objects, hash.substring(0, 2) + '/' + hash).toPath(), file.toPath(), StandardCopyOption.REPLACE_EXISTING);
  26.             }
  27.         } catch (IOException e) {
  28.             e.printStackTrace();
  29.         }
  30.     }
  31. }

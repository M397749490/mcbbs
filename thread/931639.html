如何使用
请在 `plugin.yml`内增加
  1. loadbefore:
  2.   - AutoUpData
复制代码


否则会出现null异常BUG
API接口:
  1. import updata.AutoData;
  2. /**
  3. * @parm plugin PluginBase
  4. * @parm file 更新插件的文件位置 (PluginBase 有 getFile() 方法)
  5. * @parm user GitHub用户名
  6. * @parm project 工程名称 不是插件名
  7. * @Return 是否更新完毕
  8. * 获取接口
  9. */
  10. /**
  11. * 获取接口前请先判断插件是否存在
  12. */
  13. UpData data = AutoData.get(plugin,file,user,project);
  14. /**
  15. * 在使用data参数前请先判断是否为null
  16. */
  17. /*
  18. * 判断是否存在新版本
  19. * @return Boolean
  20. */
  21. data.canUpdate();
  22. /*
  23. * 获取新版本名称
  24. * @return String 版本名 格式: 1.0.0
  25. */
  26. data.getNewVersion();
  27. /*
  28. * 获取更新内容
  29. * @return String 更新内容
  30. */
  31. data.getNewVersionMessage();
  32. /*
  33. * 使插件更新
  34. */
  35. data.toUpData();

  37. /*
  38. * 使插件更新
  39. * @param canReload 是否重启插件
  40. */
  41. data.toUpData(canReload);

  43. /**
  44.      * @param plugin 插件
  45.      * @param file 更新插件的路径
  46.      * @param user GitHub用户名
  47.      * @param project 工程名
  48.      *
  49.      * 直接对插件进行更新 如果需要更详细的设置 请用  {@link UpData} 方法
  50.      *
  51.      * @return true 为 更新成功 建议在 onEnable 直接跳出 否则会有报错 false 是更新失败
  52.      * */
  53. AutoData.defaultUpData(plugin,file,user,project);

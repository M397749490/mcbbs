配置文件
  1. #你想在哪个世界使用这个插件?
  2. disabled-worlds:
  3.   - "world_example"

  5. mobs:
  6.   chicken: &default #这个是默认配置

  8.     #这个动物能造成多少伤害（用半颗心计算）
  9.     attack-damage: 1

  11.     #动物的攻击间隔（以秒计算）
  12.     attack-speed: 0.5

  14.     #动物的攻击距离（以格计算）
  15.     attack-range: 1

  17.     #动物的移动速度(1是常速, 0.5是1的一半的速度  2是2倍速)
  18.     speed: 1.5

  20.    #追踪范围（默认情况下, 我的世界的动物可以在16格内看见玩家,但是,僵尸可以看到40格）
  21.     navigation-range: 16

  23.     #动物什么时候会打玩家
  24.     #由你来选择攻击的条件,但是满足一个条件,他就会攻击玩家
  25.     attack-conditions:

  27.       #一旦动物被攻击,他会优先去攻击多少百分比血量的玩家 (百分比计算)
  28.       health-percentage: 100

  30.       #玩家离动物多远时动物会去打玩家 (用格计算,若要使用这个条件就把前面的#去掉）
  31.       #distance: 10

  33.       #一个动物在多少范围内会帮助另一个怪物攻击玩家 (用格计算)
  34.       #group-aggro-distance: 10

  36.       #chance:
  37.       #  percentage: 25 # 动物攻击玩家的概率 (用百分比计算)
  38.       #  duration: 0 # 动物的记仇时间 (用秒计算)
  39.       #  conditions: # 触发这些机会的条件: (就是在attack-conditions中的条件)
  40.       #    health-percentage: 100

  42.       # 高级距离攻击. 距离用格计算(去掉distance前的#使用)
  43.       #distance:
  44.       #   _: 20
  45.       #   player: #只有当下面是ture，动物才会攻击玩家
  46.       #     sneaking: false
  47.       #     walking: false
  48.       #     sprinting: true #只攻击正在疾跑的玩家
  49.       #  cow: {} #如果你加上{}，那么任何奶牛都会互相攻击
  50.       #  chicken:
  51.       #    baby: true #只打小鸡
  52.       #  wolf:
  53.       #    tamed: false #不攻击被驯服的狼

  55.       #如果你希望拥有2个相同的攻击条件(在这个条件下玩家会离得很近)
  56.       #distance{your_unique_text}:
  57.       #   _: 5
  58.       #   player: {}

  60.     # 动物何时停止攻击玩家？
  61.     # 由你来选择停止攻击的条件，但满足一个条件他就会停止攻击
  62.     stop-conditions:
  63.       # 玩家距离动物多远时动物会停止攻击？(用格计算)
  64.       distance: 15

  66.       #一旦玩家的血量低于多少百分比时会停止攻击？ (百分比计算)
  67.       #health-percentage: 50

  69.       #玩家的饱和度低于多少百分比时会停止攻击玩家？(百分比计算)
  70.       #enemy-health-percentage: 50
  71.   pig:
  72.     <<: *default # 加载默认配置
  73.     # 你可以在这里覆盖所有的默认配置，比如：
  74.     #speed: 2
  75.   sheep:
  76.     <<: *default
  77.   cow:
  78.     <<: *default
  79.   mushroom_cow:
  80.     <<: *default
  81.   donkey:
  82.     <<: *default
  83.   mule:
  84.     <<: *default
  85.   ocelot:
  86.     <<: *default
  87.   rabbit:
  88.     <<: *default
  89.   parrot:
  90.     <<: *default
  91.   turtle:
  92.     <<: *default
  93.   cat:
  94.     <<: *default
  95.   panda:
  96.     <<: *default
  97.   fox:
  98.     <<: *default

  100.   horse: &horse # 为马创造一个默认配置
  101.     <<: *default
  102.     attack-range: 1.5
  103.   skeleton_horse:
  104.     <<: *horse
  105.   zombie_horse:
  106.     <<: *horse

  108.   villager: &villager
  109.     <<: *default
  110.     speed: 0.5
  111.   wandering_trader:
  112.     <<: *villager

  114.   giant:
  115.     <<: *default
  116.     attack-damage: 4
  117.     attack-speed: 1
  118.     attack-range: 3

  120.     #下面的怪是已经可以瞄准玩家的, 但你可以添加攻击的条件 (或者删除自带的瞄准)
  121.   wolf: &default_targeting

  123.     #如果填写的是 true,则敌对生物和中立生物不会主动攻击人而是触发攻击条件再去攻击人.

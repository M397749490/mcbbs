Tinkers’ Survival
匠魂生存
匠魂的附属Mod，增加在前期的难度，类似更好的开局
此Mod只能让你用相应的工具去挖掘方块，空手撸方块等于空手撸黑曜石
也有些类似无树可撸。
此Mod站内帖https://www.mcbbs.net/forum.php?mod=viewthread&tid=811651
由于没有在站内看到该mod教程，来写点比较简单教程

和匠魂强化一样，开局我们必须先找沙砾撸燧石，同样可以用三个沙砾合成一块燧石
掉落燧石的几率可以在配置文件中修改，帖后会说明。
然后使用燧石右键石头我们会获得两个燧石碎片。

然后我们就可以用它来砍木头做燧石镐啦
另外游戏内无法通过原木合成木板，需要在配置文件中更改。

配置文件详解
  1. # Configuration file

  3. general {

  5.     tools {
  6.         # List of mods that tools will always work for. All other mod tools will become wet noodles.
  7.        # 下面mod列表里的工具可以使用，不在列表里的mod的工具会无法使用
  8.         S:MOD_TOOL_WHITELIST <
  9.             opencomputers
  10.             tconstruct
  11.             tinkersurvival
  12.          >

  14.         # List of individual tools that will always work. Format tooltype-modid:item
  15.         # 此项使用添加单个工具的白名单，需要在前面打上工具类型如镐子pickaxe-物品id
  16.         S:TOOLS_WHITELIST <
  17.             shears-minecraft:shears
  18.             hammer-immersiveengineering:tool
  19.             wirecutter-immersiveengineering:tool
  20.             sword-immersiveengineering:revolver
  21.             sword-immersiveengineering:railgun
  22.             pickaxe-immersiveengineering:drill
  23.             wrench-ic2:wrench
  24.             wrench-ic2:electric_wrench
  25.          >
  26.     }

  28.     armor {
  29.         # List of all armor with temperature modifiers. Format modid:item=modifier
  30.         # 意志坚定联动，以下物品可以控制温度，1为升温 -1为降温
  31.         S:ARMOR_TEMP_MODIFIERS <
  32.             tinkersurvival:reinforced_wool_helmet=1
  33.             tinkersurvival:reinforced_wool_chestplate=1
  34.             tinkersurvival:reinforced_wool_leggings=1
  35.             tinkersurvival:reinforced_wool_boots=1
  36.             tinkersurvival:reinforced_jelled_slime_helmet=-1
  37.             tinkersurvival:reinforced_jelled_slime_chestplate=-1
  38.             tinkersurvival:reinforced_jelled_slime_leggings=-1
  39.             tinkersurvival:reinforced_jelled_slime_boots=-1
  40.          >

  42.         # List of individual armor items that will always work. Format modid:item
  43.         # 匠魂盔甲联动，适用于添加单个防具白名单,否则无法穿戴
  44.         S:ARMOR_WHITELIST <
  45.             tconstruct:piggybackpack
  46.          >

  48.         # Construct's Armory jelled slime cooling modifier amount.
  49.         # 意志坚定联动，类似降温的程度
  50.         I:CHILLING_MODIFIER=-1

  52.         # Construct's Armory wool insulation modifier amount.
  53.         # 意志坚定联动，类似升温的程度
  54.         I:INSULATED_MODIFIER=1

  56.         # List of mods that armor will always work and be craftable for. All other mod armor will not be equipable or craftable.
  57.         # 匠魂盔甲联动，添加整个mod的防具白名单，否则无法穿戴
  58.         S:MOD_ARMOR_WHITELIST <
  59.             immersiveengineering
  60.             conarm
  61.          >
  62.     }

  64.     balance {
  65.         # Enables the generation of rock piles on the surface. Default true
  66.         # 是否启动岩石碎片散落
  67.         B:ENABLE_ROCKGEN=true

  69.         # Enables rock drop from harvesting dirt. (default: false)
  70.         # 是否启动打掉泥土掉落岩石
  71.         B:ENABLE_ROCK_FROM_DIRT=false

  73.         # Chance for a successful flint knapping. Default 60% - 0.6D
  74.         # 打掉沙砾掉落燧石的几率，百分比
  75.         D:FLINT_CHANCE=0.6

  77.         # Chance for tall grass to drop plant fibers. Default 60% - 0.6D. Knives are 40% more effective.
  78.        # 高草堆掉落植物纤维的几率，使用小刀几率额外提高40%
  79.         D:GRASS_FIBER_CHANCE=0.5

  81.         # Heal rate for bandages. Crude bandages are 50% less effective.
  82.         # 绷带的回血效率，粗制绷带的效率降低50%
  83.         D:HEAL_RATE=0.14

  85.         # Chance for a rocks to generate on surface. Default 1.0D (= 100%)
  86.         # 石头掉落岩石的几率
  87.         D:ROCKGEN_CHANCE=1.0

  89.         # Chance for a rocks to drop from harvesting dirt with bare hands. Default 0.4 (= 40%)
  90.         # 打掉泥土掉落岩石的几率，上面ENABLE_ROCK_FROM_DIRT选项如果是false则不生效
  91.         D:ROCK_FROM_DIRT_CHANCE=0.4

  93.         # Option to adjust slow down on wrong tool usage. Default 1.0)
  94.         # 使用错误工具撸方块的速度
  95.         D:SLOW_DOWN_MULTIPLIER=1.0

  97.         # Chance for stick drip from breaking leaves by hand. Default 20% - 0.2F)
  98.         # 打掉叶子获得木棍的几率
  99.         D:STICK_DROP_CHANCE_HAND=0.20000000298023224

  101.         # Chance for stick drip from breaking leaves with a knife Default 50% - 0.5F)
  102.         # 小刀打树叶获得木棍的概率
  103.         D:STICK_DROP_CHANCE_KNIFE=0.5
  104.     }

  106.     features {
  107.         # Enables the saw item to craft planks. Default true
  108.         # 是否让锯子能够合成木板
  109.         B:ENABLE_SAW=true

  111.         # Removes plank and stick recipes from the game, only craftable with saw. Default true
  112.         # 是否移除游戏中木板的木棍的合成方法，只能用锯子合成
  113.         B:FORCE_SAW_FOR_PLANKS=true

  115.         # Ok, I love endermen, make them pick up this stuff. If NO_GRIEFING=true
  116.         # 小黑能够捡起的东西，需要使下面的选项为true
  117.         S:GRIEFING_WHITELIST <
  118.             minecraft:red_flower
  119.             minecraft:yellow_flower
  120.          >

  122.         # Stop enderman griefing, wtf Notch.
  123.         # 让小黑主动捡东西
  124.         B:NO_GRIEFING=true

  126.         # I cry myself to sleep at night...
  127.         # 是否禁止睡觉
  128.         B:NO_SLEEPING=true

  130.         # Includes dirt in the slowdown of mining speed with bare hands. Default true
  131.         # 徒手挖掘是否减速
  132.         B:SLOW_DOWN_DIRT_PUNCHING=true
  133.     }

  135.     client {
  136.         # Enables the fail sound if using the wrong tool.
  137.         # 如果使用错误的工具，是否发出失败的声音(这个声音是奇怪的笑声，晚上比较恐怖)
  138.         B:ENABLE_FAIL_SOUND=true
  139.     }

  141. }

打开配置文件config修改设置

   #怪物限制。
   #注意！这并不限制怪物的数量，自然产生。
   #注意！如果设置为负数，岛屿世界将使用服务器的产生限制。
   monsterspawnlimit: 100

   #动物限制。
   #注意！如果设置为负数，岛屿世界将使用服务器的产生限制。
   animalspawnlimit: 15

   #动物繁殖限制，零意味着无限制。
   #注意！动物不会被杀死或减少到这个极限。
   breedinglimit: 0

   #鱿鱼限制。
   #注意！如果设置为负数，岛屿世界将使用服务器的产生限制。
   #最大值为100
   wateranimalspawnlimit: 15

   #村民限制。
   #注意！这是每个队员的限制，所以一个有队伍的岛屿的村民限制数量是这个值的几倍。
   #注意！刷铁场只需要10个村民。
   #零是无限的。
   villagerlimit: 10

   #实体限制。
   #使用此部分限制可以将多少个实体添加到岛上。

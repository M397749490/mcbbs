如果某些用户的头像看着难受，但还不至于被勒令修改，而他还一直水帖频繁出现的话，就需要一个屏蔽脚本了。(签名档同理)
于是我写了一个屏蔽用户头像/签名的脚本(主动点进用户空间的不屏蔽)。
可自行增加需要屏蔽的UID，里面已经写了两个示例了。
未安装Tampermonkey的可自行百度安装方法及添加自定义脚本的方法
以下是脚本


  2. // ==UserScript==
  3. // @name         屏蔽MCBBS用户头像和签名
  4. // @include      *//www.mcbbs.net/thread*
  5. // @include      *//www.mcbbs.net/redirect.php*
  6. // @include      *//www.mcbbs.net/forum-redirect-tid*
  7. // @include      *//www.mcbbs.net/forum-viewthread-tid*
  8. // @include      *//www.mcbbs.net/forum.php?mod=viewthread*
  9. // @include      *//www.mcbbs.net/forum.php?mod=forumdisplay*
  10. // @include      *//www.mcbbs.net/forum-*.html
  11. // @run-at       document-body
  12. // ==/UserScript==

  14. (function() {
  15.     const linkLen = (location.origin + '/home.php?mod=space&uid=').length;

  17.     const blockList = {
  18.         // 用户UID
  19.         2704361: {
  20.             // 是否屏蔽头像
  21.             avatar: true,
  22.             // 是否屏蔽签名
  23.             sign: true
  24.         },
  25.         // 再随便来一个示例
  26.         666: {
  27.             // 是否屏蔽头像
  28.             avatar: false,
  29.             // 是否屏蔽签名
  30.             sign: false
  31.         },
  32.     };

  34.     blockAll();

  36.     var mo = new MutationObserver(function(allmutations) {
  37.         blockAll();
  38.     });
  39.     mo.observe(document.querySelector("body"), {childList: true, subtree: true});

  41.     function blockAll() {
  42.         var users = document.getElementsByClassName("favatar");
  43.         for (var user of users) {
  44.             var link = user.getElementsByTagName("a")[0];
  45.             if (link) {
  46.                 var block = blockList[link.href.substr(linkLen)];
  47.                 if (block) {
  48.                     if (block.avatar) {
  49.                         var avatar = user.getElementsByClassName("avatar")[0];
  50.                         if (avatar) avatar.remove();
  51.                     }
  52.                     if (block.sign) {
  53.                         var table = user;
  54.                         do {
  55.                             table = table.parentElement;
  56.                         } while (table.tagName !== "TABLE");
  57.                         var sign = document.getElementById(table.id).getElementsByClassName("sign")[0];
  58.                         if (sign) sign.remove();
  59.                     }
  60.                 }
  61.             }
  62.         }

  64.         users = document.getElementsByClassName("by");
  65.         for (user of users) {
  66.             link = user.getElementsByTagName("a")[0];
  67.             if (link) {
  68.                 block = blockList[link.href.substr(linkLen)];
  69.                 if (block && block.avatar) {
  70.                     link.removeAttribute("c");
  71.                 }
  72.             }
  73.         }
  74.     }
  75. })();

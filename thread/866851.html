  1. groups:
  2.   default:
  3.     worlds:
  4.     - world
  5.     - world_the_end
  6.     - ASkyBlock_nether
  7.     - world_nether
  8.     - ASkyBlock
  9.     default-gamemode: SURVIVAL
  10.   sc:
  11.     worlds:
  12.     - sc
  13.     default-gamemode: SURVIVAL

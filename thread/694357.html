植物魔法：
魔力池
  1. //Mana Infusion/Alchemy/Conjuration
  2. //OutputStack, InputStack, ManaCost
  3. mods.botania.ManaInfusion.addInfusion(<minecraft:golden_apple:1>, <minecraft:golden_apple>, 50000);
  4. mods.botania.ManaInfusion.addAlchemy(<minecraft:sand> * 4, <minecraft:sandstone:*>, 10);
  5. mods.botania.ManaInfusion.addConjuration(<minecraft:sand:1> * 2, <minecraft:sand:1>, 5);
  6. //OutputStack
  7. mods.botania.ManaInfusion.removeRecipe(<Botania:manaResource>); //Infusion
  8. mods.botania.ManaInfusion.removeRecipe(<minecraft:leather>); //Alchemy
  9. mods.botania.ManaInfusion.removeRecipe(<minecraft:redstone> * 2); //Conjuration

祭坛
  1. //Rune Altar
  2. //OutputStack, InputArray, ManaCost
  3. mods.botania.RuneAltar.addRecipe(<minecraft:stonebrick:2>, [<minecraft:stonebrick>, <minecraft:gunpowder>], 5);
  4. //OutputStack
  5. mods.botania.RuneAltar.removeRecipe(<Botania:rune>);

  3.   Use: false

  6. joinServer:
  7.   Use: false
  8.   List: []

2.编辑 config.yml
  1. Messages:
  2.   Login:
  3.     # 设置为true，玩家登陆消息不会显示
  4.     Disabled: false
  5.     # 当全服玩家达到多少时，接下来进入的玩家就不会再显示登陆消息了
  6.     # 设置为 -1 禁用
  7.     AutoHideFrom: -1
  8.     Custom:
  9.       # 设置为true，会使用自定义登入消息，有权限cmi.messages.disablelogin 的玩家可不显示登入消息
  10.       Use: false
  11.   Logout:
  12.     # 设置为true，玩家退出消息不会显示
  13.     Disabled: false
  14.     # 当全服玩家达到多少时，接下来进入的玩家就不会再显示退出消息了
  15.     # 设置为 -1 禁用
  16.     AutoHideFrom: -1
  17.     Custom:
  18.       # 设置为true，会使用自定义退出消息，有权限cmi.messages.disablequit 的玩家可不显示退出消息
  19.       Use: false
  20.   # 第一次进服显示消息   
  21.   # 可用变量: [playername], [totalUsers], [onlinePlayers]
  22.   FirstJoinMessage:
  23.     Use: false
  24.   DeathMessage:
  25.     # 当全服玩家达到多少时，玩家的死亡消息不会再显示
  26.     # 设置为 -1 禁用
  27.     AutoHideFrom: -1  
复制代码

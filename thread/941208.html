import org.bukkit.event.player.PlayerKickEvent;
import org.bukkit.plugin.java.JavaPlugin;
import org.bukkit.command.Command;
import org.bukkit.command.CommandSender;

public class Main extends JavaPlugin implements Listener
{

    public void onEnable() {
        Bukkit.getConsoleSender().sendMessage("§b[DemonJoin] §a插件成功载入Sever！");
        Bukkit.getConsoleSender().sendMessage("§b[DemonJoin] §a妖魂QQ:1763917516");
        this.saveDefaultConfig();
        Bukkit.getPluginManager().registerEvents( this, this );
    }
    public void onDisable() {
        Bukkit.getConsoleSender().sendMessage("§b[DemonJoin] §c插件已正常关闭！");
    }

    public boolean onCommand(final CommandSender sender, final Command cmd, final String Command, final String[] args) {
        if (Command.equalsIgnoreCase( "Djoin" )) {
            if (args.length == 0) {
                sender.sendMessage( "§f========= §b[DemonJoin] §f=========" );
                sender.sendMessage( "§7/djoin info §a: 查看作者信息" );
                if (sender.hasPermission( "DemonJoin.admin" )) {
                    sender.sendMessage( "§7/djoin reload §a: 重载配置文件" );
                }
                sender.sendMessage( "§f==============================" );
            } else if (args.length == 1) {
                if (args[0].equalsIgnoreCase( "info" )) {
                    sender.sendMessage( "§f========= §b[DemonJoin] §f=========" );
                    sender.sendMessage( "§7作者: §a妖魂吖" );
                    sender.sendMessage( "§7QQ号: §a1763917516" );
                    sender.sendMessage( "§f==============================" );
                }
                if (args[0].equalsIgnoreCase( "reload" )) {
                    if (!sender.hasPermission( "DemonJoin.admin" )) {
                        sender.sendMessage( "§b[DemonJoin] §c你没有权限输入该命令." );
                        return true;
                    }
                    this.reloadConfig();
                    sender.sendMessage( "§b[DemonJoin] §a配置文件重载完成." );
                }
            }
        }
        return true;
    }

    @EventHandler
    public void 玩家进入(final PlayerJoinEvent e) {
        Player p = e.getPlayer();
        e.setJoinMessage(null);
        if(p.hasPermission(this.getConfig().getString("DemonJoin.Admin.Permission"))){
            e.setJoinMessage(this.getConfig().getString("DemonJoin.Admin.JoinMessage").replace("&","§").replace("%player%",p.getName()));
        }else
        if(p.hasPermission(this.getConfig().getString("DemonJoin.MVP+.Permission"))){
            e.setJoinMessage(this.getConfig().getString("DemonJoin.MVP+.JoinMessage").replace("&","§").replace("%player%",p.getName()));
        }else
        if(p.hasPermission(this.getConfig().getString("DemonJoin.MVP.Permission"))){
            e.setJoinMessage(this.getConfig().getString("DemonJoin.MVP.JoinMessage").replace("&","§").replace("%player%",p.getName()));
        }else
        if(p.hasPermission(this.getConfig().getString("DemonJoin.SVIP+.Permission"))){
            e.setJoinMessage(this.getConfig().getString("DemonJoin.SVIP+.JoinMessage").replace("&","§").replace("%player%",p.getName()));
        }else
        if(p.hasPermission(this.getConfig().getString("DemonJoin.SVIP.Permission"))){
            e.setJoinMessage(this.getConfig().getString("DemonJoin.SVIP.JoinMessage").replace("&","§").replace("%player%",p.getName()));
        }else
        if(p.hasPermission(this.getConfig().getString("DemonJoin.VIP+.Permission"))){
            e.setJoinMessage(this.getConfig().getString("DemonJoin.VIP+.JoinMessage").replace("&","§").replace("%player%",p.getName()));
        }else
        if(p.hasPermission(this.getConfig().getString("DemonJoin.VIP.Permission"))){
            e.setJoinMessage(this.getConfig().getString("DemonJoin.VIP.JoinMessage").replace("&","§").replace("%player%",p.getName()));
        }else{
            e.setJoinMessage(this.getConfig().getString("DemonJoin.Default.JoinMessage").replace("&","§").replace("%player%",p.getName()));
        }
    }
    @EventHandler

        }else{

        }
    }
    @EventHandler
    public void 玩家断开(final PlayerKickEvent e) {
        Player p = e.getPlayer();
        e.setLeaveMessage(null);
        if(p.hasPermission(this.getConfig().getString("DemonJoin.Admin.Permission"))){
            e.setLeaveMessage(this.getConfig().getString("DemonJoin.Admin.QuitMessage").replace("&","§").replace("%player%",p.getName()));

        }else{
            e.setLeaveMessage(this.getConfig().getString("DemonJoin.Default.QuitMessage").replace("&","§").replace("%player%",p.getName()));
        }
    }
}

以上便是BukkitRunnable与粒子特效的小技巧，下面放送一个自己利用前几节课的知识加上本节的知识所写的一个特效
  1. // 会旋转的DNA
  2. public class RotatableDNA extends BukkitRunnable {

  4.     // 原点
  5.     private Location location;

  7.     private float yaw = 0;
  8.     private double y = 0D;

  10.     public RotatableDNA(Location location) {
  11.         this.location = location.clone();
  12.     }

  14.     @Override
  15.     public void run() {
  16.         for (double degree = 0, angle = 180; degree < 480; degree += 5, angle += 5) {
  17.             // 第一条
  18.             double radians = Math.toRadians(degree);
  19.             double x = Math.cos(radians);
  20.             double z = Math.sin(radians);
  21.             Vector vector = VectorUtils.getVector(location, location.clone().add(x, y, z));
  22.             Vector rotatedVector = VectorUtils.rotateVector(vector, yaw, 0);
  23.             location.add(rotatedVector);
  24.             ParticleEffect.REDSTONE.display(new ParticleEffect.OrdinaryColor(Color.RED), location, 50);
  25.             location.subtract(rotatedVector);

  27.             // 第二条
  28.             double radians2 = Math.toRadians(angle);
  29.             double x2 = Math.cos(radians2);
  30.             double z2 = Math.sin(radians2);

  32.             Vector vector2 = VectorUtils.getVector(location, location.clone().add(x2, y, z2));
  33.             Vector rotatedVector2 = VectorUtils.rotateVector(vector2, yaw, 0);
  34.             location.add(rotatedVector2);
  35.             ParticleEffect.REDSTONE.display(new ParticleEffect.OrdinaryColor(Color.RED), location, 20);
  36.             location.subtract(rotatedVector2);

  38.             //中间连线
  39.             if (degree % 30 == 0) {
  40.                 Location pointA = location.clone().add(rotatedVector);
  41.                 Location pointB = location.clone().add(rotatedVector2);
  42.                 EntityNBT.buildLine(pointA, pointB);
  43.             }

  45.             y += 0.1;
  46.         }

  48.         // 将yaw设定在0~360之间进行循环
  49.         if (yaw >= 360) {
  50.             yaw = 0;
  51.         } else {
  52.             yaw += 5;
  53.         }
  54.         y = 0;
  55.     }

  57.     public void startEffect() {
  58.         runTaskTimer(主类实例, 0L, 1L);
  59.     }
  60. }

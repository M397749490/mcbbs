是我写的插件创建和更改特效状态的例子

public static void spawnCustom(){
    Player player= Bukkit.getPlayer("xx");
    World world=player.getWorld();
    Map<String,Integer> pngName=new HashMap<>();
    pngName.put("exmple1",20);
    pngName.put("exmple2",20);
    pngName.put("exmple3",30);

    //创建一个效果的实体
    CustomParticle customParticle=new CustomParticle(
            "exmple", ParticleType.CUSTOM,
            new Vector(player.getLocation().getX(), player.getLocation().getY(),player.getLocation().getZ()),
            1f,true,3f,3,3,3,
            pngName
    );
    //设置朝某个方向移动
    customParticle.setMove(new Vector(0.1,0.2,0.1));

    //创建粒子  查看方法注释
    MuPApi.createParticle(customParticle,world,30);
    //生成粒子之后想改变粒子的属性也是可以的
    //设置你的渲染的材质,可以new一个新的放进去
    customParticle.setPngName(pngName);
    //更新了材质
    MuPApi.updateParticle(customParticle, FiledType.PNGS);
    //更新透明度
    customParticle.setAlpha(0.5f);
    MuPApi.updateParticle(customParticle,FiledType.ALPHA);
    //其余的FiledType如上述更新
}

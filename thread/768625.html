  1.         public LivingEntity getNearestLivingEntity(LivingEntity livingEntity) {
  2.                 LivingEntity nearestLivingEntity = null;
  3.                 double minDistance = Double.MAX_VALUE;
  4.                 for (LivingEntity livingEntity2 : livingEntity.getWorld().getLivingEntities()) {
  5.                         if (livingEntity.equals(livingEntity2)) {
  6.                                 continue;
  7.                         }
  8.                         double distance = livingEntity2.getLocation().distance(livingEntity.getLocation());
  9.                         if (distance < minDistance) {
  10.                                 minDistance = distance;
  11.                                 nearestLivingEntity = livingEntity2;
  12.                         }
  13.                 }
  14.                 return nearestLivingEntity;
  15.         }

  1. // ==UserScript==
  2. // @name         MCBBS Live2D
  3. // @namespace    https://i.zapic.cc/
  4. // @version      0.1
  5. // @description  Live2D!
  6. // @author       Zapic
  7. // @match        https://*.mcbbs.net/*
  8. // @run-at       document-body
  9. // @require      https://cdn.jsdelivr.net/gh/Dreamer-Paul/Pio@master/static/l2d.js
  10. // @require      https://cdn.jsdelivr.net/gh/Dreamer-Paul/Pio@master/static/pio.js
  11. // ==/UserScript==

  13. (()=>{
  14.     let $ = jQuery;
  15.     $("head").append("<link href='//cdn.jsdelivr.net/gh/Dreamer-Paul/Pio@master/static/pio.css' rel='stylesheet' type='text/css'/>");
  16.     $("body").append(`<div class="pio-container left"><div class="pio-action"></div><canvas id="pio" width="250" height="200"></canvas></div>`);
  17.     let pio = new Paul_Pio({
  18.     "mode": "fixed",
  19.     "hidden": true,
  20.     "content": {},
  21.     "model": ["https://cdn.jsdelivr.net/gh/Dreamer-Paul/Pio@master/models/pio/model.json"]
  22. });
  23. })();

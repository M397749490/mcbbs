标题: BSP的附属插件们到底怎么用？
  • GuiShopManager - 使用左键/右键点击背包或快捷栏内的指定物品打开特定菜单（BossShop专用版）
  • BS-SingleUse - 限制物品购买次数
  • GuiShopManager - 使用左键/右键点击背包或快捷栏内的指定物品打开特定菜单（BossShopPro专用版）
  • ThirdCurrency - 增加第三种货币
  • TrickyShops - 更方便地创建玩家商店
  • ItemShops - 更便捷的系统商店
  • PlayerShops GUI - 允许玩家创建自己的GUI商店
  • LimitedUses - 为商店图标设置冷却时间和使用次数限制
  • EssentialsGUI - 更方便的Kit礼包领取和Warp地标传送系统（需要Essentials插件）

GuiShopManager不用多说plugins\BossShopPro\addons\GuiShopManager文件夹下
ItemShops配置文件在BossShopPro下有一个分页的配置pagelayout，addons下是配置。在shops文件夹里有一个ItemShopExample.yml这个是ItemShop卖的东西和价格
LimitedUses是可给BSP增加了两个条件：交易(使用)次数和冷却时间举例：
# This is an example shop, demonstrating the LimitedUses addon
ShopName: LimitedUses
DisplayName: '&4&lLimitedUses'
Command: limiteduses
signs:
  text: '[LimitedUses]'
  NeedPermissionToCreateSign: true
shop:
  Diamond:
    MenuItem:
    - type:DIAMOND
    - amount:1
    - name:&dDiamond
    - 'lore1:&8You can purchase one diamond every ten hours.'
    - 'lore2:&cPrice: %price%.'
    RewardType: ITEM
    Reward:
    - - amount:1
      - type:DIAMOND
    PriceType: MONEY
    Price: 150.0
    Message: '&eYou bought a diamond for %price%.'
    ExtraPermission: ''
    InventoryLocation: 1
    Condition:
    - type:cooldown   <-----这里是冷却
    - over:36000   <-----冷却的时间
  Diamond_blocked:
    MenuItem:
    - type:BARRIER
    - amount:1
    - name:&f&lOut of stock
    RewardType: NOTHING
    PriceType: NOTHING
    Message: '&8Come back in %cooldown_LimitedUses:Diamond%.'
    ExtraPermission: ''
    InventoryLocation: 1
  StarterKit:
    MenuItem:
    - type:WOOD_PICKAXE
    - amount:1
    - name:&fStarter Kit
    - 'lore:&8Click to receive your starter kit.'
    RewardType: ITEM
    Reward:
    - - amount:1
      - type:WOOD_PICKAXE
    - - amount:1
      - type:WOOD_SPADE
    - - amount:1
      - type:WOOD_AXE
    - - amount:1
      - type:WOOD_SWORD
    - - amount:16
      - type:TORCH
    - - amount:32
      - id:4
    PriceType: NOTHING
    Message: '&eReceived your starter kit.'
    ExtraPermission: ''
    InventoryLocation: 2
    Condition:
    - type:uses   <-----这里是使用次数
    - under:1   <-----可用几次
  StarterKit_blocked:
    MenuItem:
    - type:BARRIER
    - amount:1
    - name:&f&lOut of stock
    - 'lore:&cYou have already taken your starter kit.'
    RewardType: NOTHING
    PriceType: NOTHING
    Message: ''
    ExtraPermission: ''
    InventoryLocation: 2
  Obsidian:
    MenuItem:
    - type:OBSIDIAN
    - amount:64
    - name:&964 Obsidian
    - 'lore1:&cPrice: 10000 Money'
    - 'lore2:&cBought: &b%uses%&c/&b15 &ctimes.'
    RewardType: ITEM
    Reward:
    - - type:OBSIDIAN
      - amount:64
    PriceType: MONEY
    Price: 10000
    Message: '&eYou bought %reward%! Money left: &c%left%'
    ExtraPermission: ''
    InventoryLocation: 3
    Condition:
    - type:uses   <-----一样的
    - under:15   <-----一样的

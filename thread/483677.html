public void onPlayerInteractEvent(PlayerInteractEvent event)
        {
                Player onlinep=event.getPlayer();
                Material a=onlinep.getInventory().getItemInHand().getType();
                FileConfiguration config=getConfig();
                int level=onlinep.getLevel();
                List<String> il=config.getStringList("Level");
                if(a==Material.getMaterial(config.getStringList("Item")) && level<10)            //这一段提示

        }

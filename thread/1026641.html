  17.             String name = inv.getItem(i).getItemMeta().getDisplayName();

  22.                     List<String> list = Main.map.get(s);

  24.                     if (list.size()==lore.size()){
  25.                         boolean b = DiGui(list, lore, list.size() - 1);

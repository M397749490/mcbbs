配置文件:
  1. # 开启/关闭传送(指南针)
  2. # true 开启 false 关闭
  3. compass: true
  4. compassItem: compass
  5. arenaclock: true
  6. clockItem: watch
  7. spectatorsTools: true
  8. spectatorsToolsItem: magma_cream
  9. tpToDeathTool: true
  10. tpToDeathToolShowCause: true
  11. inspector: true
  12. inspectorItem: book
  13. inspectPlayerFromTeleportationMenu: true
  14. playersHealthInTeleportationMenu: true
  15. playersLocationInTeleportationMenu: true
  16. specchat: true
  17. outputmessages: true
  18. # 开启/关闭 死亡旁观(默认关闭)
  19. deathspec: false
  20. # 列表生命显示 (?)
  21. colouredtablist: true
  22. seespecs: false
  23. blockcmds: true
  24. adminbypass: true
  25. newbieMode: true
  26. teleportToSpawnOnSpecChangeWithoutLobby: true
  27. useSpawnCommandToTeleport: true
  28. enforceArenaBoundary: true
  29. # 插件版本 请勿更改
  30. version: 2.0

public boolean onCommand(CommandSender sender, Command cmd, String label, String[] args){
if(!(sender instanceof Player)) { //看sender是不是player类的实例 预防控制台发送命令
  return false;
}
Player p1=(Player)sender; //将sender强制转换为Player类型并赋值给p1
if(cmd.getName().equals("command"){ //引号里添命令 如果玩家输入命令以后进入分支
   p1.teleport(loc);  //括号里传入location类型的数据
}
return false;
}

  3. var xr = Math.floor(Math.random()*1);//取0-1的随机数 用来判断+ 和 -
  4. var x;
  5. var y = npc.y;
  6. var z;
  7. if(xr == 1){x=npc.x+i}
  8. if(xr == 0){x=npc.x-i}
  9. if(zr == 1){x=npc.z+i}
  10. if(zr == 0){x=npc.z-i}
  11. npc.setPosition(x,npc.y,z);

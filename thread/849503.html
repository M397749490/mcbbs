如果你安装了mv多世界插件的话修改mv插件中名为world.yml配置文件，在对应世界名下面
world:
    ==: MVWorld
    hidden: 'false'
    alias: "&c>>\u751f\u5b58\u4e16\u754c<<"
    color: WHITE
    style: NORMAL
    pvp: 'true'
    scale: '1.0'
    respawnWorld: ''
    allowWeather: 'true'
    difficulty: EASY
    spawning:
      ==: MVSpawnSettings
      animals:
        ==: MVSpawnSubSettings
        spawn: 'true'    <-------这一项true 为动物生成 false为动物不生成
        spawnrate: '-1'
        exceptions: []
      monsters:
        ==: MVSpawnSubSettings
        spawn: 'true'
        spawnrate: '-1'
        exceptions: []
    entryfee:
      ==: MVEntryFee
      amount: '0.0'
      currency: '-1'
    hunger: 'true'
    autoHeal: 'true'
    adjustSpawn: 'true'
    portalForm: ALL
    gameMode: SURVIVAL
    keepSpawnInMemory: 'false'
    spawnLocation:
      ==: MVSpawnLocation
      x: 145.0
      y: 75.0
      z: 268.0
      pitch: 0.0
      yaw: 0.0
    autoLoad: 'true'

    allowFlight: 'true'

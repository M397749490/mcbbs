  1. kick_player: "&c你已被踢出 原因: &e%reason%"
  2. kick_broadcast: "%player% &e被GAC踢出 &c原因: &e%reason%"
  3. invalid_skin: "&c无效的皮肤"

  5. #GAC 配置文件
  6. version: 5 #请勿修改

  8. 简单的反作弊插件，其目的是严格，快速地检测并防止作弊。

  10. 使用这个插件,必须安装KotlinLib前置!

  12. 新raknet兼容的构建可以在这里找到

  14. GAC 可以防御:
  15. fly
  16. speed
  17. aimbots
  18. glide
  19. highjump
  20. jesus
  21. nuker
  22. speedmine
  23. teleport
  24. noclip
  25. reach
  26. spider
  27. fast ladder
  28. fast jump
  29. bhop

  31. GAC 现在不会防御:
  32. hitbox (仅仅是现在)
  33. kill aura (他还没有做出来)

  35. GAC 支持:
  36. 自定义动作（例如NXBooster或Double Jump之类的插件）
  37. 踢出 ** (bans aren't implemented due to reliability)

  39. GAC 不支持或不检查玩家:
  40. cobweb
  41. elytra
  42. vehicles
  43. creative or with fly/noclip enabled
  44. sprinting and jumping on ice (有些时候)
  45. very fast teleport (multiple times per second) might cause glitches


  48. #GAC 是否会在所有级别都处于活动状态（如果未在下面进行更改）或不是
  49. default_active: true

  51. #添加所有反作弊应处于与#default_active属性相反的状态的世界，
  52. 排除在外级别: [""]
  53. #excluded_levels: ["world1", "test", "survival world"]

  55. #如果为true，则插件会在检测到作弊的情况下踢出玩家
  56. kick_players: true

  58. #如果为true，则反作弊会检查OP玩家
  59. check_ops: false

  61. #管理所有可用的检查 (启用/关闭)
  62. checks:
  63.   speed: true #垂直和水平速度
  64.   fly: true #fly和其他无效运动检查
  65.   speedmine: true
  66.   nuker: false #检查玩家是否可以看到障碍物（此检查比其他检查使用更多的CPU资源 ）
  67.   glide: true
  68.   teleport: true
  69.   noclip: true
  70.   spam: true
  71.   reach: true
  72.   aimbot: true
  73.   bhop: true

  75. 设置:
  76.   spam_delay: 8 #两个消息之间的间隔＃当前未实现
  77.   hit_range: 4 #最大命中距离

  79. #当前GAC在使用elytra时无法执行移动检查
  80. #这意味着玩家可以使用elytra来越过反作弊
  81. #此选项禁用了elytra，因此玩家无法使用它
  82. enable_elytra: false

  84. #玩家皮肤中可以包含的透明像素点的最大百分比
  85. #具有更高百分比的玩家将被踢出
  86. #也就是反隐身
  87. max_skin_transparency: 0.5

  89. #一次移动的最大距离（以一个草方块为单位）
  90. max_movement_distance: 10

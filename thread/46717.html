\plugins\Residence\config.yml用文本编辑器打开（勿用记事本）：
  1. # These groups correspond to the Permissions groups defined in your '<worldname>.yml'
  2. Groups:
  3.     Default: #group name
  4.        # Information about the Residence zone that the player can define.
  5.        Residence:
  6.            # Determins if this group can create residences or not.  This option can be overriden with the permissions node 'residence.create'
  7.            CanCreate: true
  8.            # The maximum number of Residences a player can have.
  9.            MaxResidences: 3                        这一行的数字代表一个玩家最多能建立的领地个数
  10.            # The maximum number of physical areas a residence can have.
  11.            MaxAreasPerResidence: 2
  12.            # The maximum number of blocks a Residence can be, East to West (X).
  13.            MaxEastWest: 16
  14.            # The maximum number of blocks a Residence can be, North to South (Z).
  15.            MaxNorthSouth: 16
  16.            # The maximum number of blocks a Residence can be, High to Low (Y).
  17.            MaxUpDown: 256
  18.            # The lowest altitude this group is allowed to protect.
  19.            MinHeight: 0
  20.            # The highest altitude this group is allowed to protect.
  21.            MaxHeight: 255

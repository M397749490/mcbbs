【MC反作弊开发日记-第一章-纯Java编写Mod校验】
第一节-需求分析
在本节我们将要分析如何设计一个使用BukkitAPI以及ForgeAPI的Mod校验应用。
一.通信模型
    我们采用C-S模型即client-server（客户端-服务器模型）。
    在这个模型中需要注意5点。
        1.请求由客户端发起
        2.客户端任何信息都不可信任
        3.服务器需要能同时响应多个请求
         4.设计服务器时安全性排在第一位
        5.通信过程需要加密（解决第二点客户端不可信）
二.客户端设计
    1.分析
        1).文件校验
            (1).需要校验的文件存在多个
            (2).需要校验的文件可能存在因为某些问题而读取失败
            (3).读取文件时可能被hook（例如hook住ReadFile函数）
            (4).读取到的数据在内存中可能被篡改
                  ①.客户端被修改
                  ②.直接在文件读取完成后修改内存区域（基本没人用这种办法吧）
            (5).采用何种Hash算法
        2).数据传给服务器
            (1).传输可能加密
            (2).传输可能中断
            (3).传输可能被拦截
    2.实现
        1).文件校验
            (1).对目录进行遍历
            (2).读取失败时停止校验并且发送读取失败消息给服务器
            (3).现阶段无法防止Hook
            (4).现阶段无法防止修改
            (5).javaAPI(SE)自带的API中已经有SHA MD5 CRC等实现
        2).数据传输
            (1).加密过程以服务器为主导
            (2).中断即失败，此时需要退出在线游戏或者客户端
            (3).现阶段无法判断拦截
            (3).使用MC通信协议的plugin channel进行数据传输（单个包不大于64千字节）
            (4).使用PCD_ANTICHEAT_MOD作为plugin channel的name
        3).Forge Mod
三.服务器设计
    1.分析
        1).接收客户端数据
            (1).同时可能存在多个客户端连接
            (2).客户端数据可能有假
            (3).客户端客户端可能利用协议进行攻击
        2).对数据进行处理
            (1).校验失败
            (2).校验成功
    2.实现
        1).数据传输
            (1).bukkit服务端已经为我们很好的管理了这个情况，所以不需要处理
            (2).通信加密来防止伪造的客户端
            (3).限制客户端连接次数和流量
        2).数据处理
            (1).将玩家踢出并记录相关信息
            (2).记录相关信息

package lona.java.minecraft.bukkit.pcdtac;

import com.google.common.io.ByteArrayDataInput;
import com.google.common.io.ByteArrayDataOutput;
import com.google.common.io.ByteStreams;
import org.apache.commons.io.FileUtils;
import org.bukkit.entity.Player;
import org.bukkit.event.EventHandler;
import org.bukkit.event.Listener;
import org.bukkit.event.player.PlayerJoinEvent;
import org.bukkit.event.player.PlayerQuitEvent;
import org.bukkit.plugin.java.JavaPlugin;

import java.io.File;
import java.io.IOException;
import java.util.*;
import java.util.concurrent.ConcurrentHashMap;

public class PCDTAC extends JavaPlugin implements Listener {

    public static final String PLUGIN_CHANNEL_NAME = "PCDTAC";
    private Map<String, Integer> uncheckPlayerMap;
    private List<String> acceptableFileList;
    private byte[] requestHashPacketData;

    public enum PacketType {
        RequestClientModFileHash(0),
        ReplyClientModFileHash(1);

        PacketType(int packet_id) {
            this.packet_id = (byte) packet_id;
        }

        private byte packet_id;

        public byte getPacket_id() {
            return packet_id;
        }

        public static byte getLastPacketID() {
            return ReplyClientModFileHash.packet_id;
        }
    }

    public enum KickReason {
        BadPacket("通信错误"),
        UnacceptableMod("存在不允许安装的Mod"),
        Timeout("超时"),
        Unknow("出现未知错误");

        KickReason(String reason) {
            this.reason = reason;
        }

        private String reason;

        public String getReason() {
            return reason;
        }
    }

    public PCDTAC() {
        uncheckPlayerMap = new ConcurrentHashMap<>();

        ByteArrayDataOutput dataOutput = ByteStreams.newDataOutput();
        dataOutput.writeByte(PacketType.RequestClientModFileHash.getPacket_id());
        requestHashPacketData = dataOutput.toByteArray();
    }

    @Override
    public void onEnable() {
        try {
            acceptableFileList = FileUtils.readLines(new File("." + File.separator + "plugins" + File.separator + "PCDTAC" + File.separator + "fileList.txt"), "UTF-8");
        } catch (IOException e) {
            acceptableFileList = new ArrayList<>();
            getLogger().info("读取fileList.txt失败");
        }
        getServer().getPluginManager().registerEvents(this, this);
        getServer().getMessenger().registerOutgoingPluginChannel(this, PLUGIN_CHANNEL_NAME);
        getServer().getMessenger().registerIncomingPluginChannel(this, PLUGIN_CHANNEL_NAME, (String channel_name, Player player, byte[] payload) -> {
            ByteArrayDataInput dataInput = ByteStreams.newDataInput(payload);
            try {
                //读取包ID
                final byte packet_id = dataInput.readByte();
                //检查包ID是否正常
                if (packet_id > PacketType.getLastPacketID()) {
                    kickPlayer(player, KickReason.BadPacket);
                } else {
                    //分包
                    if (packet_id == PacketType.ReplyClientModFileHash.getPacket_id()) {
                        final short numOfFile = dataInput.readShort();
                        List<Map.Entry<String, String>> hashList = new ArrayList<>();//Map.Entry<fileName,hashValue>
                        for (short i = 0; i < numOfFile; i++) {
                            hashList.add(new AbstractMap.SimpleEntry<>(dataInput.readUTF(), dataInput.readUTF()));
                        }
                        if (isFileListAcceptable(hashList)) {
                            uncheckPlayerMap.remove(player.getName());
                        } else {
                            kickPlayer(player, KickReason.UnacceptableMod);
                        }
                    } else//不可解析的包
                    {
                        kickPlayer(player, KickReason.BadPacket);
                    }
                }

            } catch (Exception e) {
                kickPlayer(player, KickReason.Unknow);
            }
        });

        getServer().getScheduler().runTaskTimer(this, () -> {
            final Set<String> players = uncheckPlayerMap.keySet();
            for (String playerName : players) {
                int timeout = uncheckPlayerMap.get(playerName);
                if (timeout > 0) {
                    uncheckPlayerMap.put(playerName, timeout - 1);
                } else {
                    kickPlayer(playerName, KickReason.Timeout);
                }
            }
        }, 20, 20);
        getLogger().info("反作弊插件载入成功");
    }

    @Override
    public void onDisable() {

    }

    @EventHandler
    public void onPlayerJoin(PlayerJoinEvent event) {
        final String playerName = event.getPlayer().getName();
        getServer().getScheduler().runTaskLater(this, () -> {
            final Player player = getServer().getPlayer(playerName);
            player.sendPluginMessage(this, PLUGIN_CHANNEL_NAME, requestHashPacketData);
            uncheckPlayerMap.put(playerName, 5);
        }, 20 * 1);
    }

    @EventHandler
    public void onPlayerExit(PlayerQuitEvent event) {
        uncheckPlayerMap.remove(event.getPlayer().getName());
    }

    private void kickPlayer(Player player, KickReason reason) {
        player.kickPlayer(reason.getReason());
        uncheckPlayerMap.remove(player.getName());
    }

    private void kickPlayer(String playerName, KickReason reason) {
        kickPlayer(getServer().getPlayer(playerName), reason);
    }

    public boolean isFileListAcceptable(List<Map.Entry<String, String>> hashList) {
        for (Map.Entry<String, String> pair : hashList) {
            if (!isFileAcceptable(pair.getKey(), pair.getValue())) {
                return false;
            }
        }
        return true;
    }

    public boolean isFileAcceptable(String fileName, String hashValue) {
        return acceptableFileList.contains(hashValue);
    }
}


三.客户端代码
package lona.java.minecraft.forge.pcdtac;

import com.google.common.io.ByteArrayDataInput;
import com.google.common.io.ByteArrayDataOutput;
import com.google.common.io.ByteStreams;
import io.netty.buffer.ByteBuf;
import io.netty.buffer.Unpooled;
import net.minecraft.network.PacketBuffer;
import net.minecraftforge.common.MinecraftForge;
import net.minecraftforge.fml.common.FMLCommonHandler;
import net.minecraftforge.fml.common.Mod;
import net.minecraftforge.fml.common.Mod.EventHandler;
import net.minecraftforge.fml.common.event.FMLInitializationEvent;
import net.minecraftforge.fml.common.event.FMLPreInitializationEvent;
import net.minecraftforge.fml.common.eventhandler.EventPriority;
import net.minecraftforge.fml.common.eventhandler.SubscribeEvent;
import net.minecraftforge.fml.common.network.FMLEventChannel;
import net.minecraftforge.fml.common.network.FMLNetworkEvent;
import net.minecraftforge.fml.common.network.NetworkRegistry;
import net.minecraftforge.fml.common.network.internal.FMLProxyPacket;
import net.minecraftforge.fml.relauncher.Side;
import net.minecraftforge.fml.relauncher.SideOnly;
import org.apache.commons.io.FileUtils;

import java.io.File;
import java.io.IOException;
import java.lang.reflect.Field;
import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;
import java.util.AbstractMap;
import java.util.ArrayList;
import java.util.List;
import java.util.Map;

import static lona.java.minecraft.bukkit.pcdtac.PCDTAC.PLUGIN_CHANNEL_NAME;

import lona.java.minecraft.bukkit.pcdtac.PCDTAC.PacketType;

@Mod(modid = PCDTAC.MODID, name = PCDTAC.NAME, version = PCDTAC.VERSION, acceptedMinecraftVersions = "[1.8,1.12.2]")
public class PCDTAC {
    public static final String MODID = "pcdtac";
    public static final String NAME = "PCDTAC";
    public static final String VERSION = "1.0.0";
    private FMLEventChannel channel;
    private static final String modsDirPath = System.getProperty("user.dir") + File.separator + "mods";

    @EventHandler
    public void preInit(FMLPreInitializationEvent event) {
    }

    @EventHandler
    public void init(FMLInitializationEvent event) throws IOException, NoSuchAlgorithmException {
        List<Map.Entry<String, String>> modFileHashList = getModFileHashList();
        File file = new File("." + File.separator + "mods" + File.separator + "PCDTAC" + File.separator + "fileList.txt");
        if (!file.getParentFile().exists()) {
            file.getParentFile().mkdirs();
        }
        for (Map.Entry<String, String> fileHash : modFileHashList) {
            FileUtils.write(file, fileHash.getValue() + '\n', "UTF-8", true);
        }
        FMLCommonHandler.instance().bus().register(this);
        MinecraftForge.EVENT_BUS.register(this);
        (channel = NetworkRegistry.INSTANCE.newEventDrivenChannel(PLUGIN_CHANNEL_NAME)).register(this);
    }

    @SubscribeEvent(priority = EventPriority.LOWEST)
    @SideOnly(Side.CLIENT)
    public void onServerCustomPacket(FMLNetworkEvent.ClientCustomPacketEvent event) {
        try {
            final Field packetField = event.getClass().getSuperclass().getDeclaredField("packet");
            final FMLNetworkEvent.CustomPacketEvent customPacketEvent = event;
            packetField.setAccessible(true);
            final FMLProxyPacket packet = (FMLProxyPacket) packetField.get(customPacketEvent);
            ByteBuf buffer = packet.payload();
            byte[] data = new byte[buffer.readableBytes()];
            buffer.readBytes(data);
            ByteArrayDataInput dataInput = ByteStreams.newDataInput(data);
            byte packet_id = dataInput.readByte();
            if (packet_id == PacketType.RequestClientModFileHash.getPacket_id()) {
                ByteArrayDataOutput dataOutput = ByteStreams.newDataOutput();
                List<Map.Entry<String, String>> modFileHashList = getModFileHashList();
                dataOutput.writeByte(PacketType.ReplyClientModFileHash.getPacket_id());
                dataOutput.writeShort(modFileHashList.size());
                for (Map.Entry<String, String> pair : modFileHashList) {
                    dataOutput.writeUTF(pair.getKey());
                    dataOutput.writeUTF(pair.getValue());
                }
                channel.sendToServer(new FMLProxyPacket(new PacketBuffer(Unpooled.wrappedBuffer(dataOutput.toByteArray())), PLUGIN_CHANNEL_NAME));
            }
        } catch (NoSuchFieldException | IllegalAccessException | IOException | NoSuchAlgorithmException e) {
            e.printStackTrace();
        }
    }

    public String getFileHash(File file) throws IOException, NoSuchAlgorithmException {
        byte[] fileData = FileUtils.readFileToByteArray(file);
        MessageDigest messageDigest = MessageDigest.getInstance("MD5");
        messageDigest.update(fileData);
        byte[] hashBytes = messageDigest.digest();
        StringBuilder hash = new StringBuilder();
        for (byte b : hashBytes) {
            hash.append(String.format("%02X", b));
        }
        return hash.toString();
    }

    public List<Map.Entry<String, String>> getModFileHashList() throws IOException, NoSuchAlgorithmException {
        File[] files = new File(modsDirPath).listFiles((File file) -> file.getName().toLowerCase().endsWith("jar") || file.getName().toLowerCase().endsWith("zip"));
        List<Map.Entry<String, String>> modFileHashList = new ArrayList<>();
        assert files != null;
        for (File file : files) {
            String hash = getFileHash(file);
            modFileHashList.add(new AbstractMap.SimpleEntry<>(file.getPath(), hash));
        }
        return modFileHashList;
    }

}

作者: volta薯片~    时间: 2016-6-7 17:42

#Main configuration

#启用的世界，格式照抄
disabled-worlds:
- world_nether
- city

#在线人数里算不算假人，true是不算
#This will not count vanished players in the online value
hide-vanished: false

#接下来是记分版的格式
Scoreboard:
#大标题
Title: '&a&l'
#这里写的是秒数
# 理解为记分版刷新的频率吧
Update-delay: 2

Items:
#这里填写下面的内容，最多16个，格式见下
#'标题'：'变量'
#变量可在搬运贴查询
'&9Online': '%online%'
'&9Money': '%money%'


#让记分版能记录你的战绩
enable-pvpstats: false


#战绩的保存（分钟）
PvPStats-SaveIntervall: 15

#没看懂，全服统计？
Temp-Scoreboard-enabled: false

#这是全服统计的设置，基本同上
Temp-Scoreboard:
Title: '&a&lTop Kills'
#这下面是可用的变量
#%mob% | %kills% | %killstreak%
Type: '%kills%'
Color: '&9'
#多少玩家会被显示
Items: 5
Intervall-show: 300
Intervall-disappear: 300

#开启自动更新
pluginUpdate: true

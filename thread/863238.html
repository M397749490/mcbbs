public class MythicMobsApi
{
  public static MythicMob getMythicMob(String type)
  {
    return MythicMobs.inst().getMobManager().getMythicMob(type);
  }

  public static Entity spawnMythicMob(MythicMob type, Location location, int level)
    throws InvalidMobTypeException
  {
    if (type == null) {
      throw new InvalidMobTypeException("Supplied mob type cannot be null.");
    }

    return type.spawn(BukkitAdapter.adapt(location), level).getEntity().getBukkitEntity();
  }

  public static Entity spawnMythicMob(String type, Location location)
    throws InvalidMobTypeException
  {
    return spawnMythicMob(type, location, 1);
  }

  public static Entity spawnMythicMob(String type, Location location, int level)
    throws InvalidMobTypeException
  {
    MythicMob mm = MythicMobs.inst().getMobManager().getMythicMob(type);
    if (mm == null) {
      throw new InvalidMobTypeException(type + " is not a valid Mythic Mob type.");
    }

    return mm.spawn(BukkitAdapter.adapt(location), level).getEntity().getBukkitEntity();
  }

  public static boolean isMythicMob(Entity l) {
    return MythicMobs.inst().getMobManager().isActiveMob(BukkitAdapter.adapt(l));
  }

  public static boolean isMythicMob(UUID u) {
    return MythicMobs.inst().getMobManager().isActiveMob(u);
  }

  public static ActiveMob getMythicMobInstance(Entity e) {
    return MythicMobs.inst().getMobManager().getMythicMobInstance(BukkitAdapter.adapt(e));
  }

  public static boolean castSkill(Entity e, String skillName) {
    return castSkill(e, skillName, e.getLocation(), 1.0F);
  }

  public static boolean castSkill(Entity e, String skillName, float power) {
    return castSkill(e, skillName, e.getLocation(), power);
  }

  public static boolean castSkill(Entity e, String skillName, Location origin) {
    return castSkill(e, skillName, origin, 1.0F);
  }

  public static boolean castSkill(Entity e, String skillName, Location origin, float power) {
    return castSkill(e, skillName, null, origin, null, null, power);
  }

  public static boolean castSkill(Entity e, String skillName, Location origin, Collection<Entity> eTargets, Collection<Location> lTargets, float power) {
    return castSkill(e, skillName, null, origin, eTargets, lTargets, power);
  }

  public static boolean castSkill(Entity e, String skillName, Entity trigger, Location origin, Collection<Entity> eTargets, Collection<Location> lTargets, float power) {
    if (!MythicMobs.inst().getMobManager().isActiveMob(e.getUniqueId())) return false;

    Optional maybeSkill = MythicMobs.inst().getSkillManager().getSkill(skillName);
    if (!maybeSkill.isPresent()) return false;

    ActiveMob am = MythicMobs.inst().getMobManager().getMythicMobInstance(e);
    Skill skill = (Skill)maybeSkill.get();

    HashSet feTargets = new HashSet();
    HashSet flTargets = new HashSet();

    if (eTargets != null) {
      for (Entity en : eTargets) {
        feTargets.add(BukkitAdapter.adapt(en));
      }
    }

    if (lTargets != null) {
      for (Location l : lTargets) {
        flTargets.add(BukkitAdapter.adapt(l));
      }
    }

    SkillMetadata data = new SkillMetadata(SkillTrigger.API, am, BukkitAdapter.adapt(trigger), BukkitAdapter.adapt(origin), feTargets, flTargets, power);

    if (skill.usable(data, SkillTrigger.API)) {
      skill.execute(data);
    }

    return true;
  }

  public static boolean addThreat(Entity mob, LivingEntity target, double amount)
  {
    if (!MythicMobs.inst().getMobManager().isActiveMob(mob.getUniqueId())) {
      return false;
    }

    ActiveMob am = MythicMobs.inst().getMobManager().getMythicMobInstance(mob);

    if (am.getThreatTable() == null) {
      return false;
    }
    am.getThreatTable().threatGain(BukkitAdapter.adapt(target), amount);

    return true;
  }

  public static boolean reduceThreat(Entity mob, LivingEntity target, double amount) {
    if (!MythicMobs.inst().getMobManager().isActiveMob(mob.getUniqueId())) {
      return false;
    }

    ActiveMob am = MythicMobs.inst().getMobManager().getMythicMobInstance(mob);

    if (am.getThreatTable() == null) {
      return false;
    }
    am.getThreatTable().threatLoss(BukkitAdapter.adapt(target), amount);

    return true;
  }

  public static boolean taunt(Entity mob, LivingEntity target) {
    if (!MythicMobs.inst().getMobManager().isActiveMob(mob.getUniqueId())) {
      return false;
    }

    ActiveMob am = MythicMobs.inst().getMobManager().getMythicMobInstance(mob);

    if (am.getThreatTable() == null) {
      return false;
    }
    am.getThreatTable().Taunt(BukkitAdapter.adapt(target));
    return true;
  }
}

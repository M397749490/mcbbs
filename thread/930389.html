  6.                         floatExp -= toNextLevel;

  8.                 }
  9.                 while(floatExp < 0) {
  10.                         floatExp += (toNextLevel = getUpgradeExp(--level));
  11.                         if(level < 0)
  12.                                 level = floatExp = 0;

  14.                 player.setLevel(level);
  15.                 player.setExp( (float) floatExp / toNextLevel);

  19.         public int getUpgradeExp(int level) {

  21.         }

  25.         }

#Disable to stop using that material in recipes. Useful if a mod adds unobtainable placeholder ores.
"materials" {
        B:"copper"=true

        B:"tin"=true

        B:"silver"=true

        B:"lead"=true

        B:"nickel"=true

        B:"steel"=true

        B:"fluxed"=true

        B:"enderium"=true

        B:"dark_steel"=true

        B:"platinum"=false

        B:"iridium"=false
}

把true全部改成false，其他模组的矿物不需要

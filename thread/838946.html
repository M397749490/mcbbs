作者
DaNseR

兼容版本
1.4.7-1.14

下载地址

这款插件对于小游戏服务器很有用,你可以用这个插件来删除多余的玩家数据
插件自动删除配置内设定的文件或文件夹


汉化配置 · Configuration
Config.yml
  1. # 变量:
  2. # %NAME% - 玩家昵称            示例: Player
  3. # %UUID% - 玩家的UUID      示例: a01e3843-e521-3998-958a-f459800e4d11
  4. # %FIXUUID% - 玩家的UUID, 但无需 "-" 示例: a01e3843e5213998958af459800e4d11
  5. # %IP% - 玩家的IPv4地址          示例: 8.8.8.8
  6. #
  7. # 检查配置是否正确，可以查看 http://yaml-online-parser.appspot.com/
  8. #
  9. # 下载地址 https://www.spigotmc.org/resources/59029/

  11. delete:
  12. # 删除文件后是否删除玩家文件 ?
  13. # !! 文件将在2500毫秒后自动删除 (50 刻) !!
  14. onQuit: false

  16. # 如果关闭了上面这项功能以下文件就不会被删除
  17. # 示例 plugins/Essentials/userdata/%UUID%.yml
  18. # !! 文件将在2500毫秒后自动删除 (50 刻) !!
  19. onQuitFile:
  20. - world/playerdata/%UUID%.dat
  21. - world/stats/%UUID%.dat
  22. - world/advancements/%UUID%.dat
  23. - plugins/Essentials/userdata/%UUID%.yml
  24. #
  25. # 示例 plugins/AuthMe/playerdata/%UUI..
  26. # !! 文件将在2500毫秒后自动删除 (50 刻) !!
  27. onQuitFolder:
  28. - ""


  31. # 是否在开服时删除以下位置的数据?
  32. onServerStart: false

  34. # 开服时删除的文件位置
  35. onServerStartFilesInDir:
  36. - world/playerdata/
  37. - world/stats/
  38. - world/advancements/
  39. - plugins/Essentials/userdata/

  41. # 如果未使用以上功能,
  42. # 可在这里删除特定的文件
  43. # 示例 usercache.json
  44. onServerStartFiles:
  45. - usercache.json


  48. # 定时删除
  49. schedule:
  50. enabled: false
  51. tasks:
  52.   # 任务名
  53.   randomname1:
  54.    # 执行延迟
  55.    delay: 10
  56.    # 删除的文件
  57.    files:
  58.    - randomfile228538.json
  59.    # 删除的文件夹
  60.    folder:
  61.    - plugins/plugin/random/文件夹名

  4. clickEvent:

  6.   action:run_command
  7.   ,
  8.   value:"/give abc minecraft:command_block 1"
  9. }
  10. clickEvent:
  11. {
  12.   action:run_command
  13.   ,
  14.   value:"/kill abc"
  15. }
  16. }

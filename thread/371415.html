  1. DisableExtraNpcItems=false

  3. #Default Item ID range is from 26700
  4. ItemStartId=26700

  6. #Default Block ID range is from 1525
  7. BlockStartId=1525

  9. #Uses unique entities ids
  10. UseUniqueEntities=true

  12. #To use this UseUniqueEntities has to be false
  13. EntityStartId=120

  15. #Navigation search range for NPCs. Not recommended to increase if you have a slow pc or on a server
  16. NpcNavRange=32

  18. #Set to true if you want the dialog command option to be able to use op commands like tp etc
  19. NpcUseOpCommands=false

  21. InventoryGuiEnabled=true

  23. #Enables CustomNpcs startup update message
  24. EnableUpdateChecker=true

  26. #Only ops can create and edit npcs
  27. OpsOnly=false     这里的false改true

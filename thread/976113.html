本插件的优点是什么？

  • 支持自定义输入的指令，支持自定义输入内容；
  • 执行判定支持PlaceholderAPI；
  • 轻量化。



本插件如何使用？

直接在插件的配置文件中编辑，以下为配置文件，一看就懂。

#######################
#指令组插件，由煎蛋编写
#本插件的条件支持PlaceholderAPI的变量
#######################
#可用的条件类型有:
#STRING_EQUAL
#STRING_NOT_EQUAL
#STRING_EQUAL_IGNORE_CASE
#NUMBER_EQUAL
#NUMBER_NOT_EQUAL
#NUMBER_GREATER
#NUMBER_EQUAL_OR_GREATER
#NUMBER_LESS
#NUMBER_EQUAL_OR_LESS
#######################
#Actions 表示的是玩家具有所有条件后执行的
#ConditionNotEnoughActions 表示的是玩家不符合任意一个条件后执行的
#######################
#[PLAYER_COMMAND] 以玩家身份执行指令,无需加"/"
#[PLAYER_MESSAGE] 向玩家发送消息
#[CONSOLE_COMMAND] 在控制台内执行指令
#[PLAY_SOUND] 播放音乐
#######################
#温馨提示!!!
#若需判断玩家是否拥有权限
#请自己在PlaceholderAPI添加一个判断权限的javascript文件
#然后通过STRING_EQUAL或STRING_EQUAL_IGNORE_CASE进行判断
CommandGroup:
  Group1:
    #输入的指令,支持中文
    Inputs:
    - 'test'
    - 'Just_For_Test'
    - '测试'
    #条件，此处可添加多个条件
    #如果您不需要任何条件让玩家可以直接执行，此处直接改为Requirements: []
    Requirements:
      Requirement1:
        #条件类型
        Type: NUMBER_EQUAL_OR_GREATER
        #输入的条件
        Condition1: '%vault_eco_balance_fixed%'
        #要求
        Condition2: '100'
        #大意是当玩家金钱大于100时才能执行
    #玩家满足条件执行的内容
    Actions:
    - '[PLAYER_MESSAGE] 你好~'
    - '[PLAYER_COMMAND] say QWQ'
    - '[CONSOLE_COMMAND] say %player_name% 你好呀'
    - '[PLAY_SOUND] ENTITY_EXPERIENCE_ORB_PICKUP-1-1'
    #条件不足时执行的内容
    ConditionNotEnoughActions:
    - '[PLAYER_MESSAGE] 条件不足哟'
  #实例
  Developers_Introdution:
    Inputs:
    - "bukkit:pl"
    - "bukkit:plugins"
    - "bukkit:help"
    - "bukkit:?"
    - "bukkit:about"
    - "bukkit:version"
    - "bukkit:ver"
    - "pl"
    - "plugins"
    - "about"
    - "ver"
    - "version"
    - "?"
    - "help"
    #无需任何条件，只要玩家执行以上的指令都会触发
    Requirements: []
    Actions:
    - '[PLAY_SOUND] ENTITY_EXPERIENCE_ORB_PICKUP-1-1'
    - '[PLAYER_MESSAGE] &9正在准备开发者简介...'

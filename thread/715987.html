#include<cstdio>using namespace std;int n,m,a,b,op,head[200009],next[200009],dis[200009],len[200009],v[200009],l,r,team[200009],pd[100009],u,v1,e;int lt(int x,int y,int z){    op++,v[op]=y;    next[op]=head[x],head[x]=op,len[op]=z;}int SPFA(int s,int f)//SPFA……{    for(int i=1;i<=200009;i++){dis=999999999;}    l=0,r=1,team[1]=s,pd=1,dis=0;    while(l!=r)    {        l=(l+1)%90000,u=team[l],pd=0,e=head;        while(e!=0)        {            v1=v[e];            if(dis[v1]>dis+len[e])            {                dis[v1]=dis+len[e];                if(!pd[v1])                {                    r=(r+1)%90000,                    team[r]=v1,                    pd[v1]=1;                }            }            e=next[e];        }     }    return dis[f];}int main(){    scanf("%d%d",&a,&b);    lt(1,2,a);lt(2,3,b);//1到2为a，2到3为b，1到3即为a+b……    printf("%d",SPFA(1,3));    return 0;}

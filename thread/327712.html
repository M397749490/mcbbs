package com.outlook.vincenthan2009;                              

import org.bukkit.command.Command;
import org.bukkit.command.CommandSender;
import org.bukkit.entity.Player;
import org.bukkit.plugin.java.JavaPlugin;

public class Heal extends JavaPlugin{
        
        public void onEnable(){
                getLogger().info("您的插件已被加载，感谢使用！");
        }
        @Override
        public boolean onCommand(CommandSender sender, Command cmd, String label, String[] args){               
                if (label.equalsIgnoreCase("heal")){
                        if(sender instanceof Player){
                                if (((Player) sender).getHealth()!=20.0){
                                        Player player =(Player) sender;
                                        player.setHealth(20.0);
                                        sender.sendMessage("你启用了神的能力！")
                                        return true;
                                }               
                                sender.sendMessage("你的血已经是满的了！");
                                return true;
                                
                        }
                        sender.sendMessage("§a/该命令只能被玩家执行！");
                        return true;
                        
                }                                
                return false;
        }
        

        public void onDisable(){
                getLogger().info("您的插件已被卸载，感谢使用！");
        }

}


我是想要判断玩家的血量是否为满，如果为满就sender.sendMessage("你的血已经满了！")，如果不为满就补血，但是判断是否为满血的
if (((Player) sender).getHealth()!=20.0)
却报错了，报的是“The method 'getHealth()' is ambiguous for the type Player”，现在我应该怎么改？

是不是要导入什么包呀？

Damageable是管理实体的血量,伤害等等的类
实体类的方法虽然有那些方法,但是Bukkit貌似已经取消了这些方法的支持?
只能用Damageable然后强制转换来修改实体的血量等等,
不过setMaxHealth和setHealth还是有效

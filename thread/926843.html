  4.         double theta = Math.toDegrees(Math.acos(vec.getY() / r));
  5.         double fai = Math.toDegrees(Math.atan(vec.getZ() / vec.getX()));

  7.         vec.setX(newr * Math.sin(Math.toRadians(theta)) * Math.cos(Math.toRadians(fai)));
  8.         vec.setZ(newr * Math.sin(Math.toRadians(theta)) * Math.sin(Math.toRadians(fai)));
  9.         vec.setY(newr * Math.cos(Math.toRadians(theta)));

  11.         return vec;

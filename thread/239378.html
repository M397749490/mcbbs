groups:
  Default:                    《《看这个就是群组的名字
    default: true

    - -bukkit.command.kill
    inheritance:
    - g:groupmanager_default
    - g:bukkit_default
    - g:essentials_default
    - g:towny_default

      prefix: '&e'
      build: true
      suffix: ''

/ manuadd 你名字 群组名

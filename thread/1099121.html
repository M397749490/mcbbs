    ==: MVWorld
    hidden: 'false'
    alias: '&a资源世界'
    color: WHITE

    pvp: 'true'
    scale: '1.0'
    respawnWorld: ''

      ==: MVSpawnSettings

        ==: MVSpawnSubSettings
        spawn: 'true'
        spawnrate: '-1'
        exceptions: []
      monsters:
        ==: MVSpawnSubSettings
        spawn: 'true'
        spawnrate: '-1'
        exceptions: []
    entryfee:
      ==: MVEntryFee
      amount: '0.0'

    hunger: 'true'
    autoHeal: 'true'
    adjustSpawn: 'true'
    portalForm: ALL
    gameMode: SURVIVAL 需要创造的世界改成creative即可

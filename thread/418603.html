//右键使用的时候
        @Override
        public ItemStack onItemRightClick(ItemStack par1ItemStack, World par2World,EntityPlayer par3EntityPlayer)
        {
               
                if(!par2World.isRemote)
                {
                         EntitySnowman entitysnowman = new EntitySnowman(par2World);
             entitysnowman.setLocationAndAngles((double)par3EntityPlayer.prevPosX, (double)par3EntityPlayer.prevPosY+1D, (double)par3EntityPlayer.prevPosZ, 0.0F, 0.0F);
             par2World.spawnEntityInWorld(entitysnowman);
                        par1ItemStack.damageItem(20, par3EntityPlayer);
                }
                return par1ItemStack;
        }

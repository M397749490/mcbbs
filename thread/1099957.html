作者: Gently_Lord    时间: 2020-8-12 16:52
DropEffect  没有任何权限

你要解决的事情 或许在配置文件可以 解决

#插件前缀
Name: '&7[&eDropEffect&7]'
Item:
  #ID 不可重复
  0:
    #物品标记
    Lore: '&6粗糙'
    #是否发光(此选项将影响彩虹色RainBow)
    Glow: true
    #是否开启彩虹色(此项开启将无视光的颜色)
    RainBow: true
    #光的颜色
    GlowColor: 'BLUE'
    #粒子类型
    Particle: 'CRIT'
    #是否开启光环
    Halo: true
    #光环粒子类型
    HaloParticle: 'REDSTONE'
    #光环大小
    HaloSize: 0.5
    #是否开启公告
    OpenMsg: true
    #公告
    Msg: '&7恭喜玩家&e%player%&7拾取了%item%！'
    #是否开启拾取音效
    PickupSound: true
    #拾取音效
    Sound: 'ENTITY_PLAYER_LEVELUP'
    #是否开启烟花
    FireWork: true
    #烟花是否闪烁
    Filicker: true
    #烟花是否追踪
    Trail: true
    #烟花爆炸效果
    Boom_1: 'BALL'
    Boom_2: 'BALL_LARGE'
    Boom_3: 'STAR'
    #是否开启拾取Title
    PickUpTitle: true
    #第一行Title
    FristTitle: '&7你拾取了%item%'
    #第二行Title
    LastTitle: '&f2333333'
    #是否需要权限才能拾取 格式:pickup.xxx
    Permission: 'pickup.a'
    #没有权限的提示
    PermissionMsg: '你没有权限拾取'
    #是否开启拾取后的指令
    PickUpCommand: true
    Command:
    - 'say %player%拾取了%item%!&f(&e这是一个栗子&f)'

  6. import org.bukkit.plugin.java.JavaPlugin;
  7. public class PlayerExtraApi extends JavaPlugin implements Listener {
  8.         @Override
  9.         public void onEnable() {
  10.                 Bukkit.getPluginManager().registerEvents(this, this);

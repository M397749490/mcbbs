SimpleBossBar

一个简单的Boss血条公告插件。暂不支持解析与玩家相关的Papi变量。
权限的检测是在玩家进入游戏/重载插件的时候检查的。
以后有时间的话 说不定会实现一些更复杂的功能。

配置文件


  2. simplebossbar:{
  3.   //全局的时钟周期 单位为毫秒
  4.   clock-signal:1000
  5.   //尚未实现
  6.   track-player:false
  7.   //每个组只会显示第一个玩家拥有权限的BossBar
  8.   boss-bar-groups:[
  9.     //一个组
  10.     {
  11.       //组的名字
  12.       group-name:"group1"
  13.       //一个组里所拥有的BossBar
  14.       boss-bars:[
  15.         //一个BossBar
  16.         {
  17.           //BossBar的名字
  18.           //每个BossBar所需的权限为"simplebossbar.bossbar.<group-name>.<bar-name>"
  19.           //那么这个BossBar所需的权限就是"simplebossbar.bossbar.group1.bar1"
  20.           bar-name:"bar1"
  21.           //这个bossbar每帧切换的间隔，对应的时间就是interval*clock-signal
  22.           //这里就是1000*5ms
  23.           interval:5
  24.           //默认的帧属性，与instants下的可配置内容相同
  25.           default:{
  26.             //显示的文本，可以使用&作为标记的样式代码
  27.             text:"&adefault &ctext"
  28.             //血条的颜色可以使用
  29.             //BLUE,GREEN,PINK,PURPLE,RED,WHITE,YELLOW
  30.             //前面加上"minecraft:"也可以

  32.             //是否产生迷雾(没有测试效果
  33.             create-fog:false
  34.             //是否使天空变暗（没有测试效果
  35.             darken-sky:false
  36.             //血条的样式(分节)
  37.             //notched_20, notched_10, progress, notched_6, notched_12
  38.             overlay:"progress"
  39.             //（没有测试效果）
  40.             play-end-boss-music:false
  41.             //是否可见
  42.             visible:true
  43.             //与下一项共同用来计算血条显示的百分比。
  44.             health: 1
  45.             max-health: 1
  46.           }
  47.           //帧，每一帧与default项的可设置内容相同
  48.           instants:[
  49.             {
  50.               text:"&cinstant &a1"
  51.               visible:true
  52.               health: 1

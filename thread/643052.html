  2. <NuoyuCheatsConfiguration xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance" xmlns:xsd="http://www.w3.org/2001/XMLSchema">
  3. <onlinenumber>10</onlinenumber>
  4. </NuoyuCheatsConfiguration>

  1. title:
  2.   Enable: false#是否开启升级时显示title消息
  3.   Title: '&aLevel Up!'
  4.   SubTitle: '&anew level:%level%,old level: %oldlevel%'
  5.   FadeIn: 1#淡入时间
  6.   Time: 3#持续时间
  7.   FadeOut: 1#淡出时间
  8. message:
  9.   Enable: false#是否开启升级时输出文本消息
  10.   Msg: '&c==============/n&aLevel Up!/n&aold level:%oldlevel%/n&anew level:%level%/n&c=============='

  12. limit:
  13.   chat: false#是否开启聊天等级限制
  14.   allow_chat_level: 30#未达到该等级的玩家无法发言
  15.   deny_chat_msg: 'You cant speak!'#无法发言提示
  16. chat:
  17.   enable: true#是否开启强制将等级显示在聊天中,如果设置为true,那么如果你没有在essentials中加入{level},将会使用下面的格式!
  18.   format: '&7[&eLv&c.&a{level}&7]'
  19. levelcustom:
  20.   '10': #达到10级时
  21.     Cmd:#执行的指令,删除这两行则不执行指令
  22.       - 'say &a%player% Level Up!'#这里直接改成更改玩家前缀的指令就成
  23.     Msg: '&aLevel up!'#提示的消息,删除这一行则不提示消息
  24.   '30': #达到30级时
  25.     Cmd:
  26.       - 'say &a%player% Level Up!'
  27.       - 'say &aOhhhhh~'
  28.   '50': #达到50级时
  29.     Msg: '&aLevel up!/n&aOhhhhh~'

用SK写
  1. on level change:
  2.   if player's level <= 30:
  3.     set {称号::%player%} to "A"
  4.   else if player's level >=31:
  5.     set {称号::%player%} to "B"
  6. on chat:
  7.   set chat format to "[player](%{称号::%player%}%) > [message]"

首先，在QuantumRPG\modules\sets\items下创建一个.yml的文件作为一个新的套装，然后根据格式
以下只讲解重要的：
  1. # 这个是套装的名称（非物品显示名称）
  2. name: '&eWild Cat Set'
  3. # 这个是套装的前缀
  4. prefix: '&a&fBroken '
  5. # 这个是套装的后缀
  6. suffix: 'of Wild Cat'
  7. color:
  8.   active: '&a'
  9.   inactive: '&8'
  10. # 以下就是设置的套装了
  11. elements:
  12.   # 这个是头盔
  13.   helmet:
  14.     # 这个是那个部件的材质
  15.     materials:
  16.     - GOLDEN_HELMET
  17.     # 这个是那个部件的名称
  18.     name: '%prefix%Helmet %suffix%'
  19.   # 这个是胸甲（下同）
  20.   chestplate:
  21.     materials:
  22.     - GOLDEN_CHESTPLATE
  23.     name: '%prefix%Chestplate %suffix%'
  24.   # 这个是护腿
  25.   leggings:
  26.     materials:
  27.     - GOLDEN_LEGGINGS
  28.     name: '%prefix%Leggings %suffix%'
  29.   # 这个是靴子
  30.   boots:
  31.     materials:
  32.     - GOLDEN_BOOTS
  33.     name: '%prefix%Boots %suffix%'
  34. # 然后这里是套装效果
  35. bonuses:
  36.   by-elements-amount:
  37.     # 当玩家穿上两个部件时，可以添加
  38.     '2':
  39.       # 两件部件套装的显示lore
  40.       lore:
  41.       - '%c%&lSet Bonuses (2/4):'
  42.       - '%c%? +25% PvE Damage'
  43.       - '%c%? +10 Max. Health'
  44.       - '%c%? Speed I'
  45.       # 这里以及下面就是套装属性了
  46.       # 这个是比如其它属性
  47.       item-stats:
  48.         MAX_HEALTH: 10.0
  49.         PVE_DAMAGE: 25
  50.       # 这里是伤害类型，例如
  51.       # FIRE: 10
  52.       damage-types: {}
  53.       # 这里是护甲类型，例如
  54.       # WATER: 10
  55.       defense-types: {}
  56.       # 这里是药水效果
  57.       potion-effects:
  58.         SPEED: 1
复制代码

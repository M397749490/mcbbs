#coding=utf-8



import os, time

from zipfile import ZipFile,ZIP_DEFLATED

from threading import Thread



# 要备份哪个文件夹下的数据

ZipPath = "1"

# 归档保存在哪

SavePath = "2"

# 多长时间检查一次小时

Min = 1



def save_timer():

    while True:

        r = []

        # 最老的文件时间戳

        l = time.time()*100000

        # 最老的文件文件名

        ln = ""

        for path,dirnames,fs in os.walk(SavePath):

            for f in fs:

                r.append(f)

                file_creation_time = int(os.stat(os.path.join(path,f)).st_mtime*100000)

                if(file_creation_time < l):

                    l = file_creation_time

                    ln = f

        if(len(r)>7):

            # 删除最老文件

            os.remove(os.path.join(SavePath, ln))

            print("归档超过7个，删除了最早备份 "+os.path.join(SavePath, ln))



        if(time.localtime().tm_hour == 23):

            backup()



        time.sleep(Min*60)



def zipDir(dirpath,outFullName):

    zip = ZipFile(outFullName,"w",ZIP_DEFLATED)

    for path,dirnames,filenames in os.walk(dirpath):

        fpath = path.replace(dirpath,'')

        for filename in filenames:

            zip.write(os.path.join(path,filename),os.path.join(fpath,filename))

    zip.close()



def backup():

    global ZipPath, SavePath

    pathname = str(os.path.join(SavePath,"备份-" + (time.strftime("%Y-%m-%d@%H-%M-%S", time.localtime()) + ".zip")))

    zipDir(ZipPath.replace("\\","/"), pathname)

    print("备份完成 (归档路径:{0})".format(pathname))



save_timer()

    private String applyPlayerName;
    private Long dateTime;
    //根据情况再加一些字段

    public String getApplyPlayerName() {
        return applyPlayerName;
    }

    public void setApplyPlayerName(String applyPlayerName) {
        this.applyPlayerName = applyPlayerName;
    }

    public Long getDateTime() {
        return dateTime;
    }

    public void setDateTime(Long dateTime) {
        this.dateTime = dateTime;
    }
}当有A玩家发出邀请时，设置B玩家value，applyPlayerName设置成A玩家的名称，dateTime获取方式,new Date().getTime();date是java.util包下的。当B玩家输入/pvp accept的时候从map中取出对象，先判断是否为空，然后再去判断时间，public Map<String, PVPMessage> pvpMessageMap=new HashMap<>();@Override
public boolean onCommand(CommandSender sender, Command command, String label, String[] args) {
  if(label.equals("pvp")&&args.length>0){
      if("accept".equals(args[0])){
          PVPMessage pvpMessage = pvpMessageMap.get(sender.getName());
          if(pvpMessage!=null){
              //当前时间戳
              long time =new Date().getTime();
              long onlyTime=time-pvpMessage.getDateTime();
              //1秒=1000毫秒
              if(onlyTime>5000){
                  sender.sendMessage("邀请已经过期");
                  return true;
              }else{
                  //进行pvpv
                  return true;
              }
          }

      }
  }


    return false;

Function Distance_Calculation(a,b:longint):real;
Begin
    exit( sqrt(sqr(map_point[a].x-map_point[b].x) + sqr(map_point[a].y-map_point[b].y)) );
End;
Begin
    fillchar(map,sizeof(map),100);
    readln(n);
    for i:=1 to n do readln(map_point[i].x,map_point[i].y);
    readln(m);
    for i:=1 to m do begin
       readln(j,k);
       map[j,k]:=Distance_Calculation(j,k);
       map[k,j]:=map[j,k];
    end;
        for k:=1 to n do begin
       for i:=1 to n do begin
           for j:=1 to n do begin
              if (map[i,j]>map[i,k]+map[k,j]) then begin
                  map[i,j]:=map[i,k]+map[k,j];
              end;
           end;
       end;
    end;
    readln(i,j);
    writeln(map[i,j]:0:2);

导入vault，然后
  1. import net.milkbowl.vault.economy.Economy;
  2. import org.bukkit.plugin.RegisteredServiceProvider;

  4. RegisteredServiceProvider economyProvider = getServer().getServicesManager().getRegistration(Economy.class);
  5. Economy economy = (Economy)economyProvider.getProvider();

  7. double money = economy.getBalance(player);//获取钱，player为玩家名称
  8. economy.withdrawPlayer(player,money);//扣除钱
  9. economy.depositPlayer(player,money);//增加钱

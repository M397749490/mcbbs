Go：
package main

import (
    "gopkg.in/v0/qml"
    "log"
)

// 用于注入qml的go结构
type GoObject struct {
}

var root qml.Object

func (GoObject) MOTDChanged(text string) {
    log.Println("go: ", text)
    obj := root.ObjectByName("mainShow")
    obj.Set("text", text)
}

func (GoObject) CopyMOTD(a string) {
    log.Println("go: copy MOTD clicked")
}

func (GoObject) ColorChanged(color string) {
    log.Println("go: ", color)
    switch color {
    case "4":
        log.Println("红")
    case "c":
        log.Println("粉红")
    case "6":        log.Println("金黄")
    case "e":
        log.Println("黄")
    case "2":
        log.Println("绿")
    case "a":
        log.Println("浅绿")
    case "b":
        log.Println("青")
    case "3":
        log.Println("浅蓝")
    case "1":
        log.Println("蓝")
    case "9":
        log.Println("蓝紫")
    case "d":
        log.Println("浅粉")
    case "5":
        log.Println("紫")
    case "f":
        log.Println("白")
    case "7":
        log.Println("浅灰")
    case "8":
        log.Println("灰")
    case "0":
        log.Println("黑")
    default:
        log.Println("无法识别颜色代码！")    }}
func main() {    // 初始化qml    qml.Init(nil)
    // 创建引擎    engine := qml.NewEngine()
    // 加载qml    component, err := engine.LoadFile("MOTDTool.qml")    if err != nil {        panic(err)    }
    // 获得上下文    context := engine.Context()
    // 将一个go对象注入进qml上下文    goObject := GoObject{}    context.SetVar("goObject", goObject)
    // 创建qml窗口
    window := component.CreateWindow(nil)
    // 获得根控件
    root = window.Root()
    // 显示窗口
    window.Show()
    // 等待退出
    window.Wait()}

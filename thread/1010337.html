来 我教你 自己处理读写流
然后读取的时候把注释都变成配置节点
保存的时候恢复成注释
如果你要代码的话
给你核心代码 继承 YamlConfiguration 重写这两玩意儿

  2.     // 新增保留注释字段
  3.     protected static String commentPrefixSymbol = "'注释 ";
  4.     protected static String commentSuffixSymbol = "': 注释";

  6.     protected static String fromRegex = "( *)(#.*)";
  7.     protected static Pattern fromPattern = Pattern.compile(fromRegex);

  9.     protected static String toRegex = "( *)(- )*" + "(" + commentPrefixSymbol + ")" + "(#.*)" + "(" + commentSuffixSymbol + ")";
  10.     protected static Pattern toPattern = Pattern.compile(toRegex);

  12.     protected static Pattern countSpacePattern = Pattern.compile("( *)(- )*(.*)");

  14.     protected static int commentSplitWidth = 90;

  16.    @Override
  17.     public void loadFromString(String contents) throws InvalidConfigurationException {
  18.         String[] parts = contents.split(newLine);
  19.         List<String> lastComments = new ArrayList<>();
  20.         StringBuilder builder = new StringBuilder();
  21.         for (String part : parts) {
  22.             Matcher matcher = fromPattern.matcher(part);
  23.             if (matcher.find()) {
  24.                 String originComment = matcher.group(2);
  25.                 String[] splitComments = split(originComment, commentSplitWidth);
  26.                 for (int i = 0; i < splitComments.length; i++) {
  27.                     String comment = splitComments[i];
  28.                     if (i == 0) {
  29.                         comment = comment.substring(1);
  30.                     }
  31.                     comment = COMMENT_PREFIX + comment;
  32.                     lastComments.add(comment.replaceAll("\\.", "．").replaceAll("'", "＇").replaceAll(":", "："));
  33.                 }
  34.             } else {
  35.                 matcher = countSpacePattern.matcher(part);
  36.                 if (matcher.find() && !lastComments.isEmpty()) {
  37.                     for (String comment : lastComments) {
  38.                         builder.append(matcher.group(1));
  39.                         builder.append(this.checkNull(matcher.group(2)));
  40.                         builder.append(commentPrefixSymbol);
  41.                         builder.append(comment);
  42.                         builder.append(commentSuffixSymbol);
  43.                         builder.append(newLine);
  44.                     }
  45.                     lastComments.clear();
  46.                 }
  47.                 builder.append(part);
  48.                 builder.append(newLine);
  49.             }
  50.         }
  51.         super.loadFromString(builder.toString());
  52.     }

  54.     @Override
  55.     public String saveToString() {
  56.         yamlOptions.setIndent(options().indent());
  57.         yamlOptions.setDefaultFlowStyle(DumperOptions.FlowStyle.BLOCK);
  58.         yamlRepresenter.setDefaultFlowStyle(DumperOptions.FlowStyle.BLOCK);
  59.         String header = buildHeader();
  60.         String dump = yamlz.dump(getValues(false));
  61.         if (dump.equals(BLANK_CONFIG)) {
  62.             dump = "";
  63.         }
  64.         String contents = header + dump;
  65.         StringBuilder savcontent = new StringBuilder();
  66.         String[] parts = contents.split(newLine);
  67.         for (String part : parts) {
  68.             Matcher matcher = toPattern.matcher(part);
  69.             if (matcher.find() && matcher.groupCount() == 5) {
  70.                 part = this.checkNull(matcher.group(1)) + matcher.group(4);
  71.             }
  72.             savcontent.append(part.replaceAll("．", ".").replaceAll("＇", "'").replaceAll("：", ":"));
  73.             savcontent.append(newLine);
  74.         }
  75.         data = savcontent.toString();
  76.         return data;
  77.     }

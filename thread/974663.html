  5. //注意要在tick事件使用
  6. function tick(event) {
  7.   //让当前NPC吃饱
  8.   Satiate(event.npc);
  9. }

  11. /**
  12. * void **Satiate(ICustomNpc npc)
  13. * @param {"ICustomNpc"} npc 要喂饱的npc
  14. * @description 让这个该死的NPC吃饱吧
  15. */
  16. function Satiate(npc) {
  17.   var RoleCompanion = Java.type("noppes.npcs.roles.RoleCompanion");
  18.   var CompanionFoodStats = Java.type("noppes.npcs.roles.companion.CompanionFoodStats");
  19.   var role = npc.getMCEntity().roleInterface;
  20.   if (!(role instanceof RoleCompanion)) {
  21.     return;
  22.   }
  23.   var foodstats = role.foodstats;
  24.   var CompanionFoodStatsClass = CompanionFoodStats.class;
  25.   var foodLevelField = CompanionFoodStatsClass.getDeclaredField("foodLevel");
  26.   var foodTimerField = CompanionFoodStatsClass.getDeclaredField("foodTimer");
  27.   foodLevelField.setAccessible(true);
  28.   foodTimerField.setAccessible(true);
  29.   foodLevelField.set(foodstats, 20);
  30.   foodTimerField.set(foodstats, 0);
  31. }

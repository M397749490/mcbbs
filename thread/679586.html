AD_gainai#取什么都可以，只有不重复:
  NAME: '随身工作台'
  ID: 1#显示的物品
  KEEP-OPEN: true#点击是否关闭菜单
  COMMAND: 'workbench'
  POSITION-X: 1#位置|列
  POSITION-Y: 1#位置|行


AD_gainai111#取什么都可以，只有不重复:
  NAME: '&e随身末影箱·'
  ID: 2#显示的物品
  KEEP-OPEN: true#点击是否关闭菜单
  COMMAND: 'ec'
  POSITION-X: 3#位置|列
  POSITION-Y: 1#位置|行

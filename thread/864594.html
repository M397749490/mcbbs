举个栗子主菜单案例
menu-settings:
  name: '&e&l快捷菜单'
  rows: 1
  command: 'cd'  //快捷指令
  open-with-item:
    id: '347' //设置默认可以打开这个菜单的物品id
    left-click: true //左键打开
    right-click: true //右键打开
a2:
  NAME: '打开子菜单'
  LORE:
    - '打开子菜单'
  ID: 310
  KEEP-OPEN: true //保持菜单开启状态(一定要)
  COMMAND: 'op:cc open 子菜单'
  POSITION-X: 1
  POSITION-Y: 1

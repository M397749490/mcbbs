这是一个PVP 插件
    本插件并不高端，当玩家使用弓箭/鱼竿射到玩家时，将会提示玩家的血量信息。玩家被射死时提示距离。

  9. package net.kar.bowshot;

  11. import java.text.NumberFormat;
  12. import org.bukkit.Bukkit;
  13. import org.bukkit.ChatColor;
  14. import org.bukkit.Location;
  15. import org.bukkit.entity.Arrow;
  16. import org.bukkit.entity.Damageable;
  17. import org.bukkit.entity.Entity;
  18. import org.bukkit.entity.FishHook;
  19. import org.bukkit.entity.Player;
  20. import org.bukkit.event.EventHandler;
  21. import org.bukkit.event.Listener;
  22. import org.bukkit.event.entity.EntityDamageByEntityEvent;
  23. import org.bukkit.event.entity.EntityDamageEvent;
  24. import org.bukkit.event.entity.PlayerDeathEvent;
  25. import org.bukkit.plugin.java.JavaPlugin;

  27. public class Main extends JavaPlugin //继承JavaPlugin，使本类"成为一个插件"
  28. implements Listener/*实现监听器接口(使本类成为一个监听器)*/{

  30.     //精度，指精确到小数点后几位
  31.     int accuracy = 1;
  32.     //是否开启声音
  33.     boolean sound = false;
  34.     //声音的名字
  35.     String soundName;
  36.    
  37.     @Override
  38.     public void onEnable(){//插件的入口
  39.         getServer().getConsoleSender().sendMessage(ChatColor.translateAlternateColorCodes('&', "&6&l载入插件KarBowShot"));//将符号'&'替换成颜色符号"§",否则不会显示出颜色
  40.         
  41.         Bukkit.getPluginManager().registerEvents(this/*注册的监听器对象*/, this/*主类地址*/);//注册事件(由于本类是监听器的子类(或者说本类就是一个监听器),所以这里用一个this表示)
  42.         //如果监听器在其他类，将第一个this改成 new 监听器类名() 即可，就像这样
  43.         Bukkit.getPluginManager().registerEvents(new OtherListener(), this);
  44.         
  45.         //读取配置文件
  46.         this.accuracy = getConfig().getInt("Accuracy");
  47.         this.sound = getConfig().getBoolean("Sound");
  48.         this.soundName = getConfig().getString("SoundName");
  49.         
  50.         //为防止某些腐竹错误地填写精度,我们要对其进行判断
  51.         if(this.accuracy > 7 || this.accuracy < 1){
  52.             this.accuracy = 1;
  53.             getServer().getConsoleSender().sendMessage(ChatColor.translateAlternateColorCodes('&', "&6&l错误的精度"));
  54.         }
  55.         
  56.         //生成配置文件
  57.         saveDefaultConfig();
  58.         reloadConfig();
  59.     }
  60.    
  61.    
  62.     @EventHandler
  63.     public void onEntityDamageEntity(EntityDamageByEntityEvent event) {
  64.         //实现鱼竿勾到人时的效果
  65.         if ((event.getDamager() instanceof FishHook)) {
  66.             FishHook arrow = (FishHook) event.getDamager();
  67.             if ((arrow.getShooter() instanceof Player)) {
  68.                 Player shooter = (Player) arrow.getShooter(); //得到使用鱼竿的人
  69.                 Damageable targetEntity = (Damageable) event.getEntity();//得到被鱼竿勾中的实体
  70.                 if ((targetEntity instanceof Player)) { //判断被鱼竿勾中的实体是否为玩家
  71.                     Player targetPlayer = (Player) targetEntity; //得到被鱼竿勾中的玩家
  72.                     double targetPlayer_Health = targetEntity.getHealth();
  73.                     Double damage = (double) event.getFinalDamage(); //得到实际伤害值(如果用getDamage()的话，得到的是没有被减免的伤害)
  74.                     if (!targetEntity.isDead()) { //如果对方已经死亡，那么就没必要知道对方的血量了
  75.                         Double realHealth = (double) (targetPlayer_Health - damage.intValue());
  76.                         
  77.                         //对伤害值的精度(即小数点后的位数)进行限制
  78.                         NumberFormat nf = NumberFormat.getInstance();
  79.                         nf.setMaximumFractionDigits(this.accuracy);
  80.                         
  81.                         if (realHealth.intValue() > 0) {
  82.                             shooter.sendMessage("§a§l" + targetPlayer.getName() + "§7剩余血量§f§l" + nf.format(realHealth) + "/"+targetPlayer.getMaxHealth()+"§c?");
  83.                         }
  84.                     }
  85.                 }
  86.             }
  87.         }
  88.         
  89.         //实现箭射到人时的效果，同理
  90.         if ((event.getDamager() instanceof Arrow)) {
  91.             Arrow arrow = (Arrow) event.getDamager();
  92.             if ((arrow.getShooter() instanceof Player)) {
  93.                 Player shooter = (Player) arrow.getShooter();
  94.                 Entity targetEntity = event.getEntity();
  95.                 if ((targetEntity instanceof Player)) {
  96.                     Player targetPlayer = (Player) targetEntity;
  97.                     double targetPlayer_Health = targetPlayer.getHealth();
  98.                     Double damage = (double) event.getFinalDamage();
  99.                     
  100.                     double flc = KarMath.getDistance(shooter.getLocation(), targetPlayer.getLocation());//算出两名玩家之间的距离
  101.                     
  102.                     if (!targetEntity.isDead()) {
  103.                         Double realHealth = (double) (targetPlayer_Health - damage.intValue());
  104.                         
  105.                         NumberFormat nf = NumberFormat.getInstance();
  106.                         nf.setMaximumFractionDigits(this.accuracy);
  107.                         
  108.                         if (realHealth.intValue() > 0) {
  109.                             shooter.sendMessage("§7你在§f§l"+nf.format(flc)+"§7米外对 §a§l" + targetPlayer.getName() + " §7造成了§f§l" + nf.format(event.getFinalDamage()) + "§7点伤害,剩余血量§f" + nf.format(realHealth) + "/"+targetPlayer.getMaxHealth()+"§c?");
  110.                         }
  111.                     }
  112.                 }
  113.             }
  114.         }
  115.     }
  116.    
  117.     @EventHandler
  118.     public void onPlayerDeath(PlayerDeathEvent event){
  119.         Player p = event.getEntity();//得到死亡的玩家,注意这里不是getPlayer();
  120.         EntityDamageEvent ede = p.getLastDamageCause(); // 得到玩家受到的最后一次伤害事件
  121.         EntityDamageEvent.DamageCause dc = ede.getCause(); //得到玩家受到的最后一次伤害
  122.         
  123.         if (dc == EntityDamageEvent.DamageCause.PROJECTILE) { //判断玩家受到的最后一次伤害是否为弓箭伤害
  124.             if (p.getKiller() != null) { //判断杀手是否存在(有可能会是发射器射出的箭杀死的玩家)，防止NullPointerException的发生
  125.                 Player killer = p.getKiller();//得到杀手
  126.                
  127.                 double flc = KarMath.getDistance(p.getLocation(), killer.getLocation());//得到两名玩家之间的距离
  128.                
  129.                 //对伤害值的精度(即小数点后的位数)进行限制
  130.                 NumberFormat nf = NumberFormat.getInstance();
  131.                 nf.setMaximumFractionDigits(this.accuracy);
  132.                
  133.                 double health = killer.getHealth();
  134.                 event.setDeathMessage("§c§l§m" + p.getDisplayName() + "§7 被 §a§l" + killer.getDisplayName() + "§7(§c§l" + nf.format(health) + "§c?§7) 射死了(§f§l" + nf.format(flc) + "§7米远)");
  135.             }
  136.         }
  137.         
  138.         if(sound){//给杀手发送一段声音
  139.             if(p.getKiller() != null){
  140.                 p.getKiller().playSound(p.getKiller().getLocation(), soundName, 1, 1);
  141.             }
  142.         }
  143.         
  144.     }
  145. }

  147. //写一个工具类，如果工具(方法)数量多，建议这样做
  148. class KarMath{
  149.    
  150.     //工具类中全是静态函数，为防止该类被实例化，将其构造函数私有化
  151.     private KarMath(){}
  152.    
  153.     public static double getDistance(Location lc1,Location lc2){
  154.         //关于三维坐标系中两点的距离算法，请自行百度
  155.         return Math.sqrt(Math.pow(lc1.getX() - lc2.getX() , 2)+ Math.pow(lc1.getY() - lc2.getY() , 2)+ Math.pow(lc1.getZ() - lc2.getZ() , 2));
  156.     }
  157.    
  158. }

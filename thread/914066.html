@EventHandler
        public void onPlayerRespawn(PlayerRespawnEvent e) {
                new Thread(()->{
                        while(e.getPlayer().isDead())
                        {
                                try {
                                        Thread.sleep(1);
                                }catch(InterruptedException exc) {return;}
                        }
                        e.getPlayer().setHealth(e.getPlayer().getHealth()*0.8);
                }).start();
        }

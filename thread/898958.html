  12. /**
  13. * 保存玩家物品栏
  14. * @param yml 存储物品栏的yml
  15. * @param player 玩家
  16. */
  17. public void savePlayerInv(FileConfiguration yml, Player player) {
  18.         // 首先将yml中玩家对应的数据清除
  19.         yml.set(player.getName(), null);
  20.        
  21.         Inventory inv = player.getInventory();
  22.                
  23.         // 获取玩家背包里的物品
  24.         ItemStack[] items = inv.getContents();
  25.         // 注意1.8及以下版本getContents方法不包含玩家身上穿的装备
  26.         // 需要另外使用p.getEquipment().getHelmet/Chestplate/Boots/Leggings来获取装备保存
  27.         // 检查是否包含身上装备可以使用items.length>=40来检查，如果返回true则包含
  28.        
  29.         for(int index = 0; index < items.length; index++) {
  30.                 ItemStack item = items[index];
  31.                
  32.                 // 因为玩家背包中可能有空位，所以需要检查是否为null以及材质是否为空
  33.                 if(item != null && item.getType() != Material.AIR) {
  34.                         // 因为ItemStack实现了ConfigurationSerializable接口
  35.                         // 所以我们可以直接调用yml的set方法保存ItemStack
  36.                         // 使用index作为节点名，能方便我们载入其物品栏
  37.                         yml.set(player.getName() + "." + index, item);
  38.                 }
  39.         }
  40.        
  41.         // 处理完后记得保存yml
  42. }
  43.        
  44. /**
  45. * 载入玩家物品栏
  46. * @param yml 存储物品栏的yml
  47. * @param player 玩家
  48. * [url=home.php?mod=space&uid=491268]@Return[/url] boolean 是否加载成功
  49. */
  50. public boolean loadPlayerInv(FileConfiguration yml, Player player) {
  51.         // 检查yml是否储存了玩家的物品栏
  52.         if(yml.contains(player.getName())) {
  53.                 Inventory inv = player.getInventory();
  54.                 // 建议清空玩家物品栏
  55.                 inv.clear();
  56.                
  57.                 // 获取yml中该玩家物品栏对应的节点入口
  58.                 ConfigurationSection ymlSection = yml.getConfigurationSection(player.getName());
  59.                 // 使用yml的getKeys(false)方法来获取一层深度的子节点名
  60.                 // 获取到的节点名同时也是该节点下物品在物品栏中对应的位置
  61.                 for(String index : ymlSection.getKeys(false) ) {
  62.                         // 调用yml的getItemStack(String path)方法我们就能直接获取ItemStack了
  63.                         inv.setItem(Integer.parseInt(index), ymlSection.getItemStack(index));
  64.                 }
  65.                 return true;
  66.         }
  67.         return false;
  68. }

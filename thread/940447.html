  1. /give @p stone 1 0 {minecraft:can_place_on:{blocks:["stone"]}}

  1. /give @p stone 1 0 {minecraft:can_destroy:{blocks:["stone"]},minecraft:can_place_on:{blocks:["stone"]}}

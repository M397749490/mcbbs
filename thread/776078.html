  1. Groups:
  2.     Default: #用户组名称
  3.        # 有关玩家可以定义的信息.
  4.        # 去除下面的'#'可以镜像复制此配置到其他组.
  5.        #Mirror:
  6.          #- 'group1'
  7.          #- 'group2'
  8.        Residence:
  9.            # 玩家是否可以创建领地.这个设置可被 'residence.create' 权限覆盖.
  10.            CanCreate: true
  11.            # 玩家允许拥有最大领地数.
  12.            MaxResidences: 3
  13.            # 一个领地的最大物理区域数目.
  14.            MaxAreasPerResidence: 15
  15.            # 一个领地在东西方向最大距离 (X坐标).                      【这里】
  16.            MaxEastWest: 300
  17.            # 一个领地在南北方向最大距离 (Z坐标).                         【这里】
  18.            MaxNorthSouth: 300
  19.            # 一个领地在上下方向最大高度 (Y坐标).                             【这里】
  20.            MaxUpDown: 256
  21.            # 允许创建领地的最低高度.
  22.            MinHeight: 0
  23.            # 允许创建领地的最高高度.  
  24.            MaxHeight: 255
  25.             # 子区域领地最大深度.
  26.             # 当一个领地子区域包含另一个子区域时,最大深度为2.
  27.             # 设置为0将禁用领地子区域.
  28.            SubzoneDepth: 3
  29.            # 是否允许传送到领地.
  30.            CanTeleport: true
  31.            # 是否允许使用'/res unstuck'命令, 当你卡在领地内时可以传送出去
  32.            Unstuck: true
  33.            # 是否允许使用'/res select '命令, 如果禁用,那么只能使用工具选取领地.
  34.            SelectCommandAccess: true
  35.            # 领地主人是否允许使用 黑名单/屏蔽列表 .
  36.            ItemListAccess: true
  37.        # 进入和离开领地时的消息设置.

楼上说的是旧版的办法，但新版的同理。打开以下目录的文件
\plugins\Residence\groups.yml
MaxEastWest、MaxNorthSouth更改这两个就是最大圈地范围了。剩下的那个高度不需要动的
翻到同文件中，最下方的BuyCost，这是圈地中，每一个方块的价格，默认大概为0.05，可自行更改的。

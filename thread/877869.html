  1. servers:
  2.   lobby:
  3.     motd: '&1Just another BungeeCord - Forced Host'
  4.     address: localhost:25565
  5.     restricted: false
  6.     player_limit: 2018
  7.   prison:
  8.     motd: '&1Just another BungeeCord - Forced Host'
  9.     address: localhost:40008
  10.     restricted: false
  11.     player_limit: 2018
  12.    bedwar:
  13.     motd: '&1Just another BungeeCord - Forced Host'
  14.     address: localhost:25567
  15.     restricted: false

作者: 红菜头糕    时间: 2020-8-8 09:37
function died(e){
var n = e.npc;
var time = 100;  // 100 tick
var radius = 10; // 半径
var x = n.getHomeX(),y = n.getHomeY,z = n.getHomeZ();
var ps = n.getWorld().getNearbyEntities(x,y,z,radius,1);
for(var i = 0; i < ps.length; i++){
ps[i].message("boss击杀成功,5秒后返回主城..");
}
n.getTimers().forceStart(1,time,false);
}

function timer(e){
var n = e.npc;
var x = n.getHomeX(),y = n.getHomeY,z = n.getHomeZ();
var ps = n.getWorld().getNearbyEntities(x,y,z,10,1);
for(var i = 0; i < ps.length; i++){
n.executeCommand("spawn "+ps[i].getName());
}
}

上面代码放进去就行，time是时间，radius是半径
效果就是boss死亡5秒后，会把自己出生点(也就是你最开始放置NPC的位置)周围10格的玩家
传送回主城，房间的问题适当把半径调大到覆盖boos战区域就好

一些原理

一、如何定位要塞
简单来说，当末影之眼扔出去时会朝要塞方向飞去。在两个不同位置投掷末影之眼，其飞行方向的延长线将交于要塞的位置。据此可以确定要塞位置。
本原版模组使用以下公式计算要塞大致位置：
  1. x=((z1-z2)*tan(a1)*tan(a2)+x1*tan(a2)-x2*tan(a1))/(tan(a2)-tan(a1));
  2. z=(x1-x2+z1*tan(a1)-z2*tan(a2))/(tan(a1)-tan(a2));

其中，
x1,z1为位置1坐标，a1为位置1朝向；
x2,z2为位置2坐标，a2为位置2朝向；
x,z为要塞坐标。

详情请参见https://www.mcbbs.net/thread-799313-1-1.html

  2. #include<cmath>
  3. using namespace std;
  4. double pi=3.1415926535897932384626433832795;
  5. double rad(double x)
  6. {
  7.         return x*pi/180;
  8. }
  9. int main()
  10. {
  11.         ofstream f;
  12.         f.open("load_tans.mcfunction");
  13.         for(double x=360.00;x>=-360.00;x-=0.01)
  14.         {
  15.                 f<<"scoreboard players set "<<int(100*x)<<" slTans "<<int(100*tan(rad(x)))<<" stTans"<<endl;
  16.         }
  17.         f.close();
  18.         return 0;
  19. }

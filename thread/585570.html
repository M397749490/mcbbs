  26. public class Event implements Listener {

  31.         public Event(Main plugin) {
  32.                 this.plugin = plugin;

  34.                 Dye d = new Dye();
  35.                 d.setColor(DyeColor.BLUE);
  36.                 lapis = d.toItemStack();
  37.                 lapis.setAmount(3);
  38.         }

  40.         @EventHandler
  41.         public void onOpenInventory(InventoryOpenEvent e) {
  42.                 if (e.getInventory() instanceof EnchantingInventory) {
  43.                         e.getInventory().setItem(1, lapis);
  44.                         plugin.inventories.add((EnchantingInventory) e.getInventory());
  45.                 }
  46.         }

  48.         @EventHandler
  49.         public void onCloseInventory(InventoryCloseEvent e) {
  50.                 if (e.getInventory() instanceof EnchantingInventory) {
  51.                         if (plugin.inventories.contains((EnchantingInventory) e.getInventory())) {
  52.                                 e.getInventory().setItem(1, null);
  53.                                 plugin.inventories.remove((EnchantingInventory) e.getInventory());
  54.                         }
  55.                 }
  56.         }

  58.         @EventHandler
  59.         public void cancelSteal(InventoryClickEvent e) {
  60.                 if (e.getClickedInventory() instanceof EnchantingInventory) {
  61.                         if ((plugin.inventories.contains((EnchantingInventory) e.getInventory())) && (e.getSlot() == 1)) {
  62.                                 e.setCancelled(true);
  63.                         }
  64.                 } else if ((e.getView().getTopInventory() instanceof EnchantingInventory) && (e.getClickedInventory() instanceof PlayerInventory)) {
  65.                         if ((e.getCurrentItem().getType() == lapis.getType()) && ((e.getAction() == InventoryAction.MOVE_TO_OTHER_INVENTORY) || (e.getAction() == InventoryAction.PICKUP_ALL))) {
  66.                                 e.setCancelled(true);
  67.                         }
  68.                 }
  69.         }
  70.        
  71.         @EventHandler
  72.         public void cancelSteal(InventoryMoveItemEvent e) {
  73.                 if (e.getSource() instanceof EnchantingInventory) {
  74.                         if (plugin.inventories.contains((EnchantingInventory) e.getSource())) {
  75.                                 if ((e.getItem().getType() == lapis.getType()) && (e.getItem().getAmount() > 1)) {
  76.                                         e.setCancelled(true);
  77.                                 }
  78.                         }
  79.                 }
  80.         }

  82.         @EventHandler
  83.         public void freeLapis(EnchantItemEvent e) {
  84.                 if (plugin.inventories.contains((EnchantingInventory) e.getInventory())) {
  85.                         e.getInventory().setItem(1, lapis);
  86.                 }
  87.         }
  88.        
  89.         @EventHandler
  90.         public void correctLevelCost(EnchantItemEvent e) {
  91.             e.getEnchanter().setLevel(e.getEnchanter().getLevel() - e.getExpLevelCost() + (e.whichButton() + 1));
  92.             e.setExpLevelCost(0);
  93.         }

  95.         @EventHandler
  96.         public void setRandomEnchntLevel(PrepareItemEnchantEvent e) {
  97.                 ContainerEnchantTable table = (ContainerEnchantTable) ((CraftInventoryView) e.getView()).getHandle();
  98.                 table.f = r.nextInt();
  99.         }

  101.         @EventHandler
  102.         public void hideEnchants(PrepareItemEnchantEvent e) {
  103.                 ContainerEnchantTable table = (ContainerEnchantTable) ((CraftInventoryView) e.getView()).getHandle();
  104.                 plugin.getServer().getScheduler().scheduleSyncDelayedTask(plugin, () -> {
  105.                         table.h[0] = (-1 | 0 << 8);
  106.                         table.h[1] = (-1 | 0 << 8);
  107.                         table.h[2] = (-1 | 0 << 8);
  108.                 } , 1);
  109.         }
  110.        
  111.         @EventHandler
  112.         public void cancelRename(InventoryClickEvent event) {
  113.                 if (event.getInventory() instanceof AnvilInventory) {
  114.                         if (event.getSlotType() == SlotType.RESULT) {
  115.                                 if((event.getCurrentItem() != null) && (event.getCurrentItem().getItemMeta().getDisplayName() != null) && (!(event.getCurrentItem().getItemMeta().getDisplayName().equals("")))) {
  116.                                         event.setCancelled(true);
  117.                                         Player p = (Player) event.getWhoClicked();
  118.                                 p.setExp(p.getExp());
  119.                                 }
  120.                         }
  121.                 }
  122.         }
  123. }

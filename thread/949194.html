求解MythicMobs 如何设置怪物的攻击距离
gls:
  Type: PIG_ZOMBIE
  Display: '&b&l冻死的剑士哥拉斯'
  Disguise:
    Type: player
    Player: '&b&l冻死的剑士哥拉斯'
    Skin: 'walker2000'
    Blocking: true
    Invisible: false
    ShowName: true
    Sprinting: true
    ModifyBoundingBox: false
  Bossbar:
    Enabled: true
    Title: <mob.name>
    Range: 8
    Color: GREEN
    Style: SEGMENTED_20
  Health: 1000
  Damage: 30
  Attack: 6
  Modules:
    ThreatTable: true
  Skills:
   - explosion{yield=2;bd=false;fire=false} ~onDamaged 0.2
  Equipment:
   - diamond_helmet:4
   - diamond_sword:0
  Options:
    MovementSpeed: 0.1
    PreventOtherDrops: true   
    AlwaysShowName: true
  Drops:
  -

配置文件
  1. #Configuration file for BigBrother plugin

  3. #Do not change the interface if you don't know what you are doing. It's good as it is now
  4. interface: 0.0.0.0 ← 修改IP
  5. port: 25565 ← 端口

  7. network-compression-threshold: 256

  9. #Sets the server description on the server list
  10. motd: "§bPocketMine-MP server using §6§lBigBrother§r§b plugin\n§aConnect to Minecraft: PE servers from PC clients" ← 服务器motd介绍

  12. #Use Mojang services to authenticate players
  13. online-mode: true ← 正版验证，盗版的话改成false

  15. #Automatically authenticate players on SimpleAuth that were authenticated via Mojang servers
  16. online-mode-simpleauth: false

  18. #Action when there is a nick conflict
  19. #kickpe: kicks the Pocket Edition player
  20. #kickpc: kicks the PC player
  21. #kickold: kicks the player that was on the server
  22. #kicknew: kicks the player that is connecting
  23. #none: do nothing, leave it up to PocketMine-MP and other plugins
  24. action-on-conflict: none

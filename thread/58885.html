  1. # 这是Bukkit的主要设置文件。
  2. # 正如你所能看到的，实际上没有给服务器添加插件的话，也就没有多少需要配置的项。
  3. # 作为一个对该文件内变量的参考，你可以浏览Bukkit WIKI来获得必要信息：
  4. # http://wiki.bukkit.org/Bukkit.yml
  5. #
  6. # 如果在该文件方面需要帮助，请放心加入我们（指Bukkit）的IRC或者
  7. # 在官方论坛上发帖寻求建议。
  8. #
  9. # IRC: #bukkit @ esper.net
  10. #    (如果你看不懂这个，直接浏览该链接就好：http://webchat.esper.net/?channels=bukkit )
  11. # 论坛：http://forums.bukkit.org/forums/bukkit-help.6/
  12. # Twitter：http://twitter.com/Craftbukkit
  13. # 漏洞记录器：http://leaky.bukkit.org/

  15. # 基本设置
  16. settings:

  18.   # 是否开启末路之地。默认：true
  19.   allow-end: true

  21.   # 是否当服务器过载时提示"[WARNING] Can't keep up! Did the system time change, or is the server overloaded?"默认：true
  22.   warn-on-overload: true

  24.   # 出生点半径，不包括出生点本身，因此直径为33，范围内将存在出生点保护。默认：16
  25.   spawn-radius: 16

  27.   # 权限文件名称，该文件无需权限插件支持，可独立工作。默认：permissions.yml
  28.   permissions-file: permissions.yml

  30.   # 用于升级插件的文件夹名称，重载时该文件夹内的插件将会替换原插件。默认：update。注意：该文件夹必须在插件文件夹下；绝对路径无效。
  31.   update-folder: update

  33.   # 目前该选项已被取消，之前的含义：每秒多少个数据包可以被游戏中的ping列表所使用，最大值。默认：100
  34.   ping-packet-limit: 100

  36.   # 自从Minecraft添加了当一个实体陷入方块时，会将它排挤出方块的特性（实体包括地面物品、各种动物、飞行的雪球、火球和箭，以及玩家自己等），
  37.   # 玩家经常发现自己登入时地点比上次退出游戏的地点高（比如在空间很小的房子内退出，重进后出生在房顶上），
  38.   # bukkit.yml可以让你设置当玩家登入游戏时，开启或关闭这项特性，使玩家出生在确切的位置上。
  39.   # 设置为true，当玩家进入游戏时原版服务器的冲突检查功能，如果有需要就跳过
  40.   # 设置为false，则不跳过该检查功能。默认：false
  41.   use-exact-login-location: false

  43.   # 测量插件对于事件的工作时间，将开启 /timings 命令。默认：false
  44.   plugin-profiling: false

  46.   # 客户端重连服务器的间隔，这也就是显示“End of stream”的原因，
  47.   # 设置为0可以取消该功能，但会使服务器更易受到攻击（仅推荐测试用服务器使用）。默认：4000
  48.   connection-throttle: 4000

  50.   # 当远程查询时是否返回插件列表。默认：true
  51.   query-plugins: true

  53.   # 是否显示警告如果插件使用了过时事件，"true"和"false"控制是否开启，
  54.   # "default"为一直显示警告，除非插件开发者注册事件时告诉服务器不必提醒。默认：default
  55.   deprecated-verbose: default

  57. # 以下内容正在翻译
  58. # 官方没有给出解释，没理解错的话，应该是每chunk内的生物数量限制
  59. spawn-limits:

  61.   # 怪物的生物数量限制。默认：70
  62.   monsters: 70

  64.   # 动物的生物数量限制。默认：15
  65.   animals: 15

  67.   # 水生动物的数量限制。默认：5
  68.   water-animals: 5

  70. # 每次尝试生成该活动的tick间隔（1秒=20 tick），生物产生几率由原版服务器提供默认值。
  71. ticks-per:

  73.   # 尝试生成动物的tick间隔，0为关闭，推荐使用server.properties下的 spawn-animals 控制开启或关闭。默认：400
  74.   animal-spawns: 400

  76.   # 尝试生成怪物的tick间隔，0为关闭，推荐使用server.properties下的 spawn-monsters 控制开启或关闭。默认：1
  77.   monster-spawns: 1

  79.   # 自动保存服务器的tick间隔，建议使用备份插件来进行备份，0为关闭。默认：0
  80.   autosave: 0

  82. # 版本检查器相关设置
  83. auto-updater:

  85.   # 是否开启版本检查。默认：true
  86.   enabled: true

  88.   # 当正在使用的版本被检查出有巨大漏洞时，应该提醒谁？
  89.   # 可填"warn-ops"（警告OP）和"warn-console"（警告控制台），格式为字符串列表。默认：[warn-console, warn-ops]
  90.   on-broken:
  91.   - warn-console
  92.   - warn-ops

  94.   # 当正在使用的版本被检查出过期时，应该提醒谁？
  95.   # 可填"warn-ops"（警告OP）和"warn-console"（警告控制台），格式为字符串列表。默认：[warn-console, warn-ops]
  96.   on-update:
  97.   - warn-console
  98.   - warn-ops

  100.   # 服务器优先检查哪种最新版本服务端？可填："rb"、"beta"或"dev"。
  101.   preferred-channel: rb

  103.   # 版本检查器检查地址。除非你知道你在做什么，否则不要更改这个选项。默认：dl.bukkit.org
  104.   host: dl.bukkit.org

  106.   # 给服务器建议频道，不知道干什么用的，官方上也没有相关说明，待更新
  107.   suggest-channels: true

  109. # 这部分不必解释了，懂数据库的都懂这个，不懂的也不必修改
  110. database:
  111.   username: bukkit
  112.   isolation: SERIALIZABLE
  113.   driver: org.sqlite.JDBC
  114.   password: walrus
  115.   url: jdbc:sqlite:{DIR}{NAME}.db

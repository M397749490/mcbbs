最后终于解决了  不应该是您说的那么简单的
首先一个继承Block的类  要添加两个参数private Icon field_94393_a;
                                                      private Icon field_94392_b;
然后就是重写
@Override
        public void registerIcons(IconRegister par1IconRegister) {
                // TODO Auto-generated method stub
        this.blockIcon     = par1IconRegister.registerIcon(this.getTextureName() + "_side");
        this.field_94393_a = par1IconRegister.registerIcon(this.getTextureName() + "_top");
        this.field_94392_b = par1IconRegister.registerIcon(this.getTextureName() + "_bottom");
        }

        @Override
        public Icon getIcon(int par1, int par2) {
                // TODO Auto-generated method stub
                 return par1 == 0 ? this.field_94392_b : (par1 == 1 ? this.field_94393_a : this.blockIcon);
        }

这两个方法。在registerIcons（）这个方法里面定义了 png图片的名字格式。这是我仿照TNT那个方块写的。还是感谢你给了灵感

最后的最后我重写了一个继承自Block的类
之后继承自这个类的Block类就能设置六面不同的了


package com.reno.mod.blocks;

import com.reno.mod.renoDatas;
import com.reno.mod.renoMod;

import net.minecraft.block.Block;
import net.minecraft.block.material.Material;
import net.minecraft.client.renderer.texture.IconRegister;
import net.minecraft.entity.player.EntityPlayer;
import net.minecraft.util.Icon;
import net.minecraft.world.World;

public class renoSixBaseBlocks extends Block {

        private Icon blockIcon_top;
        private Icon blockIcon_bottom;
        private Icon blockIcon_front;
        private Icon blockIcon_back;
        private Icon blockIcon_left;
        private Icon blockIcon_right;

        public renoSixBaseBlocks(int par1, Material par2Material) {
                super(par1, par2Material);

               
                // TODO Auto-generated constructor stub
        }

        @Override
        public void registerIcons(IconRegister par1IconRegister) {
                // TODO Auto-generated method stub
                this.blockIcon_back = par1IconRegister.registerIcon(this.getTextureName()
                                + "_back");
                this.blockIcon_top = par1IconRegister.registerIcon(this
                                .getTextureName() + "_top");
                this.blockIcon_bottom = par1IconRegister.registerIcon(this
                                .getTextureName() + "_bottom");
               
                this.blockIcon_left = par1IconRegister.registerIcon(this.getTextureName()
                                + "_left");
                this.blockIcon_right = par1IconRegister.registerIcon(this
                                .getTextureName() + "_right");
                this.blockIcon_front = par1IconRegister.registerIcon(this
                                .getTextureName() + "_front");
        }

        @Override
        public Icon getIcon(int side, int par2) {
                // TODO Auto-generated method stub
                switch (side) {
                case 0:
                        return this.blockIcon_bottom;
                case 1:
                        return this.blockIcon_top;
                case 2:
                        return this.blockIcon_left;
                case 3:
                        return this.blockIcon_right;
                case 4:
                        return this.blockIcon_front;
                case 5:
                        return this.blockIcon_back;
                default:
                        break;
                }
                return this.blockIcon_bottom;

        }

}

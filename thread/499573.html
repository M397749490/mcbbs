作者: 2396794736    时间: 2015-9-5 21:39
groups:
  Default:
    default: false     #设置问false
    permissions:
    - -bukkit.command.kill
    inheritance:
    - g:groupmanager_default
    - g:bukkit_default
    - g:essentials_default
    - g:towny_default
    info:
      prefix: '&e'
      build: false
      suffix: ''
  Builder:
    default: true     #这里设置成true
    permissions:
    - InfoBoard.View
    - levels.user
    - scriptblock.interact.use
    - scriptblock.walk.use
    inheritance:
    - default
    - g:essentials_builder
    - g:towny_builder
    info:
      prefix: '&2'
      build: true
      suffix: ''
在服务器文件中这样配置：
打开/plugins/GroupManager/worlds/world/groups.yml

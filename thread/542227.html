标题: 求vb.net,自动获取java路径代码
前些日子看到了mcmny大大开源的启动器MMCLS，但是其中寻找JAVA代码貌似失效了，WIN7，WIN10均测试无法自动获取，求各位大神告知代码！
以下是mcmny大大的寻找JAVA代码：
If djavapath = "" Then
            Try
                Dim rk As RegistryKey = Registry.LocalMachine

                rk = rk.OpenSubKey("SOFTWARE")
                rk = rk.OpenSubKey("JavaSoft")

                rk = rk.OpenSubKey("Java Runtime Environment")

                Dim javaver As String = ""

                javaver = rk.GetSubKeyNames(0)
                rk = rk.OpenSubKey(javaver)

                djavapath = rk.GetValue("JavaHome") + "\bin\javaw.exe"

            Catch ex As NullReferenceException
                If MessageBox.Show("程序未发现java主程序文件，可能您没有安装java(MC启动必需)，也可能是程序判断错误！如果是判断错误，请选择否并手动选择java路径，如果不是，是否前往java官网下载？", "提示", MessageBoxButtons.YesNo) = Windows.Forms.DialogResult.Yes Then

                    System.Diagnostics.Process.Start("http://www.java.com/")
                    djavapath = "javaw"
                    End
                Else
                    djavapath = "javaw"
                    Form2.Show()
                End If
            End Try

Dim checkjava As Object
                checkjava = CreateObject("wscript.shell")
                Dim a As String
                a = checkjava.regread("HKEY_LOCAL_MACHINE\SOFTWARE\Microsoft\Windows\CurrentVersion\App Paths\javaws.exe\Path")
                djavapath = a + "\javaw.exe"

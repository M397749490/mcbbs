  1. # 新建记分版 foo
  2. scoreboard objectives add foo dummy
  3. # 将一只猪 @e[type=pig,limit=1] 的血量 Health 存为 #temp 的 foo 记分版分数
  4. execute store result score #temp foo run data get entity @e[type=pig,limit=1] Health
  5. # 如果 #temp 的 foo 分数小于 20，则运行命令 tellraw @a {"text":"血量小于10颗心"}
  6. execute if score #temp foo matches ..20 run tellraw @a {"text":"血量小于10颗心"}

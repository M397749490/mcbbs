          - "/money Nijikokun"
    iConomy.top:
        介绍: 查看资金排行榜
        默认拥有权限者: op
        指令: "/money top"
    iConomy.payment:
        介绍: 发送钱给其它人
        默认拥有权限者: op
        指令: "/money pay <名字> <数额>"
        例:
          - "/money pay Nijikokun 30"
    iConomy.accounts.create:
        介绍: 创建账户
        默认拥有权限者: op
        指令: "/money create <名字> [数额]"
        例:
          - "/money create Nijikokun"
          - "/money create Nijikokun 30"
    iConomy.accounts.remove:
        介绍: 移除一个存在的账户
        默认拥有权限者: op
        指令: "/money remove <名字>"
        例:
          - "/money remove Nijikokun"
    iConomy.accounts.give:
        介绍: 给予他人一个指定的数额
        默认拥有权限者: op
        指令: "/money give <名字> <数额>"
        例:
          - "/money give Nijikokun 30"
    iConomy.accounts.take:
        介绍: 从他人账户中移除一定数额资金
        默认拥有权限者: op
        指令: "/money take <account-name> <amount>"
        例:
          - "/money take Nijikokun 30"
    iConomy.accounts.set:
        介绍: 给他人账户设定指定数额的资金
        默认拥有权限者: op
        指令: "/money set <名字> <数额>"
        例:
          - "/money set Nijikokun 30"
    iConomy.accounts.status:
        介绍: 查看账户状态 (可见, 隐藏, 其它)
        默认拥有权限者: op
        指令: "/money status <名字>"
        例:
          - "/money status Nijikokun"
    iConomy.accounts.status.set:
        介绍: 设置账户状态 (可见 (1), 隐藏 (0))
        默认拥有权限者: op
        指令: "/money status <名字>"
        例:
          - "/money status Nijikokun"
    iConomy.accounts.purge:
        介绍: 清空所有账户到初始值
        默认拥有权限者: op
        指令: "/money purge"
    iConomy.accounts.empty:
        介绍: 清空所有账户数据
        默认拥有权限者: op
        指令: "/money empty"

  1. Spawn:
  2.   # Forces players to login in defined spawn point when logging into server
  3.   SpawnOnJoin: true   #这里是登录服务器时是否强制重生主城
  4.   # Defines players spawn point after death if set to true, if not, then it will be used only for /cmi spawn command
  5.   # RespawnLocation will indicate if you want to use this location as possible respawn point for player after death
  6.   Main:   #这是重生点设置的
  7.     World: bskyblock_world
  8.     X: 0.51540633481784
  9.     Y: 125.0
  10.     Z: 1.300000011920929
  11.     Pitch: 30.149595
  12.     Yaw: 4.350186
  13.     RespawnLocation: false   #这是重生点时候强制
  14.     Rng: 0
  15.   # Defines players first spawn point when he logs into server for the first time
  16.   FirstSpawn:
  17.     Use: true

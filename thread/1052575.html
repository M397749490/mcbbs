作者: wjwrh    时间: 2020-5-28 21:50
  1. @echo OFF
  2. :head
  3. set a=-1
  4. :start
  5. set /a a+=1
  6. set b=3
  7. color 3
  8. title Server [已重新启动 %a% 次]
  9. java -Xms3G -Xmx3G -XX:+UnlockExperimentalVMOptions -XX:TargetSurvivorRatio=90 -XX:G1MaxNewSizePercent=80 -XX:G1NewSizePercent=50 -XX:+DisableExplicitGC -XX:G1MixedGCLiveThresholdPercent=50 -XX:+AlwaysPreTouch -jar paperclip.jar nogui
  10. ping -n %b% 127.1>nul
  11. goto start

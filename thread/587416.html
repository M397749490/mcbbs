  5. loop="loop"
  6. # Core
  7. echo -e "\033[34;49;1m(C) 2015 MoeCraft All Rights Reserved\e[0m"
  8. echo "MineCraft Launcher V1.5 // Written by Kenvix @ 2015/8/8"
  9. #if [ $1 == ${loop} ];
  10. #then
  11. #    echo "Loop-Running Mode Enabled"
  12. #fi
  13. echo -e "\033[31;49;1m------------------------------------------------------------\e[0m"
  14. echo -e "\033[32;49;1mMineCraft Server Start\e[0m"
  15. echo -e "Using Jar File: ${jarfile}"
  16. echo -e ${comm}
  17. echo -e "\033[31;49;1m------------------------------------------------------------\e[0m"
  18. java ${comm} -jar ${jarfile}
  19. echo ""
  20. echo -e "\033[31;49;1m------------------------------------------------------------\e[0m"

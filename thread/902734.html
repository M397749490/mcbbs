配置文件

  1. #聊天组编辑
  2. Groups:
  3.     default:
  4.         #是否处于禁言状态
  5.         Mute: false
  6.         #前缀
  7.         Default-Prefix: '&7'
  8.         #聊天格式
  9.         ChatFormat: '{player_prefix}{player_displayname}: {message}'
  10.         #是否允许彩字聊天
  11.         Allow-ChatColor: true
  12.         #聊天冷却
  13.         Chat-Delay: 2
  14.         #命令冷却
  15.         Commands-Delay: 2
  16.         #私聊设置
  17.         Private-Chat:
  18.             #是否允许私聊
  19.             Enabled: true
  20.             #发送对象信息
  21.             Target-Message: '&dFrom {sender_prefix}{sender_displayname}: &f{message}'
  22.             #发送者信息
  23.             Sender-Message: '&dTo {target_prefix}{target_displayname}: &f{message}'
  24.             #冷却时间
  25.             PrivateChat-Delay: 1
  26.         #标题聊天
  27.         Title-Chat:
  28.             #是否允许使用/chat title 和/title 发送全部玩家都能看见的标题发言
  29.             Enabled: false
  30.             #上行
  31.             Title: '{player_prefix}{player_displayname}:'
  32.             #下行
  33.             SubTitle: '&7{message}'
  34.             #淡入时间(秒)
  35.             FadeIn: 0
  36.             #停留时间
  37.             Stay: 5
  38.             #淡出时间
  39.             FadeOut: 0
  40.             #冷却时间
  41.             Title-Delay: 20
  42.         #动作栏信息聊天
  43.         ActionBar-Chat:
  44.             #是否允许使用/chat actionbar 和/actionbar 发送全部玩家都能看见的动作栏信息发言
  45.             Enabled: false
  46.             #发送内容设置
  47.             #支持多行信息
  48.             Messages:
  49.             - '{player_prefix}{player_displayname}&7: {message}'
  50.             #发送速度
  51.             Sending-speed: 0.4
  52.             #冷却时间
  53.             ActionBar-Delay: 20
  54.         #加入游戏时的信息设置
  55.         Join-Settings:
  56.             #开启后，默认的登录信息将会被删除
  57.             Enabled: true
  58.             #登录时的聊天信息设置
  59.             #如果你不想要有登录信息， 可以删除掉它
  60.             Messages:
  61.             - '&6玩家 &f{player_prefix}{player_displayname} &6加入了服务器。'
  62.             #标题信息
  63.             Title:
  64.                 #是否开启
  65.                 Enabled: false
  66.                 #上行
  67.                 Title: '{player_prefix}{player_displayname}'
  68.                 #下行
  69.                 SubTitle: '&6加入了服务器。'
  70.                 #淡入、停留、淡出
  71.                 FadeIn: 0.4
  72.                 Stay: 3
  73.                 FadeOut: 0
  74.             #动作栏信息
  75.             ActionBar:
  76.                 #是否开启
  77.                 Enabled: false
  78.                 #信息
  79.                 Messages:
  80.                 - '&6玩家 &f{player_prefix}{player_displayname} &6加入了服务器。'
  81.                 #发送速度
  82.                 Sending-speed: 0.4
  83.         #加入游戏时的信息设置
  84.         Quit-Settings:
  85.             #同上设置
  86.             Enabled: true
  87.             Messages:
  88.             - '&6玩家 &f{player_prefix}{player_displayname} &6退出了服务器。'
  89.         #TAB显示设置
  90.         Show-to-TAB:
  91.             #是否显示到TAB上
  92.             Enabled: true
  93.             #格式
  94.             Format: '{player_prefix}{player_displayname}'
  95.         #前缀列表，玩家若拥有/chat prefix的权限将可以自由切换以下前缀，包括他个人拥有的前缀
  96.         Prefix-List:
  97.         - '&7'
  98.         #脏话关键词屏蔽
  99.         Anti-Key-Words:
  100.             Enabled: true
  101.             #是否无视空格
  102.             Ignore-Blanks: true
  103.             #执行命令
  104.             Run-commands:
  105.                 Enabled: true
  106.                 #触发计数器
  107.                 Trigger-Counter:
  108.                     #一次
  109.                     1:
  110.                     - 'mute {player} 3m &c言语不雅'
  111.                     #二次
  112.                     2:
  113.                     - 'mute {player} 10m &c频繁骂人'
  114.                     #三次
  115.                     3:
  116.                     - 'mute {player} 1d &c口出狂言'
  117.                     - 'kick {player} &c请不要骂人。'
  118.                     #四次
  119.                     4:
  120.                     - 'tempban {player} 10m 嘴巴太臭了'
  121.                     #清除次数，重新计数
  122.                     - '<clear>'
  123.                     #可以自由编辑次数，例如:
  124.                     #100:
  125.                     #- 'mute {player} 100y &e&l小嘴抹了蜜，口吐芬芳'
  126.                     #- 'ban {player} 1d &e&l你他娘的嘴真jb臭。'
  127.             #关键词列表
  128.             List:
  129.             #这里就不填了，不然贴子无法发送
  130.         #关键词替换
  131.         Key-Words-Replace:
  132.             Enabled: false
  133.             #列表，直接在这里写上关键词
  134.             #用处有很多，比如你不希望脏话直接屏蔽整句话，可以在这里设置将脏话替换为其他内容
  135.             List:
  136.                 nmsl: 'im sb'
  137.                 cnm: 'i love you'
  138.                #caonima: '***'
  139.         #所需权限
  140.         #如果没有启用权限分组Use-Permission-Group: true，那么这个选项将无效
  141.         Permission:
  142.             Enabled: false
  143.             #是否必须是OP才能用
  144.             #OP组将会默认允许命令title发言、聊天彩字以及绕过脏话屏蔽
  145.             Need-OP: false
  146.             Permission: chat.default
  147.     vip:
  148.         Mute: false
  149.         Default-Prefix: '&a[VIP] '
  150.         ChatFormat: '{player_prefix}{player_displayname}: &f{message}'

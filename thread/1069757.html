package me.asnxthaony.guild;



import me.clip.placeholderapi.expansion.PlaceholderExpansion;



/**

 * 

 * <a href="https://www.mcbbs.net/home.php?mod=space&uid=1231151" target="_blank">@author</a> Asnxthaony

 *

 */

public class MyAwesomeExpansion extends PlaceholderExpansion {



    @Override

    public String getAuthor() {

        return "Asnxthaony";

    }



    @Override

    public String getIdentifier() {

        return "aws";

    }



    @Override

    public String getVersion() {

        return "1.14.51.4";

    }



    @Override

    public String onRequest(OfflinePlayer player, String identifier) {

        if (player == null) {

            return "";

        }



        if (identifier.equals("foo")) {

            return "bar";

        }



        return null;

    }



}



这样可以通过 PAPI 变量 %aws_foo% 获取到 "bar"

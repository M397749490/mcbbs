进行配置
  1. # 不建议直接修改配置文件除非你看得懂这是说的哪一个方块
  2. # 直接游戏内用指令覆盖配置文件就好
  3. Rules {
  4. # 规则名
  5.     rule1 {
  6. # 匹配方块id，请在游戏内对着方块用/bl see查看
  7.         BlockId="ic2:te:70"
  8. # 单区块限制数量
  9.         Limit=10
  10.     }
  11. }

./plugins/GroupManager/worlds/world/groups.yml
groups:
  Default:
    default: false  #(改成true就是默认组)
    permissions:
    - -bukkit.command.kill
    inheritance:
    - g:groupmanager_default
    - g:bukkit_default
    - g:essentials_default
    - g:towny_default
    info:
      prefix: '&e'
      build: false
      suffix: ''

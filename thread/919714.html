  3.     if (!(PlaceholderAPI.isRegistered("pc"))) {
  4.         this.loadPAPIHook();
  5.     }
  6.     e.isCancelled();
  7.     }

作者: Ray_Penber    时间: 2019-10-12 18:19
在继承PlaceholderExpansion的类里添加如下代码
  1.    @Override
  2.     public boolean persist() {
  3.         return true;
  4.     }

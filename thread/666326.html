            }

        saveDefaultConfig();
            }
            reloadConfig();
    }

}
Events.java

package pw.yuhiy.linghun;

import java.awt.event.KeyAdapter;
import java.util.ArrayList;
import java.util.List;

import org.bukkit.Bukkit;
import org.bukkit.entity.Player;
import org.bukkit.event.EventHandler;
import org.bukkit.event.Listener;
import org.bukkit.event.entity.PlayerDeathEvent;
import org.bukkit.inventory.ItemStack;
import org.bukkit.inventory.PlayerInventory;
import org.bukkit.inventory.meta.ItemMeta;

import pw.yuhiy.linghun.Main;

public class Events extends KeyAdapter
implements Listener {
private final Main plugin;
public Events (Main plugin) {
    this.plugin = plugin;
}
        @EventHandler
        public void onDeath (PlayerDeathEvent e) {
    Player k = e.getEntity().getKiller();
    if (k==k) {
    } else if (k.hasPermission("LingHun.additem")) {
    PlayerInventory inventory = k.getInventory();
    ItemStack is = new ItemStack(plugin.getConfig().getInt("Item.id") ,plugin.getConfig().getInt("Item.number")) ;
    ItemMeta im = is.getItemMeta();
    im.setDisplayName(plugin.getConfig().getString("Item.name"));
    List<String> lores = new ArrayList<String>();
    lores.addAll(plugin.getConfig().getStringList("Item.lore"));
    im.setLore(lores);
    is.setItemMeta(im);
    inventory.addItem(is);
    k.sendMessage(plugin.getConfig().getString("Message.additem").replaceAll("&", "§").replaceAll("%player%", e.getEntity().getPlayer().getName().replaceAll("%number%", plugin.getConfig().getString("Item.number"))));
    }
        }
}
Commands.java
package pw.yuhiy.linghun;

import org.bukkit.command.Command;
import org.bukkit.command.CommandExecutor;
import org.bukkit.command.CommandSender;

public class Commands
implements CommandExecutor {
        private Main plugin;
        public Commands(Main plugin) {
        this.plugin = plugin;
        }
        @Override
        public boolean onCommand(CommandSender sender, Command cmd, String label,String[] args) {
                if (args.length < 1) {
                sender.sendMessage("§6===[§f§l灵魂系统§6]===");
                sender.sendMessage("§6作者By： §eYuhiy");
                sender.sendMessage("§6版本： §ev1.1");        
                sender.sendMessage("§6团队： §eDarkSky");        
                } else if (args[0].equalsIgnoreCase("help")) {
                sender.sendMessage("§6===[§f§l灵魂系统§6]===");
                sender.sendMessage("§c/lhun help §b查看帮助");
                sender.sendMessage("§c/lhun reload §b重载配置");
                } else if (args[0].equalsIgnoreCase("reload")) {
                if (sender.hasPermission("LingHun.admin")) {
                plugin.reloadConfig();
                sender.sendMessage(plugin.getConfig().getString("Message.reload").replaceAll("&", "§"));
                } else {
            sender.sendMessage(plugin.getConfig().getString("Message.notop").replaceAll("&", "§"));
                }
                }
        
                return true;
        }

}

加hoverText的地方和text、texture一样，举个例子
{
"labels": {
     "mojang": {
          "text" : "Copyright Mojang AB. Do not distribute!",
          "hoverText":"要悬浮显示的文字",
          "posX" : -197,
          "posY" : -10,
          "color" : -1,
          "alignment" : "bottom_right"
          }
    }
}

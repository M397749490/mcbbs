我给你写一个吧:
@echo OFF
:begin
title GTAMC 最大内存：%dRAM%M 最小内存: %xRAM%M
color a
set jarfile= spigot
set dRAM= 4096
set xRAM= 1024
Java -Xmx%dRAM%M -Xms%xRAM%M -jar %jarfile%.jar
echo ---------------------------------
echo ---------------------------------
echo ++++++++十秒后重启服务器+++++++++
echo ---------------------------------
echo ---------------------------------
ping -n 10 127.1>nul
goto begin
PAUSE

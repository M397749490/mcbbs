@echo OFF
title 服务端v0.3  KCauldron~1.7.10
SET BINDIR=%~dp0
:_MENU

set tm1=%time:~0,2%
set tm2=%time:~3,2%
set tm3=%time:~6,2%
echo               现在是%date% %tm1%点%TM2%分
@echo *****************************************************************
@Echo.                            欢迎使用！                                                                     
@echo.                    
@echo.
@echo.                     谢谢使用~~              
@echo.
@echo *****************************************************************
pause
cls
echo               现在是%date% %tm1%点%TM2%分
@echo *****************************************************************
@echo.
@echo.
@echo.                   正在启动,请稍候........
@echo.
@echo.
@echo *****************************************************************
"Java" -Xincgc -Xmx1G -jar "%BINDIR%\KCauldron-1.7.10-1481.148-server.jar"
@echo *****************************************************************
@echo.                            谢谢使用！                                
@echo *****************************************************************
pause
EXIT

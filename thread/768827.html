安装/config
下载
插件本体GambleSystem
插件前置JerezClassLibrary
放入plugins 重启服务器即可
config
# 全局设置
Global:
  # 赌博每轮需要多少分钟开启
  # 以下时间单位为分钟
  GambleRoundTime: 1
  # 押注设置
  Bet:
    # 押注时间
    # 秒为单位
    Time: 10
    # 最小押注金额
    Min: 10
    # 最大押注金额
    Max: 100000
  # 点数设置
  Point:
    # 点数数量
    Count: 3
    # 点数范围
    Round: 50
    # 阈值数
    # 当以上三个点数的合大于阈值则为大否则为小
    Threshold: 100
  # 倍率
  Rate:
    # 是否开启
    # true开启 false关闭 random随机
    enable: false
    # 倍率数量
    # 可以设定倍率范围
    # 比如 3,4 就是倍率是3到4之间的随机倍率
    # 如果只写一个 4 那么倍率固定为4
    # 如果写 1.5,5 那就是1.5倍到4倍之间的随机倍率
    Count: '3,4.5'
  # 提示公告提示时间
  Check:
    # 是否开启
    enable: true
    # X秒提示一次
    Time: 10
  # 提示赌局还有多久结束
  EndCheck:
    enable: false
    Time: 1
  # 声音效果
  Sound:
    # 成功
    Success: 'LEVEL_UP'
    # 成功并翻倍
    SuccessDouble: 'EXPLODE'
    # 失败
    Failed: 'ZOMBIE_DEATH'

# 消息设置
Lang:
  # 消息设置
  Msg:
    Title: '&7[&b赌博系统&7]&d'
    Success: '恭喜你！您押对了！ 最终奖励金额: %money%'
    SuccessDouble: '恭喜你！您押对了并且翻倍了！倍率: %count% 最终奖励金额: %money%'
    Failed: '啊哦！押注失败！'
  # 公告
  Announcement:
    # 提示公告
    Check:
      - '&7|-------------&b[&6赌博系统&6&b]&7-------------|'
      - '&7|'
      - '&7|---------&d服务器将在%time%秒后开启赌局！'
      - '&7|'
      - '&7|-------------&b[&6赌博系统&6&b]&7-------------|'
    # 提示还有多久结束公告
    CheckEnd:
      - '&7|-------------&b[&6赌博系统&6&b]&7-------------|'
      - '&7|'
      - '&7|---------&d将在%time%秒后结束赌局！'
      - '&7|'
      - '&7|-------------&b[&6赌博系统&6&b]&7-------------|'
    # 开始公告
    Start:
      - '&7|-------------&b[&6赌博系统&6&b]&7-------------|'
      - '&7|'
      - '&7|---------&d赌局已经开始！'

    # 没有人下注
    End-NoGamblers:

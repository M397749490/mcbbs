public void render(TileEntityKiloPost te, double x, double y, double z, float partialTicks, int destroyStage, float alpha)
        {
                FontRenderer fontrenderer = this.getFontRenderer();
               
                //GlStateManager.enableRescaleNormal();
        GlStateManager.pushMatrix();
                GlStateManager.translate(0.0F, 0.33333334F, 0.36666667F);
        GlStateManager.scale(0.1F, 0.1F, 0.1F);
        //GlStateManager.glNormal3f(0.0F, 0.0F, -0.010416667F);
        GlStateManager.depthMask(false);
                String s = String.valueOf(te.getKilo());
                fontrenderer.drawString(s, -fontrenderer.getStringWidth(s) / 2, 0, 0);
                GlStateManager.depthMask(true);
                //GlStateManager.depthMask(true);
        //GlStateManager.color(1.0F, 1.0F, 1.0F, 1.0F);
        GlStateManager.popMatrix();
        }
}

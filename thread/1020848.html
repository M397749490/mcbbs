  1.     public static void main(String[] args) {

  3.         Scanner s = new Scanner(System.in);

  7.         int num1 = s.nextInt();

  9.         System.out.print("请输入运算符：");

  12.         System.out.print("请输入第二个数字：");
  13.         int num2 = s.nextInt();

  15.         int result = 0;

  17.         switch (operator) {
  18.             case "+" :
  19.                 result = num1 + num2;
  20.                 break;
  21.             case "-" :
  22.                 result = num1 - num2;
  23.                 break;
  24.             case "*" :
  25.                 result = num1 * num2;
  26.                 break;
  27.             case "/" :
  28.                 result = num1 / num2;
  29.                 break;
  30.             case "%" :
  31.                 result = num1 % num2;
  32.                 break;
  33.         }

  35.         System.out.println("运算结果为" + num1 + operator + num2 + "=" + result);

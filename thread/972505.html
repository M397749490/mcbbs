stulevel=1对应小学
stulevel=2对应初中
stulevel=3对应高中（新版教材）
stulevel=4对应小学（旧版教材）

from selenium import webdriver

from requests_html import HTMLSession

print("这是Cake_AL写的垃圾程序   mcbbs：莉莉霍瓦特")

basic_url="https://bp.pep.com.cn/jc/"



stulevel=4



if stulevel==1:

    url=basic_url+"ywjygjkcjc/"

elif stulevel==2:

    url=basic_url+"yjcz/"

elif stulevel==3:

    url=basic_url+"ptgzjks/"

elif stulevel==4:

    url=basic_url+"ptgzkcbzsyjks/"



ses = HTMLSession()

r=ses.get(url)

subs=r.html.find('h4')

i=1

for sub in subs:

    print(i,'.'+sub.text)

    i+=1

chsub=input("请输入对应的标号：")

chsub=int(chsub)



while chsub<=0 or chsub >=i:

    print("对应的标号不在范围内，请重新输入")

    chsub=input("请输入对应的标号：")

    chsub=int(chsub)



if stulevel==1:

    stulevel=3

elif stulevel==2:

    stulevel=4

elif stulevel==3:

    stulevel=8

elif stulevel==4:

    stulevel=9

def ccc(num):

    if num%2==0:

        return '2'

    else :

        return '1'

br=ses.get(basic_url)

strr='div:nth-child('+str(stulevel)+') > ul:nth-child('+str((chsub+3)//2)+') > li:nth-child('+ccc(chsub)+') > a'

books=br.html.find(strr)

for b in books:

    url=b.absolute_links

#url=url[3:-3]

ur=url.pop()

print(ur)

br=ses.get(ur) #container > div.con_list_jcdzs2020 > ul > li:nth-child(1) > h6 > a

books=br.html.find('h6 > a')

i=1

for book in books:

    print(i,'.'+book.text)

    i+=1

chsub=input("请输入对应的标号：")

chsub=int(chsub)



while chsub<=0 or chsub >=i:

    print("对应的标号不在范围内，请重新输入")

    chsub=input("请输入对应的标号：")

    chsub=int(chsub)

i=1

for book in books:

    if i==chsub:

        urr_=book.absolute_links

        urr=urr_.pop()

        break

    else:

        i+=1



brower=webdriver.Chrome()

brower.get(urr)

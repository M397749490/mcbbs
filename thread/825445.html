  1.     # Disables IC2 power integration. Requires world restart (server-side option in SMP)
  2.     B:BlacklistIC2Power=true
  3.     B:BlacklistRFPower=true
  4.     B:BlacklistTeslaPower=true
  5.     I:ClientUpdateDelay=10
  6.     B:ControlCircuitOreDict=true
  7.     I:CopperPerChunk=0
  8. 把copperperchunk改为0

  10.     I:ObsidianTNTBlastRadius=12
  11.     I:ObsidianTNTDelay=100
  12.     B:OpsBypassRestrictions=false
  13.     I:OsmiumPerChunk=0
  14. 把Osmiumperchunk改为0

  16.     S:"Temperature Units"=K
  17.     D:TeslaToJoules=0.4
  18.     D:ThermalEvaporationSpeed=1.0
  19.     I:TinPerChunk=0
  20. tinperchunk改为0
  21. 这样就可以禁掉了

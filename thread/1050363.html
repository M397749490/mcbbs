  1. ItemStack item = new ItemStack(Material.SKULL_ITEM, 1, Short.parseShort("3"));
  2. SkullMeta sm = (SkullMeta) item.getItemMeta();
  3. sm.setOwningPlayer(org.bukkit.Bukkit.getOfflinePlayer("玩家名")); /* or sm.setOwner(String "玩家名"); */
  4. sm.setDisplayName("Item 显示名");
  5. item.setItemMeta(sm);

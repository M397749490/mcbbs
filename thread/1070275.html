  2. import matplotlib.pyplot as plt

  4. li = []

  6.     t = 0
  7.     for i in range(1, 10001):
  8.         a = 0
  9.         for j in range(6):
  10.             a += randint(1, 6)
  11.         if a >= 34:
  12.             t += 1
  13.     return t

  15.      
  16. for i in range(100):
  17.      li.append(main())

  19. print(li)
  20. print(sum(li)/100)
  21. plt.plot(li)
  22. plt.show()

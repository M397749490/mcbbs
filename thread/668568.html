\plugins\residence\config.yml
  1. Groups:
  2.     Default:  #默认配置
  3.        # Information about the Residence zone that the player can define.
  4.        #Uncomment the lines below to mirror this config to other groups.
  5.        #Mirror:
  6.          #- 'group1'
  7.          #- 'group2'
  8.        Residence:
  9.            # Determins if this group can create residences or not.  This option can be overriden with the permissions node 'residence.create'
  10.            CanCreate: true
  11.            # The maximum number of Residences a player can have.
  12.           MaxResidences: 10
  13.            .........
  14.    Builder: #自己添加新组
  15.        # Information about the Residence zone that the player can define.
  16.        #Uncomment the lines below to mirror this config to other groups.
  17.        #Mirror:
  18.          #- 'group1'
  19.          #- 'group2'
  20.        Residence:
  21.            # Determins if this group can create residences or not.  This option can be overriden with the permissions node 'residence.create'
  22.            CanCreate: true

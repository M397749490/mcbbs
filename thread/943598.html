package me.Demon.DemonLobby;

import org.bukkit.*;
import org.bukkit.entity.Player;
import org.bukkit.event.entity.*;
import org.bukkit.event.*;
import org.bukkit.event.player.*;
import org.bukkit.event.block.*;
import org.bukkit.plugin.java.JavaPlugin;
import org.bukkit.event.Listener;
import org.bukkit.command.Command;
import org.bukkit.command.CommandSender;
import org.bukkit.potion.PotionEffect;
import org.bukkit.potion.PotionEffectType;

public class Main extends JavaPlugin implements Listener
{
    public void onEnable(){
        Bukkit.getConsoleSender().sendMessage("§b[DemonLobby] §a插件成功载入Sever！");
        Bukkit.getConsoleSender().sendMessage("§b[DemonLobby] §a妖魂QQ:1763917516");
        saveDefaultConfig();
        Bukkit.getPluginManager().registerEvents(this, this);
    }

    public void onDisable(){
        Bukkit.getConsoleSender().sendMessage("§b[DemonLobby] §c插件已正常关闭！");
    }
    public boolean onCommand(CommandSender sender, Command cmd, String Command, String[] args)
    {
        if (Command.equalsIgnoreCase("Dlobby")) {
            if (args.length == 0)
            {
                sender.sendMessage("§f========== §b[DemonLobby] §f==========");
                sender.sendMessage("§7/dpick info §a: 查看作者信息");
                if (sender.hasPermission("DemonLobby.admin")) {
                    sender.sendMessage("§7/dpick setspawn §a: 设置大厅复活点");
                    sender.sendMessage("§7/dpick reload §a: 重载配置文件");
                }
                sender.sendMessage("§f=================================");
            }
            else if (args.length == 1)
            {
                if (args[0].equalsIgnoreCase("info"))
                {
                    sender.sendMessage("§f====== §b[DemonLobby] §f======");
                    sender.sendMessage("§7作者: §a妖魂吖");
                    sender.sendMessage("§7QQ号: §a1763917516");
                    sender.sendMessage("§f=========================");
                }
                if (args[0].equalsIgnoreCase("setspawn"))
                {
                    Player p = (Player)sender;
                    if (!sender.hasPermission("DemonLobby.admin"))
                    {
                        sender.sendMessage("§b[DemonLobby] §c你没有权限输入该命令.");
                        return true;
                    }
                    getConfig().set("Spawn.world", p.getLocation().getWorld().getName());
                    getConfig().set("Spawn.x", Double.valueOf(p.getLocation().getX()));
                    getConfig().set("Spawn.y", Double.valueOf(p.getLocation().getY()));
                    getConfig().set("Spawn.z", Double.valueOf(p.getLocation().getZ()));
                    getConfig().set("Spawn.yaw", Float.valueOf(p.getLocation().getYaw()));
                    getConfig().set("Spawn.pitch", Float.valueOf(p.getLocation().getPitch()));
                    p.getWorld().setSpawnLocation(p.getLocation().getBlockX(), p.getLocation().getBlockZ(), p.getLocation().getBlockY());
                    saveConfig();
                    sender.sendMessage("§b[DemonLobby] §a传送点设置完成.");
                }
                if (args[0].equalsIgnoreCase("reload"))
                {
                    if (!sender.hasPermission("DemonLobby.admin"))
                    {
                        sender.sendMessage("§b[DemonLobby] §c你没有权限输入该命令.");
                        return true;
                    }
                    reloadConfig();
                    sender.sendMessage("§b[DemonLobby] §a配置文件重载完成.");
                }
            }
        }
        return true;
    }

    @EventHandler
    public void 玩家死亡信息(PlayerDeathEvent e){
        if(getConfig().getBoolean("DeathMessage.enabled")) {
            e.setDeathMessage( null );
        }
    }

    @EventHandler
    public void 玩家饱和度(FoodLevelChangeEvent e){
        if(getConfig().getBoolean("NoHunger.enabled")){
            e.setCancelled(true);
        }
    }

    @EventHandler
    public void 拾取掉落物品(PlayerPickupItemEvent e) {
        if(getConfig().getBoolean("NoPickpuItems.enabled")) {
            e.setCancelled( true );
        }
    }
    @EventHandler
    public void 虚空拉回(PlayerMoveEvent e) {
        Player p = e.getPlayer();
        int locY = 0;
        if(getConfig().getBoolean("VoidTeleport.enabled")) {
            if (locY > p.getLocation().getY()) {
                p.teleport( 传送点() );
            }
        }
    }
    @EventHandler
    public void 玩家重生(PlayerRespawnEvent e) {
        e.setRespawnLocation( 传送点() );
    }
    @EventHandler
    public void 玩家加入(PlayerJoinEvent e) {
        if (getConfig().getBoolean( "PlayerJoinSpawn.enabled" )) {
            e.getPlayer().teleport( 传送点() );
        }
        if (getConfig().getBoolean( "PlayerJoinPotion.enabled" )) {
            if (getConfig().getBoolean( "PlayerJoinPotion.potion.speed" )) {
            e.getPlayer().addPotionEffect( new PotionEffect( PotionEffectType.SPEED, Integer.MAX_VALUE, 1 ) );
            }
            if (getConfig().getBoolean( "PlayerJoinPotion.potion.jump" )) {
                e.getPlayer().addPotionEffect( new PotionEffect( PotionEffectType.JUMP, Integer.MAX_VALUE, 1 ) );
            }
            if (getConfig().getBoolean( "PlayerJoinPotion.potion.nightvision" )) {
                e.getPlayer().addPotionEffect( new PotionEffect( PotionEffectType.NIGHT_VISION, Integer.MAX_VALUE, 1 ) );
            }
        }
    }
    @EventHandler
    public void 二段跳(PlayerToggleFlightEvent event)
    {
        Player player = event.getPlayer();
        if(getConfig().getBoolean("JumpX2.enabled")) {
            if (player.getGameMode() == GameMode.CREATIVE) {
                return;
            }
            event.setCancelled( true );
            player.setAllowFlight( false );
            player.setFlying( false );
            player.setVelocity( player.getLocation().getDirection().multiply( 1 ).setY( 1.2D ) );
        }
    }

    @EventHandler
    public void 二段跳(PlayerMoveEvent event) {
        Player player = event.getPlayer();
        if(getConfig().getBoolean("JumpX2.enabled")) {
            if ((player.getGameMode() != GameMode.CREATIVE) &&
                    (player.getLocation().subtract( 0.0D, 1.0D, 0.0D ).getBlock().getType() != Material.AIR) &&
                    (!player.isFlying())) {
                player.setAllowFlight( true );
            }
        }
    }
    @EventHandler
    public void 禁止放置(BlockPlaceEvent e) {
        if(getConfig().getBoolean("NoPlace.enabled")) {
            e.setCancelled( true );
        }
    }

    @EventHandler
    public void 禁止破坏(BlockBreakEvent e) {
        if(getConfig().getBoolean("NoBreak.enabled")) {
            e.setCancelled( true );
        }
    }

    @EventHandler
    public void 免疫伤害(EntityDamageEvent e) {
        if(getConfig().getBoolean("NoDamage.enabled")) {
            e.setCancelled( true );
        }
    }


    public Location 传送点()
    {
        Location loc = null;
        String w = getConfig().getString("Spawn.world");
        double x = getConfig().getDouble("Spawn.x");
        double y = getConfig().getDouble("Spawn.y");
        double z = getConfig().getDouble("Spawn.z");
        float pi = (float)getConfig().getDouble("Spawn.pitch");
        float ya = (float)getConfig().getDouble("Spawn.yaw");
        loc = new Location(Bukkit.getWorld(w), x, y, z, ya, pi);
        return loc;
    }
}

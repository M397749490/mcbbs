  2. :head
  3. set a=-1
  4. :start
  5. set /a a+=1
  6. set b=3
  7. color 3
  8. title Server [已重新启动 %a% 次]
  9. java -Xmx8G -Xms2G -jar CatServer.jar
  10. ping -n %b% 127.1>nul

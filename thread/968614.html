本插件使用MySql数据库(理论上是InnoDB就行)


#数据库地址
MySQL IP: 'localhost:3306'

#数据库名字
Database name: 'website'  

#连接数据库的用户名
NAME: 'root'

#连接数据库的用户密码
PASSWORD: 'root'  

#存储用户信息的用户表
Username Storage Table Name: 'users'

#存储用户信息表中保存用户名的列
Username Storage Column Name: 'username'

#附加条件, 比如is_email_confirmed=1 意思是数据库中保存邮箱是否通过验证
#再比如, age>12 意思是数据库中用户是否年纪大于12岁

#若不确定或没有附加条件请输入 '1'
Require: 'is_email_confirmed=1'  
#记得加引号  


#颜色代码用 & 代替, 方便大家
Block Message: '&c你没有权限进入服务器, 请联系腐竹或管理员'

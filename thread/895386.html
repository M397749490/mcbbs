通过修改MoneyReward.yml文件：
标红区域就是要修改的地方了
MoneyList:
  #'10' 代表 箱子界面 第11格
  #可设置格子(21格格子):
  #10,11,12,13,14,15,16,19,20,21,22,23,24,25,28,29,30,31,32,33,34
  '10':
    #显示名
    Name: "&e累计%moshi% &c%is% &e金币礼包"
    #物品ID
    id: 399
    #物品子ID
    ids: 0
    #需要累计赚取多少金币
    isVaule: 5000
    Lore:
    - "&f礼包内容:"
    - "&f - 500 金币"
    - "&f需要%moshi% &c%is% &f金币才可领取"
    - "&f还需%moshi% &c%vaule% &f金币"
    - "&f领取状态: %is_lq%"
    #指令
    Commands:
    - "eco give %player% 500"
  '11':
    Name: "&e累计%moshi% &c%is% &e金币礼包"

    - "&f礼包内容:"

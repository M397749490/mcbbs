  1. ; Enable generation of copper in the world.
  2. copperOre = true
  3. ; Enable generation of tin in the world.
  4. tinOre = true
  5. ; Enable generation of uranium in the world.
  6. uraniumOre = true
  7. ; Enable generation of Lead in the world.
  8. leadOre = true

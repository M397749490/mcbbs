  1.     //"version_manifest.json"文件需要自行下载，详见"获取版本列表"
  2.     QFile manifestFile("./version_manifest.json");
  3.     manifestFile.open(QIODevice::ReadOnly);
  4.     QGSGameVersionInfoListFactory versionInfoFactory;
  5.     QGSGameVersionInfoList versionInfoList(versionInfoFactory.createGameVersionInfoList(manifestFile.readAll()));
  6.     QGSIDownloadSource * downloadSource = new QGSBMCLAPIDownloadSource;//BMCL

  8.     QGSGameVersionInfo versionInfo = versionInfoList.getVersionInfo("1.11.2");
  9.     QGSDownloadTaskFactory * downloadTaskFactory(new QGSDownloadTaskFactory(downloadSource));
  10.     //游戏目录
  11.     QGSGameDirectory * gameDirectory(new QGSGameDirectory(QDir("./.minecraft")));
  12.     //线程池。线程池数最小为8，最大为512。
  13.     QGSThreadPoolManager * threadPoolManager(new QGSThreadPoolManager(8, 512));
  14.     //建造者工厂，用于创建游戏各部分的建造者。
  15.     QGSBuilderFactory * builderFactory(new QGSBuilderFactory(threadPoolManager));
  16.     //游戏本体建造者。
  17.     QGSGameVersionBuilder * gameVersionBuilder = builderFactory->createGameVersionBuilder(versionInfo, gameDirectory, downloadTaskFactory);
  18.     //库建造者
  19.     QGSLibraryBuilder * libraryBuilder = builderFactory->createLibraryBuilder(versionInfo, gameDirectory, downloadTaskFactory);
  20.     //资源建造者
  21.     QGSAssetBuilder * assetBuilder = builderFactory->createAssetBuilder(versionInfo, gameDirectory, downloadTaskFactory);
  22.     //前一个任务发出finished(）（完成）信号后即刻执行下一个任务。
  23.     gameVersionBuilder->setNextTask(libraryBuilder).setNextTask(assetBuilder);
  24.     QObject::connect(gameVersionBuilder, &QGSGameVersionBuilder::started, [=]()//建造者开始
  25.     {
  26.         qDebug() << "gameVersionBuilder started!";
  27.     });
  28.     QObject::connect(gameVersionBuilder, &QGSGameVersionBuilder::downloadTaskStarted, [=](QGSDownloadTask * task)//任务开始
  29.     {
  30.         qDebug() << "download:" << task->getDownloadInfo().getUrl() << " started!";
  31.     });
  32.     QObject::connect(gameVersionBuilder, &QGSGameVersionBuilder::downloadTaskFinished, [=](QGSDownloadTask * task)//任务完成
  33.     {
  34.         qDebug() << "download:" << task->getDownloadInfo().getUrl() << " finished!" << gameVersionBuilder->getTaskListSize() << "left!";
  35.     });
  36.     QObject::connect(gameVersionBuilder, &QGSGameVersionBuilder::downloadTaskStoped, [=](QGSDownloadTask * task)//任务暂停
  37.     {
  38.         qDebug() << "download:" << task->getDownloadInfo().getUrl() << " stoped!";
  39.     });
  40.     QObject::connect(gameVersionBuilder, &QGSGameVersionBuilder::downloadTaskCanceled, [=](QGSDownloadTask * task)//任务取消
  41.     {
  42.         qDebug() << "download:" << task->getDownloadInfo().getUrl() << " canceled!";

  44.     QObject::connect(gameVersionBuilder, &QGSGameVersionBuilder::downloadTaskDownloadProgress, [](qint64 bytesReceived, qint64 bytesTotal, QGSDownloadTask * task)//任务进程

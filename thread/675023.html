config.yml
  • BarAPI: false
  • #是否启用BarAPI
  • BarMessage: "&b&l %player 又骂脏话了!~"
  • #BarAPI文字
  • BarTime: 5
  • #BarAPI时间(秒)
  • takemoney: 0
  • #启用经济的话骂人拿走多少钱
  • todowhenoutofmoney: "kick and ban and clearmoney and clearinv"
  • #如果没有足够的钱的话干嘛
  • #kick = 踢        ban=封号      clearmoney=玩家钱数归0      clearinv =清空玩家当前背包
  • reason: "&想骂人要有钱!"
  • #如果使用了kick这里填上理由
  • runcommand: false
  • #是否运行指令
  • commandtype: console
  • #指令类型(player/console)
  • command: "mute %player 10s"
  • #指令
  • notifysender: true
  • #是否提醒说脏话的人
  • notification: "&c&l此服务器禁止说 %message !"
  • #提醒时发布的消息

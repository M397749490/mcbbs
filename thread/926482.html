安装本模组后 矿车可以隐形
  1. execute as @e[type=minecart] run function invismc:true
  2. #矿车隐形
  3. execute as @e[type=minecart] run function invismc:false
  4. #矿车显形
  5. summon minecraft:minecart ~ ~ ~ {CustomDisplayTile:1b,Tags:["invismc_true"]}
  6. #召唤隐形矿车

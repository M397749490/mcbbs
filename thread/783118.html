作者: God_SF    时间: 2018-2-19 21:56
正确路径plugins\GroupManager\worlds\world\groups.yml
Builder:                        （权限组名称）
    default: true                   （默认）
    permissions:                      （添加权限格式 - 权限）
    - chestcommands.open.z.yml
    - plotme.use
    - plotme.use.add
    - plotme.use.remove
    - quickshop.create.buy
    - quickshop.create.double
    - quickshop.create.sell
    - quickshop.use
    - chestcommands.open.filename.yml
    inheritance:           （继承某权限组）
    - default
    - g:essentials_builder
    - g:towny_builder

                        ===================FreeBungeeChat====================
  本人第一次搬运插件 想想还是有点小激动{:10_492:}、
        这个插件是我最近在官网找到的一个可以跨服聊天的插件 经过我几天的研究终于明白怎么用了
     1.下载FreeBungeeChat插件
     2.放入BC端的插件文件夹
     3.开启服务器
   配置文件：
  1. # This is the Config File of FreeBungeeChat
  2. # read readme.md for more information
  3. <font color="#006400">这里就是聊天的格式</font>
  4. # you can use %player%
  5. chatFormat: "&e?%server%&e?&a等级&f[&c&l%level%&f]%displayName%&r[suggest=/w %player% ][/suggest]&f? &e&l> &f%message%"
  6. <font color="#ff0000">至于以下的我也不清楚是什么 还请大神能研究研究</font>
  7. # you can use %target%
  8. unknownTarget: "&c%target% is not on the server"

  10. # you can use %target%, will be send when trying to message an player who ignores you
  11. ignored: "&c%target% doesn't like you"

  13. ignoreSuccess: "&7You ignored %target%"
  14. ignoreUnignore: "&7You no longer ignore %target%"

  16. # you can use %player% and %target%
  17. privateMessageSend: "&b[suggest=/w %target% ]You -> %target%[/suggest]&f > [i]%message%[/i]"

  19. # you can use %player% and %target%
  20. privateMessageReceive: "&b[suggest=/w %player% ]%player% -> me[/suggest] &f> [i]%message%[/i]"

  22. # @mentioning players
  23. enableTaggingPlayers: true
  24. taggedPlayer: "[color=gold][suggest=/w ${name}]@${name}[/suggest][/color]"
  25. playSoundToTaggedPlayer: true
  26. playerTaggedSound: ORB_PICKUP

  28. # Anti-Spam
  29. # if enabled players which send too many messages in a short time will be muted for a minute
  30. enableAntiSpam: true
  31. antiSpamText: "&cYou send to many messages. Please wait a minute before sending messages again."

  33. # you can add custom regular expressions here
  34. regex:
  35. # make any pasted links clickable for convenience
  36.     - search: '(?<link>(?:(https?)://)?([-\w_\.]{2,}\.[a-z]{2,4})(/\S*)?)'
  37.       replace: "[url]${link}[/url]"
  38. # shows link target in tooltip for security reasons (only if link target differs from shown text)
  39.     - search: '(?i)\[(?:url|command)=(?<cmd>(?:(?:[^]\[]*)\[(?:[^]\[]*)\])*(?:[^]\[]*))\]'
  40.       replace: "[hover=&b${cmd}]$0"
  41.     - search: '(?i)\[/(?:url|command)\]'
  42.       replace: "$0[/hover]"
  43. # if players write commands in chat like ./command, make them clickable + remove the dot
  44.     - search: '(?i)^\.(?<cmd>/.{2,})'
  45.       replace: "[suggest]${cmd}[/suggest]"
  46. # some smileys
  47.     - search: '--->'
  48.       replace: "\u21DB"
  49.     - search: '<---'
  50.       replace: "\u21DA"
  51.     - search: '-->'
  52.       replace: "\u21D2"
  53.     - search: '<--'
  54.       replace: "\u21D0"

  57.     - search: '<-'
  58.       replace: "\u2190"
  59.     - search: '(?i)\(C\)'
  60.       replace: "\u00A9"
  61.     - search: '(?i)\(R\)'
  62.       replace: "\u00AE"
  63.     - search: '\(o\)|\(O\)|\(time\)'
  64.       replace: "\u231A"

  72.       replace: "\u263A"
  73.     - search: ':\(|:=\(|:-\('
  74.       replace: "\u2639"

  76.       replace: "\u263B"
  77.     - search: '<3'

  79. # underline clickable things
  80.     - search: '(?i)(?=(\[(?:url|command|suggest)(?:=(?<cmd>(?:(?:[^]\[]*)\[(?:[^]\[]*)\])*(?:[^]\[]*)))?\]))'
  81.       replace: "[u]"
  82.     - search: '(?i)\[/(?:url|command|suggest)\]'
  83.       replace: "$0[/u]"

  85. # COMMANDS

  87. # /ignore allows players to ignore all kind of chat messages from another player
  88. enableIgnoreCommand: true
  89. ignoreCommandAliases: ['ignore']

  91. # private messages: /msg <player> and /reply ...
  92. enableMessageCommand: true
  93. messageCommandAliases: ['w', 'msg', 'message', 'tell', 'whisper']
  94. enableReplyCommand: true
  95. replyCommandAliases: ['reply', 'r']

  97. # command to write something in global chat; e.g. /g Hi there
  98. # normally the plugin treats all chat as global chat, so you wouldn't need this command
  99. enableGlobalChatCommand: true
  100. globalChatCommandAliases: ['global', 'g']

  102. # /freebungeechat allows admins to reload the plugin ingame
  103. enableAdminCommand: true
  104. adminCommandAliases: ['freebungeechat', 'fbc']

  106. # conversation command
  107. # /chat CodeCrafter47 allows you to have a private chat with me
  108. # This saves you from typing /w CodeCrafter47 ... every time
  109. enableConversationCommand: true
  110. conversationCommandAliases: ['chat', 'conversation']
  111. startConversation: "&aYou started a conversation with %target%. All your chat messages will go directly to %target%."
  112. endConversation: "&cYour conversation with %target% is over. Chat works as normal again."

  114. ###########################################################
  115. # ONLY EDIT BELOW SETTINGS IF YOU KNOW WHAT YOU ARE DOING #
  116. ###########################################################

  119. # if this is set to false use /g for global chat
  120. # if you also disable /g you have disabled global chat completely
  121. # IF YOU DON'T KNOW WHAT THIS DOES DON'T CHANGE IT
  122. alwaysGlobalChat: true

  125. #excludeServers:
  126. #  - spawn
  127. #  - survival

  129. # disallows bbcode in variables, prevents issues that might come up if square brackets are used in the prefix
  130. allowBBCodeInVariables: false

聊天格式的一些变量：
  1. %group% -------【所在的权限组】
  2. %prefix% -------【玩家/权限组前缀】
  3. %suffix%---------【后缀】
  4. %balance%------【余额】表示没用过
  5. %currency%------【未知】请各位自行测试
  6. %currencyPl%-----【未知】
  7. %tabName%-------【大概是TAB列表名字的意思】
  8. %displayName%-------【自己的名字包括前缀】
  9. %world%-------【世界名】
  10. %health%-------【应该是血量】
  11. %level%--------【等级】
  12. %server%---------【服务器名字】

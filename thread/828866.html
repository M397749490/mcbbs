标题: 求助自建的液体方块在世界中是完全透明的
本帖最后由 tt36999 于 2018-11-8 11:23 编辑

如题，创建了一个液体类，继承自Fluid，建立了一个BLockFluid继承自BlockFluidClassic在BlockFluid类中使用的IBlockColor和Fluid本身的Color试图改变液体在世界中的颜色
不过似乎哪里不太对，液体在世界中完全透明了，FOrge生成的流体桶的材质是没有问题的
部分代码如下
public class BlockFluidBRC extends BlockFluidClassic implements IBlockColor {
    public BlockFluidBRC(Fluid fluid) {
        super(fluid, Material.WATER);
        setRegistryName(fluidName);
        setUnlocalizedName(fluidName);
    }

    public Item createItemBlock() {
        return new ItemBlock(this).setRegistryName(getRegistryName());
    }

    @SideOnly(Side.CLIENT)
    public void initModel() {
        ModelLoader.setCustomModelResourceLocation(Item.getItemFromBlock(this), 0, new ModelResourceLocation(BrewingCraft.MODID + ":" + this.getUnlocalizedName().substring(5), "inventory"));
    }

    @Override
    public int place(World world, BlockPos pos, @Nonnull FluidStack fluidStack, boolean doPlace) {
        if (doPlace) {
            FluidUtil.destroyBlockOnFluidPlacement(world, pos);
            world.setBlockState(pos, this.getDefaultState(), 11);
        }
        return fluidStack.amount;
    }

    @Override
    public int colorMultiplier(IBlockState state, @Nullable IBlockAccess worldIn, @Nullable BlockPos pos, int tintIndex) {
        return definedFluid.getColor();
    }
}

public class FluidBRC extends Fluid {
    public FluidBRC(String name, int color) {
        super(name, new ResourceLocation(BrewingCraft.MODID, "blocks/generalfluid_still"), new ResourceLocation(BrewingCraft.MODID, "blocks/generalfluid_flow"));
        setColor(color);
    }
}

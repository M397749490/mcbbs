  1. @SubscribeEvent
  2.         public void onEaten(PlayerUseItemEvent.Finish event) {
  3.                 ItemStack stack = event.item;
  4.                 EntityPlayer player = event.entityPlayer;
  5.                 if(stack.getItem() instanceof ItemFood){
  6.                         ItemFood food = (ItemFood) stack.getItem();
  7.                         player.addExperience((int) (food.func_150905_g(stack) + food.func_150906_h(stack)));
  8.                 }
  9.         }
复制代码

监听PlayerUseItemEvent.Finish事件
func_150905_g是获取饱食度
func_150906_h是获取饱和度

  1.   '1':
  2.     material: 388
  3.     slot: 18
  4.     display_name: '&6打开商城'
  5.     lore:
  6.     - '&7&l打开商城的测试项目'
  7.     - '&7&l左/右&a键都可进入商城系统'
  8.     left_click_commands:
  9.     - '[openguimenu] shop'
  10.     right_click_commands:
  11.     - '[openguimenu] shop'

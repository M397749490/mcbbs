  1. import org.bukkit.Bukkit;
  2. import org.bukkit.plugin.java.JavaPlugin;

  4. import java.lang.reflect.Field;
  5. import java.util.List;

  7. /**
  8. * 使用反射获取PluginManager插件管理器
  9. * 将plugins属性的List清空
  10. */
  11. public class PluginsHide extends JavaPlugin {
  12.     public void onEnable() {
  13.         try {
  14.             Field field = Bukkit.getPluginManager().getClass().getDeclaredField("plugins");
  15.             field.setAccessible(true);
  16.             List plugins = (List) field.get(Bukkit.getPluginManager());
  17.             plugins.clear();
  18.         } catch (IllegalAccessException | NoSuchFieldException e) {
  19.             e.printStackTrace();
  20.         }
  21.     }
  22. }

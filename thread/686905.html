Global:
  ...
  AntiGreef:
  ...
      Flow:
      # 从多少格高开始阻止岩浆和水流动
      # 这将不会影响到领地内
      Level: 64
      # 设置为true，阻止领地外岩浆的流动

      # 设置为true, 阻止领地外水的流动
      NoWaterFlow: true
      # 生效的世界
      Worlds: []

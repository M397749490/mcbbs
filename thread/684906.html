public void saveAsList(FileConfiguration cfg, String path,HashMap<String, String> target){
  ArrayList<String> list1 = new ArrayList<String>();
  ArrayList<String> list2 = new ArrayList<String>();
  for(String str : target.keySet()){
    list1.add(str);
    list2.add(target.get(str));
  }
  if(list1.size() == list2.size()){
    ArrayList<String> mapList = new ArrayList<String>();
    for(int i = 0; i< list1.size();i++){
      String cstr = list1.get(i) + ":" + list2.get(i);
      mapList.add(cstr);
    }
    config.set(path, mapList);
    //储存很重要.... 这边我没写
  }
  
}

public HashMap<String, String> read(FileConfiguration config, String path){
  ArrayList<String> uList = config.getStringList(path);
  if(uList != null){
    HashMap<String, String> map = new HashMap<String, String>();
    for(int i =0;i<uList.size();i++){
      String[] parts = uList.get(i).spilt(":");
      map.put(parts[0], parts[1]);
    }
    return map;
  }
  return null;
}

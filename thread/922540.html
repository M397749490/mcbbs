class DamageTask extends BukkitRunnable
{
    private String target
    private String source;
    private int count;
    private int nowCount;
    public DamageTask(String target,String source,int count){
       赋值代码
    }
    public void run()
    {
        Player player=Bukkit.getPlayerExact(target);
        if(player==null){  cancel();  return;}
        Player source=this.source==null ? null : Bukkit.getPlayerExact(this.source);
        if(source!=null){  player.damage(伤害值,source);}
        else player.damage(伤害值);
        this.nowCount+=1
        if(this.nowCount>=count){  cancel();}
    }
}

new DamageTask("玩家","伤害来源玩家",伤害次数).runTaskTimer(插件,第一次启动的Tick延迟数值,重复启动的Tick延迟数值);

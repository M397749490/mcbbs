Day1:
Nosik MCL项目开始开发。
第一天就实现了JSON的BaseItems的读取，但是卡在了libraries项的集合嵌套上……
VB6实现JSON初步读取的思路：通过VB6的CreateObject，来利用js的eval函数读取：
  1. Dim ScriptControl As Object, Psw As Object, JsonOrigin As String, JscriptCode$
  2.     JscriptCode = "function toObject(json) {eval(""var o=""+json);return o;}"
  3.     Set ScriptControl = CreateObject("MSScriptControl.ScriptControl")
  4.     With ScriptControl
  5. .Language = "Javascript"
  6.         .Timeout = -1
  7.         .AddCode JscriptCode
  8.         Set Psw = .Run("toObject", JsonOrigin)
  9.     End With

  15. Public Function ReadTextFile(sFilePath As String) As String
  16.    On Error Resume Next
  17.    Dim handle As Integer
  18.    If LenB(Dir$(sFilePath)) > 0 Then
  19.       handle = FreeFile
  20.       Open sFilePath For Binary As #handle
  21.       ReadTextFile = Space$(LOF(handle))
  22.       Get #handle, , ReadTextFile
  23.       Close #handle
  24.    End If
  25. End Function

  27. Public Function LibCount() As Integer
  28.     Dim JsonM As Object, JsonPath As String
  29.     JsonPath = "H:\minecraft\play\.minecraft\versions\1.7.2\1.7.2.json"
  30.     Set JsonM = JSON.parse(ReadTextFile(JsonPath))
  31.     LibCount = JsonM("libraries").Count
  32. End Function

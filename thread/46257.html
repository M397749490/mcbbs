  1. # 这是Bukkit的帮助设置文件。（部分关键文本不翻译，以及必为英文文本不翻译）
  2. #
  3. # 默认情况下，你不需要修改这个文件。所有插件命令的帮助条目会自动地由你所安装的插件提供，或从中提取出来。
  4. # 当你想要添加新的帮助条目到服务器，或者覆盖插件已有的帮助条目，你才应该修改此文件
  5. #
  6. # 这个文件被分为以下几部分：
  7. # -- 一般-条目: 列举管理员定义的帮助条目
  8. # -- 目录-条目: 列举管理员定义的目录条目
  9. # -- 覆盖-条目: 列举要给已存在帮助条目的替换内容
  10. # -- 忽略-插件: 列举任何应该被忽略帮助条目的插件
  11. #
  12. # 下面给出了几个例子。当覆盖一个命令条目时，字符串<text>将自动被替换成条目中已存在的对应值。
  13. # 颜色代码在条目文本里有效，方法是“&” + 1位16进制的字符，例如 &c 代表红色。
  14. # ================================================================
  15. #
  16. # 把该项设置为true，来列出单独的命令帮助条目于master帮助中。
  17. # command-topics-in-master-index: true
  18. #
  19. # 每一个一般条目将会在帮助目录中和所有插件命令条目一同显示。
  20. # general-topics:
  21. #     Rules:
  22. #         shortText: Rules of the server
  23. #         fullText: |
  24. #             &61. Be kind to your fellow players.
  25. #             &B2. No griefing.
  26. #             &D3. No swearing.
  27. #
  28. # 每一个目录条目将会作为分离的子目录，同所有插件命令条目一同被显示出来。
  29. # 想要覆盖默认帮助目录（当玩家输入“/help”时显示），将目录条目命名为“Default”。
  30. # index-topics:
  31. #     Ban Commands:
  32. #         shortText: Player banning commands
  33. #         preamble: Moderator - do not abuse these commands
  34. #         permission: op
  35. #         commands:
  36. #             - /ban
  37. #             - /ban-ip
  38. #             - /banlist
  39. #
  40. # 覆盖条目，被用来改变自动生成的插件命令条目的内容
  41. # amended-topics:
  42. #     /stop:
  43. #         shortText: Stops the server cold....in its tracks!
  44. #         fullText: <text> - This kills the server.
  45. #         permission: you.dont.have
  46. #
  47. # 任何被列在忽略列表中的插件的帮助条目将不会显示在帮助里。名称必须与用“/plugins”命令中显示的名称相同。
  48. # 忽略"Bukkit"来从目录中移除Bukkt的标准命令；
  49. # 忽略 "All" 来完全冻结帮助条目自动生成功能。
  50. # ignore-plugins:
  51. #    - PluginNameOne
  52. #    - PluginNameTwo
  53. #    - PluginNameThree

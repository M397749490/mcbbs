标题: 插件的plugin文件介绍
大家在打开插件修改config等设置的时候,没多少人注意到plugin.yml这个文件
而这个文件是每个插件必备的,里面还隐藏着一些插件的信息 值得看看
我用authme为例


name: AuthMe #插件的名字 plugman加载或卸载就是用这个名字的,加载可以用文件名,但卸载必须用这个英文名字
author: Xephi59 #作者
website: http://dev.bukkit.org/bukkit-plugins/authme-reloaded/ #插件的网站,一般可以从这里找到插件的网站
description: AuthMe prevents people, which aren't logged in, from doing stuff like placing blocks, moving, typing commands or seeing the inventory of the current player. #插件的功能描述 可改
main: fr.xephi.authme.AuthMe #主类路径
version: 3.3.6 #版本
softdepend: [Vault, ChestShop, Spout, Multiverse-Core, Notifications, Citizens, CombatTag, Essentials, EssentialsSpawn] #可选用的扩展插件
commands: #指令
    register: #注册指令/register 这个改了报错 甚至无法加载插件
        description: Register an account #说明
        usage: /register password confirmpassword #用法
        aliases: reg #指令别名,输入指令也可以输入别名 这个可以更改 比如把reg改成/r 就可以输入/r来注册了
    unregister:
        description: unregister your account
        usage: /unregister password #这个没有aliases这个选项 所以大家别自作聪明添加个进去 插件坏了别找我
permissions: #插件的权限 在这里基本能找到插件的所有权限 有的插件的权限并不一定都在这
    authme.player.*: #母权限
        description: Gives access to all authme player commands
        default: true #所有人是否有这个权限
        children: #子权限
            authme.register: true
            authme.login: true
            authme.changepassword: true
            authme.logout: true
            authme.unregister: true
            authme.passpartu: true
            authme.l: true
            authme.reg: true
    authme.admin.*:
         description: Gives access to all authme admin commands
         children:
             authme.admin.reload: true
             authme.admin.register: true
             authme.admin.changepassword: true
             authme.admin.unregister: true
             authme.admin.purge: true
             authme.seeOtherAccounts: true
             authme.admin.lastlogin: true
             authme.admin.getemail: true
             authme.admin.chgemail: true
             authme.admin.purgelastpos: true
             authme.admin.switchantibot: true
             authme.bypassantibot: true
             authme.admin.getip: true
    authme.register:
        description: Register an account
        default: true #这个改成OP就默认只有OP可用了 改成false即 默认不给所有人权限 但OP也能获得权限

配置文件，
  1. interact:  # 右键点击
  2.     enable: true   # 是否开启限制
  3.     maxInterval: 110 # 允许的最高右击频率，毫秒为单位，the smaller, the faster
  4. hit:    # 左键点击
  5.     enable: false    # 是否开启限制
  6.     maxInterval: 90    # 允许的最高攻击频率，毫秒为单位，the smaller, the faster
  7. shootArrow:    # 射箭速度
  8.     enable: false    # 是否开启限制
  9.     maxInterval: 190     # 允许的最高射击速度，毫秒为单位，more smaller, more faster

注意：有些小游戏插件有强制玩家超过正常频率射箭的功能，因此可能造成无法正常游戏。此功能默认为关。
另外，shootArrow选项只适用于弓箭，因为据我所知只有弓箭有冷却时间，其他雪球什么的是包含在interact内的。插件使玩家被动发射的雪球、鸡蛋等Projectile不会拦截，因为不会触发右击事件。

  42.     @Override
  43.     public void onEnable() {
  44.         saveDefaultConfig();
  45.         reloadConfig();
  46.         if(getConfig().getBoolean("interact.enable")){
  47.             final int maxInterval = getConfig().getInt("interact.maxInterval", 110);
  48.             Bukkit.getPluginManager().registerEvents(new Listener() {
  49.                 private final HashMap<Integer, Record> interact = new HashMap<>(64, 0.8f);

  51.                 @EventHandler(priority = EventPriority.LOWEST, ignoreCancelled = false)
  52.                 public void onInteract(PlayerInteractEvent event){
  53.                     Action action = event.getAction();
  54.                     if((action == Action.RIGHT_CLICK_BLOCK && !event.isCancelled() &&
  55.                             event.getMaterial() != Material.MINECART && event.getMaterial() != Material.BOAT) ||
  56.                             action == Action.RIGHT_CLICK_AIR)
  57.                         if(computeInterval(interact, event.getPlayer().getUniqueId())
  58.                                 < maxInterval)
  59.                             event.setCancelled(true);
  60.                         else
  61.                             event.setCancelled(false);
  62.                 }
  63.             }, this);
  64.         }

  66.         if(getConfig().getBoolean("hit.enable")){
  67.             final int maxInterval = getConfig().getInt("hit.maxInterval", 90);
  68.             Bukkit.getPluginManager().registerEvents(new Listener() {
  69.                 private final HashMap<Integer, Record> hit = new HashMap<>(64, 0.8f);

  71.                 @EventHandler(priority = EventPriority.LOWEST, ignoreCancelled = true)
  72.                 public void onHit(EntityDamageByEntityEvent event){
  73.                     if(event.getDamager() instanceof HumanEntity)
  74.                         if(computeInterval(hit, event.getDamager().getUniqueId())
  75.                                 < maxInterval)
  76.                             event.setCancelled(true);
  77.                 }
  78.             }, this);
  79.         }
  80.         if(getConfig().getBoolean("shootArrow.enable")){
  81.             final int maxInterval = getConfig().getInt("shootArrow.maxInterval", 190);
  82.             Bukkit.getPluginManager().registerEvents(new Listener() {
  83.                 private final HashMap<Integer, Record> shoot = new HashMap<>(64, 0.8f);

  85.                 @EventHandler(priority = EventPriority.LOWEST, ignoreCancelled = true)
  86.                 public void onShoot(ProjectileLaunchEvent event){
  87.                     Projectile projectile;
  88.                     if((projectile = event.getEntity()) instanceof Arrow &&
  89.                             projectile.getShooter() instanceof HumanEntity)
  90.                         if(computeInterval(shoot, ((Entity) projectile.getShooter()).getUniqueId())
  91.                                 < maxInterval)
  92.                             event.setCancelled(true);
  93.                 }
  94.             }, this);
  95.         }
  96.     }
  97.    
  98.     private static long computeInterval(Map<Integer, Record> map, Object obj){
  99.         int key = obj.hashCode();

  101.         long curTime = System.currentTimeMillis();
  102.         Record last = map.get(key);
  103.         if(last == null)
  104.             map.put(key, last = new Record(curTime));
  105.         
  106.         try{
  107.             return curTime - last.time;
  108.         }finally{
  109.             last.time = curTime;
  110.         }
  111.     }

  113.     public static class Record{
  114.         private long time;

  116.         public Record(long time) {
  117.             this.time = time;
  118.         }

  120.         @Override
  121.         public boolean equals(Object obj) {
  122.             return this == obj ||
  123.                     (obj != null && getClass() != obj.getClass() && this.time == ((Record) obj).time);
  124.         }

  126.         @Override
  127.         public int hashCode() {
  128.             return 31 * 7 + (int) (this.time ^ (this.time >>> 32));
  129.         }

  131.         @Override
  132.         public String toString() {
  133.             return Long.toString(time);
  134.         }
  135.     }
  136. }

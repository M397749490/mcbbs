/chguser help [指令]
# 重置玩家数量到正常值
/chguser reset
# 重载配置
/chguser reload
# 自动增加或减少玩家数量(减用"-"号)(服务器列表)
/chguser autoadd [数量]
# 设置永久在线玩家数(服务器列表)
/chguser setplayers [数量]
# 从服务器隐藏一个玩家
/chguser hide [玩家名称]
# 显示一个隐藏的玩家
/chguser show [玩家名称]
# 创建一个虚假玩家
/chguser fake [玩家名称]
# 移除一个虚假玩家
/chguser remfake [玩家名称]

  2.         new PacketAdapter(this, ConnectionSide.SERVER_SIDE, Packets.Server.CHAT) {
  3.             @Override
  4.             public void onPacketSending(PacketEvent event) {
  5.                 String message = event.getPacket().getStrings().read(0); // Get the message

  7.                 // Check if the message is the permission message
  8.                 if (message.equalsIgnoreCase("You don't have permission")) {
  9.                     // Modify the message
  10.                     event.getPacket().getStrings().write(0, ChatColor.RED + "Unknown Command");
  11.                 }
  12.             }

  15. );

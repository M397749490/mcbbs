作者: 神奈川归尘    时间: 2020-3-1 04:33
  1.   NewPlayer:
  2.     Use: true
  3.     # Setting this to true, residence will be created for free
  4.     # By setting to false, money will be taken from player, if he has them
  5.     Free: true
  6.     # Range from placed chest o both sides. By setting to 5, residence will be 5+5+1 = 11 blocks wide
  7.     Range:
  8.       X: 10
  9.       Y: 10
  10.       Z: 10

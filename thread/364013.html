Player p=Bukkit.getPlayer(a);
int s=998;
Scoreboard score=Bukkit.getScoreboardManager().getNewScoreboard();
Objective obj=p.getScoreboard().getObjective(DisplaySlot.SIDEBAR);
obj=score.registerNewObjective("积分信息", "dummy");
obj.setDisplaySlot(DisplaySlot.SIDEBAR);
obj.setDisplayName("§a积分信息");
p.setScoreboard(score);
Score d=obj.getScore(Bukkit.getOfflinePlayer("积分"));
d.setScore(s);

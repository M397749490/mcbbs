你能不能把config文件里的   更多附魔的配置文件 发一下。

配置文件如下：

    B:Enable=true
    I:Rarity=200
}


tools {
    B:HarvestEnable=true
    I:HarvestID=81
}


weapons {
    B:CleaveEnable=true
    I:CleaveID=74
    B:DefusingEnable=true
    I:DefusingID=73
    B:DisjunctionEnable=true
    I:DisjunctionID=83
    B:DowsingEnable=true
    I:DowsingID=82
    B:ExecutionEnable=true
    I:ExecutionID=88
    B:ExplosiveEnable=true
    I:ExplosiveID=89
    B:FrostEnable=true
    I:FrostID=85
    B:IceAspectEnable=true
    I:IceAspectID=86
    B:LeechEnable=true
    I:LeechID=87
    B:MendingEnable=true
    I:MendingID=70
    B:PoisonEnable=true
    I:PoisonID=71
    B:QuickdrawEnable=true
    I:QuickdrawID=94
    B:SilenceEnable=true
    I:SilenceID=92
    B:SpellbaneEnable=true
    I:SpellbaneID=93
    B:SteadfastEnable=true

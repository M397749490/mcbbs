function damaged(e) {
    var n = e.npc;
    var damage = e.damage;
    var max = n.getMaxHealth();
    var hp = n.getHealth();
    var type = n.getAi().getDoorInteract();
    if (hp - damage < max * 0.4 && type == 2) {
        var w = n.getWorld();
        var p = w.getClosestEntity(n.getPos(), 10, 1);
        if (p != null) {
            n.getAi().setDoorInteract(1)
            n.setPos(p.getPos());
            w.explode(p.getX(), p.getY(), p.getZ(), 2, false, false);
            p.setHealth(p.getHealth() - 2);
            p.message("受到爆炸伤害");
        }
    }
    if (hp - damage < max * 0.1) {
        if (n.isAlive()) {
            var w = n.getWorld();
            w.explode(n.getX(), n.getY(), n.getZ(), 2, false, false);
            n.setHealth(0);
        }
    }
}

function init(e) {
    e.npc.getAi().setDoorInteract(2);
}

效果是NPC的血量低于40%时会瞬移到10格内最近的玩家，爆炸并且对其扣血
p.setHealth()是直接设置血量，所以是固伤，你也可以不要
然后NPC默认与门交互是关闭的，如果触发40%的爆炸以后会把这一项改成开启
这个会跟血量一起当成条件判断
也就是这个技能只会触发一次

低于10%的那个我就当你NPC自爆好了，爆炸以后NPC直接会死亡
不过掉落物可能会被炸没(

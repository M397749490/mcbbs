标题: forge1.12.2 物品打破后紫黑块和物品特性应用不上的问题
方块类
public class blockLoveBlock extends Block {

    public blockLoveBlock() {
        super(Material.GRASS);
        this.setSoundType(SoundType.STONE);
        this.setHardness(2.5F);
        // Forge 的采掘等级机制，幻数 2 代表与钻石矿同级，magic string "pickaxe" 代表镐
        this.setHarvestLevel("pickaxe", 2);
        this.setRegistryName("story", "loveblock");
        this.setUnlocalizedName("loveblock");
        this.setLightLevel(1);
        this.setCreativeTab(CreativeTabsLoader.tabFMLTutor);
    }
}

//红色的特性均应用不上


方块注册类
@Mod.EventBusSubscriber(modid = "story")
public class BlockInitializer{

     static Block loveblock = new blockLoveBlock();
     static Item loveblockitem = new ItemBlock(loveblock).setRegistryName("story", "loveblock");


    @SubscribeEvent
    public static void registerBlock(RegistryEvent.Register<Block> event) {
        // 和物品一样，每一个方块都有唯一一个注册名，不能使用大写字母。
        event.getRegistry().register(loveblock);
    }
    @SubscribeEvent
    public static void registerItem(RegistryEvent.Register<Item> event) {
        // 注意这个 ItemBlock 使用了和它对应的方块一样的注册名。
        // 对于所有有物品形态的方块，其物品的注册名和它自己的注册名需要保持一致。
        event.getRegistry().register(loveblockitem);
    }

    @SubscribeEvent
    public static void onModelReg(ModelRegistryEvent event) {
        ModelLoader.setCustomModelResourceLocation(Item.getItemFromBlock(loveblock), 0, new ModelResourceLocation(loveblockitem.getRegistryName(), "inventory"));
    }
}
打破方块后还是呈现紫黑块，其他状态材质正常。

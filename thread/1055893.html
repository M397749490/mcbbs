参考我的配置修改一下authmebungee
  1. # List of servers which required to be authenticated
  2. authServers:
  3. - login
  4. # Consider every server as an auth server
  5. allServersAreAuthServers: false
  6. commands:
  7.     # Allows or not commands to be performed if user is not logged in
  8.     requireAuth: true
  9.     # List of commands allowed to be perform without being authenticated
  10.     whitelist:
  11.     - /login
  12.     - /register
  13.     - /l
  14.     - /reg
  15.     - /email
  16.     - /captcha
  17. # Allows or not user to talk in chat if he is not logged in
  18. chatRequiresAuth: true
  19. serverSwitch:
  20.     # Kick all players who switch servers without being authenticated (eg. plugin teleport)
  21.     requiresAuth: true
  22.     kickMessage: Authentication required.
  23. # Enable auto-login between servers
  24. autoLogin: true
  25. # If enabled, unlogged users will be sent to the unloggedUserServer server!
  26. sendOnLogout: true
  27. # If sendOnLogout is enabled, unlogged users will be sent to this server!
  28. unloggedUserServer: login

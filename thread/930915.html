  1. private ItemStack testItem; // 測試物品
  2.     private Inventory testInventory;

  4.     @Override
  5.     public void onEnable() {
  6.         FactoryAPI api = FactoryLib.getApi();
  7.         testItem = api.getItemStackFactory() // 獲取物品工廠 (一次一個實例 !
  8.                 .name("&b測試物品") // 名稱
  9.                 .lore("&e第一行", "&b第二行", "&c第三行") // 敘述
  10.                 .material(Material.DIAMOND_AXE) // 類型
  11.                 .durability(20) // 耐久度
  12.                 .build(); // 建造

  14.         ItemStack previousItem = api.getItemStackFactory().material(Material.WOODEN_AXE).name("&a上一頁").build();
  15.         int previousSlot = InventoryFactory.ONE_ROW; // 第二行的格數 = 9

  17.         ItemStack nextItem = api.getItemStackFactory().material(Material.DIAMOND_AXE).name("&b下一頁").build();
  18.         int nextSlot = InventoryFactory.ONE_ROW + InventoryFactory.END; // 第二行最尾的格數 =  9 + 8

  20.         int centerSlot = InventoryFactory.ONE_ROW + InventoryFactory.CENTER; // 第二行中央的格數 = 9 + 4
  21.         ItemStack backItem = api.getItemStackFactory().material(Material.BEDROCK).name("&d返回主頁").build();

  23.         Inventory previousInv = api.getInventoryFactory() // 獲取背包工廠 (一次一個實例
  24.                 .setRow(3) // 設置最大行數
  25.                 .setTitle("&a這是上一頁") // 標題
  26.                 .ring(new ItemStack(Material.APPLE)) // 環狀圍繞物品 (必須有三行以上
  27.                 .addItem(centerSlot, backItem) // 添加物品 (格, 物品)
  28.                 .changePage(centerSlot, () -> testInventory) // 特定格數點擊換頁
  29.                 .build(); // 建造

  31.         Inventory nextInv = api.getInventoryFactory()
  32.                 .setRow(3)
  33.                 .setTitle("&b這是下一頁")
  34.                 .ring(new ItemStack(Material.BREAD))
  35.                 .addItem(centerSlot, backItem)
  36.                 .changePage(centerSlot, () -> testInventory)
  37.                 .build();

  39.         testInventory = api.getInventoryFactory()
  40.                 .setRow(3)

指令(玩起了orm)


  1. @Command -> 绑在Class上
  2. @MainCommand -> 主指令, 绑在方法上
  3. @SubCommand -> 副指令，绑在方法上

  5. 1. 任何被绑方法必须拥有唯一两个参数: CommandSender 和 String[]

  7. 2. 主指令不要绑多于一个

  9. 3. 任何被绑方法返回必须是CommandResult

复制代码
command -> 指令名称 | description -> 指令介绍 | placeholders -> 指令参数 | alias -> 指令别称 | parent -> 父指令名称(给副指令的副指令用)

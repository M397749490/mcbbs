作者: 桃渊林    时间: 2015-3-25 20:02
AuthMe的MySQL配置部分:
  1. DataSource:
  2.     mySQLColumnName: username
  3.     # 表名，保持默认
  4.     mySQLTablename: authme
  5.     # MySQL用户名，请先创建一个
  6.     mySQLUsername: authme
  7.     # 这里设置为mysql来启用MySQL数据库
  8.     backend: mysql
  9.     # 玩家最后登录信息字段，保持默认
  10.     mySQLColumnLastLogin: lastlogin
  11.     # 数据库名，请先创建一个数据库
  12.     mySQLDatabase: authme
  13.     # MySQL端口，保持默认即可
  14.     mySQLPort: '3306'
  15.     # 玩家ip字段，保持默认
  16.     mySQLColumnIp: ip
  17.     # 数据库连接地址，根据需要字体，若把MySQL部署在本地保持默认
  18.     mySQLHost: 127.0.0.1
  19.     # 玩家密码字段，保持默认
  20.     mySQLColumnPassword: password
  21.     # 数据库密码，请修改''里面的
  22.     mySQLPassword: '12345'
  23.     # 缓存数据库查询，建议改为false以保证数据实时性（假如这个服改了密码另一个服没生效）
  24.     caching: false
  25.     # 下面是玩家退出的坐标及世界，保持默认
  26.     mySQLlastlocX: x
  27.     mySQLlastlocY: y
  28.     mySQLlastlocZ: z
  29.     mySQLlastlocWorld: world
  30.     # 玩家邮箱字段，保持默认
  31.     mySQLColumnEmail: email
  32.     # 玩家id字段，保持默认
  33.     mySQLColumnId: id
  34.     # 玩家登录状态字段，保持默认
  35.     mySQLColumnLogged: isLogged
  36.     # 至此，您已完成AuthMe的MySQL配置

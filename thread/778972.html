  1. # 配置文件版本,重要,请勿修改
  2. # 插件升级配置的时候需要用到
  3. version: '1.0'
  4. # 设置消息输出等级
  5. # INFO<WARN<SEVERE<DEBUG<DEVELOP
  6. LogLevel: SEVERE
  7. # 是否记录错误堆栈
  8. LogStackTrace: true
  9. # 插件大部分消息前缀
  10. MsgPrefix: §7[§aBanItem§7]§3
  11. # 是否开启服务器信息收集,只是用于插件使用情况统计
  12. # 收集的信息: 使用版本 服务器ID 服务器IP 服务器端口 服务器版本 Java版本 系统版本
  13. EnableMetrics: true
  14. # 是否启用线程安全检查
  15. # 当前的安全检查有[命令是否异步运行]
  16. IsThreadSafe: true
  17. # 权限节点配置
  18. Permission:
  19.   # 是否启用物品ID权限支持
  20.   # 未启用之前只有BanItem.bypass.use这样是最细的权限
  21.   # 启用之后可以更细分到BanItem.bypass.use.54的权限
  22.   CheckItemId: false
  23.   # 是否启用物品子ID权限支持,必须先启用CheckItemId
  24.   # 启用之后可以更细分到BanItem.bypass.use.54@1的权限
  25.   CheckItemDamage: false
  26. # BanItem插件主配置
  27. BanItem:
  28.   # 是否从NBT中移除设置了没收的物品
  29.   RemoveItemFromNBT: false
  30.   # 是否在方块的NBT中搜索禁用物品(检查inv权限)
  31.   CheckBanItemInBlockNBT: true
  32.   # 当玩家登录的时候,检查登录的玩家背包里是否有禁用的物品(检查inv权限)
  33.   CheckInvWhenPlayerJoin: true
  34.   # 当玩家切换世界的时候,检查登录的玩家背包里是否有禁用的物品(检查inv权限)
  35.   CheckInvWhenChangeWorld: true
  36.   # 是否在玩家切换手中物品的时候检禁用物品(检查hold,inv权限)
  37.   CheckWhenPlayerSwitchItem: true
  38.   # 玩家登陆多少Ticks后检查背包
  39.   # 太小可能会在没收东西时造成客户端崩溃
  40.   CheckInvDelayTicks: 50
  41.   # 是否在玩家放置方块时检查方块周围是否是或者有包含禁用物品的方块(检查inv权限)
  42.   CheckBlockAroundWhenPlace: false
  43.   # 新增禁用物品时,默认的禁用理由
  44.   DefaultReason: 该物品已经被禁用
  45.   # 新增禁用物品时,没收选项的默认值
  46.   DefaultConfiscateOption: true
  47.   # 在进行方块检查时,不没收的地上方块,不支持子id
  48.   # 例如,ban了基岩,并且设置了没收
  49.   # 但是这样会导致在玩家右键基岩时,基岩方块会直接消失
  50.   # 所以只要把基岩方块的名字或者数字id添加到此处即可
  51.   BlocksNotSetToAir:

  62. #     # 什么是自定义物品列表? 例如你把ic2 mod里ban掉的物品整理出来放到了ic2.yml中,那么换周目的时候,只要把ic2.yml放到PreferItemList/,这样新周目ic2的物品就不需要重新去ban了
  63. #     Prefer:

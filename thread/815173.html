修改server.properties 文件中的

motd=A Minecraft Server  //服务器讯息
  1. #Minecraft server properties
  2. #Wed Jan 04 01:33:25 SGT 2017
  3. generator-settings=  //自定义超平坦世界
  4. op-permission-level=4  //OP权限等级 （1-无视重生点 2-可用作弊指令 3-可用管理指令 4-可用所有指令）
  5. allow-nether=true  //是否开启下界
  6. level-name=world  //地图文件夹的使用名字
  7. enable-query=false  //是否允许使用GameSpy4协议的服务器监听器  用于收集服务器信息
  8. allow-flight=false  //是否允许飞行
  9. announce-player-achievements=true  //是否在服务器中显示获得成就
  10. prevent-proxy-connections=false  //
  11. server-port=25565  //服务器端口号
  12. max-world-size=29999984  //世界边界的最大半径
  13. level-type=DEFAULT  //地图生成类型
  14. #DEFAULT - 标准的世界带有丘陵，河谷，海洋等
  15. #FLAT - 一个没有特色的平坦世界，适合用于建设
  16. #LARGEBIOMES - 如同预设世界，但所有生态系都更大（1.3之后可用）
  17. #AMPLIFIED - 如同预设世界，但世界生成高度提高（1.7.2之后可用）
  18. #CUSTOMIZED - 自定义地形。使用方法是在generator-settings贴上代码（1.9之后可用）
  19. enable-rcon=false  //是否允许远程访问服务器控制台
  20. force-gamemode=false  //是否让玩家加入时强制变为默认模式
  21. level-seed=  //自定义/锁定地图种子
  22. server-ip=  //绑定服务器IP
  23. network-compression-threshold=256  //
  24. max-build-height=256  //游戏中能建造的最大高度
  25. spawn-npcs=true  //是否生成村民
  26. white-list=false  //是否开启白名单
  27. spawn-animals=true  //是否生成动物
  28. snooper-enabled=true  //是否允许数据采集 http://snoop.minecraft.net.
  29. hardcore=false  //是否开启极限模式
  30. resource-pack-sha1=  //资源包的sha-1值
  31. online-mode=true  //是否开启正版验证
  32. resource-pack=  //输入URL链接的资源包
  33. pvp=true  //是否允许PVP
  34. difficulty=1  //难度调整（0-和平  1-简单  2-普通  3-困难）
  35. enable-command-block=false  //是否允许命令方块
  36. player-idle-timeout=0  //玩家空闲超时的时间 0为无限制
  37. gamemode=0  //自定义游戏模式 （0-生存  1-创造  2-冒险  3-旁观）
  38. max-players=20  //服务器能容纳的最大玩家数量
  39. spawn-monsters=true  //是否生成怪物
  40. view-distance=10  //设置服务端传送给客户端的数据量 同时也是客户端视野距离的上限
  41. generate-structures=true  //是否生成结构（如：村庄）
  42. motd=A Minecraft Server  //服务器讯息
  43. max-tick-time=60000  //毫秒计数的最大服务器看门狗消息监测周期，超过该周期看门狗将停止服务器程序。一个服务器周期为60.00秒（最大密度0.05）
  44. spawn-protection=16  //出生点保护区半径

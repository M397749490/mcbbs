白嫖了那么久的mcbbs，今天终于轮到自己出一份力了。。
在我还是小白的时候，就总是找不到能够强化附魔等级的插件mod之类的。所以我就特意写了一份能够高度自定义强化附魔等级的脚本。

前置：CustomNPCmod/bukkit服务端
版本：1.7.10/（1.12.2需要自行补充完善脚本）
用法：将代码放进npc钩子里就行，更具体的可以去看npc脚本教程
下面上脚本代码
  1. /*食用准备：可以通过对话钩子设置player.getHeldItem().setTag("itensify","damage")的方法。
  2. 通过拿着手上的物品右键npc来为你选定的强化石物品进行标记。
  3. ps:除"damage"外还有"durability"和"thorns"两种默认类型;
  4.   食用方法：可以把该脚本放在对话钩子；然后手拿你要强化的物品，右键npc即可（背包要有标记了的强化石）
  5.   Time:2020.6.20
  6.   author:Coolla
  7.   */

  9. if(player.getHeldItem()!==null){
  10. var success =100;//基础成功率设置
  11. var maxlvl=200;//最大等级限制
  12. var searchitem=player.getInventory();

  14. var itemid=search(searchitem);
  15. if(itemid!==-1){
  16. var tickets=identify(searchitem[itemid]);//判断强化石类型
  17. if(tickets!==0){
  18. var enchment= Java.type("org.bukkit.enchantments.Enchantment");
  19. switch (tickets){//三种附魔类型
  20. case 1:var ench=enchment.DAMAGE_ALL;break;
  21. case 2:var ench=enchment.DURABILITY;break;
  22. case 3:var ench=enchment.THORNS;break;
  23. };

  25. var bukkit =Java.type("org.bukkit.Bukkit");
  26. var playerName = player.getName();
  27. var playerB=bukkit.getPlayer(playerName);
  28. var hand=playerB.getItemInHand();

  30. success=matchlvl(hand,ench,maxlvl);//检查附魔的等级
  31. if(success>0){
  32. remove(searchitem,itemid);

  34. levelup(hand,ench,success);//强化附魔
  35. }else npc.say("你已经超过附魔等级限制")}
  36. }else npc.say("你没有强化石");
  37. }else npc.say("请把需要强化的东西放在手上");

  39. function search (itemlist){//寻找强化石
  40. for(var i=0;i<36;i++){
  41. if(itemlist[i]!==null){
  42.          if(itemlist[i].hasTag("itensify")==true)return i;
  43. }}return -1}


  46. function identify(matchitem){//鉴别附魔强化石
  47.          var name=matchitem.getTag("itensify");
  48. switch(name){
  49. case "damage":var id =1;break;
  50. case "durability":var id =2;break;
  51. case "thorns":var id =3;break;
  52. };return id
  53. }

  55. function matchlvl(hand,ench,maxlvl){//检查附魔的等级
  56. var lvl=hand.getEnchantmentLevel(ench);
  57. var i=0;
  58. if(lvl>=maxlvl)return 0;
  59. if(lvl>=0)i=100;
  60. if(lvl>10)i= 90;
  61. if(lvl>20)i=80;
  62. if(lvl>50)i=50;
  63. if(lvl>100)i=40;//强化几率
  64. return i;
  65. }

  67. function levelup (item,ench,success){//强化方法
  68. var random=parseInt(Math.random()*100+1);
  69. if(random<success){
  70. var lvl=1+item.getEnchantmentLevel(ench);
  71. item.addUnsafeEnchantment(ench,lvl);
  72. npc.say("&6恭喜你，强化成功");}
  73. else {npc.say("很遗憾，强化失败");}
  74. }

  76. function remove(matchitem,i)//移除强化石方法
  77.          {var num=matchitem[i].getStackSize();
  78.                 if(num==1){
  79. player.getMCEntity().field_71071_by.field_70462_a[i]=null;
  80. player.sendMessage("&6&l★★已消耗强化石*1★★");
  81. }else{
  82.                 num-=1;
  83.                matchitem[i].setStackSize(num);
  84.                player.sendMessage("&6&l★★已消耗强化石*1★★");
  85.                }}

  1. String str = "<r:0_9>".replace("<r:","").replace(">","");
  2. String[] s = str.split("_");
  3. int first = Integer.parseInt(s[0]);
  4. int last = Integer.parseInt(s[1]);
  5. Random random = new Random();
  6. int i = random.nextInt() * (last - first + 1) + first;

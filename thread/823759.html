可以做到，但十分复杂
  1. #以下命令以1.13格式书写
  2. #以下一次性执行
  3. /summon armor_stand ~ ~ ~ {Invulnerable:1b,Invisible:1b,Tag:[{"1"}]}
  4. /summon armor_stand ~ ~ ~ {Invulnerable:1b,Invisible:1b,Tag:[{"2"}]}
  5. /scoreboard objectives add random dummy
  6. /scoreboard players set @e[tag=1] random 1
  7. /scoreboard players set @e[tag=2] random 2

  9. #以下命令高频执行
  10. /execute as @e at @s if score @e[tag=1,tag=2] random matches 1 run data merge entity @e[type=Zombie,tag=!armed] {ArmorItems:{0：{id:"minecraft:diamond_boots"},1:{id:"minecraft:diamond_leggings"},2:{id:"minecraft:diamond_chestplate"},3:{"id:"minecraft:diamond_helmet"}}
  11. /execute as @e at @s if score @e[tag=1,tag=2] random matches 2 run data merge entity @e[type=Zombie,tag=!armed] {ArmorItems:{0:{id:"minecraft:golden_boots"},1:{id:"minecraft:golden_leggings"},2:{id:"minecraft:golden_chestplate"},3:{"id:"minecraft:golden_helmet"}}
  12. /data merge entity @e[type=Zombie,tag=!armed] {HandItems:{0:{id:"minecraft:diamond_sword"}}
  13. /tag @e[type=Zombie] add armed

  15. #效果：僵尸随机穿戴黄金/钻石甲，手持钻石剑
  16. #如有错误，请各位dalao不吝指教

使用DeluxeMenus

    slot: 1 #这是显示的位置，从零开始从左往右从上至下计数
    priority: 2 #可无视
    view_requirement:
      requirements:
        delay:
          type: has permission
          permission: 权限节点
    material: chest
    display_name: '&6点我获取VIP礼包' #只有VIP可以看见！
    update: true
    lore:
    - ''

           # 玩家可以创建最大的子领地数.

           MinEastWest: 3

           MaxNorthSouth: 512
           # (可选) 领地的最小宽度, 从南至北算 (Z).
           MinNorthSouth: 3
           # 领地的最大高度，从高到低起 (Y).
           MaxUpDown: 256
           # (可选) 领地的最小高度，从高到低起(Y).
           MinUpDown: 3
           # 保护该权限组的最矮距离.
           MinHeight: 0
           # 保护该权限组的最高距离.
           MaxHeight: 255
            # 当前区域可以拥有最大的子领地数.
           MaxSubzonesInArea: 3
            # # 子区域领地最大深度.
            # 当一个领地子区域包含另一个子区域时,最大为2.
            # 设置为0将禁用领地子区域.
           SubzoneDepth: 3
           # (可选) 子领地的最大距离, 从东到西算 (X).
           SubzoneMaxEastWest: 256
           # (可选) 子领地的最小距离, 从东到西算 (X).
           SubzoneMinEastWest: 3
           # (可选) 子领地的最大宽度, 从南至北算 (Z).
           SubzoneMaxNorthSouth: 256
           # (可选) 子领地的最小宽度, 从南至北算 (Z).
           SubzoneMinNorthSouth: 3
           # (可选) 子领地的最大高度，从高到低起 (Y).
           SubzoneMaxUpDown: 256
           # (可选) 子领地的最小高度，从高到低起 (Y).
           SubzoneMinUpDown: 3
           # 是否允许传送到领地
           CanTeleport: true
           # 是否运行使用 /res unstuck 指令, 这个指令可以让卡在领地的玩家自己离开
           Unstuck: true
           # 是否使用 /res kick 指令, 这可以把玩家踢出领地.
           Kick: true
           # 是否允许使用'/res select '命令, 如果禁用,那么只能使用工具选取领地.
           SelectCommandAccess: false
           # 领地主人是否允许使用 黑名单/屏蔽列表
           ItemListAccess: true
       # 进入和离开领地时的消息设置.
       Messaging:
           # 玩家是否可以改变领地进出提示.
           CanChange: true
           # 这个文本是当前用户组新建领地后的默认进入信息.
           # 消息留空将禁用信息.
           DefaultEnter: "欢迎 %player 来到 %residence, 领地主人是 %owner."
           # 这个文本是当前用户组新建领地后的默认离开信息.
           # 消息留空将禁用信息.
           DefaultLeave: "你已经离开了 %residence."
       # 有关租赁系统的设置.
       Lease:
          # 可租给他人最大天数..
          MaxDays: 365
          # 当使用 '/res lease renew' 命令时可增加租时多少天.
          RenewIncrement: 0
       #有关租赁系统的设置
       Rent:
           #决定玩家一次可以租多少个领地
           MaxRents: 3
           #决定玩家一次可以设置为暂时性租给他人的领地个数
           MaxRentables: 3
           #决定玩家一次可以设置租地的最大暂时性zu给他人天数
           MaxRentDays: 365
       #有关领地经济系统的设置.
       Economy:
          # 是否可以购买待购的领地
          CanBuy: true
          # 是否可以出售领地
          CanSell: true
          # 购买领地时，是否忽略领地面积/数量的限制
          IgnoreLimits: false
          # 创建一个新领地或添加物理区域时的每个方块大小花费.
          BuyCost: 0.01
          #  移除一个领地时返还每个方块大小价格.
          SellCost: 0
          # 租赁领地时每个方块大小花·费 (如果租赁系统开启).
          RenewCost: 0
       #  Flag检查顺序:
       # 1: Player
       # 2: Group
       # 3: Owner
       # 4: 如果其他未定义,则恢复默认值 (默认为 true).
       # Flag 的权限默认是 false.
       # 这些 flag 设置覆盖总体设置.
       Flags:
          # 特别允许或拒绝这个组权限的Flag.
          Permission:
               #build: true
          # 特指为这个组的成员在领地内是否可以进行建造的Flag.
          Default:
              #build: true
          # 特指领地的创建者在领地内是否可以进行建造的Flag.
          CreatorDefault:
              #build: true
          # 特指其他组的成员在领地内是否可以进行建造的Flag.
          GroupDefault:
               #default: #group name
                   #build: false
          #这些Flag将应用于这个组, 当他们在领地之外时将会被使用.
          #这些Flag将会覆盖上面的Flag, 均为全局选项.
          World:
                Global: #这些Flag将会应用于所有世界.
                     #build: false
                WorldNameHere: #或者你也可以单独设置每个世界的情况.
                     #build: false

  1. <div> #一定需要是斧子才能采集
  2. NeedAxe: false
  3. #必须是有精准采集附魔工具
  4. NeedSilkTouch: true
  5. #需要特定的工具名字才能采集,空字符(两个英文格式的双引号,"")为不需要(可以用颜色字符"&",来达到RPG效果)
  6. #会自动替换%Player%为玩家的名字,如 &f&l%Player%&a的神镐 或 搬箱器
  7. ToolName: ""
  8. #需要按住shift
  9. NeedShift: false
  10. #需要的权限,空字符(两个英文格式的双引号,"")为不需要权限(可以弄成VIP才能使用)
  11. Permission: ""
  12. #箱子物品的名字,会自动替换%Player%为玩家的名字,空字符(两个英文格式的双引号,"")为默认名字(即不更改箱子的名字)
  13. ItemName: "%Player%的箱子"
  14. #在掉落的箱子物品的注释(Lore)显示文字,会自动替换%Player%为玩家的名字,留空白为不显示
  15. ItemLore:
  16.   - "&7这是一个装了物品的箱子"
  17. #在掉落的箱子物品的注释(Lore)中部分显示箱子中的物品
  18. DisplayTheItemsInTheChest: true
  19. #在注释中显示物品的最大数量(1~10)
  20. DiscribtionsAmountLimit: 8
  21. #若超出最大数量,在最后一行显示的文字,会自动替换%Amount%为物品的总数量(占的总格数)
  22. OutOfLimit: "&7等%Amount%个物品"
  23. #若箱子为空箱子,则不显示上述名字和注释
  24. EmptyChestFix: true
  25. #无法在箱子中放置的箱子中放置箱子(允许在箱子中放置箱子，但只能是一层，这是为了防止无限嵌套箱子)
  26. AntiMultiChest: true
  27. #支持陷阱箱
  28. TrappedChest: true
  29. </div>

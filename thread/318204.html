gun文件的内容是：
  1. {
  2.   "itemName" : "【弓的名字】",
  3.   "inventoryIcon" : "【弓的小图标】",
  4.   "dropCollision" : 【这次我搞懂了，是碰撞箱，不懂自己去百度,格式跟前剑的一样】，
  5.   "maxStack" : 【填1】,
  6.   "rarity" : "【品质，我不重复了】",
  7.   "description" : "【介绍，Unicode转化地址照样在下面】",
  8.   "shortdescription" : "【名称】",
  9.   "image" : "【图标】.png:<frame>",
  10.   "imageFrameSelector" : "【填bowImageSelector】",
  11.   "handPosition" : 【手的位置，中括号括起来的有序数对】,
  12.   "firePosition" : 【应该是攻击区域的，格式同handPosition】,
  13.   "level" : 【等级】,
  14.   "fireTime" : 【区域判定时间，跟剑一样，当然要小一些】,
  15.   "twoHanded" : 【双持，科学点一般是true】,

  17.   "fireOnRelease" : 【射出去以后才有区域判定，这里就是弓与枪的区别了，弓的话填true】,
  18.   "walkWhileFiring" : 【走路时也能射（hehe），一般是true】,
  19.   "projectileType" : "【箭的类型，弓一般是woodenarrow】",
  20.   "powerProjectileType" : "【会心一击（就是拉到最满那一瞬间）的类型，默认为chargedarrow（闪一下）（我打飞碟的时候经常会心一击year~）】",
  21.   "powerProjectileTime" : 【会心一击的时间段】,
  22.   "projectile" : {
  23.     "speed" : 【速度，也不知道是什么速度】,

  25.     "power" : 【伤害】,
  26.     "powerCurve" : "【跟speedCurve一样就行了】",        
  27.     "color" : [【R】, 【G】, 【B】]【箭的轨迹的颜色】
  28.   },

  30.   "muzzleEffect" : {
  31.     "fireSound" : [ { "file" : "/sfx/gun/【声音】.wav" } ]

  33. }

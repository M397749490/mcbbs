在插件的exp.myl文件夹中的。named下。按格式加入，怪物名（需要带上颜色代码）@EXP:经验值
作者: 弥猫うた    时间: 2019-2-15 14:29
本帖最后由 Mucrazy 于 2019-2-15 14:31 编辑


exp.yml


CONFIG_VERSION: 1.1
# 版本号,请勿改动
MobLevel:
  Enable: true
  LevelSplitter: ', '
  LevelDifference: 5
# 一些插件会创建有等级的怪物
# 这个设定可以帮你移除低等级怪物掉落经验
# 等级怪物例子: "&2Zombie [&fLevel, 5]"
Named:
  - 'TEST@EXP: 20'
  - 'Test Name@EXP: 50'
  - '&2Custom Name@EXP: 100'
  - '&2Super &4Boss Name@EXP: 150'
  - 'Pig! Level, 10@EXP: 10'
# 设置掉落的经验当怪物被杀死 [Integer]
# MOB NAME : EXP [Integer] ||| 和MythicMobs或其他插件联动的怪物掉落示例
Exp:
  BAT: 5
  BLAZE: 15
  CAVE_SPIDER: 15
  CHICKEN: 3
  COW: 4
  CREEPER: 12
  ENDER_DRAGON: 200
  ENDERMAN: 15
  ENDERMITE: 5
  GHAST: 20
  GIANT: 50
  GUARDIAN: 20
  HORSE: 5
  IRON_GOLEM: 30
  MAGMA_CUBE: 5
  MUSHROM_COW: 30
  OCELOT: 5
  PIG: 5
  PIG_ZOMBIE: 15
  PLAYER: 0
  RABBIT: 3
  SHEEP: 5
  SILVERFISH: 5
  SKELETON: 10
  SLIME: 3
  SNOWMAN: 10
  SPIDER: 8
  SQUID: 5
  VILLAGER: 6
  WITCH: 20
  WITCHER: 200
  WOLF: 13
  ZOMBIE: 8  MOB_TYPE: 8
  DEFAULT: 3
# MOB_TYPE : EXP [Integer]# 原版生物的经验掉落设定
# DEFAULT设定数值将应用到未在上列列表中出现的生物

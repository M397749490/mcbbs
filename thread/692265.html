1。报错先试试你的防作弊插件有误！你看看是不是本版本不对【neverlag1.3.0】，或者是你的登陆插件设置了上线物品不可见易造成这个情况，然后就是肯呢过没有添加地图保存插件！
2.领地插件residence配置文件cong下修改配置文件，添加相应权限组，配置汉化文件如下，具体就像编辑vip权限组一样！
Default: #用户组名称
        # 有关玩家可以定义的信息.
        # 去除下面的'#'可以镜像复制此配置到其他组.
        #Mirror:
         #- 'group1'
         #- 'group2'
       Residence:
           # 玩家是否可以创建领地.这个设置可被 'residence.create' 权限覆盖.
           CanCreate: true
           # 玩家允许拥有最大领地数.
           MaxResidences: 5
           # 一个领地的最大物理区域数目.
           MaxAreasPerResidence: 3
           # 一个领地在东西方向最大距离 (X坐标).
           MaxEastWest: 64
           # 一个领地在东西方向最小距离 (X坐标).
           MinEastWest: 3
           # 一个领地在南北方向最大距离 (Z坐标).
           MaxNorthSouth: 64
           # 一个领地在南北方向最小距离 (Z坐标).
           MinNorthSouth: 3
           # 一个领地在上下方向最大高度 (Y坐标).
           MaxUpDown: 256
           # 一个领地在上下方向最小高度 (Y坐标).
           MinUpDown: 3
           # 允许创建领地的最低高度.
           MinHeight: 0
           # 允许创建领地的最高高度.
           MaxHeight: 255
            # 子区域领地最大深度.
            # 当一个领地子区域包含另一个子区域时,最大深度为2.
            # 设置为0将禁用领地子区域.
           SubzoneDepth: 3
           # 一个子领地在东西方向最大距离 (X坐标).
           SubzoneMaxEastWest: 16
           # 一个子领地在东西方向最小距离 (X坐标).
           SubzoneMinEastWest: 3
           # 一个子领地在南北方向最大距离 (Z坐标).
           SubzoneMaxNorthSouth: 16
           # 一个子领地在南北方向最小距离 (Z坐标).
           SubzoneMinNorthSouth: 3
           # 一个子领地在上下方向最大高度 (Y坐标).
           SubzoneMaxUpDown: 256
           # 一个子领地在上下方向最小高度 (Y坐标).
           SubzoneMinUpDown: 3
           # 是否允许领地传送
           CanTeleport: true
           # 是否允许使用'/res unstuck'命令, 当你卡在领地内时可以传送出去
           Unstuck: true
           # 是否允许使用'/res kick'命令, 去踢出领地内的玩家
           Kick: False
           # 是否允许使用'/res select '命令, 如果禁用,那么只能使用工具选取领地.
           SelectCommandAccess: true
           # 领地主人是否允许使用 黑名单/屏蔽列表 .
           ItemListAccess: true
       # 进入和离开领地时的消息设置.
       Messaging:
          #玩家是否可以改变领地进出提示.
          CanChange: true
          # 这个文本是当前用户组新建领地后的默认进入信息.
          # 消息留空将禁用信息.
          DefaultEnter: '欢迎 %player 来到 %owner 的领地： %residence.'
          # 这个文本是当前用户组新建领地后的默认离开信息.
          # 消息留空将禁用信息.
          DefaultLeave: '你离开了 %residence.'
       # 有关出租系统的设置.
       Lease:
          # 可出租最大天数.
          MaxDays: 16
          # 当使用 '/res lease renew' 命令时可续租多少天.
          RenewIncrement: 14
       # 有关租用系统的设置
       Rent:
           # 决定玩家一次可以租用多少个领地
           MaxRents: 3
           # 决定玩家一次可以设置为出租的领地个数
           MaxRentables: 3
           # 决定玩家一次可以设置租地的最大出租天数
           MaxRentDays: 31
       # 有关领地经济系统的设置.
       Economy:
          # 是否可以购买出售中的领地.
          CanBuy: true
          # 是否可以出售领地.
          CanSell: false
          # 在购买领地时,是否忽略领地限制.
          IgnoreLimits: false
          # 创建一个新领地或添加物理区域时的每个方块大小花费.
          BuyCost: 0.5
          # 移除一个领地时返还每个方块大小价格.
          SellCost: 0.25
          # 出租领地时每个方块大小花费 (如果出租系统开启).
          RenewCost: 0.2
       # Flag检查顺序:
       # 1: Player           玩家
       # 2: Group            用户组
       # 3: Owner            拥有者
       # 4: 如果其他都未定义将使用默认(一般是true).
       # Flag 的权限默认是 false.
       # 这些 flag 设置覆盖总体设置.
       Flags:
          # 特别允许或拒绝这个组权限的Flag.
          Permission:
              #build: true
          # 特指为这个组的成员在领地内是否可以进行建造的Flag.
          Default:
              #build: true
          # 特指领地的创建者在领地内是否可以进行建造的Flag.
          CreatorDefault:
              #build: true
          # 特指其他组的成员在领地内是否可以进行建造的Flag.
          GroupDefault:
              #default: #组名
                  #build: false
          # 这些Flag将应用于这个组, 当他们在领地之外时将会被使用.
          # 这些Flag将会覆盖上面的Flag, 均为全局选项.
          World:
                Global: # 这些Flag将会应用于所有世界.
                    #build: false
                WorldNameHere: # 或者你也可以单独设置每个世界的情况.
                    #build: false
    NextGroup: #组名称, 有 residence.group.nextgroup 权限节点的玩家有权使用这个领地组
       Residence:
           CanCreate: true
           MaxResidences: 4
           MaxAreasPerResidence: 2
           MaxEastWest: 20
           MaxNorthSouth: 20
           MaxUpDown: 20
           MinHeight: 0
           MaxHeight: 255
           SubzoneDepth: 3
           CanTeleport: true
           Unstuck: true
           Kick: true
           SelectCommandAccess: true
           ItemListAccess: true
       Messaging:
           CanChange: true
           DefaultEnter: Welcome %player to %residence, owned by %owner.
           DefaultLeave: Now leaving %residence.
       Lease:
          MaxDays: 16
          RenewIncrement: 14
       Rent:
           MaxRents: 3
           MaxRentables: 3
           MaxRentDays: 31
       Economy:
          CanBuy: true
          CanSell: false
          IgnoreLimits: false
          BuyCost: 0.7
          SellCost: 0.3
          RenewCost: 0.3

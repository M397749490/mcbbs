  59.     @EventHandler
  60.     public void onClick(PlayerInteractEvent e) {
  61.         if (e.getAction() == Action.RIGHT_CLICK_BLOCK || e.getAction() == Action.RIGHT_CLICK_AIR || e.getPlayer().hasPermission("tntthrow.use")) {
  62.             if (e.getPlayer().getItemInHand() != null && e.getPlayer().getItemInHand().getType().equals(Material.TNT)) {
  63.                 e.setCancelled(true);

  65.                 if (e.getPlayer().getItemInHand().getAmount() > 1) {
  66.                     e.getPlayer().getItemInHand().setAmount(e.getPlayer().getItemInHand().getAmount() - 1);
  67.                 } else {
  68.                     e.getPlayer().setItemInHand(null);
  69.                 }

  71.                 TNTPrimed tntPrimed = (TNTPrimed) e.getPlayer().getWorld().spawnEntity(e.getPlayer().getLocation(), EntityType.PRIMED_TNT);
  72.                 tntPrimed.setFuseTicks(fuseTicks);

  74.                 BlockIterator blockIterator = new BlockIterator(e.getPlayer());
  75.                 for (int i = 0; i < blockRadius; i++) {
  76.                     blockIterator.next();
  77.                 }

  79.                 VectorUtils.entityPush(tntPrimed, blockIterator.next().getLocation(), blockVelocity);

  81.                 new BukkitRunnable() {
  82.                     @Override
  83.                     public void run() {
  84.                         if (tntPrimed == null || tntPrimed.isDead() || tntPrimed.getFuseTicks() == 0) {
  85.                             cancel();
  86.                         } else {
  87.                             particleType.display(offsetX, offsetY, offsetZ, speed, amount, tntPrimed.getLocation(), radius);
  88.                         }
  89.                     }
  90.                 }.runTaskTimer(this, 0, 1);
  91.             }
  92.         }
  93.     }

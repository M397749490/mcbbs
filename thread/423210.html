  1. 晴天:
  2.     ExtraPermission: ''
  3.     MenuItem:
  4.     - name:&d&l神の放晴 &6&l[**][**][**]
  5.     - lore:&8控制天气停雨#&8神の力！喝大力！#&c代价:200沫币
  6.     - id:10
  7.     - amount:1
  8.     InventoryLocation: 1
  9.     Message: '&6你使用神力使天空放晴！代价为 200 沫币！'
  10.     PriceType: money
  11.     Price: 200
  12.     Reward:
  13.     - sun TMD sun
  14.     - sun world sun
  15.     RewardType: command

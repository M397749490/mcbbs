  2. <b>SoulClear</b>
  3. <hr class="l">
  4. <font size="3">此插件为Sponge插件，作用是清理掉落物以及怪物，目前可以自定义清理的世界，指定ID物品不清理，指定怪物不清理，掉落物与怪物分别计算时间间隔，触发清理的怪物数量，清理的掉落物可使用GUI界面回购，自定义语言文件。</font>

  28. <div class="blockcode"><blockquote>#清理地上掉落物
  29. clearItem:
  30.   #清理间隔  单位：分
  31.   interval: 10
  32.   #运行的世界
  33.   #all 代表所有世界
  34.   worlds:
  35.     - "all"
  36.   #清理黑名单
  37.   #前缀
  38.   #[all]  代表ID全称
  39.   #[contains]  代表所有包含的ID
  40.   blacklist:
  41.     - "[all]minecraft:sandstone"
  42.     - "[all]minecraft:stone"
  43.     - "[contains]iron"


  46. #清理敌对生物
  47. clearHostile:
  48.   #检测间隔  单位：分
  49.   interval: 8
  50.   #需要超过这个数量才进行清除
  51.   maxLimit: 200
  52.   #运行的世界
  53.   #all 代表所有世界
  54.   worlds:
  55.     - "all"
  56.   #清理黑名单
  57.   #[all]  代表ID全称
  58.   #[contains]  代表所有包含的ID
  59.   blacklist:
  60.     - "[all]minecraft:witch"
  61.     - "[all]minecraft:skeleton"
  62.     - "[contains]creeper"

  64. #清理高频红石
  65. clearRedstone:
  66.   #检测间隔  单位：秒
  67.   interval: 20
  68.   #每次间隔内同红石刷新次数
  69.   maxChange: 12

  71. #回购垃圾
  72. repo:
  73.   #每项物品所需价格
  74.   price: 100
  75.   #回购界面名称
  76.   title: "&9&l回购垃圾"
  77.   #回购物品Lore最下方的信息   \n表示换行
  78.   tailLore:
  79.     - ""
  80.     - ""
  81.     - "  &a回购价格：&6%price%"

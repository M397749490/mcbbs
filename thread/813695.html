改成这样，会概率刷黑曜石，并且岩浆从任何方向入水都正常刷矿。
该文件路径：.minecraft\config\flowstone.cfg

  1. # Configuration file

  3. general {

  5.     cobblestone {
  6.         I:"minecraft:cobblestone"=2
  7.         I:"minecraft:diamond_ore"=2
  8.         I:"minecraft:gold_ore"=4
  9.         I:"minecraft:iron_ore"=64
  10.         I:"minecraft:lapis_ore"=2
  11.         I:"minecraft:obsidian"=2
  12.         I:"minecraft:redstone_ore"=16
  13.         I:"minecraft:stone"=200
  14.     }

  16.     stone {
  17.         I:"minecraft:diamond_ore"=2
  18.         I:"minecraft:gold_ore"=4

/give @p minecraft:oak_sign{BlockEntityTag:{Text1:'["",{"text":"Mi","color":"dark_green"},{"text":"ne","color":"dark_blue"},{"text":"cr","color":"yellow"},{"text":"af","color":"light_purple"},{"text":"t "},{"text":"To","color":"dark_aqua"},{"text":"ol"},{"text":"s","color":"gray"}]'},display:{Name:'{"text":"Custom Sign"}'}}

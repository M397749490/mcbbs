//变身技能
function damaged(event) {
    var health = event.npc.health;
    //这里是判断npc的生命值，我这里写的是如果生命值小于一半的时候触发
    if (health < event.npc.getMaxHealth() * 0.5) {
        event.npc.tempdata.put("nbt", true);
        //这里写变身后的提示语
        event.source.message("[提示]你激怒我了!!!");
        //这里写变身后的最大生命值
        event.npc.setMaxHealth(500);
        event.npc.health = event.npc.getMaxHealth();
        //这里写变身后的攻击伤害
        event.npc.setStrength?(10);
        //这里写变身后的体积
        event.npc.setSize(8);
    }
}

function targetLost(event) {
    if (event.npc.tempdata.has("nbt")) {
        //丢失目标后设置的生命,攻击,体积，依次往下
        event.npc.setMaxHealth(200);
        event.npc.setStrength?(20);
        event.npc.setSize(5);
    }
}

function died(event) {
    //死亡后设置的生命,攻击,体积，依次往下
    event.npc.setMaxHealth(200);
    event.npc.setStrength?(20);
    event.npc.setSize(5);
}

AntiAchievement - 成就拦截

很轻量的一个插件
这个插件可以完整的拦截成就
而服务器端自带的成就拦截只能拦截消息而不能拦截成就

指令:
/ach - 开/关此插件

配置:
enable: true #开/关

下载:
MoecraftFS
百度云

源码:
  1. public class AntiAchievement extends JavaPlugin implements Listener{
  2.     boolean enable;
  3.     public void onEnable(){
  4.         getServer().getPluginManager().registerEvents(this, this);
  5.         getLogger().info("[AntiAchievement]加载完成,插件作者:kilpikonna");
  6.         getCommand("ach").setExecutor(this);
  7.         getConfig().options().copyDefaults(true);
  8.         saveConfig();
  9.         this.enable = getConfig().getBoolean("enable");
  10.     }
  11.     public boolean onCommand(CommandSender sender, Command cmd, String label, String[] args){
  12.         Player player = (Player)sender;
  13.         if(player.hasPermission("ke.ach.toggle")){
  14.             if(cmd.getName().equalsIgnoreCase("ach")){
  15.                 this.enable = !this.enable;
  16.                 player.sendMessage(ChatColor.GREEN + "成功调整为" + this.enable);
  17.                 getConfig().addDefault("enable",this.enable);
  18.                 saveConfig();
  19.             }
  20.         }
  21.         return true;
  22.     }
  23.     @EventHandler
  24.     public void antiAchievement(PlayerAchievementAwardedEvent e){
  25.         if(this.enable){
  26.             e.setCancelled(true);   
  27.         }
  28.     }
  29. }

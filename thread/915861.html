  281.     private void checkUpdate() {
  282.         try {
  283.             final String u = "https://api.github.com/repositories/121261530/releases";
  284.             HttpsURLConnection con = (HttpsURLConnection) new URL(u)
  285.                     .openConnection();
  286.             con.setRequestMethod("GET");
  287.             con.getResponseCode();
  288.             try (InputStreamReader reader = new InputStreamReader(con.getInputStream(), Charsets.UTF_8)) {
  289.                 JsonObject json = new JsonParser().parse(reader).getAsJsonArray().get(0).getAsJsonObject();
  290.                 String v = json.get("tag_name").getAsString()
  291.                         .replace("version", "")
  292.                         .replace("build", "");
  293.                 String preKey = "pre";
  294.                 if (!v.contains(preKey)) {
  295.                     int c = new ComparableVersion(v).compareTo(new ComparableVersion(VERSION));
  296.                     if (c > 0) {
  297.                         logger.info("found a latest version:" + v + ".Your version now:" + VERSION);
  298.                     } else if (c < 0) {
  299.                         logger.info("the latest version in github.com:" + v + "[Your version:" + VERSION + "]");
  300.                     }
  301.                 }
  302.             }
  303.         } catch (Throwable e) {
  304.             logger.info("check for updating failed");
  305.             logger.debug("check update error", e);
  306.         }
  307.     }

package seesaw.arenam ;

import java.io.File;            
import java.io.IOException;
import org.bukkit.Material;
import org.bukkit.command.Command;
import org.bukkit.command.CommandSender;
import org.bukkit.configuration.file.FileConfiguration;
import org.bukkit.configuration.file.YamlConfiguration;
import org.bukkit.entity.Player;
import org.bukkit.event.EventHandler;
import org.bukkit.event.Listener;
import org.bukkit.event.block.BlockBreakEvent;
import org.bukkit.inventory.ItemStack;
import org.bukkit.inventory.meta.ItemMeta;
import org.bukkit.plugin.java.JavaPlugin;
import org.bukkit.event.player.PlayerInteractEvent;



public class ArenaM extends JavaPlugin implements Listener
{
private final File customArenaConfig = new File(getDataFolder(),"Arena.yml");
public FileConfiguration AConfig = YamlConfiguration.loadConfiguration(customArenaConfig);
private final File customCacheConfig = new File(getDataFolder(),"Cache.yml");
public FileConfiguration CacheC = YamlConfiguration.loadConfiguration(customCacheConfig);
String StickName = "§2 [设置棒棒]";
//----------------------------------------------------------------------------//

    @Override
public void onEnable()
{   
    DataFolder();
    SetupSooConfig();
    SetupCacheConfig();
    getServer().getPluginManager().registerEvents(this,this);      
    getLogger().info("[ArenaM] 插件已启用");
}

    @Override
public void onDisable()
{
    saveArenaConfig();
    saveCacheConfig();
    getLogger().info("[ArenaM] 插件已停用");
}     

@Override
public boolean onCommand(CommandSender sender, Command cmd, String label, String[] args )
{
if(cmd.getName().equalsIgnoreCase("am") )
{
  if(args.length == 0)
  {}
  if(args.length == 1)
  {
      if (sender instanceof Player) {} else
      {
          Player p = (Player)sender;
          if(args[0].equalsIgnoreCase("getstick"))
            {
            getStick(sender);
            p.sendMessage("已给予你[设置棒棒]");
            }     
      }  
  }  
return true;
}
return false;
}        

public void getStick(CommandSender sender)
{

ItemStack s = new ItemStack(Material.STICK ,1) ;
ItemMeta si = s.getItemMeta();
Player gp = (Player)sender;
si.setDisplayName(StickName);
s.setItemMeta(si);
gp.getInventory().addItem(s);

}

@EventHandler
public void IBB(BlockBreakEvent event)   
{
Player kp = event.getPlayer();
org.bukkit.Material type = kp.getItemInHand().getType();
if(kp.getItemInHand().getItemMeta().getDisplayName().equalsIgnoreCase(StickName) )
{  if (type == org.bukkit.Material.STICK)
     { event.setCancelled(true); }
}
}

@EventHandler
public void PIE(PlayerInteractEvent event)  
{
org.bukkit.event.block.Action TickType = event.getAction();
Player kp = event.getPlayer();
  org.bukkit.Material type = kp.getItemInHand().getType();
if(kp.hasPermission("set") && type == org.bukkit.Material.STICK)
  {
     if(kp.getItemInHand().getItemMeta().getDisplayName().equalsIgnoreCase(StickName) )
     {
         if(TickType == org.bukkit.event.block.Action.LEFT_CLICK_BLOCK)
         {
             int lx = event.getClickedBlock().getLocation().getBlockX();
             int lz = event.getClickedBlock().getLocation().getBlockZ();
             getCacheConfig().set("TicktLocation.Frist.x",lx);
             getCacheConfig().set("TicktLocation.Frist.z",lz);
             saveCacheConfig();
             kp.sendMessage(" ");
             kp.sendMessage("§1 你已设置第一个点");
             kp.sendMessage("§1 X坐标:" + "§2 " +lx);
             kp.sendMessage("§1 Z坐标:" + "§2 " +lz);
             kp.sendMessage(" ");
         }
         if(TickType == org.bukkit.event.block.Action.RIGHT_CLICK_BLOCK)
         {
             int rx = event.getClickedBlock().getLocation().getBlockX();
             int rz = event.getClickedBlock().getLocation().getBlockZ();
             getCacheConfig().set("TicktLocation.Last.x",rx);
             getCacheConfig().set("TicktLocation.Last.z",rz);
             saveCacheConfig();
             kp.sendMessage(" ");
             kp.sendMessage("§1 你已设置第二个点");
             kp.sendMessage("§1 X坐标:" + "§2 " +rx);
             kp.sendMessage("§1 Z坐标:" + "§2 " +rz);
             kp.sendMessage(" ");
         }
     }
}
}

public YamlConfiguration SetupArenaConfig()
{
                if (!(customArenaConfig.exists()))  //假如文件不存在
                {
                 try   //尝试保存文件
                {
                   customArenaConfig.createNewFile() ;
                  }
                 catch(IOException none)   //捕捉异常(保存不成功)
                   {
                      getLogger().info("文件无法保存");
                   }
                }
                return YamlConfiguration.loadConfiguration(customArenaConfig);

}
public void saveArenaConfig()
{
    if (!(customArenaConfig.exists()))   //假如文件不存在
     { SetupArenaConfig(); }
    else
     {
        try
        {
            getArenaConfig().save(customArenaConfig);
        } catch (IOException none)       //捕捉异常(保存不成功)
        {
          getLogger().info("无法保存文件");
        }
     }
}
public FileConfiguration getArenaConfig()
    {
     return AConfig;
    }


public YamlConfiguration SetupCacheConfig()
{
                if (!(customCacheConfig.exists()))  //假如文件不存在
                {
                 try   //尝试保存文件
                {
                   customCacheConfig.createNewFile() ;
                  }
                 catch(IOException none)   //捕捉异常(保存不成功)
                   {
                      getLogger().info("文件无法保存");
                   }
                }
                return YamlConfiguration.loadConfiguration(customCacheConfig);

}
public void saveCacheConfig()
{
    if (!(customCacheConfig.exists()))   //假如文件不存在
     { SetupCacheConfig(); }
    else
     {
        try
        {
            getArenaConfig().save(customCacheConfig);
        } catch (IOException none) {
          getLogger().info("无法保存文件");
        }
     }
}
public FileConfiguration getCacheConfig()
    {
     return  CacheC;
    }

public void DataFolder()  
    {
      if(!getDataFolder().exists())   // 如果没有工作目录
      { getDataFolder().mkdir(); }    // 创建一个新的(与插件同名)
    }
}

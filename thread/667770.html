public class ParticleIcesmoke extends Particle{
        //两种写法都可以
        //public static final ResourceLocation icon = new ResourceLocation(Name.MOD_ID,"textures/particle/icesmoke.png");
        public static final ResourceLocation icon = new ResourceLocation("soa:particle/icesmoke");


        public ParticleIcesmoke(World worldIn, double posXIn, double posYIn, double posZIn, double velocityX, double velocityY, double velocityZ) {
                super(worldIn, posXIn, posYIn, posZIn);
                motionX = velocityX;             //x轴向量
                motionY = velocityY;             //y轴向量
                motionZ = velocityZ;             //z轴向量
                particleMaxAge = 10;             //存活时间
                particleAlpha = 1f;                  //透明度1为完全不透明,0为完全透明,可输入中间值
                particleScale = 1f;                  //显示大小
                particleGravity = -0.001f;       //重力影响,等一下在更新位置的时候y轴坐标可以用到这个参数来实现重力效果
                TextureAtlasSprite sprite = Minecraft.getMinecraft().getTextureMapBlocks().getAtlasSprite(icon.toString());
                setParticleTexture(sprite);      //绑定贴图
        }

}

他是protected,所以我们没办法通过new一个TextureAtlasSprite来传入值.
Minecraft.getMinecraft().getTextureMapBlocks().getAtlasSprite(icon.toString());
的意思是去mc的贴图素材库里寻找一个贴图而这个回传值就是一个 TextureAtlasSprit
但是我们的路径并不是指向mc的贴图素材库,而是自己mod的贴图,呢则么办呢
所以现在我们就要把自己的这个贴图路径注册的mc贴图库内

然后是 onUpdate()
    @Override
    public void onUpdate(){
      prevPosX = posX;
      prevPosY = posY;
      prevPosZ = posZ;
      particleScale -= particleScale/particleMaxAge; //让粒子大小随着时间慢慢缩小(不可取的逻辑写法)
      particleAlpha -= particleAlpha/particleMaxAge;//让粒子透明度随着时间慢慢增加(不可取的逻辑写法)
      motionY -= particleGravity;//通过设置的重力参数让y向量不断向下增加
      moveEntity(motionX, motionY, motionZ);


      if (isCollided) {//碰撞了就消失
        this.setExpired();
      }


      if (this.particleMaxAge-- <= 0) {//年寿到了就消失
        this.setExpired();
      }
    }


这是一个客户端单边方法
至于则么测试的其实只要在方法里写一句
system.out.println("yoooooooooooo")
然后看是Client呼出的还是Server呼出的还是两个都呼出了
这也是个每tick执行的方法(或者是每帧,懒得测试了...反正就是不断刷新的方法)
这样写算式稍微简化了父类的方法,把重力加速度去掉了
并且加了2行新代码
让粒子随着时间而改变自己的属性
然而这是一个不可取的写法,那时我脑抽写出来的
如果要让让粒子大小随着时间慢慢缩小
应该这样的写
int scale;
scale =  particleScale*(particleAge/particleMaxAge)
这样scale才会随着时间慢慢变小直到为0,而且时间和大小是等比例缩放的
不过由于刚刚好我设定的参数用这个垃圾逻辑不会出错,而且能
出现非常好的效果,所以我就没改了

接下来最后一个是重点
        @Override
        public void renderParticle(VertexBuffer vertexBuffer, Entity entity, float partialTick,float edgeLRdirectionX, float edgeUDdirectionY, float edgeLRdirectionZ,float edgeUDdirectionX, float edgeUDdirectionZ){
            //client only
            //下面4句通过获取绑定贴图的大小,定义贴图最左上角的点和最右下角的点
            double minU = particleTexture.getMinU();         
            double maxU = particleTexture.getMaxU();
            double minV = particleTexture.getMinV();
            double maxV = particleTexture.getMaxV();


           //和父类方法一样,通过partialTick和interpPos补足在两tick内的多帧显示坐标变化量
            double scale = 0.1F * particleScale;
            final double scaleLR = scale;
            final double scaleUD = scale;
            double x = prevPosX + (posX - prevPosX) * partialTick - interpPosX;
            double y = prevPosY + (posY - prevPosY) * partialTick - interpPosY;
            double z = prevPosZ + (posZ - prevPosZ) * partialTick - interpPosZ;


          //和父类方法一样,关于亮度的一些设定
            int combinedBrightness = getBrightnessForRender(partialTick);
            int skyLightTimes16 = combinedBrightness >> 16 & 65535;
            int blockLightTimes16 = combinedBrightness & 65535;


            //绑定4个点(在之前的冰弹render里就出现过这个方法了)
            //pos的三个参数,通过输入的参数来让渲染粒子图片永远面对玩家
            //tex的两个参数,绑定贴图点位
            //color的前3个参数,可以设置贴图颜色的RBG偏移,最后一个为设置透明度
            //lightmap用来设置亮度
            vertexBuffer.pos(
                        x - edgeLRdirectionX * scaleLR - edgeUDdirectionX * scaleUD,
                    y - edgeUDdirectionY * scaleUD,
                    z - edgeLRdirectionZ * scaleLR - edgeUDdirectionZ * scaleUD)
                    .tex(maxU, maxV)
                    .color(particleRed, particleGreen, particleBlue, particleAlpha)
                    .lightmap(skyLightTimes16, blockLightTimes16)
                    .endVertex();
            vertexBuffer.pos(
                        x - edgeLRdirectionX * scaleLR + edgeUDdirectionX * scaleUD,
                    y + edgeUDdirectionY * scaleUD,
                    z - edgeLRdirectionZ * scaleLR + edgeUDdirectionZ * scaleUD)
                    .tex(maxU, minV)
                    .color(particleRed, particleGreen, particleBlue, particleAlpha)
                    .lightmap(skyLightTimes16, blockLightTimes16)
                    .endVertex();
            vertexBuffer.pos(
                    x + edgeLRdirectionX * scaleLR + edgeUDdirectionX * scaleUD,
                    y + edgeUDdirectionY * scaleUD,
                    z + edgeLRdirectionZ * scaleLR + edgeUDdirectionZ * scaleUD)
                    .tex(minU, minV)
                    .color(particleRed, particleGreen, particleBlue, particleAlpha)
                    .lightmap(skyLightTimes16, blockLightTimes16)
                    .endVertex();
            vertexBuffer.pos(
                        x + edgeLRdirectionX * scaleLR - edgeUDdirectionX * scaleUD,
                    y - edgeUDdirectionY * scaleUD,
                    z + edgeLRdirectionZ * scaleLR - edgeUDdirectionZ * scaleUD)
                    .tex(minU, maxV)
                    .color(particleRed, particleGreen, particleBlue, particleAlpha)
                    .lightmap(skyLightTimes16, blockLightTimes16)
                    .endVertex();
        }

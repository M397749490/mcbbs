下面这串代码放在Init接口中
  1. npc.clearTempData();
  2. npc.setTempData('backHome-count',{});
  3. npc.setTempData('backHome-cooldown',{});
复制代码


下面这串代码放在Update接口中
  1. // 冷却时长毫秒
  2. var cd = 5000;
  3. // 传送到的坐标
  4. var pos = '~ ~3 ~';

  6. players = world.getAllServerPlayers();
  7. for(var i in players){
  8.     var player = world.getPlayer(players[i].name);
  9.     if(player != null && player.isSneaking() && player.getHeldItem() != null){
  10.         var held = player.getHeldItem().getDisplayName();
  11.         if(held.indexOf('回城') > -1){
  12.             if(player.name in npc.getTempData('backHome-count')){
  13.                 var now = npc.getTempData('backHome-count')[player.name] += 0.5;
  14.                 if(now > 2.5){
  15.                     var time = new Date().getTime();
  16.                     delete npc.getTempData('backHome-count')[player.name];
  17.                     if(player.name in npc.getTempData('backHome-cooldown') && time < npc.getTempData('backHome-cooldown')[player.name]){
  18.                         player.sendMessage('回城冷却中···');
  19.                     }else{
  20.                         var cmd = 'tp {name} {pos}';
  21.                         npc.executeCommand(cmd.replace('{name}',player.name).replace('{pos}',pos));
  22.                         npc.getTempData('backHome-cooldown')[player.name] = time + cd;
  23.                         cooldown(player);
  24.                     }
  25.                 }
  26.             }else{
  27.                 npc.getTempData('backHome-count')[player.name] = 0.5;
  28.             }
  29.         }
  30.     }else if(players[i].name in npc.getTempData('backHome-count')){
  31.         delete npc.getTempData('backHome-count')[players[i].name];
  32.     }
  33. }

  35. function cooldown(player){
  36.     var Thread = Java.type("java.lang.Thread");
  37.     var Runnable = Java.extend(Java.type("java.lang.Runnable"),{
  38.         run: function(){
  39.             try{
  40.                 Thread.sleep(cd);
  41.                 player.sendMessage('冷却完毕');
  42.                 delete npc.getTempData('backHome-cooldown')[player.name];
  43.             }catch(e){}
  44.         }
  45.     });
  46.     new Thread(new Runnable()).start();
  47. }
复制代码


以上代码放入对应的接口中，请让npc处于一直被加载的状态。
手持带有"回城"关键词的物品潜行3秒会执行命令将玩家传送到指定的坐标。

  37.             HttpURLConnection conn = (HttpURLConnection) downloadUrl   
  38.                     .openConnection();

  40.             conn.setRequestMethod("GET");

  42.             conn.setConnectTimeout(5 * 1000);

  44.             String filename = parseFilename(path);   
  45.             /* 获取下载文件的总大小 */   
  46.             int dataLen = conn.getContentLength();   
  47.             if (dataLen < 0) {   
  48.                 System.out.println("获取数据失败");

  50.             }

  53.             RandomAccessFile accessFile = new RandomAccessFile(localFile, "rwd");   
  54.             /* 这时候，其实本地目录下，已经创建好了一个大小为下载文件的总大小的文件 */   
  55.             accessFile.setLength(dataLen);   
  56.             accessFile.close();   
  57.             /* 计算每条线程要下载的数据大小 */

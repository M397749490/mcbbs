#include <iostream>

using namespace std;

int main()
{
        //控制e的数量的变量
        unsigned int e{ 0 };

        for(;;)
        {
                cout << "自动刷e器1.0" << endl << endl;

                cout << "请输入你要多少个e（最高4294967295个e，建议913500个e）" << endl;

                e = 0;
                cin >> e;

                if (cin.fail())
                {
                        system("cls");

                        cout << "输入错误，应该输入一个小于或者等于4294967295的整数" << endl;

                        system("pause");
                        system("cls");

                        return 0;
                }
                else
                {
                        system("cls");

                        for (unsigned int a{ 0 }; a < e; a += 1)
                        {
                                cout << "e";
                        }
                        cout << endl;
                        system("pause");

                        system("cls");
                }
        }
        return 0;
}

  1. tag @s add self
  2. execute as @e[tag=doll] if score @s id = @a[tag=self,limit=1] id run tag @a[tag=self,limit=1] add find
  3. execute if entity @s[tag=!find] run xxx
  4. tag @s remove find
  5. tag @s remove self

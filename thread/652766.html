我会用一个随机礼物的插件来举例子。

  1. <?php

  3. namespace LDX\RandomItem;

  5. use pocketmine\command\Command;
  6. use pocketmine\command\CommandSender;
  7. use pocketmine\Player;
  8. use pocketmine\plugin\PluginBase;
  9. use pocketmine\Server;
  10. use pocketmine\utils\TextFormat;
  11. use pocketmine\item\Item;

  13. class Main extends PluginBase {

  15.   public function onEnable() {
  16.     $this->saveDefaultConfig();
  17.     $c = $this->getConfig()->getAll();
  18.     $t = $c["Interval"] * 1200;
  19.     $num = 0;
  20.     foreach ($c["Items"] as $i) {
  21.       $r = explode(":",$i);
  22.       $this->itemdata[$num] = array("id" => $r[0],"meta" => $r[1],"amount" => $r[2]);
  23.       $num++;
  24.     }
  25.     $this->getServer()->getScheduler()->scheduleRepeatingTask(new Gift($this),$t);
  26.   }

  28.   public function onCommand(CommandSender $issuer,Command $cmd,$label,array $args) {
  29.     if((strtolower($cmd->getName()) == "gift") && isset($args[0])) {
  30.       if($this->getServer()->getPlayer($args[0]) instanceof Player) {
  31.         $d = $this->generateData();
  32.         $p = $this->getServer()->getPlayer($args[0]);
  33.         $this->give($p,$d);
  34.         $p->sendMessage("Random item given! (" . $data["id"] . ":" . $data["meta"] . ")");
  35.       } else {
  36.         $issuer->sendMessage("Player not connected.");
  37.       }
  38.       return true;
  39.     } else if((strtolower($cmd->getName()) == "gift") && !(isset($args[0]))) {
  40.       $this->giveAll();
  41.       return true;
  42.     } else {
  43.       return false;
  44.     }
  45.   }

  47.   public function give($p,$data) {
  48.     if($p instanceof Player && ($p->hasPermission("randomitem") || $p->hasPermission("randomitem.receive"))) {
  49.       $item = new Item($data["id"],$data["meta"],$data["amount"]);
  50.       $p->getInventory()->addItem($item);
  51.     }
  52.   }

  54.   public function giveAll() {
  55.     $data = $this->generateData();
  56.     $this->broadcast("Random item given! (" . $data["id"] . ":" . $data["meta"] . ")");
  57.     foreach($this->getServer()->getOnlinePlayers() as $p) {
  58.       $this->give($p,$data);
  59.     }
  60.   }

  62.   public function broadcast($m) {
  63.     foreach($this->getServer()->getOnlinePlayers() as $p) {
  64.       $p->sendMessage($m);
  65.     }
  66.     $this->getLogger()->info(TextFormat::YELLOW . $m);
  67.   }

  69.   public function generateData() {
  70.     return $this->itemdata[mt_rand(0,(count($this->itemdata) - 1))];
  71.   }

  73. }

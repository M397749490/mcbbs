@EventHandler
public void PutSpawnBox(BlockPlaceEvent event){
    Player player=event.getPlayer();
    //获取被放置的方块
    Block block=event.getBlockPlaced();
    //获取玩家放置的方块的物品形式
    ItemStack itemStack=event.getItemInHand();
    //如果是刷怪笼类型
    if(itemStack.getType()==Material.SPAWNER){
        //转换成刷怪笼类对象
        CreatureSpawner creatureSpawner=(CreatureSpawner) block.getState();
        //获取物品的Meta
        ItemMeta itemMeta=itemStack.getItemMeta();
        //查看预先设定的刷怪类型
        assert itemMeta != null;
        //获取lore(刷怪类型存在里面)
        ArrayList<String> lore= (ArrayList<String>) itemMeta.getLore();
        if(lore != null){
            //通过正则得出刷怪类型的名字
            player.sendMessage(lore.toString());
            String select="(^.+?) 刷怪笼";
            Pattern pattern=Pattern.compile(select);
            for(String i:lore){
                Matcher matcher=pattern.matcher(i);
                if(matcher.find()){
                    //检测到的预设刷怪类型
                    EntityType entityType=EntityType.valueOf(matcher.group(1));
                    //修改刷怪笼的刷怪类型
                    creatureSpawner.setSpawnedType(entityType);
                    
                }
            }
        }
    }
}//不知道为啥没起作用,求大佬帮忙看看

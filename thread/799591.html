SignItem


介绍
      SignItem是一款适用于职业类型的服务器模式
      能够将玩家使用的不符合职业的武器自动删除或丢弃（职业，实际上是判断权限而已）

展示
      


配置
  1. all:
  2. # 1、2是ID 命令需要，记住了
  3.   '1':
  4.     # Lore
  5.     # 推荐使用隐藏 如&a&1
  6.     sign: '&b测试第一个'
  7.     # 如果为true 则删除
  8.     remove: true
  9.     # 如果 remove为false
  10.     # 并且 drop为true 则丢弃
  11.     drop: false
  12.     # 如果 remove为false
  13.     # 如果 drop为false
  14.     # 并且 user为true 则物品切换到其它槽位
  15.     use: false
  16.     # 文字消息
  17.     message:
  18.       # 是否启用
  19.       enable: true
  20.       # 文本
  21.       text: '&a不符合条件使用该物品'
  22.     # 弹幕 需要安装ProtocolLib插件
  23.     title:
  24.       # 是否启用
  25.       enable: true
  26.       # 主弹幕
  27.       mainText: '&f无法使用'
  28.       # 小弹幕
  29.       subText: '&c不符合条件使用'
  30.     # 如果拥有权限，则不受限制
  31.     permission: test.one.item
  32.   '2':
  33.     sign: '测试第二个'
  34.     remove: false
  35.     drop: true
  36.     use: false
  37.     message:
  38.       enable: true
  39.       text: '&a不符合条件使用该物品'
  40.     title:
  41.       enable: false
  42.       mainText: '&f无法使用'
  43.       subText: '&c不符合条件使用'
  44.     permission: haha.lala

  1. //迷宫大小

  6. //tmp = 二维数组(0是空气,1是墙)
  7. //StartX = 起点坐标X

  10. //EndZ = 终点坐标Z
  11. function A_Star(e, tmp, StartX, StartZ, EndX, EndZ) {
  12. //寻路计时器(可省略)
  13. var test = new Date().getTime();
  14. var p = e.player;
  15. var openList = [], //开启列表
  16.   closeList = [], //关闭列表
  17.   result = [], //结果数组
  18.   result_index; //结果数组在开启列表中的序号
  19. openList.push({
  20.   x: StartX,
  21.   y: StartZ,
  22.   G: 0
  23. });
  24. do {
  25.   //返回最后一个元素
  26.   var currentPoint = openList.pop();
  27.   closeList.push(currentPoint);
  28.   var surroundPoint = SurroundPoint(currentPoint);
  29.   for (var i in surroundPoint) {
  30.    var item = surroundPoint[i]; //获得周围的四个点
  31.    if (
  32.     item.x >= 0 && //判断是否在地图上
  33.     item.y >= 0 &&
  34.     item.x < dungeon_setting.x &&
  35.     item.y < dungeon_setting.z &&
  36.     tmp[item.x][item.y] != 1 && //判断是否是障碍物
  37.     !existList(item, closeList) && //判断是否在关闭列表中
  38.     tmp[item.x][currentPoint.y] != 1 && //判断之间是否有障碍物，如果有障碍物是过不去的
  39.     tmp[currentPoint.x][item.y] != 1) {
  40.     //g 到父节点的位置
  41.     //如果是上下左右g等于10
  42.     var g = currentPoint.G + 10;
  43.     if (!existList(item, openList)) { //如果不在开启列表中
  44.      //计算H，通过水平和垂直距离进行确定
  45.      item['H'] = Math.abs(EndX - item.x) * 10 + Math.abs(EndZ - item.y) * 10;
  46.      item['G'] = g;
  47.      item['F'] = item.H + item.G;
  48.      item['parent'] = currentPoint;
  49.      openList.push(item);
  50.     } else { //存在在开启列表中，比较目前的g值和之前的g的大小
  51.      var index = existList(item, openList);
  52.      //如果当前点的g更小
  53.      if (g < openList[index].G) {
  54.       openList[index].parent = currentPoint;
  55.       openList[index].G = g;
  56.       openList[index].F = g + openList[index].H;
  57.      }
  58.     }
  59.    }
  60.   }
  61.   //如果开启列表空了，没有通路，结果为空
  62.   if (openList.length == 0) {
  63.    e.API.executeCommand(e.npc.world, 'title ' + p.name + ' subtitle ["",{"text":"§4?"}]');
  64.    p.message('§3无路径')
  65.    p.message("§a寻路耗时: " + (new Date().getTime() - test) + "ms");
  66.    break;
  67.   }
  68.   openList.sort(sortF); //这一步是为了循环回去的时候，找出 F 值最小的, 将它从 "开启列表" 中移掉
  69. } while (!(result_index = existList({
  70.    x: EndX,
  71.    y: EndZ
  72.   }, openList)));
  73. //判断结果列表是否为空
  74. if (!result_index) {
  75.   result = [];
  76. } else {
  77.   var currentObj = openList[result_index];
  78.   do {
  79.    //把路劲节点添加到result当中
  80.    result.unshift({
  81.     x: currentObj.x,
  82.     y: currentObj.y
  83.    });
  84.    currentObj = currentObj.parent;
  85.   } while (currentObj.x != StartX || currentObj.y != StartZ);
  86.   e.API.executeCommand(e.npc.world, 'title ' + p.name + ' subtitle ["",{"text":"§a?"}]')
  87.   p.message("§a寻路耗时: " + (new Date().getTime() - test) + "ms");
  88.   p.message('§2有路径');
  89. }
  90. return result;
  91. }

  118. function existList(point, list) {

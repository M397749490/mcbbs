package dev.ginyai.getworlddir;



import org.bukkit.Bukkit;

import org.bukkit.World;

import org.bukkit.command.Command;

import org.bukkit.command.CommandSender;

import org.bukkit.entity.Player;

import org.bukkit.plugin.java.JavaPlugin;



import java.lang.invoke.MethodHandle;

import java.lang.invoke.MethodHandles;

import java.lang.invoke.MethodType;

import java.lang.reflect.Method;

import java.util.Arrays;

import java.util.Collection;

import java.util.Collections;

import java.util.List;

import java.util.Locale;

import java.util.stream.Collectors;

import java.util.stream.Stream;



public class GetWorldDir extends JavaPlugin {



    @Override

    public void onEnable() {

        super.onEnable();

    }



    private String getUsage(String label, CommandSender sender) {

        if (sender instanceof Player) {

            return label + " [player|world]";

        } else {

            return label + " <player|world>";

        }

    }



    @Override

    public boolean onCommand(CommandSender sender, Command command, String label, String[] args) {

        World world;

        if (args.length == 0 && sender instanceof Player) {

            world = ((Player) sender).getWorld();

        } else if (args.length == 1) {

            if (Bukkit.getWorld(args[0]) != null) {

                world = Bukkit.getWorld(args[0]);

            } else {

                Player player = Bukkit.getPlayer(args[0]);

                if (player != null) {

                    world = player.getWorld();

                } else {

                    sender.sendMessage(getUsage(label, sender));

                    return true;

                }

            }

        } else {

            sender.sendMessage(getUsage(label, sender));

            return true;

        }

        try {

            sender.sendMessage("DimensionId: " + getDimensionId(world));

        } catch (Throwable t) {

            sender.sendMessage("Failed to get dimensionId");

            t.printStackTrace();

        }

        sender.sendMessage(world.getWorldFolder().toString());

        return true;

    }



    private MethodHandle getOnlinePlayerStream;



    @Override

    public List<String> onTabComplete(CommandSender sender, Command command, String alias, String[] args) {

        if (getOnlinePlayerStream == null) {

            MethodHandles.Lookup lookup = MethodHandles.lookup();

            try {

                MethodHandle getOnlinePlayers = lookup.findStatic(Bukkit.class, "getOnlinePlayers", MethodType.methodType(Collection.class));

                MethodHandle stream = lookup.findVirtual(Collection.class, "stream", MethodType.methodType(Stream.class));

                getOnlinePlayerStream = MethodHandles.collectArguments(stream, 0, getOnlinePlayers);

            } catch (NoSuchMethodException e) {

                try {

                    MethodHandle getOnlinePlayerArray = lookup.findStatic(Bukkit.class, "getOnlinePlayers", MethodType.methodType(Player[].class));

                    MethodHandle stream = lookup.findStatic(Arrays.class, "stream", MethodType.methodType(Stream.class, Object[].class));

                    getOnlinePlayerStream = MethodHandles.collectArguments(stream, 0, getOnlinePlayerArray);

                } catch (NoSuchMethodException | IllegalAccessException ex) {

                    ex.printStackTrace();

                    return Collections.emptyList();

                }

            } catch (IllegalAccessException e) {

                e.printStackTrace();

                return Collections.emptyList();

            }

        }

        if (args.length == 1) {

            try {

                Stream<Player> playerStream = (Stream<Player>) getOnlinePlayerStream.invoke();

                Stream<World> worldStream = Bukkit.getWorlds().stream();

                String prefix = args[0].toLowerCase(Locale.ROOT);

                return Stream.concat(playerStream.map(Player::getName), worldStream.map(World::getName))

                    .filter(s -> s.toLowerCase(Locale.ROOT).startsWith(prefix))

                    .collect(Collectors.toList());

            } catch (Throwable t) {

                t.printStackTrace();

            }

        }

        return Collections.emptyList();

    }



    private MethodHandle getDimensionId;



    public int getDimensionId(World world) throws Throwable {

        if (getDimensionId == null) {

            MethodHandles.Lookup lookup = MethodHandles.lookup();

            Method getHandle = world.getClass().getMethod("getHandle");

            MethodHandle getDimId = lookup.findGetter(getHandle.getReturnType(), "dimension", int.class);



            getDimensionId = MethodHandles.collectArguments(

                getDimId, 0, lookup.unreflect(getHandle)

            ).asType(MethodType.methodType(int.class, World.class));

        }

        return (int) getDimensionId.invokeExact(world);

    }

}

  1. Process p = new Process();

  3. p.StartInfo.UseShellExecute = false;
  4. p.Start();
  5. Random rand = new Random();
  6. rand = new Random((int)DateTime.Now.Ticks);
  7. Thread.Sleep(rand.Next(最短时间，最长时间);
  8. Process p2 = new Process();
  9. p2.StartInfo.FileName = "cmd.exe";
  10. p2.StartInfo.UseShellExecute = false;
  11. p2.StartInfo.CreateNoWindow = true;
  12. p2.StartInfo.RedirectStandardInput = true;
  13. p2.Start();
  14. p2.StandardInput.WriteLine("taskkill -f -im (刚才那个进程的文件名，如chrome.exe) ");
  15. p2.StandardInput.WriteLine("exit");

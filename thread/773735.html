/title @a[x=10,y=10,z=10,dx=5,dy=3,dz=5,tag=!abc] title {"text":"战斗开始","color":"red"}
/scoreboard players tag @a[x=10,y=10,z=10,dx=5,dy=3,dz=5] add abc


/tellraw @a[x=20,y=10,z=20,dx=5,dy=3,dz=5,tag=!def] {"text":"欢迎光临"}
/scoreboard players tag @a[tag=def] remove def
/scoreboard players tag @a[x=20,y=10,z=20,dx=5,dy=3,dz=5] add def

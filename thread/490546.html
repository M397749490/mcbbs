# 最大领地数量
MaxResidences: 3
# 东西向最大距离
MaxEastWest: 16
# 南北向最大距离
MaxNorthSouth: 16
# 最大领地高度
MaxUpDown: 256
# 最低领地高度
MinHeight: 0
# 领地允许的高度
MaxHeight: 255
# 领地每格花费
BuyCost: 0.05

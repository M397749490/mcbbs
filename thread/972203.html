#include <stdio.h>
#include <math.h>
int max(int x, int y)
{
        if (x > y) return x;
        else return y;
}
int main()
{
        int n, floor;
        for (n = 1; n <= 20; n++)
        {
                floor = int(n * 2 / sqrt(3) - (1 - sqrt(3) / 2));
                printf("n=%d,数量=%d\n", n, max(n*n, floor*n - floor / 2));
        }
}

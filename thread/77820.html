  2. SET BINDIR=%~dp0
  3. "%ProgramFiles%\Java\jre7\bin\java.exe" -Xincgc -Xmx1G -jar "%BINDIR%\craftbukkit-1.4.5-R1.0.jar"
  4. PAUSE

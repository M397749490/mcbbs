命令/权限
命令说明权限
/dchat reload重新加载配置文件delaychat.admin
配置
  1. # 是否对指令也同样生效?
  2. # 如果设置为false，则插件只会对聊天有效
  3. doCmd: false
  4. # doCmd设置为false时，这项配置才有效
  5. # 插件会将这里列出的指令视为聊天
  6. # 你可以将一些私聊指令添加进来
  7. # 或者添加其它你想添加的指令
  8. otherCmd:
  9. - 'tell'
  10. - 'msg'
  11. - 't'
  12. - 'm'
  13. # 延迟精度
  14. # 较大的数值会让聊天的延迟时间产生误差
  15. # 较小的数值会给服务器带来轻微负担
  16. # 推荐50-150，单位是毫秒
  17. Accuracy: 100
  18. # 玩家的聊天队列中，最多积压多少条聊天?
  19. maxList: 5
  20. # 当玩家聊天队列的积压达到上限时，发送消息的提示信息
  21. overlist: '§e请勿刷屏'
  22. # 此处可配置不同权限的玩家拥有不同的聊天延迟
  23. # 当玩家拥有多个权限时，使用聊天延迟最小的那个
  24. # 权限可按照此格式随意添加
  25. permission:
  26. # 当玩家拥有delaychat.vip1权限时，延迟为2000毫秒
  27. - 'vip1:2000'
  28. # 当玩家拥有delaychat.vip2权限时，延迟为1000毫秒
  29. - 'vip2:1000'
  30. # 当玩家拥有delaychat.vip3权限时，无延迟
  31. # 注意，当延迟>0时，需要同时大于Accuracy这一项配置的数值
  32. - 'vip3:0'
  33. # 默认的延迟，玩家无以上权限时的聊天延迟，单位是毫秒
  34. defaultTime: 3000

作者: Tommer    时间: 2017-9-10 15:21
是领地

    Flow:
      # 这是限制水和岩浆流动时高度的设置
      # 这个选项不会限制领地内水和岩浆的高度
      Level: 63
      # 如果设置为true, 则会限制岩浆的高度
      NoLavaFlow: true
      # 如果设置为true, 则会限制水的高度
      NoWaterFlow: true
      Worlds:
      - world
    Place:
      # 这里将会限制水和岩浆开始放置的地方
      # 这个选项不会限制领地内水和岩浆的高度
      Level: 63
      # 如果设置为true, 则会限制岩浆的高度
      NoLavaPlace: true
      # 如果设置为true, 则会限制水的高度
      NoWaterPlace: true
      Worlds:
      - world

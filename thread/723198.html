  44.         RegisteredServiceProvider<Economy> rsp = getServer().getServicesManager().getRegistration(Economy.class);
  45.         if (rsp == null) {
  46.             return false;

  48.         econ = rsp.getProvider();
  49.         return econ != null;
  50.     }
  51.    
  52.     private boolean setupChat() {
  53.         RegisteredServiceProvider<Chat> rsp = getServer().getServicesManager().getRegistration(Chat.class);

  55.         return chat != null;
  56.     }

  58.     private boolean setupPermissions() {
  59.         RegisteredServiceProvider<Permission> rsp = getServer().getServicesManager().getRegistration(Permission.class);
  60.         perms = rsp.getProvider();
  61.         return perms != null;
  62.     }
  63.    
  64.     public boolean onCommand(CommandSender sender, Command command, String commandLabel, String[] args) {
  65.         if(!(sender instanceof Player)) {

  67.             return true;
  68.         }
  69.         
  70.         Player player = (Player) sender;
  71.         
  72.         if(command.getLabel().equals("test-economy")) {
  73.             // Lets give the player 1.05 currency (note that SOME economic plugins require rounding!)
  74.             sender.sendMessage(String.format("You have %s", econ.format(econ.getBalance(player.getName()))));
  75.             EconomyResponse r = econ.depositPlayer(player, 1.05);
  76.             if(r.transactionSuccess()) {
  77.                 sender.sendMessage(String.format("You were given %s and now have %s", econ.format(r.amount), econ.format(r.balance)));
  78.             } else {
  79.                 sender.sendMessage(String.format("An error occured: %s", r.errorMessage));
  80.             }
  81.             return true;
  82.         } else if(command.getLabel().equals("test-permission")) {

  84.             if(perms.has(player, "example.plugin.awesome")) {

  86.             } else {
  87.                 sender.sendMessage("You suck!");
  88.             }
  89.             return true;
  90.         } else {
  91.             return false;
  92.         }
  93.     }

年先把资源放在Resources.resx里

这是一个释放Resources里面的文件的方法
  1. string path =".minecraft\versions\1.14.4-forge-28.1.56";
  2. if (!File.Exists(path + @"\释放的资源.扩展名"))
  3. {
  4. byte[] Save = Properties.Resources.资源的名字;
  5. FileStream fsObj = new FileStream(path + @"\释放的资源.扩展名", FileMode.CreateNew);
  6. fsObj.Write(Save, 0, Save.Length);
  7. fsObj.Close();
  8. }

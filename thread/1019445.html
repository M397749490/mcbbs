配置文件
#你认为的超速频率

suspicious-threshold: 0.8

# 当超速时执行的指令，若禁用则""，不需要加斜杠"/"，可支持多条指令

# 变量 %player% (玩家名字)

# 例如 "say This is an example command!"suspicious-command: ""

# 超速是否记录
logger: true

# Metrics (bStats.org)
metrics: true

命令&权限

dnotify.toggle - /dnotify toggle
dnotify.version - /dnotify version

找你计分板的配置文件出来先，找到所选行代码，看看是不是打错了，有乱码 ...

配置文件如下，没有设置字符长度的地方，也没有中文需要转码，而且编码正确。
Scoreboard:
#Name of the scoreboard
  title: '&cScoreBoard'
#You can use placeholders without PlaceHolderAPI:
#{Player} - Get?s playername.
#{Online} - Get?s count of online players.
#{MaxPlayers} - Get?s max count of online players. (Server slots)
#{Group} Get?s player rank/group (Vault needed)
#{Money} Get?s player money (Essentials needed)
#Rows of scoreboard placeholders from PlaceHolderAPI format like original.
#PlaceHolderAPI format for example PlayerPoints: %playerpoints_points%
  text:
    - '&aNick §7?'
    - '{Player}'
    - ''
    - '&eMoney §7?'
    - '{Money}'
    - ''
    - '&cGroup §7?'
    - '{Group}'
    - ''
    - '&dOnline §7?'
    - '&5{Online}&8/&7{MaxPlayers}'
    - ''
    - '&cPlugin by SouthCraft'
    - '&c(LukixCZE)'
    - '&cNice and Simple ;)'

Settings:
#World list, where will Scoreboard disabled.
  World-Blacklist:
    - world1
#Refresh rate of Scoreboard in seconds.
  Refresh-rate: 7
#Here you can customize messages.
Messages:
ScoreBoard-Hide: '§f(§aScoreBoard§f) §7? §eScoreBoard is now hiden.'
ScoreBoard-Show: '§f(§aScoreBoard§f) §7? §eScoreBoard is now show.'
No-Permission: '§f(§aScoreBoard§f) §7? §cYou do not have permission to use this command.'
Config-Reload: '§f(§aScoreBoard§f) §7? §aConfig successfully reloaded.'
ScoreBoard-Toggle-Disabled-World: '§f(§aScoreBoard§f) §7? §cThe scoreboard is disabled in this world.'
#Permissions:
#sb.toggle - Allow to use /sb toggle
#sb.reload - Allow to use /sb reload

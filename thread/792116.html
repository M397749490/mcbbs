Minecraft聊天栏不支持制表符，你需要使用空格处理字符串并输出。David Toledo写了一个函数帮助解决这个问题。
  1. /********************************************************
  2. * Fix string spaces to align text in minecraft chat
  3. *
  4. * @author David Toledo ([EMAIL]david.oracle@gmail.com[/EMAIL])
  5. * @param String to be resized
  6. * @param Size to align
  7. * @Return New aligned String
  8. */
  9. public static String fixFontSize (String s, int size) {

  11. String ret = s.toUpperCase();

  13. if ( s != null ) {

  15. for (int i=0; i < s.length(); i++) {
  16. if ( s.charAt(i) == 'I' || s.charAt(i) == ' ') {
  17. ret += " ";
  18. }
  19. }

  21. int faltaEspacos = size - s.length();
  22. faltaEspacos = (faltaEspacos * 2);

  24. for (int i=0; i < faltaEspacos; i++) {
  25. ret += " ";
  26. }
  27. }

  29. return (ret);
  30. }

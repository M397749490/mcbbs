        因为一直在网上找不到满意的解决方案，花了几天时间，终于自己解决了1.7.10的网页白名问题。QAQ

        下面就分享下，说不定有作用呢~哈哈~~~

       方案一：使用程序模拟人工从控制台输入的方式。
                优点：不需要服务端任何改动
                缺点：稳定性不强，需要淘宝上买那种带控制面板的服务姬

        （楼主的话：这是我一开始想到的方案，因为发现在网页控制面板可以输入命令，于是想到是否可以写个程序自动完成命令的输入，于是顺着思路做了下。由于需要让机器自动发Http请求，然而发送命令后无法直接通过服务器返回的信息判断是否成功……所以理论上稳定性不强，但经过测试……发现还可以的qwq下面就介绍下~~）

        首先，这个方案的难点是使用php（其他语言一样啦~）发送HTTP的Get和Post请求，并能做到session的存储。
        这里我在网上找到了一个大神写的HttpClient.php的组件，直接拿来用~~

  141.     /**
  142.      * Set a HTTP header for the next request.
  143.      * @param string the name of the request header
  144.      * @param string the value of the request header
  145.      * If the value is NULL, the header will be dropped.
  146.      * Note: special key 'x-server-addr' will force to use instead of gethostbyname(host)

  148.     function setHeader($key, $value = null)

  150.         $this->_reset();
  151.         $key = strtolower($key);
  152.         if (is_null($value)) unset($this->headers[$key]);
  153.         else $this->headers[$key] = strval($value);

  155.    
  156.     /**
  157.      * Get one or more HTTP headers of the last request.
  158.      * @param string the name of the header to be fetched.
  159.      * If is NULL, return the all headers of the last request.
  160.      * @return mixed fetched header value or headersas key-value array.

  163.     function getHeader($key = null)
  164.     {
  165.         if (is_null($key)) return $this->headers;
  166.         $key = strtolower($key);
  167.         if (!isset($this->headers[$key])) return null;
  168.         return $this->headers[$key];
  169.     }

  171.     /**
  172.      * Add a HTTP cookie sent for the next request.
  173.      * @param string the name of the cookie to be added
  174.      * @param string the value of the cookie to be added
  175.      */
  176.     function setCookie($key, $value)
  177.     {
  178.         $this->_reset();
  179.         if (!isset($this->headers['cookie'])) $this->headers['cookie'] = array();
  180.         $this->headers['cookie'][$key] = $value;
  181.     }

  183.     /**
  184.      * Get a HTTP cookie item by name
  185.      * @param string the name of the cookie to be fetched
  186.      * If the name is NULL, all matched cookies are returned as key-value array.  
  187.      * @param string host of all saved cookies (include expired)
  188.      * If the host is NULL, fetch the cookie from last request.
  189.      * @return mixed fetched cookie item or cookies as key-value array.
  190.      * Every cookie item is a assoc array, keys include: value, expires, path, host
  191.      * If the cookie dose not exists, NULL is returned.   
  192.      */
  193.     function getCookie($key = null, $host = null)
  194.     {
  195.         // fetch from last request
  196.         if (!is_null($key)) $key = strtolower($key);
  197.         if (is_null($host))
  198.         {
  199.             if (!isset($this->headers['cookie'])) return null;
  200.             if (is_null($key)) return $this->headers['cookie'];
  201.             if (!isset($this->headers['cookie'][$key])) return null;
  202.             return $this->headers['cookie'][$key];
  203.         }
  204.         // fetch from all saved cookies.
  205.         $host = strtolower($host);
  206.         while (true)
  207.         {
  208.             if (isset($this->cookies[$host]))
  209.             {
  210.                 if (is_null($key)) return $this->cookies[$host];
  211.                 if (isset($this->cookies[$host][$key])) return $this->cookies[$host][$key];
  212.             }
  213.             // search for next sub-domain
  214.             $pos = strpos($host, '.', 1);
  215.             if ($pos === false) break;
  216.             $host = substr($host, $pos);
  217.         }
  218.         return null;
  219.     }

  221.     /**
  222.      * Save all cookies to a file.
  223.      * @param string the file path that cookies will be saved to.
  224.      * @return boolean save result, return true on success and false on faiulre.
  225.      * Note: all cookies are serialized before saving.
  226.      */
  227.     function saveCookie($fpath)
  228.     {
  229.         if (false === ($fd = @fopen($fpath, 'w')))
  230.             return false;
  231.         $data = serialize($this->cookies);
  232.         fwrite($fd, $data);
  233.         fclose($fd);
  234.         return true;
  235.     }

  242.     function loadCookie($fpath)

  244.         if (file_exists($fpath) && ($cookies = @unserialzie(file_get_contents($fpath))))
  245.             $this->cookies = $cookies;
  246.     }

  249.      * Add a post field for the next request
  250.      * @param string the name of the field.

  254.     function addPostField($key, $value)

  260.         {

  262.             foreach ($value as $tmpk => $tmpv)
  263.             {
  264.                 $tmpk = $key . '[' . $tmpk . ']';
  265.                 $this->post_fields[$tmpk] = strval($tmpv);
  266.             }
  267.         }
  268.     }

  270.     /**
  271.      * Add a multipart post file for the next request

  273.      * @param string the filename or filepath to be uploaded
  274.      * @param string content the file content
  275.      * If the content is null and fname is a valid filepath,
  276.      * content will be set to the file content.
  277.      */
  278.     function addPostFile($key, $fname, $content = '')
  279.     {
  280.         $this->_reset();
  281.         if ($content === '' && is_file($fname)) $content = @file_get_contents($fname);
  282.         $this->post_files[$key] = array(basename($fname), $content);
  283.     }

  285.     /**
  286.      * Do a http request via get method

  296.     /**
  297.      * Do a http request via head method

  300.      * @return mixed all respond HTTP header or false on failure before server respond.
  301.      */
  302.     function Head($url, $redir = false)
  303.     {
  304.         if ($this->_do_url($url, 'head', null, $redir) !== false)
  305.             return $this->getHeader(null);
  306.         return false;
  307.     }

  309.     /**
  310.      * Do a http request via post method
  311.      * @param string the absolute URL
  312.      * @param boolean handle redirected requests automatically or not
  313.      * @return string respond body data or false on failure before server respond.
  314.      * Note: post request variable should be set by ::addPostField() and ::addPostFile()
  315.      */
  316.     function Post($url, $redir = true)
  317.     {
  318.         $data = '';
  319.         if (count($this->post_files) > 0)
  320.         {
  321.             $boundary = md5($url . microtime());
  322.             foreach ($this->post_fields as $tmpk => $tmpv)
  323.             {
  324.                 $data .= "--{$boundary}\r\nContent-Disposition: form-data; name="{$tmpk}"\r\n\r\n{$tmpv}\r\n";
  325.             }
  326.             foreach ($this->post_files as $tmpk => $tmpv)
  327.             {
  328.                 $type = 'application/octet-stream';
  329.                 $ext = strtolower(substr($tmpv[0], strrpos($tmpv[0],'.')+1));
  330.                 if (isset($GLOBALS['___HC_MIMES___'][$ext])) $type = $GLOBALS['___HC_MIMES___'][$ext];
  331.                 $data .= "--{$boundary}\r\nContent-Disposition: form-data; name="{$tmpk}"; filename="{$tmpv[0]}"\r\nContent-Type: $type\r\nContent-Transfer-Encoding: binary\r\n\r\n";
  332.                 $data .= $tmpv[1] . "\r\n";
  333.             }
  334.             $data .= "--{$boundary}--\r\n";
  335.             $this->setHeader('content-type', 'multipart/form-data; boundary=' . $boundary);
  336.         }
  337.         else
  338.         {
  339.             foreach ($this->post_fields as $tmpk => $tmpv)
  340.             {
  341.                 $data .= '&' . rawurlencode($tmpk) . '=' . rawurlencode($tmpv);
  342.             }
  343.             $data = substr($data, 1);
  344.             $this->setHeader('content-type', 'application/x-www-form-urlencoded');
  345.         }
  346.         $this->setHeader('content-length', strlen($data));
  347.         return $this->_do_url($url, 'post', $data, $redir);
  348.     }

  350.     /**
  351.      * Download a file to local via get method with range support
  352.      * @param string the absolute URL
  353.      * @param string local filepath to saved, default is the same filename on current working directory.
  354.      * @param boolean weather to overwrite the exists file
  355.      * when filepath exists and not a valid partially-downloaded file.
  356.      * @return boolean true on success and false on failure.
  357.      * Note: this method can be used to resume getting a partially-downloaded file.
  358.      */
  359.     function Download($url, $filepath = null, $overwrite = false)
  360.     {
  361.         // get filepath & head
  362.         if ($filepath === true)
  363.         {
  364.             $overwrite = true;
  365.             $filepath = null;
  366.         }
  367.         if (is_null($filepath) || empty($filepath)) $filepath = '.';
  368.         // get normal headers first
  369.         $savehead = $this->getHeader(null);
  370.         if (!$this->Head($url, true))
  371.         {
  372.             if ($this->verbose) echo "[ERROR] failed to get headers for downloading file.\n";
  373.             return false;
  374.         }
  375.         else if ($this->getStatus() != 200)
  376.         {
  377.             if ($this->verbose) echo "[ERROR] can not get a valid 200 HTTP respond status.\n";
  378.             return false;
  379.         }
  380.         // get filename & filesize
  381.         $url = $this->getUrl();
  382.         if ($this->verbose) echo "[INFO] real download url is: $url\n";
  383.         if (is_dir($filepath))
  384.         {
  385.             if (substr($filepath, -1, 1) != DIRECTORY_SEPARATOR) $filepath .= DIRECTORY_SEPARATOR;        
  386.             if (($disposition = $this->getHeader('content-disposition'))
  387.                 && preg_match('/filename=[\'"]?([^;\'" ]+)/', $disposition, $match))
  388.             {
  389.                 $filename = $match[1];
  390.                 if ($this->verbose) echo "[INFO] fetch filename from disposition header: $filename\n";
  391.             }
  392.             else
  393.             {
  394.                 $tmpstr = ($pos = strpos($url, '?')) ? substr($url, 0, $pos) : $url;
  395.                 $pos = strrpos($tmpstr, '/');
  396.                 $filename = substr($tmpstr, $pos + 1);
  397.                 if ($filename == '') $filename = 'index.html';
  398.                 if ($this->verbose) echo "[INFO] fetch filename from URL: $filename\n";
  399.             }
  400.             while (true)
  401.             {
  402.                 $filepath .= $filename;
  403.                 if (!is_dir($filepath)) break;
  404.                 $filepath .= DIRECTORY_SEPARATOR . $filename;
  405.             }
  406.         }
  407.         // check filepath
  408.         if (!file_exists($filepath) || !($fsize = @filesize($filepath)))
  409.         {
  410.             $savefd = @fopen($filepath, 'w');
  411.             if ($this->verbose) echo "[INFO] save file directly to: $filepath\n";
  412.         }
  413.         else
  414.         {
  415.             $length = $this->getHeader('content-length');
  416.             $accept = $this->getHeader('accept-ranges');
  417.             if ($length && $fsize < $length && stristr($accept, 'bytes'))
  418.             {
  419.                 // range request used
  420.                 $this->setHeader('range', 'bytes=' . $fsize . '-');
  421.                 $savefd = @fopen($filepath, 'a');
  422.                 if ($this->verbose) echo "[INFO] range download used, range: {$fsize}-\n";
  423.             }
  424.             else if ($overwrite)
  425.             {
  426.                 $savefd = @fopen($filepath, 'w');
  427.                 if ($this->verbose) echo "[INFO] overwrite the exists file: $filepath\n";
  428.             }
  429.             else
  430.             {
  431.                 // auto append filename '.1, .2, ...'
  432.                 for ($i = 1; @file_exists($filepath . '.' . $i); $i++);
  433.                 $filepath .= '.' . $i;
  434.                 $savefd = @fopen($filepath, 'w');
  435.                 if ($this->verbose) echo "[INFO] auto skip exists file, last save to: $filepath\n";
  436.             }
  437.         }
  438.         // check the savefd
  439.         if (!$savefd)
  440.         {
  441.             if ($this->verbose) echo "[ERROR] can not open the file to save data: $filename\n";
  442.             return false;
  443.         }
  444.         // do real download via get method
  445.         foreach ($savehead as $hk => $hv) $this->setHeader($hk, $hv);
  446.         if ($this->_do_url($url, 'get', null, false, $savefd) !== false)
  447.         {
  448.             $this->filepath = $filepath;
  449.             fclose($savefd);
  450.             if ($this->verbose) echo "[INFO] downloaded file saved in: $filepath\n";
  451.             return true;
  452.         }
  453.         else
  454.         {
  455.             if ($this->verbose) echo "[ERROR] can not download the URL: $url\n";
  456.             return false;
  457.         }
  458.     }

  460.     // -------------------------------------------------

  463.     // read data from socket
  464.     function _sock_read($fd, $maxlen = 4096, $wfd = false)
  465.     {
  466.         $rlen = 0;
  467.         $data = '';
  468.         $ntry = HC_MAX_RETRIES;
  469.         while (!feof($fd))
  470.         {
  471.             $part = fread($fd, $maxlen - $rlen);
  472.             if ($part === false || $part === '') $ntry--;
  473.             else $data .= $part;

  9.     }

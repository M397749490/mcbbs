  1. Pattern PATTERN = Pattern.compile("((:?§[0-9a-fklmnor])+)", CASE_INSENSITIVE);
  2. String lore = "§at1§B§nt2";
  3. String s1 = PATTERN.matcher(lore).replaceAll("$1§7§m");//变成灰色删除线
  4. String s2 = s1.replaceAll("§7§m","");//恢复

output
  1. field String lore = "§at1§B§nt2"
  2. field String s1 = "§a§7§mt1§B§n§7§mt2"
  3. field String s2 = "§at1§B§nt2"

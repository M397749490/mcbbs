如果是要实现单纯往前不往上下的话：
  1. Vector vector = player.getLocation().getDirection();
  2. vector.setY(0d);
  3. //如果玩家面朝大地或90°仰望天空，X 和 Z会是0
  4. if(vector.getX() == 0 && vector.getY() == 0)
  5.   vector.setX(1d);

  7. //使用normalize() 将vector伸长到一个单位的长度
  8. vector.normalize();

  10. //使用multiply() 将vector缩放到指定的长度
  11. //一单位的长度缩小到0.09倍大概就是一方块的长度
  12. vector.multiply(0.09);

  14. player.setVelocity(vector);

怎么设置服务器  VIP组圈地便宜 和 圈地 数量  说详细点啦谢谢
作者: SmartMadio    时间: 2014-8-25 10:30
首先，你需要在圈地插件的Config.yml找到Groups:字段，然后，在下面加上如下内容（此处假设你的VIP组的组名是VIP）：

  1.     VIP: #VIP组名
  2.        # Information about the Residence zone that the player can define.
  3.        #Uncomment the lines below to mirror this config to other groups.
  4.        #Mirror:
  5.          #- 'group1'
  6.          #- 'group2'
  7.        Residence:
  8.            # Determins if this group can create residences or not.  This option can be overriden with the permissions node 'residence.create'
  9.            CanCreate: true
  10.            # VIP组圈地数量上限
  11.            MaxResidences: 3
  12.            # The maximum number of physical areas a residence can have.
  13.            MaxAreasPerResidence: 2
  14.            # The maximum number of blocks a Residence can be, East to West (X).
  15.            MaxEastWest: 256
  16.            # The maximum number of blocks a Residence can be, North to South (Z).
  17.            MaxNorthSouth: 256
  18.            # The maximum number of blocks a Residence can be, High to Low (Y).
  19.            MaxUpDown: 256
  20.            # The lowest altitude this group is allowed to protect.
  21.            MinHeight: 0
  22.            # The highest altitude this group is allowed to protect.
  23.            MaxHeight: 255
  24.             # The maximum recursive depth that subzones can go.
  25.             # A subzone within a subzone within the Residence zone would be a depth of 2.
  26.             # Set to 0 to disable subzones.
  27.            SubzoneDepth: 3
  28.            # Whether or not to allow teleporting to Residences.
  29.            CanTeleport: true
  30.            # Allow or Disallow the use of /res unstuck, to get yourself out of a residence if stuck in one somehow
  31.            Unstuck: true
  32.            # Allow or Disallow the use of the /res select command, if disabled they can only use the selection tool.
  33.            SelectCommandAccess: true
  34.            # Allow or Disallow the group to access the Blacklist/Ignorelist for residences they own.
  35.            ItemListAccess: true
  36.        # Options relating to enter and leave messages on the residence.
  37.        Messaging:
  38.            # Whether or not the player can change the Residence enter and leave messages.
  39.            CanChange: true
  40.            # The default enter message to apply to new Residences created by players in this group.
  41.            # Leaving the message blank will disable it.
  42.            DefaultEnter: Welcome %player to %residence, owned by %owner.
  43.            # The default leave message to apply to new Residences created by players in this group.
  44.            # Leaving the message blank will disable it.
  45.            DefaultLeave: Now leaving %residence.
  46.        # Options relating to the Lease system.
  47.        Lease:
  48.           # The maximum number of days to allow leases.
  49.           MaxDays: 16
  50.           # The number of days to add to the lease following a '/res lease renew' command.
  51.           RenewIncrement: 14
  52.        #Options related to the rent system
  53.        Rent:
  54.            #determines how many residences can be rented by players of this group at once
  55.            MaxRents: 3
  56.            #determines how many residences can be set for rent by players of this group at once
  57.            MaxRentables: 3
  58.        # Options relating to the Residence Economy.
  59.        Economy:
  60.           # Whether or not players in this group can buy other Residences that are for sale.
  61.           CanBuy: true
  62.           # Whether or not players in this group can sell their Residences.
  63.           CanSell: false
  64.           # Whether or not to ignore the Residence size/count limits when buying a residence.
  65.           IgnoreLimits: false
  66.           # VIP组圈地时，每个方块的价格。
  67.           BuyCost: 0.05
  68.           # The cost, per block, of renewing a Residence lease (if the lease system is enabled).
  69.           RenewCost: 0.02
  70.        # Flags are checked in the order:
  71.        # 1: Player
  72.        # 2: Group
  73.        # 3: Owner
  74.        # 4: If all others are undefined, reverts to default value (usually true).
  75.        # Flag permissions default to false.
  76.        # These flags settings override the globals.
  77.        Flags:
  78.           # Specifically allow or deny this group from changing certain flags.
  79.           Permission:
  80.                #build: true
  81.           # Specifys the flags that are applied at residence creation for players of this group.
  82.           Default:
  83.               #build: true
  84.           # Specifys the flags that are applied at to the creator at residence creation.
  85.           CreatorDefault:
  86.               #build: true
  87.           # Specifys the flags that are applied to other groups for this residence, at creation.
  88.           GroupDefault:
  89.                #default: #group name
  90.                    #build: false
  91.           #these flags are applied specifically to this group whenever they are outside a residence
  92.           #these flags will override the flags at the top, under the Global section.
  93.           World:
  94.                 Global: #these flags will apply to all worlds
  95.                      #build: false
  96.                 WorldNameHere: #or you can apply them by each individual world
  97.                      #build: false

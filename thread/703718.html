现在这个按钮是没有用的，接下来我们就开始对按钮设置事件
代码如下:
import java.io.File;
import java.io.FileWriter;
import java.io.IOException;
import java.util.logging.Level;
import java.util.logging.Logger;
import javafx.application.Application;
import javafx.event.EventHandler;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.control.Label;
import javafx.scene.control.TextArea;
import javafx.scene.input.MouseEvent;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;
public class HelloJavafx extends Application{
    public static void main(String[] args) {
        launch(args);
    }
    @Override
    public void start(Stage stage) throws Exception {
    stage.setTitle("TxtHelper");
    stage.setWidth(300);
    stage.setMinHeight(150);
    Button complete = new Button("完成");
    TextArea text = new TextArea();
    complete.setOnMouseClicked(new EventHandler<MouseEvent>() {//内部类设置事件
        @Override
        public void handle(MouseEvent t) {//重写父类方法
            try {
            File target = new File("C://MyJavaFx/test.txt"); //以下方法我觉得应该都懂
            target.getParentFile().mkdirs();
            FileWriter writer = new FileWriter(target);
            writer.write(new String(text.getText().getBytes(),"utf-8"));//转个码好
            writer.close();//关闭流
            target.createNewFile();            
            } catch (IOException ex) {
              ex.printStackTrace();
            }
        }
    });
    VBox vbox = new VBox(10);
    vbox.getChildren().addAll(text,complete);
    stage.setScene(new Scene(vbox));
    stage.show();
    }   
}



现在在文本框里写入任意内容，点击完成，就会在C://MyJavaFx/test.txt路径下生成你所写的内容啦

插件名:Laborus
版本:Laborus-1.6.2 ,sponge API:7 (经测试spongeforge-1.12.2-2768-7.1.6-RC3643可稳定运行)
插件官网地址:https://ore.spongepowered.org/Lergin/Laborus
githubwiki:https://github.com/Lergin/Laborus/wiki/Configuration第三方可视化编辑器:https://laborus.lergin.de  (可能需要梯子)【请注意,版本已过时,需要自己魔改】
请先允许我介绍一下这个插件和写教程的契机。

众所周知sponge上插件数量比起spigot少太多,想要搞个依赖插件的sponge服务器实属不易,在我苦心钻研并且决定自闭nt-rpg数日之后,逛ore之余,发现了这款Laborus(作者:Lergin)
简单来说,这就是一个类似mcjobs一样的插件,不过没有右键技能,取而代之的是可以自定义药水效果和一些其他巴拉巴拉的。

【在实际使用时请自行参考wiki,那边有更全的“属性”和“方法”】
我在BBS找了找,发现没人搬运这个。希望有能力的大佬搬运下(？)顺带把完整的wiki汉化了(？)

为何推荐这款插件:
1、sponge上难得的简单易用RPG类插件
2、优秀的可扩展性、优秀的易用性(相对而言)
3、汉化起来很简单,用语魔改也就很简单
4、功能上,等级系统很棒(可以根据不同等级启用不同阶段的技能和技能效果)
5、可以瞎折腾
先上汉化的config,由于是我一边研究一边写汉化,备注并不工整,只能说...能懂得人肯定一眼就理解,不懂(指没有任何编程常识)的人看起来会很吃力

# activated the anti replace farming system,启用一个防止通过重复刷作物来刷经验的系统
antiReplaceActive=false
# the time after that a location is removed from the anit replace farming list,待机上述系统的时间阈值(将会作用于一个活跃区块)
antiReplaceTime=48
# settings for the commands,命令部分
commands {
    # starts the job ability
    ability {
        # command
        command=ability
        # description of the command
        description {
            text=启用职业技能
        }
        # description of the job parameter
        paramJobDescription {
            text=职业
        }
        # permission needed to use the command,你可以自由修改权限,建议不要改变根节点
        permission="laborus.commands.ability"
    }
    # adds job xp
    addXp {
        # command
        command=addXp
        # description of the command
        description {
            text=增加某人的职业经验
        }
        # description of the job parameter
        paramJobDescription=职业
        # description of the player parameter
        paramPlayerDescription=玩家
        # permission needed to use the player parameter
        paramPlayerPermission="laborus.commands.addXp.other_player"
        # description of the xp parameter
        paramXpDescription="[xp]"
        # permission needed to use the command
        permission="laborus.commands.addXp"
    }
    # changes the job
    change {
        # command
        command=change
        # description of the command
        description {
            text=变更职业
        }
        # description of the action parameter
        paramActionDescription=行为
        # description of the job parameter
        paramJobDescription=职业
        # description of the join parameter
        paramJoinDescription=Join
        # description of the leave parameter
        paramLeaveDescription=Leave
        # permission needed to use the command
        permission="laborus.commands.change"
    }
    # shows information about jobs
    info {
        # command
        command=info
        # description of the command
        description {
            text=显示职业信息
        }
        # description of the job parameter
        paramJobDescription=职业
        # permission needed to use the command
        permission="laborus.commands.info"
    }
    # command needed to place before each of the commands
    mainCommand=jobs
    # command to reload the config
    reload {
        # command
        command=reload
        # description of the command
        description {
            text="Reloads the configuration of the plugin"
        }
        # permission needed to use the command
        permission="laborus.commands.reload"
    }
    # sets the job xp
    setXp {
        # command
        command=setXp
        # description of the command
        description {
            text=设置职业经验
        }
        # description of the job parameter
        paramJobDescription=job
        # description of the player parameter
        paramPlayerDescription=player
        # permission needed to use the player parameter
        paramPlayerPermission="laborus.commands.setXp.other_player"
        # description of the xp parameter
        paramXpDescription=xp
        # permission needed to use the command
        permission="laborus.commands.setXp"
    }
    # command to toggle the active status of the job system (only affects the player itself)
    toggle {
        # command
        command=toggle
        # description of the command
        description {
            text=切换职业系统开关状态
        }
        # permission needed to use the command
        permission="laborus.commands.toggle"
    }
}

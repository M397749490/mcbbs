很简单，发送一个
  1. http://login.minecraft.net/?user=用户邮箱&password=用户密码&version=13
复制代码
然后会获得一个类似于
  1. 1374223516000:deprecated:用户名:ad7f29e67be84d9d99f7b9b3c6fde15e:08587d233ea64f2233c112cdb13bce3f
复制代码
这里面的“用户名”就是绑定邮箱的正版用户名
而“ad7f29e67be84d9d99f7b9b3c6fde15e”
就是要取得的token
把这一段提取出来放到启动附加参数“ --session ”就行
以上就是大致思路
Golang源码：
package main

import (
    "fmt"
    "io/ioutil"
    "log"
    "net/http"
    "strings"
)

func main() {
    var (
        email    string
        password string
    )
    fmt.Println("请输入邮箱:")
    fmt.Scanf("%s", &email)
    fmt.Println("请输入密码:")
    fmt.Scanf("%s", &password)
    res, err := http.Get("http://login.minecraft.net/?user=" + email + "&password=" + password + "&version=15")
    if err != nil {
        log.Fatal(err)
    }
    t, err := ioutil.ReadAll(res.Body)
    res.Body.Close()
    if err != nil {
        log.Fatal(err)
    }
    switch string(t) {
    case "":
        fmt.Print("ERROR\n")
        return
    case "Bad Login":
        fmt.Print("Bad Login\n")
        return
    default:
        fmt.Printf("Return: %s\n", t)
    }
    text := strings.Split(string(t), ":")
    fmt.Printf("Server Time: %q\n", text[0])
    fmt.Printf("Name: %q\n", text[2])
    fmt.Printf("Token: %q\n", text[3])
    fmt.Printf("UUID: %q\n", text[4])
}

  2.     public static double getTPS() {
  3.         String version = org.bukkit.Bukkit.getServer().getClass().getPackage().getName().replace(".", ",").split(",")[3];
  4.         try {
  5.             Class<?> minecraftserver = Class.forName("net.minecraft.server." + version + ".MinecraftServer");
  6.             Method getserver = minecraftserver.getMethod("getServer");

  8.             Field h = null;
  9.             for (Field field : minecraftserver.getFields()) {
  10.                 if (field.getType().getName().equalsIgnoreCase("[J")) {
  11.                     h = field;
  12.                     break;
  13.                 }
  14.             }

  16.             long average = 0;
  17.             for (long b : (long[]) h.get(getserver.invoke(minecraftserver))) {
  18.                 average += b;
  19.             }
  20.             average /= ((long[]) h.get(getserver.invoke(minecraftserver))).length;


  23.             return Math.min(1000 / (average * 1e-6), 20.0);
  24.         } catch (Exception a) {
  25.             return 0;
  26.         }
  27.     }

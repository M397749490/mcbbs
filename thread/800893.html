创建四个计分板：
scoreboard objectives add lightning1 dummy
scoreboard objectives add lightning2 dummy
scoreboard objectives add lightning3 stat.damageDealt
scoreboard objectives add lightning4 dummy
scoreboard objectives add ice dummy
-----
scoreboard players set @a lightning2 2 {SelectedItem:{id:"minecraft:iron_sword",Count:1b,Damage:0s,tag:{display:{Name:"冰封剑"}}}}
（这条指令探测玩家手持名为冰封剑的铁剑，并设定改lightning2玩家为2分。）
----------
scoreboard players remove @a[score_lightning2_min=1] lightning2 1 （手不持分数归零）
---------------
scoreboard players set @e[type=!player] lightning1 1 {HurtTime:10s} (探测生物受伤)
--------------
execute @a[score_lightning3_min=1,score_lightning2_min=1,score_lightning2=2] ~ ~ ~ scoreboard players set @e[score_lightning1_min=1,r=5,type=!player] lightning4 1 【探测手持冰封剑，并且造成伤害的玩家的周围的受伤的生物(玩家除外）】
--------------------------------
scoreboard players add @e[score_lightning4_min=1] lightning4 1  (加分)
----------------
execute @e[score_lightning4_min=10,score_lightning4=10] ~ ~ ~ fill ~-2 ~ ~2 ~2 ~5 ~-2 ice   (0.5秒后冰封)
-------------------------------
execute @e[score_lightning4_min=10,score_lightning4=10] ~ ~ ~ entitydata @e[score_lightning4_min=10,type=creeper] {ignited:0b,ExplosionRadius:0,fire:-1,NoAI:1}    (如果攻击的是爬行者，就停止点燃，爆炸范围变0，不火烧，并不动）
--------------------------
execute @e[score_lightning4_min=10,score_lightning4=10] ~ ~ ~ entitydata @e[score_lightning4_min=10] {Fire:-1,NoAI:1}
（攻击的生物不动）
-------------------------------------------------------
scoreboard players set @e[score_lightning4_min=12] lightning4 0     (清零)
---------------------------------------------------
(冰封剑合成)
scoreboard players set @e[type=item] ice 1 {Item:{id:"minecraft:iron_sword",Count:1b,Damage:0s},OnGround:1b}
scoreboard players set @e[type=item] ice 2 {Item:{id:"minecraft:ice",Count:64b,Damage:0s},OnGround:1b}
execute @e[type=item,score_ice=1,score_ice_min=1] ~ ~ ~ execute @e[type=item,score_ice=2,score_ice_min=2,r=1] ~ ~ ~ summon item ~ ~0.1 ~ {Item:{id:"minecraft:iron_sword",Count:1b,tag:{ench:[{lvl:2s,id:19s}],Unbreakable:1,display:{Name:"冰封剑"}},Damage:0s},OnGround:1b}
execute @e[type=item,score_ice=1,score_ice_min=1] ~ ~ ~ execute @e[type=item,score_ice=2,score_ice_min=2,r=1] ~ ~ ~ kill @e[type=item,score_ice_min=1,r=1]

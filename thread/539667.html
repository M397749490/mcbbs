# 菜单配置文件

# 开启/关闭兼容模式(像healthbar，coloredtags，幽灵玩家或mccombatlevel之类的插件)
# 不懂是什么，貌似就是开了之后能兼容其他记分牌插件
compatibilityMode: false

# 关闭记分牌显示的世界
disabled-worlds:
  - city

# 开启/关闭隐身玩家显示
hide-vanished: false

Scoreboard:
  Title: '&a&l昼夜军团欢迎您'
  # 秒
  # 即时更新（您填1的话，它会每一秒更新）
  Update-delay: 2
  Items:
    # 标题必须有以下48个字符
    # 标题；类型
    '&9服务器在线': '%online%'
    '&9你的金币': '%money%'

# 让scoreboardstats记录数据（杀死，死亡，mobkills，killstreak）你需要没有这个插件
enable-pvpstats: false

#Temp菜单开启/关闭
Temp-Scoreboard-enabled: false

#Temp菜单
Temp-Scoreboard:
  Title: '&a&lTop Kills'
  # %mob% | %kills% | %killstreak%
  Type: '%kills%'
  Color: '&9'
  # 展示了多少玩家
  Items: 5
  Intervall-show: 300
  Intervall-disappear: 300

#插件自动更新开启/关闭
pluginUpdate: true

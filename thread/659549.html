标题: 服务器linux脚本带崩溃重启自动备份
要求:
shell或者py都行能崩溃重启.
每到2点定时关闭服务端备份.
如果金粒不够我还可以加。

作者: bzts123456    时间: 2017-1-4 21:42
#!/bin/bash
DIR=$HOME/mc # Minecraft Directory. Plz Modify this line before running. 先在这里修改你的MC目录，然后运行此脚本
M_OLDPWD=$OLDPWD
M_PWD=$PWD
LOGDIR=$DIR/srvlog
LOG=$LOGDIR/launcher.log
[[ -d "$LOGDIR" ]] || ( mkdir "$LOGDIR" ; echo "$(date)[Info]Created log directory.">>$LOG )
cd "$DIR"
echo "$(date)[Info]Launching.">>$LOG
MCSTAT=0
false
while [[ "$?" != "0" || -f "$DIR/restart.lck" ]]; do
    if [[ "$MCSTAT" == "0" ]]; then
        echo "$(date)[Info]Start server.">>$LOG
        MCSTAT=1
    elif [[ -f "$DIR/restart.lck" ]]; then
        echo "$(date)[Info]Restart server.">>$LOG
        sleep 2
    else
        echo "$(date)[Warn]Crashed. Restart server.">>$LOG
        sleep 10
    fi
    if [[ -f "$DIR/restart.lck" ]]; then
        rm "$DIR/restart.lck"
    fi
    # Commands here will execute before launch
    java -server -Xmx1024M -Djava.awt.headless=true -jar $DIR/craftbukkit.jar #这里看你的情况来改
done
echo "$(date)[Info]Exit.">>$LOG
cd "$M_PWD"
OLDPWD=$M_OLDPWD

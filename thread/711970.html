  2. ItemStack item = new ItemStack(Material.DIAMOND_SWORD);

  13. //  数值为0 百分比为1(7-2号修订=w=)
  14. //UUID是用于区分这些Attribute的建议可以设置大一些
  15. //Slot值该Attribute在玩家的哪个 部位 才会生效，不填写则所有部位都会生效
  16. damage.set("AttributeName", new NBTTagString("Damage"));
  17. damage.set("Name", new NBTTagString("generic.attackDamage"));
  18. damage.set("Amount", new NBTTagInt(20)); //值
  19. damage.set("Operation", new NBTTagInt(0));
  20. damage.set("UUIDLeast", new NBTTagInt(20000));
  21. damage.set("UUIDMost", new NBTTagInt(1000));
  22. damage.set("Slot", new NBTTagString("mainhand"));
  23. modifiers.add(damage);
  24. compound.set("AttributeModifiers", modifiers);
  25. nmsItem.setTag(compound);

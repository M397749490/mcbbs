LockIt——麻麻再也不怕我的东西被偷了
用途（usage）：你可以锁上你的箱子，容器和门
指令：
/lock - 锁上一个箱子（输入完指令后再右键）
/unlock - 解锁一个箱子
/public - 使该物品 共享（人们可以打开它，但是不可以破坏它）
/passlock <密码> <密码> - 创建密码保护
/modify <add|remove|removeall|public> [players...]
add - 给某个玩家该保护的权限
remove - 去除某个玩家的权限
removeall - 去除所有玩家的权限
public - 使物品共享
/password <密码> - 为你自己解锁物品
配置文件：
#protectable blocks
#保护的方块
blocks:
    chest: true
    furnace: true
    brewing: true
    dispenser: true
    hopper: true
    dropper: true
    item_frame: true
    sign: true
    oak_door: true
    iron_door: true
    spruce_door: true
    birch_door: true
    jungle_door: true
    acacia_door: true
    dark_oak_door: true
    wooden_trap_door: true
    iron_trap_door: true
    oak_fence_gate: true
    birch_fence_gate: true
    spruce_fence_gate: true
    jungle_fence_gate: true
    acacia_fence_gate: true
    dark_oak_fence_gate: true
    trapped_chest: true
#每个被保护的方块当玩家放置时都会被保护
#如果是false, 只有带有 lockit.autolock permission 权限的玩家才能使用
enable_auto_lock: false
#是否开启自动保护
#自动保护间隔（s）
auto_save_interval: 600
#如果带有 lockit.info permission 权限的玩家用这个物品右击上锁的方块, 就会显示出保护信息
info_item: leather
权限
lockit:
  description: "LockIt中最主要权限"
  default: op #op默认拥有
  children: #子权限
    lockit.command:
      description: "允许玩家使用所有LI的指令"
      default: op
      children:
        lockit.command.lock:
          description: "允许玩家给方块上锁"
          default: true
        lockit.command.public:
          description: "允许玩家使用公共保护"
          default: true
        lockit.command.unlock:
          description: "允许玩家解锁方块"
          default: true
    lockit.command.passlock:
          description: "允许玩家用密码上锁箱子"
          default: true
    lockit.command.modify:
          description: "允许玩家修改他们上锁的箱子"
          default: true
    lockit.command.password:
          description: "允许玩家用正确的密码解锁被保护的箱子"
          default: true
    lockit.access:
      description: "允许玩家使用所有方块"
      default: op
    lockit.deny:
      description: "不允许玩家使用所有方块"
      default: false
    lockit.autolock:
      description: "允许玩家自动锁箱"
      default: false
    lockit.info:
      description: "允许玩家查看被保护方块的信息"
      default: op

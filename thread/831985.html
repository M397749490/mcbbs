  1. #聊天字体符号
  2. chatFormatSymbol: '&'
  3. #是否启用后台说话前缀,设置为false时为不启用
  4. overrideSayCommand: true
  5. enableSignBroadcast: true
  6. signBroadcastTool: SIGN

  8. prefix:
  9. #控制台使用say时说话的样子
  10.   console: '&o&6[控制台]&r'
  11. #玩家默认使用say时说话的样子
  12.   default: '[&l默认消息]&r'
  13. #玩家使用say时说话的样子
  14.   players:
  15.     CB_CCL: '&o[CB_CCL]&r' .

  17. #自动更新检查
  18. updateCheck: true
复制代码

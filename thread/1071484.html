同时，对脚本进行二次开发也很简单，只需创建一个继承MinecraftServer的类即可。
代码进行语法高亮如下：
# coding: utf-8
import subprocess as subp
import io


class MinecraftServer:
    def __init__(self, jar, java="java"):
        self.jar_path = jar
        self.java = java or "java"
        self.process = None

    def start(self):
        self.process = subp.Popen(
            [self.java, "-jar", self.jar_path],
            stdin=subp.PIPE,
            stdout=subp.PIPE,
            stderr=subp.PIPE,
        )

    @property
    def stdout(self) -> io.TextIOWrapper:
        if not self.process:
            return "服务器未启动，请启动服务器后执行命令！"
        return self.process.stdout

    @property
    def stderr(self) -> io.TextIOWrapper:
        if not self.process:
            return "服务器未启动，请启动服务器后执行命令！"
        return self.process.stderr

    @property
    def stdin(self) -> io.TextIOWrapper:
        if not self.process:
            return "服务器未启动，请启动服务器后执行命令！"
        return self.process.stdin

    def exec_command(self, command, get_output=False):
        if not self.process:
            return "服务器未启动，请启动服务器后执行命令！"
        self.stdin.write(command)
        self.stdin.write("\n")
        self.stdin.flush()
        if get_output:
            return self.stdout.read()
        else:
            return None


class Spigot(MinecraftServer):
    def __init__(self, jar, java="java"):
        super(self, MinecraftServer).__init__(jar, java)


class BungeeCord(MinecraftServer):
    def __init__(self, jar, java="java", servers=None):
        super(self, MinecraftServer).__init__(jar, java)
        self.servers = servers or []

    def start(self):
        for server in self.servers:
            server.start()
        return super(self, MinecraftServer).start()

    def add_server(self, server: MinecraftServer):
        self.servers.append(server)

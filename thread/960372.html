标题: org.omg.CORBA包下的各种Holder类是个好东西
RT，众所周知java只支持按引用传递，而利用这个Holder类，可以实现类似于按引用传递的功能，为我们的编程开发提供了很大的便利。代码见隐藏内容

  1. package com.github.polaristella.demos;

  3. import org.omg.CORBA.*;

  5. public class HolderDemo
  6. {
  7.         public static void main(String[] args)
  8.         {
  9.                 int x = 5;
  10.                 IntHolder xHolder = new IntHolder(5);
  11.                 notUseHolder(x);
  12.                 useHolder(xHolder);
  13.                 System.out.println("Not use holder:" + x +"\nUse holder:" + xHolder.value);
  14.         }

  16.         private static void notUseHolder(int x)
  17.         {
  18.                 x *= 5;
  19.         }

  21.         private static void useHolder(IntHolder x)
  22.         {
  23.                 x.value *= 5;
  24.         }
  25. }

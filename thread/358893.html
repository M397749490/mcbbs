建议所有1.6.4~1.7.10的腐竹都安装这个插件！明明装了假矿插件、装了反矿追插件，结果玩家才来一天，就一大箱子的钻石块，是否感到疑惑不解？
近日利用服务器BUG无限刷物品的方法广为流传，影响范围包括1.6.4~1.7.10的所有纯净、MOD服务器。而方法简单到令人发指，仅需*********************，部分无法卡bug的服务器仅需额外的一块红石即可继续无限刷物品，危害十分之大。于是制作了这个反作弊插件共享出来，共同抵御熊孩子！


1. 插件并非禁止使用树苗或者骨粉
2. 安装插件后依然可以用骨粉对橡树催熟
3. 只有当符合作弊条件的情况下使用骨粉，才会禁止催熟，不影响正常游戏

以下为源代码，省得别有用心的人诬陷我植入后门
防御原理很简单，玩家用骨粉催熟树苗时，判断树苗上方是否有方块遮挡，遮挡则取消事件。不但可以防御现在的这个BUG，还能防御利用树苗破坏别人房屋。


  1. import org.bukkit.ChatColor;
  2. import org.bukkit.Location;
  3. import org.bukkit.Material;
  4. import org.bukkit.World;
  5. import org.bukkit.entity.Player;
  6. import org.bukkit.event.EventHandler;
  7. import org.bukkit.event.EventPriority;
  8. import org.bukkit.event.Listener;
  9. import org.bukkit.event.block.Action;
  10. import org.bukkit.event.player.PlayerInteractEvent;
  11. import org.bukkit.plugin.java.JavaPlugin;

  13. public class NoBonemealCheat extends JavaPlugin implements Listener {
  14.         public void onEnable() {
  15.                 getServer().getPluginManager().registerEvents(this, this);
  16.                 getLogger().info("=========================");
  17.                 getLogger().info("NoBonemealCheat is enable!");
  18.                 getLogger().info("反骨粉无限刷物品插件成功开启!");
  19.                 getLogger().info("=========================");
  20.         }

  22.         public void onDisable() {
  23.                 getLogger().info("NoBonemealCheat is disable!");
  24.                 getLogger().info("反骨粉无限刷物品插件成功关闭!");
  25.         }

  27.         @SuppressWarnings("deprecation")
  28.         @EventHandler(priority = EventPriority.HIGHEST, ignoreCancelled = true)
  29.         // 监听
  30.         public void AntiCheat(PlayerInteractEvent e) {
  31.                 // 是否已经取消
  32.                 if(e.isCancelled()){
  33.                         return;
  34.                 }
  35.                 // 是否是右键方块
  36.                 if(e.getAction() == Action.RIGHT_CLICK_BLOCK){
  37.                         Player p = e.getPlayer();
  38.                         // 手上拿的是否是骨粉
  39.                         if(!p.getItemInHand().getType().equals(Material.INK_SACK)){
  40.                                 return;
  41.                         }
  42.                         int typeid = e.getClickedBlock().getTypeId();
  43.                         // 点击的是否不为空气
  44.                         if(typeid == 0){
  45.                                 return;
  46.                         }
  47.                         // 点击的是否为树苗
  48.                         if(typeid == 6 || typeid == 2175){
  49.                                 Location loc = e.getClickedBlock().getLocation();
  50.                                 World world = e.getClickedBlock().getWorld();
  51.                                 int x = loc.getBlockX();
  52.                                 int y = loc.getBlockY() + 1;
  53.                                 int z = loc.getBlockZ();
  54.                                 Location top_loc = new Location(world,x,y,z);
  55.                                 // 树苗上方是否有方块阻挡
  56.                                 if(top_loc.getBlock().getTypeId() != 0){
  57.                                         // 取消事件
  58.                                         e.setCancelled(true);
  59.                                         // 警告
  60.                                         p.sendMessage(ChatColor.RED + "[系统提示]树苗上方有方块阻挡，你无法对它使用骨粉！");
  61.                                 }
  62.                         }
  63.                 }
  64.                
  65.         }
  66. }

数据库安装
  1. use-mysql: true                                  #是否使用MySQL
  2. table-prefix: co_                                 #数据前缀
  3. mysql-host: localhost                          #数据库地址 localhost表示本地
  4. mysql-port: 3306                               #数据库端口
  5. mysql-database: database                   #数据库名
  6. mysql-username: root                         #数据库登陆名
  7. mysql-password: 123456                    #数据库密码

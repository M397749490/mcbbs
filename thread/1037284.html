  3. import (

  12. func main() {
  13.         f1, err := os.Open("./1.txt")
  14.         defer f1.Close()
  15.         if err != nil {
  16.                 fmt.Println(err)
  17.                 return
  18.         }
  19.         read1 := bufio.NewScanner(f1)
  20.         var t []string
  21.         for read1.Scan() {
  22.                 if strings.Contains(read1.Text(), "{{code|") {
  23.                         start := strings.Index(read1.Text(), "{{code|") + 7
  24.                         end := strings.Index(read1.Text()[start:], "}}")
  25.                         t = append(t, read1.Text()[start:][:end])
  26.                 }
  27.         }
  28.         f2 := files("2.txt")
  29.         defer f2.Close()
  30.         f3 := files("3.txt")

  32.         for id := range t {
  33.                 id := id
  34.                 io.WriteString(f2, writeid(t[id]))
  35.                 io.WriteString(f3, awriteid())
  36.                 i++
  37.         }
  38. }

  40. var i int

  42. func awriteid() string {
  43.         a := "scoreboard players add @a[scores={loymine_" + strconv.Itoa(i) + "=1..}] loymine_add 1\n"
  44.         b := "scoreboard players set @a[scores={loymine_" + strconv.Itoa(i) + "=1..}] loymine_" + strconv.Itoa(i) + "\n"
  45.         return a + b
  46. }
  47. func writeid(id string) string {
  48.         return "scoreboard objectives add " + "loymine_" + strconv.Itoa(i) + " minecraft.mined:" + id + "\n"
  49. }
  50. func files(filesname string) *os.File {

  52.         if err != nil {
  53.                 panic(err)
  54.         }
  55.         return f
  56. }

作者: CR_019    时间: 2020-7-22 10:28
建议把每个计分板的设定单列成一个计分板比如：
data/loymine/functions/loymineload/loymine_0.mcfunction:
  1. scoreboard objectives add loymine_0 minecraft.mined:acacia_button
复制代码
然后用一个load函数将这几百条函数运行一遍

因为错误的函数或未知函数在tag里面运行会被认定为语法错误而导致整个tag无法运行，而在另一个mcfunction文件里用function命令调用则不会，使用之顶多这条命令没有任何效果。这样就可以解决版本兼容问题，一个数据包兼容1.13以上版本

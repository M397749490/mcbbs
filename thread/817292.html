@echo OFF
title   『凛冬』◆服务器
SET BINDIR=%~dp0
:_MENU
CLS
set tm1=%time:~0,2%
set tm2=%time:~3,2%
set tm3=%time:~6,2%
echo                    现在是%date% %tm1%点%TM2%分
@Echo. *****************************************************************
@echo.                                                                                               
@echo.                    服务器即将开启,请等待......  
@echo.
@echo.             注意:关服请输入stop         
@echo.               
@echo.*****************************************************************
@echo.
@echo.请按任意按键来启动服务器！！！
@echo.
pause
cls
@echo.                  现在是%date% %tm1%点%TM2%分
@echo. *****************************************************************
@echo.
@echo.
@echo.                   服务器正在启动中,请稍候......
@echo.
@echo.
@echo. *****************************************************************
java -Xms2048m -Xmx2048m -jar cauldron-1.7.10-server.jar
@echo. *****************************************************************
@echo.                 提示 --- 关服成功！                                
@echo. *****************************************************************
pause
EXIT

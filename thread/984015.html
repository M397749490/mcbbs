安装Random Patches模组，其配置文件95-114行可以修改窗口图标及标题

  1.     window {
  2.         # The path to the 16x16 Minecraft window icon.
  3.         # Leave this and the 32x32 icon blank to use the default icon.
  4.         # Default:
  5.         S:icon16=

  7.         # The path to the 32x32 Minecraft window icon.
  8.         # Leave this and the 16x16 icon blank to use the default icon.
  9.         # Default:
  10.         S:icon32=

  12.         # The path to the 256x256 window icon which is used on Mac OS X.
  13.         # Leave this, the 16x16 icon and the 32x32 icon blank to use the default icon.
  14.         # Default:
  15.         S:icon256=

  17.         # The Minecraft window title.
  18.         # Default: Minecraft 1.12.2
  19.         S:title=Minecraft 1.12.2
  20.     }

  2. public static IService getService(ClassLoader classLoader) {
  3.     if (classLoader == null) {
  4.         classLoader = Thread.currentThread().getContextClassLoader();
  5.         // 当没有指定 classLoader 的时候, 大部分(不排除有部分弱智库不这样做) 库
  6.         // 都会使用 Thread.currentThread().getContextClassLoader() 作为目标 ClassLoader
  7.         //
  8.         // 由于 Java 的类加载本身并不依赖 Thread.currentThread().getContextClassLoader(),
  9.         // 而是依赖需要使用的类他自己的唯一类加载器, 所以, 线程上下文加载器你设置成什么, 对java本身并没有任何影响
  10.         // 换句话来说, 线程上下文类加载器无法影响类本身的加载机制
  11.         // 上下文类加载器只是各种类库的缺省设置
  12.         // 这个值影响的只有类库的加载
  13.         //
  14.         // 对于 Bukkit Plugin System
  15.         // Bukkit 加载插件的时候, 并不会去修改 Thread.currentThread().getContextClassLoader(),
  16.         // 也就是说, 默认情况下, contextClassLoader = ClassLoader.getSystemClassLoader()
  17.         //
  18.         // 此时, SystemCL 没法找到 PluginCL,
  19.         // 你需要临时把上下文加载器设置成 PluginCL
  20.         // 然后, 为了避免系统不稳定性, 我们在调用完毕后需要设置回去
  21.         //
  22.         // 就像下面这样
  23.         /*
  24.         ClassLoader old = Thread.currentThread().getContextClassLoader();
  25.         try {
  26.             Thread.currentThread().setContextClassLoader(Bootstrap.class.getClassLoader());
  27.             // Service loading.
  28.         } finally {
  29.             Thread.currentThread().setContextClassLoader(old);
  30.         }
  31.         */
  32.     }

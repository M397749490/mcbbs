  4. if(a > b)
  5.   test(c, a, b);
  6. else
  7.   printf("%d,%d,%d\n", a, b, c);
  8. }
  9. int main()
  10. {
  11. int a, b, c;

题目2

  1. #include<stdio.h>
  2. #define MAX 100
  3. void solve(char first[], int spos_f, int epos_f, char mid[], int spos_m, int epos_m)
  4. {
  5. int i, root_m;
  6. if(spos_f > epos_f)
  7.   return;
  8. for(i = spos_m; i <= epos_m; i++)
  9.   if(first[spos_f] == mid[i])
  10.   {
  11.    root_m = i;
  12.    break;
  13.   }
  14. solve(first, spos_f + 1, spos_f + (root_m - spos_m), mid, spos_m, root_m - 1);
  15. solve(first, spos_f + (root_m - spos_m) + 1, epos_f, mid, root_m + 1, epos_m);
  16. printf("%c", first[spos_f]);
  17. }
  18. int main()
  19. {
  20. char first[MAX], mid[MAX];
  21. int len;
  22. scanf("%d", &len);
  23. scanf("%s", first);
  24. scanf("%s", mid);
  25. solve(first, 0, len - 1, mid , 0, len - 1);
  26. printf("\n");
  27. return 0;
  28. }

build.xml的内容可以参考官方wiki，网址http://mcp.ocean-labs.de/index.php/Ant_Builds_with_MCP。
  1. <?xml version="1.0" encoding="UTF-8" ?>
  2. <project name="Sample" basedir="." default="all">

  4.    <!-- Your name here -->
  5.    <property name="author" value="C. Reeper" />

  7.    <!-- Your release version here -->
  8.    <property name="version" value="1.0.0" />

  10.    <!-- Choose ".zip" or ".jar" as the dist file extension -->
  11.    <property name="filetype" value="zip" />

  13.    <!-- Update if mcp directory isn't in same location as this file -->
  14.    <property name="mcp.dir" location="./mcp" />

  16.    <!-- Update if dist directory isn't in same location as this file -->  
  17.    <property name="dist.dir" location="./dist" />

  19.    <!-- Update only if build directory isn't in same location as this file -->
  20.    <property name="build.dir" location="./build" />

  22.    <!-- Update to point to your Python executable. Example: /usr/bin/python -->
  23.    <!-- The following location is for Windows: -->
  24.    <property name="python" location="${mcp.dir}/runtime/bin/python/python_mcp.exe" />
  25.    <target name="init" description="Initialize build environment">
  26.       <echo message="Cleaning old build and dist, MCP reobf directories" />
  27.       <delete dir="${build.dir}" />
  28.       <delete dir="${dist.dir}" />
  29.       <delete dir="${mcp.dir}/reobf" />

  31.       <echo message="Creating build and dist dirs" />
  32.       <mkdir dir="${build.dir}" />
  33.       <mkdir dir="${dist.dir}" />
  34.    </target>

  36.    <target name="recompile" description="MCP recompile" >
  37.       <exec executable="${python}" dir="${mcp.dir}">
  38.           <arg value="runtime/recompile.py"/>
  39.           <arg value="%*"/>
  40.       </exec>
  41.    </target>

  43.    <target name="reobfuscate" description="MCP reobfuscate">
  44.       <exec executable="${python}" dir="${mcp.dir}">
  45.           <arg value="runtime/reobfuscate.py"/>
  46.           <arg value="%*"/>
  47.       </exec>
  48.    </target>

  50.    <target name="stage" description="Stage resources into build directory">
  51.       <copy todir="${build.dir}">
  52.          <fileset dir="${mcp.dir}/reobf/minecraft" excludes=".svn/**, *.txt" />
  53.       </copy>
  54.       <!-- Copy other stuff specific to your project -->
  55.    </target>

  57.    <target name="all" description="Perform all tasks for a build" depends="init, recompile, reobfuscate, stage, jarzip">
  58.       <echo message="MCP Ant Build Completed!" />
  59.    </target>

  61.    <target name="jarzip" description="Jar (or zip) the project's dist output">
  62.       <mkdir dir="${dist.dir}" />
  63.       <jar destfile="${dist.dir}/${ant.project.name}${version}.${filetype}" duplicate="preserve" index="true" manifestencoding="UTF-8">
  64.          <manifest>
  65.             <attribute name="Built-By" value="MCP (http://mcp.ocean-labs.de)"/>
  66.             <attribute name="Implementation-Vendor" value="${author}"/>
  67.             <attribute name="Implementation-Title" value="${ant.project.name}"/>
  68.             <attribute name="Implementation-Version" value="${version}"/>
  69.          </manifest>
  70.          <fileset dir="${build.dir}" />
  71.       </jar>
  72.    </target>

  74. </project>

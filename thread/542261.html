个人用的是水龙头1.8.8api开发的
// 方块为玩家交互事件中点击的方块，当然其他实现也可以
Block chest = event.getClickedBlock();
net.minecraft.server.v1_8_R3.World world = ((org.bukkit.craftbukkit.v1_8_R3.CraftWorld)chest.getWorld()).getHandle();
net.minecraft.server.v1_8_R3.BlockPosition bp = new net.minecraft.server.v1_8_R3.BlockPosition(chest.getX(), chest.getY(), chest.getZ());
net.minecraft.server.v1_8_R3.TileEntityChest tec = (net.minecraft.server.v1_8_R3.TileEntityChest)world.getTileEntity(bp);
world.playBlockAction(bp, tec.w(), 1, 1);  // 参数4的值1为打开，0为关闭


// 过段时间后关闭箱子
Bukkit.getServer().getScheduler().runTaskLater(主类, new Runnable() {
        @Override
        public void run() {
                // 关闭箱子
                world.playBlockAction(bp, tec.w(), 1, 0);  // 值为0，则关闭箱子
        }
}, 秒 * 20L);  // minecraft 中的tick，1秒为 20tick

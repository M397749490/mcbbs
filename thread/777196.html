#需要 SkQuery (附属)

on right click with grass block:
        {Grass::%player%} == NULL:
                set {Grass::%player%} to true
                wait 1 minutes
                player is online:
                        give a grass block to player
                        delete {Grass::%player%}
on any movement:
        {Grass::%player%} == NULL:
                stop
        cancel event

on tool change:
        {Grass::%player%} == NULL:
                stop
        cancel event

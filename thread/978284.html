  2. execute @e[type=armor_stand,tag=mark] ~ ~ ~ function minecraft:next unless @s[score_clock=-1,score_clock_min=-1]
  3. execute @e[type=armor_stand,tag=mark] ~ ~ ~ function minecraft:play unless @s[score_clock=-1,score_clock_min=-1]

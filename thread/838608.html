全球游戏规则
此模组可以在客户端/服务端加载时自动对所有的世界进行规则修改
无需再一一修改，例如要让每个世界都死亡不掉落，苦力怕不破坏地形
关闭火焰传播，禁止观察者生成地形等等服务器必改的选项
某些服务器加了些跟维度有关的模组或自己另外创建了其他世界
导致维度很多，一个个修改太麻烦了，本模组就可以一次性全部修改好。

配置文件示例

  1. # Configuration file

  3. ##########################################################################################################
  4. # gamerules
  5. #--------------------------------------------------------------------------------------------------------#
  6. # Set the values to ('true'/'false'/an integer or a string defaultValue) depending if you want to have the GameRule (enabled/disabled or have a different defaultValue)
  7. ##########################################################################################################

  9. gamerules {
  10.     # If an announcement when a player gets an advancement should be done
  11.     B:announceAdvancements=true

  13.     # Whether command blocks should notify admins when they perform commands
  14.     B:commandBlockOutput=true

  16.     # Whether the server should skip checking player speed when the player is wearing elytra.
  17.     B:disableElytraMovementCheck=false

  19.     # Whether the day-night cycle and moon phases progress
  20.     B:doDaylightCycle=true

  22.     # Whether entities that are not mobs should have drops
  23.     B:doEntityDrops=true

  25.     # Whether fire should spread and naturally extinguish
  26.     B:doFireTick=true

  28.     # Whether players should only be able to craft recipes that they've unlocked first
  29.     B:doLimitedCrafting=false

  31.     # Whether mobs should drop items
  32.     B:doMobLoot=true

  34.     # Whether mobs should naturally spawn
  35.     B:doMobSpawning=true

  37.     # Whether blocks should have drops
  38.     B:doTileDrops=true

  40.     # Whether the weather will change
  41.     B:doWeatherCycle=true

  43.     # The function to run every game tick
  44.     S:gameLoopFunction=-

  46.     # Whether the player should keep items in their inventory after death
  47.     B:keepInventory=false

  49.     # Whether to log admin commands to server log
  50.     B:logAdminCommands=true

  52.     # Determines the number at which the chain command block acts as a "chain".
  53.     I:maxCommandChainLength=65536

  55.     # The maximum number of other pushable entities a mob or player can push, before taking 3 suffocation damage
  56.     # per half-second. Setting to 0 disables the rule. Damage affects survival-mode or adventure-mode players, and all mobs but bats.
  57.     # Pushable entities include non-spectator-mode players, any mob except bats, as well as boats and minecarts.
  58.     I:maxEntityCramming=24

  60.     # Whether creepers, zombies, endermen, ghasts, withers, ender dragons, rabbits, sheep, and villagers should be able to change blocks
  61.     # and whether villagers, zombies, skeletons, and zombie pigmen can pick up items
  62.     B:mobGriefing=true

  64.     # Whether the player can regenerate health naturally if their hunger is full enough (doesn't affect external healing, such as
  65.     # golden apples, the Regeneration effect, etc.)
  66.     B:naturalRegeneration=true

  68.     # How often a random block tick occurs (such as plant growth, leaf decay, etc.) per chunk section per game tick. 0 will disable random
  69.     # ticks, higher numbers will increase random ticks
  70.     I:randomTickSpeed=3

  72.     # Whether the debug screen shows all or reduced information; and whether the effects of F3+B (entity hitboxes) and F3+G (chunk boundaries) are shown.
  73.     B:reducedDebugInfo=false

  75.     # Whether the feedback from commands executed by a player should show up in chat. Also affects the default behavior of whether command blocks store their output text
  76.     B:sendCommandFeedback=true

  78.     # Whether a message appears in chat when a player dies
  79.     B:showDeathMessages=true

  81.     # The number of blocks outward from the world spawn coordinates that a player will spawn in when first joining a server or when dying without a spawnpoint.
  82.     I:spawnRadius=10

  84.     # Whether players in spectator mode can generate chunks
  85.     B:spectatorsGenerateChunks=true
  86. }


  89. ##########################################################################################################
  90. # misc
  91. #--------------------------------------------------------------------------------------------------------#
  92. # A collection of misc configs
  93. ##########################################################################################################

  95. misc {
  96.     # Syntax is: name-enabled-command
  97.     # enabled is either true or false, all @p will be replaced with the playername
  98.     S:customGamerules <
  99.         GGRExample-false-/tell @p Hello World
  100.      >

  102.     # If gamerules and world difficulty should be saved to config on world unload
  103.     B:saveGameRules=true

  105.     # Sets the difficulty of a world when loaded, respects it the difficulty is locked or not for the world
  106.     # -1: Disabled
  107.     # 0: Peaceful
  108.     # 1: Easy
  109.     # 2: Normal
  110.     # 3: Hard
  111.     I:worldDifficulty=-1

  113.     # If a world's difficulty should be locked when loaded, if world already is locked it can't be change
  114.     # If the global world difficulty is enabled it's set first
  115.     B:worldDifficultyLocked=false
  116. }

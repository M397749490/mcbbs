翻译软件翻译  将就看吧！！

\plugins\AuthMe\config.yml

______________________________________________________

DataSource:
    # file or mysql
    backend: mysql
    # MySQL database name
    mySQLDatabase: authme
    # MySQL username
    mySQLUsername: authme
    # MySQL password
    mySQLPassword: '12345'
    # MySQL host
    mySQLHost: 127.0.0.1
    # MySQL port
    mySQLPort: '3306'
    # MySQL table name
    mySQLTablename: authme
    # name of the username column in the mysql table
    mySQLColumnName: username
    # name of the password column in the mysql table
    mySQLColumnPassword: password
    # name of the IP column in the mysql table
    mySQLColumnIp: ip
    # name of the lastlogin column in the mysql table
    mySQLColumnLastLogin: lastlogin
    # do you want caching? Only needed when the file
    # backend is used or the MySQL database is on
    # another host than the mc server
    caching: false
settings:
    sessions:

        #当设置为true  玩家IP将被绑定到该玩家在登录的用户名。IP和用户名不对应时无法登陆
        enabled: false

        # 多少分钟后会话超时，设置为0的时候 永远不超时
        timeout: 10

    restrictions:
        # 是否允许为注册玩家聊天
        allowChat: false

        # 是否把未注册玩家立即踢掉
        kickNonRegistered: false

        # 不能登录的玩家被传送到spawn？再他们登录后将被传送回正常位置
        teleportUnAuthedToSpawn: true

        # 名字最大允许长度
        maxNicknameLength: 20

        # 是否允许未注册玩家移动
        allowMovement: false

        # 名字最短长度
        minNicknameLength: 3

        # 无法登录或注册在多少分钟后被踢。设置为0来禁用。

        timeout: 30

        # 未注册用户能移动多远。设置0为无限远。
        allowedMovementRadius: 100

    security:
        # 加密可能的值：MD5，SHA1，SHA256，XAuth
        passwordHash: SHA256

    registration:
        # 是否启用注册？
        enabled: true

        # 每X秒发送提醒玩家要登录/注册
        messageInterval: 5

        # 只有注册和登录的玩家可以玩。未注册的只能看。
        force: true

        # 不能登录同时在线的用户名
        ForceSingleSession: false

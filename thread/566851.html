  1. Email:
  2.     # 电子邮件的 SMTP 服务器主机
  3.     mailSMTP: smtp.gmail.com
  4.     # 电子邮件的 SMTP 服务器端口
  5.     mailPort: 465
  6.     # 电子邮件账户，用于发送邮件
  7.     mailAccount: ''
  8.     # 邮箱账户的密码
  9.     mailPassword: ''
  10.     # 自定义发送人名称, 会替代电子邮件账户显示
  11.     mailSenderName: ''
  12.     # 随机密码长度
  13.     RecoveryPasswordLength: 8
  14.     # 邮件主题
  15.     mailSubject: 'Your new AuthMe Password'
  16.     # 邮件内容
  17.     mailText: 'Dear <playername>, <br /><br /> This is your new AuthMe password for the server <br /><br /> <servername> : <br /><br /> <generatedpass><br /><br />Do not forget to change password after login! <br /> /changepassword <generatedpass> newPassword'

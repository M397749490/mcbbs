测试方案设计：
第一组实验：楼主计划用 pthreads 开 50 个线程，使用 while 循环，循环之前先使用 microtime 获得线程启动的精确时间，并用变量计算出循环的次数，每循环一次变量的值加1。循环5秒后输出计算每秒的循环次数的平均值
第二组实验：继续开 50 个线程，但是这次使用随机函数+sin正弦三角函数以增加计算量
第三组实验：用 pthreads 开 100 个线程，使用 while 循环，循环之前先使用 microtime 获得线程启动的精确时间，并用变量计算出循环的次数，每循环一次变量的值加1。最后输出计算每秒的循环次数的平均值 (不加上 sin)
第四组实验：开 100 个线程，加上随机与sin

测试脚本源代码：
GitHub地址：https://github.com/SuperMarcus/PHPSpeedTester
第一组实验：

<?php
class imtry extends Thread{
    private $stop, $time;

    public function __construct(){}
    public function run(){
        $this->time = microtime(true);
        echo "Thread: ".$this->getThreadId()." is running.\n";
        $this->stop = false;
        $this->loop = 0;
        while($this->stop === false){
            $this->loop++;
        }
    }
    public function stop(){
        $this->stop = true;
        echo "Thread: ".$this->getThreadId()." ticks: ".($this->loop / (microtime(true) - $this->time))."ticks/sec.\n";
        usleep(10);
        var_dump($this->join());
    }
}

    $threads = [];

    while(count($threads) < 50){
        $threads[] = new imtry();
    }

    foreach($threads as $thread){
        $thread->start();
    }

    sleep(10);

    foreach($threads as $thread){
        $thread->stop();
    }
?>

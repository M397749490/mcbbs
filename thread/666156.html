  2.   # 总体的物理现象或行为的设置

  5.     water-flow: true                      # 阻止 水的流动
  6.     water-bucket-flow: true               # 阻止 水桶水的流动
  7.     fire-spread: true                      # 阻止 火的传播
  8.     lava-fire-spread: true                 # 阻止 岩浆引起的火的传播
  9.     flint-fire: false                      # 阻止 打火石产生火
  10.     lightning-fire-spread: true            # 阻止 闪电引起的火的传播
  11.     portal-creation: false                 # 阻止 传送门的创造

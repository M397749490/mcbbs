插件下载地址：https://poggit.pmmp.io/p/GuessTheNumber
本贴不提供任何插件下载只提供汉化文件



介绍：

插件支持以下游戏运算符
加法，减法，除法，乘法，次方，阶乘
支持查看某局游戏答案

  1. #这个我就不翻译了，我觉得大家都能看懂23333
  2. language: eng

  4. #玩家输入答案后要等多久之后才能验证答案
  5. #默认3秒
  6. timer: 3

  8. #若回答正确某个运算方式后奖励物品
  9. #注意物品格式“物品ID:数据值:数量”
  10. #例：12:0:1指的是回答正确后给予物品“12”，数量"1"个
  11. #例子：35:2:2指的是回答正确后给予物品“35:2”，数量"2"个
  12. prizes:
  13.     #命令[/numbergame 随机数模式]的获奖物品设置
  14.     randomIntegerItems: ['265:0:10']

  16.     #命令[/numbergame 次方模式]的获奖物品设置
  17.     exponentItems: ['322:0:1']

  19.     #命令[/numbergame 加法模式]的获奖物品设置
  20.     additionItems: ['351:4:22']

  22.     #命令[/numbergame 减法模式]的获奖物品设置
  23.     subtractionItems: ['264:0:1']

  25.     #命令[/numbergame 乘法模式]的获奖物品设置
  26.     multiplicationItems: ['264:0:1']

  28.     ##命令[/numbergame 除法模式]的获奖物品设置
  29.     divisionItems: ['264:0:1']

  31.     #命令[/numbergame 阶乘模式]的获奖物品设置
  32.     factorialItems: ['388:0:1']

  34. #命令[/numbergame 随机数]模式随机数最小和最大设置
  35. randomInteger:
  36. #最小的数
  37.     min: 1
  38. #最大的数
  39.     max: 100

  41. #命令[/numbergame 次方模式]设置
  42. # (数)^指数
  43. #以下为解释:
  44. # base = 5表示数设置的5
  45. # exponent = 2表示次方为2
  46. #上述介绍
  47. # (5)^2 = 25
  48. exponent:
  49. #指数最小值
  50.     exponentMin: 0
  51. #指数最大值
  52.     exponentMax: 6
  53. #数的最小值
  54.     baseIntMin: 1
  55. #数的最大值
  56.     baseIntMax: 20

  58. #命令[/numbergame 加法模式]设置
  59. addition:
  60. #加法的数子最小值
  61.     min: 0
  62. #加法的数字最大值
  63.     max: 2000

  65. subtraction:
  66.     min: 0
  67.     max: 2000
  68.     allowNegativeSolution: false

  70. multiplication:
  71.     min: 0
  72.     max: 100

  74. # dividend / divisor
  75. division:
  76.     dividendMin: 1
  77.     dividendMax: 1000
  78.     divisorMin: 1
  79.     divisorMax: 10

  81. #阶乘设置
  82. # 5! = 5*4*3*2*1 = 120
  83. factorial:
  84. #阶乘数最小值
  85.     min: 0
  86. #阶乘数最大值
  87.     max: 7

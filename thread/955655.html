function doyo_front:uid

  2. #将 player_info 存到 player_info_store
  3. execute store result storage doyo_front:default player_info_transfer.Order long 0.5 run function doyo_front:player/order
  4. #递归, 结果 × 0.5 存到 player_info_transfer.Order
  5. execute store result storage doyo_front:default player_info_transfer.Uid long 1 run data get storage doyo_front:default player_info_transfer.Order
  6. #将 player_info_transfer.Order 存到 player_info_transfer.Uid
  7. setblock ~ ~ ~ minecraft:acacia_sign{Text1:'{"selector":"@p"}'}
  8. #将玩家名字存到 Text1

  10. #将 Text1 存到 player_info_transfer.Name

  12. #将 player_info_transfer 追加到 player_info

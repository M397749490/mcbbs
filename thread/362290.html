  1. <div><span style="font-family: 宋体; line-height: 1.5;">blacklist:</span></div><font face="宋体">
  2.     # </font><font face="宋体">阻止玩家放置以下方块
  3.     placement: 10,11,46,327

  5.     # 阻止玩家使用以下物品（这个地方写上你的电脑的物品ID）
  6.     usage: 327

  8.     # 阻止玩家破坏以下方块
  9.     break:</font>

[注意事项]============
    [1]这款插件重制于RongTeam!
    [2]这款插件可能支持全版本服务器
    [3]欢迎提出Bug，拒绝伸手党!

[插件功能]============
  • 对于BedwarsRel插件支持
  • 语言信息可自定义
  • 队伍聊天系统
  • 队伍聊天设置
  • 队伍人数设置
  • MySQL设置

[命令权限]============
  - /zd plugin - 查看当前插件信息
  - /zd help - 查看当前插件帮助
  - /zd create - 创建队伍
  - /zd invite <玩家名字> - 邀请玩家进入自己的队伍
  - /zd accept - 同意加入队伍
  - /zd deny - 拒绝加入队伍
  - /zd kick <玩家名字> - 踢出玩家

  - NewTeam.admin - 管理员权限

[配置解读]============
MySQL:
#MySQL设置
  ip: 127.0.0.1:3306
  db: 'NewTeam'
  table: 'NewTeam'
  user: 'root'
  password: '123456'
Team:
#队伍设置
  Members: 3
  #队伍最大人数
  #发送邀请间隔
  wait: 30
TeamChat:
#队伍聊天设置
  Chat: '@'
  #触发队伍聊天符号
  ChatMessage: '&b队伍聊天 &7| &3%player% &6>> &f%message%'
  #队伍聊天格式
BedwarsRel:
#对于BedwarsRel的支持设置
  enable: true
  #是否开启对于BedwarsRel的支持设置
  TeamJoinGame: true
  #队长加入游戏则队员一起加入游戏
  TeamJoinTeam: false
  #队长选择队伍则队友选择跟队长一样的队伍
  TeamLeaveGame: true
  #队长离开游戏则队员一起离开游戏
Message:
#信息设置
  prefix: '&7[&b&l队伍系统&7] '
  #信息前缀
  plugin:
  #插件信息
  - '&e-=*&7[&b&l队伍系统&7]&e*=-'
  - '&f作者By: Yuhiy'
  - '&f版本: 3.0'
  - '&f命令: /zd'
  - '&f权限: NewTeam.admin'
  - '&f描述: 一款好用的组队插件'
  help:
  #插件帮助
  - '&e-=*&7[&b&l队伍系统&7]&e*=-'
  - '&f/zd plugin - 查看当前插件信息'
  - '&f/zd help - 查看当前插件帮助'
  - '&f/zd create - 创建队伍'
  - '&f/zd invite <玩家名字> - 邀请玩家进入自己的队伍'
  - '&f/zd accept - 同意加入队伍'
  - '&f/zd deny - 拒绝加入队伍'
  - '&f/zd kick <玩家名字> - 踢出玩家'
  create: '&a成功创建了队伍'
  #创建队伍
  inteam: '&c你已经在队伍里面了'
  #正在队伍里
  invitewait: '&a已经发送邀请信息了'
  #已发送邀请信息
  frominvite: '&a邀请组队信息已经发送给 &e%player% '
  #发送邀请信息
  toinvite:
  #接收邀请信息
  - '&a玩家 &e%player% &a邀请你加入他的队伍里'
  - '&e输入 &c/zd accept &e来接受邀请'
  - '&e输入 &c/zd deny &e来拒绝邀请'
  hasteam: '&c对方已经在其他队伍里了!'
  #已经拥有队伍
  notfoundplayer: '&c没有找到该玩家!'
  #没有找到玩家
  noteamowner: '&c你不是队长,你不能这样做!'
  #不是队长
  noteam: '&c你当前不在某个队伍里,&e输入 &c/zd create &e来创建队伍'
  #没有队伍
  notfoundcommand: '&c没有找到命令,输入&e /zd help &c查看帮助'
  #没有找到指令
  accept: '&a成功加入&e %player% &a的队伍里 &7(%now%/%max%)'
  #同意加入
  joinedteam: '&a玩家 &e%player% &a加入了队伍 &7(%now%/%max%)'
  #同意加入后
  fullteam: '&c该队伍已经满人了!'
  #队伍满人
  notinvite: '&c对方没有邀请你!'
  #没有邀请
  playernoteam: '对方没有队伍!'
  #对方没有队伍
  deny: '&a拒绝加入 &e%player% &a的队伍'
  #拒绝
  denyteam: '&c玩家&e %player% &c拒绝了邀请'

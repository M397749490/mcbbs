你看下我服务器的配置吧。这个是bat代码

你只需要设置前面几个应该就可以了。

goto restart这个是重启代码


@echo OFF
:restart
SET BINDIR=%~dp0
rem #######下面这个是设置服务器名字的########
set Name=倾世之城 - BC前置
rem #######下面这个是设置Jar文件名字的#######
set bin=BungeeCord.jar
rem #######下面这个是设置启动内存的##########
set Xms=128M
rem #######下面这个是设置最大内存的##########
set Xmx=1024M
set /a reloadnu=%reloadnu%+0
title 当前正常运行：%Name%  已经重启了丨%reloadnu%丨次
echo                   现在是%date% %time%
@echo *****************************************************************
@echo.
@echo.正在启动%Name%,请稍候........
@echo.当前启动内存为 丨%Xmx%
@echo.启动的bat文件是丨%bin%
@echo.当前时间：%date%丨%time%
@echo.
@echo *****************************************************************
echo ┌───────┬────┬────────┬────┬──────┐>>启动日志.txt
echo 丨自动启动服务器丨执行日期丨%DATE% 丨执行时间丨 %time%丨>>启动日志.txt
echo └───────┴────┴────────┴────┴──────┘>>启动日志.txt
java -Xms%Xms% -Xmx%Xmx% -jar %bin%
@echo *****************************************************************
@echo.
@echo.%Name%,已经崩溃或关闭
@echo.当前时间:%date% %time%
@echo.当前启动设置的内存为丨%Xmx%
@echo.本次启动的bat文件是 丨%bin%
@echo.
@echo *****************************************************************
ping -n 2 -w 200 0.0.0.1>nul
ping -n 2 -w 200 0.0.0.1>nul
set /a reloadnu=%reloadnu%+1
title 当前正常运行：%Name%  已经重启了丨%reloadnu%丨次
@echo.*****************************************************************
@echo.
@echo.正在为重新启动%Name%做准备...
@echo.准备完毕，正在重启，请稍后....
@echo.
@echo.*****************************************************************
goto restart
pause
EXIT

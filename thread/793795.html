用java的反射功能制作开服工具的时候发现存档位置会和jar包的位置相同,我想对他们分离
  1. import java.io.File;  
  2. import java.lang.reflect.Method;  
  3. import java.net.URL;  
  4. import java.net.URLClassLoader;

  6. public class Test {
  7.         @SuppressWarnings({ "unchecked", "rawtypes" })
  8.         public static void main(String[] args) throws Exception {
  9.         File file=new File(".\\serverfile\\Spigot-1.9.jar");//jar包的路径  
  10.         URL url=file.toURI().toURL();  
  11.         URLClassLoader urlClassLoader= (URLClassLoader) ClassLoader.getSystemClassLoader(); //获取系统加载器
  12.         Method add = URLClassLoader.class.getDeclaredMethod("addURL", new Class[] { URL.class });
  13.         add.setAccessible(true);
  14.         add.invoke(urlClassLoader, new Object[] {url});
  15.         Class c=urlClassLoader.loadClass("org.bukkit.craftbukkit.Main");//服务端主类的名字
  16.         Method mainmethod = c.getMethod("main",String[].class);
  17.         mainmethod.invoke(null,(Object)new String[]{});
  18.         }

  20. }

标题: sender.sendmessage问题
我想编一个插件，目的是输入/suicide时自杀，很简单。写完主类之后写命令执行器，代码如下：



package com.outlook.vincenthan2009;

import org.bukkit.command.CommandExecutor;
import org.bukkit.command.CommandSender;
import org.bukkit.entity.Player;

public class Command {       //声明类
       
        public boolean onCommand(CommandSender sender, Command cmd, String label, String[] args){ //OnCommand
                if (((CommandSender) cmd).getName().equalsIgnoreCase("suicide")){                     //判断命令
                        if (!(sender instanceof Player)) {                                                //判断是否为玩家
                                sender.sendMessage("该命令只能被玩家执行！");
                           return true;
                        }else{
                                Player player=(Player) sender;
                                player.setHealth(0.0);
                                return true;
                        }                                                                                 //判断是否为玩家
                }                                                                                     //判断命令
                return false;                                    
        }                                                                                         //OnCommand

}                          //声明类



以上为命令执行器的代码，不过我在服务端内输入/suicide时没有提示“该命令只能被玩家执行”，这是怎么回事？IDE

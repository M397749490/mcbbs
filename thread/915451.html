package cn.mcres.karlatemp.mxlib.test;

import org.bukkit.configuration.ConfigurationSection;
import org.bukkit.configuration.file.YamlConfiguration;
import org.bukkit.entity.Player;

import java.util.ArrayList;
import java.util.List;
import java.util.logging.Level;
import java.util.logging.Logger;

public class Booter {
    public static final List<Perm> perms = new ArrayList<>(); // 定义全部需要检测的列表

    public static final class Perm { //定义一个容器
        public String perm;
        public String msg;

        public Perm() {
            this(null, null);
        }

        public Perm(String perm, String msg) {
            this.perm = perm;
            this.msg = msg;
        }
    }

    public static final Logger logger = Logger.getGlobal(); // 这是日志系统

    public static void main(String[] args) {
        logger.setLevel(Level.ALL);
        reloadConfig();// 加载配置内容
        send(null); // 发送信息
    }

    public static void send(Player player) { // 运行监测程序
        for (Perm p : perms) { // 使用缓存的数据加快运行效率
            if (player.hasPermission(p.perm))
                player.sendMessage(p.msg);
        }
    }

    public static void reloadConfig() { // 重载/加载配置内容
        perms.clear(); // 清理缓存
        final ConfigurationSection config = new YamlConfiguration(); // 换成 getConfig

        final ConfigurationSection join = config.getConfigurationSection("join"); // 获取join节点
        if (join == null) { // 配置没有此节点
            logger.warning("Warming: No \"join\" in config.yml");
            return; // 中断
        }
        for (String k : join.getKeys(false)) {// 循环遍历 join 节点的全部子节点
            logger.fine("[DEBUG] 正在遍历的子节点为 [" + k + ']'); //Debug信息
            final Object value = join.get(k); // 取子节点的值
            if (!(value instanceof ConfigurationSection)) { // 如果不是树
                // join.get(k); 取原始值
                logger.warning("在 config.yml/join." + k + " 中含有错误的配置格式: " + value);
                continue; // 开始下一轮循环
            }
            final ConfigurationSection section = (ConfigurationSection) value;// 上面已经判断了。直接强转
            // 取得 join[k].permission 和 join[k].message 的值
            String perm = section.getString("Permission", null);
            String mess = section.getString("Message", null);
            if (perm == null) {
                logger.warning("在 config.yml/join." + k + " 中含有错误的配置格式: 缺少 \"Permission\"");
                continue;
            }
            if (mess == null) {
                logger.warning("在 config.yml/join." + k + " 中含有错误的配置格式: 缺少 \"Message\"");
                continue;
            }
            perms.add(new Perm(perm, mess));// 使用List缓存加快执行速度
        }
    }
}

就是如何获取到不同颜色的粒子和更多的粒子效果... 求助...

这是我的代码所使用的是org.bukkit.Effect
但只是单纯把org.bukkit.Effect 有的效果附上在玩家身上
我查了org.bukkit.Effect; 发现所写的Effect 很少
所描写的内容也不合乎MineCraft 内粒子效果数量

@EventHandler
public void Effect(PlayerMoveEvent PM){
Player p = PM.getPlayer();
if((p.hasPermission("P.Effect") &&
(this.plugin.hashmap.contains(PM.getPlayer().getName())))){
if (this.plugin.FLAME.contains(PM.getPlayer().getName())){
p.getWorld().playEffect(p.getLocation(), Effect.LAVA_POP, 5);
}if (this.plugin.SMOKE.contains(PM.getPlayer().getName())){
p.getWorld().playEffect(p.getLocation(), Effect.SMOKE, 5);
}if (this.plugin.WITCH.contains(PM.getPlayer().getName())){
p.getWorld().playEffect(p.getLocation(), Effect.WITCH_MAGIC, 5);
}if (this.plugin.PORTAL.contains(PM.getPlayer().getName())){
p.getWorld().playEffect(p.getLocation(), Effect.PORTAL, 5);
}if (this.plugin.HEART.contains(PM.getPlayer().getName())){
p.getWorld().playEffect(p.getLocation(), Effect.HEART, 5);
}if (this.plugin.ENDER.contains(PM.getPlayer().getName())){
p.getWorld().playEffect(p.getLocation(), Effect.ENDER_SIGNAL , 5);
}if (this.plugin.NOTE.contains(PM.getPlayer().getName())){
p.getWorld().playEffect(p.getLocation(), Effect.NOTE , 5 , 30);
}if (this.plugin.happyVillager.contains(PM.getPlayer().getName())){
p.getWorld().playEffect(p.getLocation(), Effect.HAPPY_VILLAGER , 5 , 30);
}if (this.plugin.LAVADRIP.contains(PM.getPlayer().getName())){
p.getWorld().playEffect(p.getLocation(), Effect.LAVADRIP , 5 , 30);
}if (this.plugin.THUNDERCLOUD.contains(PM.getPlayer().getName())){
p.getWorld().playEffect(p.getLocation(), Effect.VILLAGER_THUNDERCLOUD , 5 , 30);
}
}
}
}

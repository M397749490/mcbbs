  1. citizenscmd.* #管理员权限

  3. /npcmd  
  4. #查看所有指令 使用此指令的权限  citizenscmd.npcmd

  6. /npcmd add  <console/noPerms/custom*> 绑定的指令
  7. #给NPC绑定指令  使用此指令的权限  citizenscmd.add
  8. #例子
  9. #/npcmd add console fly %player%
  10. #右键后，后台会执行指令/fly 右键的人的名字
  11. #/npcmd add noPerms fly
  12. #右键后，拥有essentials.fly权限的人会执行指令fly，反之则提示无权限
  13. #/npcmd add fly.233 fly
  14. #右键后，拥有essentials.fly和fly.233权限的人会执行指令fly，反之则提示无权限

  16. /npcmd remove
  17. #删除NPC绑定的所有指令 使用此指令的权限 citizenscmd.remove

  19. /npcmd cooldown  秒数
  20. #添加冷却时间 使用此指令的权限 citizenscmd.cooldown
  21. #例子/npcmd cooldown 100
  22. #然后这个npc就会有100秒的冷却，从右键时开始算

  24. /npcmd reload
  25. #重载插件  使用此指令的权限 citizenscmd.reload

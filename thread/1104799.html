1.8的话spigot没有封装attribute，用nms改武器本身的attribute吧

  1.          ItemStack stack = new ItemStack(Material.IRON_SWORD);
  2.          net.minecraft.server.v1_8_R1.ItemStack nmsStack = CraftItemStack.asNMSCopy(stack);
  3.          NBTTagCompound compound = nmsStack.getTag();
  4.          if (compound == null) {
  5.             compound = new NBTTagCompound();
  6.              nmsStack.setTag(compound);
  7.              compound = nmsStack.getTag();
  8.          }
  9.          NBTTagList modifiers = new NBTTagList();
  10.          NBTTagCompound healthboost = new NBTTagCompound();
  11.          healthboost.set("AttributeName", new NBTTagString("generic.attackDamage"));
  12.          healthboost.set("Name", new NBTTagString("generic.attackDamage"));
  13.          healthboost.set("Amount", new NBTTagInt(20));
  14.          healthboost.set("Operation", new NBTTagInt(0));
  15.          healthboost.set("UUIDLeast", new NBTTagInt(894654));
  16.          healthboost.set("UUIDMost", new NBTTagInt(2872));
  17.          modifiers.add(healthboost);
  18.          compound.set("AttributeModifiers", modifiers);
  19.          nmsStack.setTag(compound);
  20.          stack = CraftItemStack.asBukkitCopy(nmsStack);

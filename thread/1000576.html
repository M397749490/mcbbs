每天自动备份worlds文件夹，方便回档
这里用的是pm2后台管理服务进程，算是screen之外又一个选择吧。
因为是bdlauncher启动的，所以停服是通过向进程发送stop命令，官服的话直接pm2 stop 0即可
上脚本

  1. #!/bin/bash

  3. #启动命令：pm2 start -x "./bdlauncher" --restart-delay=10000 -n mc
  4. #需要备份的世界名，多个文件夹用空格分开
  5. WDLEVEL='zzbd default'
  6. #pm2进程ID
  7. PID=0
  8. #备份路径（当前目录）
  9. BACKPATH=$(dirname $(readlink -f $0))

  11. #准备目录
  12. WDPATH="${BACKPATH}/worlds"
  13. BKPATH="${BACKPATH}/bak"
  14. [ ! -d "${WDPATH}" ] && exit 1
  15. [ ! -d "${BKPATH}" ] && mkdir -p "${BKPATH}"

  17. #停止服务器
  18. pm2 send $PID "stop" 2>> ${BKPATH}/worldbak.log
  19. [ $? -ne 0 ] && { echo "服务器通信异常!! 中止.." >> ${BKPATH}/worldbak.log; exit 1; }
  20. echo "备份时间为 `date +%Y-%m-%d/%H:%M:%S`,备份level为 ${WDLEVEL}" >> ${BKPATH}/worldbak.log
  21. #检查服务器是否关闭
  22. sleep 3
  23. NEXT_WAIT_TIME=0
  24. until [ "$(pm2 ls $PID -m | grep -c "status : waiting restart")" -ge 1 ] || [ $NEXT_WAIT_TIME -eq 4 ]; do
  25. sleep $(( NEXT_WAIT_TIME++ ))
  26. done
  27. [ $NEXT_WAIT_TIME -eq 4 ] && { echo "服务器关闭失败!! 中止.."; exit 1; }
  28. #正式备份
  29. cd ${BKPATH}
  30. for x in $WDLEVEL; do
  31. #世界不存在跳过
  32. [ ! -d "${WDPATH}/${x}" ] && continue
  33. #存档数据未更新跳过
  34. currtime=`date -r ${WDPATH}/${x} +%Y-%m-%d_%H%M%S`
  35. [ -n "`find -maxdepth 1 -name "${x}*.tar.gz"`" ] && lastime=`ls -t ${x}*.tar.gz | head -1` || lastime=''
  36. [ "${lastime}" == "${x}${currtime}.tar.gz" ] && continue
  37. #压缩文件夹
  38. tar fczP ${x}latest.tar.gz ${WDPATH}/${x} > /dev/null
  39. echo "世界 ${x} 备份$([ $? -eq 0 ] && echo "成功" || echo "失败")!!" >> worldbak.log
  40. #只保存前三个备份
  41. ls -A1t $(find -name "${x}*.tar.gz" -type f) | tail -n +4 | xargs rm > /dev/null 2>&1
  42. done
  43. #检查服务器是否重启
  44. sleep 5
  45. NEXT_WAIT_TIME=0
  46. until [ "$(pm2 ls $PID -m | grep -c "status : online")" -ge 1 ] || [ $NEXT_WAIT_TIME -eq 4 ]; do
  47. sleep $(( NEXT_WAIT_TIME++ ))
  48. done
  49. [ $NEXT_WAIT_TIME -eq 4 ] && { echo "服务器重启失败!! 中止.."; exit 1; }
  50. #备份文件名修改为当前存档时间戳
  51. for x in $WDLEVEL; do
  52. [ ! -f "${x}latest.tar.gz" ] && continue
  53. currtime=`date -r ${WDPATH}/${x} +%Y-%m-%d_%H%M%S`
  54. mv -f ${x}latest.tar.gz ${x}${currtime}.tar.gz
  55. done

放到服务器根目录，通过crontab每天自动执行，然后可以愉快的关掉作弊选项了
2020.4.13更新
增加三种模式，冷备份（备份时服务器会重启），热备份，游戏内控制台输入关键词触发备份，上脚本
  1. #!/bin/bash

  3. set -e
  4. PATH=/root/.nvm/versions/node/v12.16.1/bin:/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
  5. export PATH

  7. #参数默认值
  8. MAX_BACKUPS=4
  9. LOG_TO_FILE=false
  10. #需要备份的世界名，多个文件夹用空格分开
  11. WDLEVEL='zzbd default'
  12. #pm2进程ID
  13. PID=0
  14. #触发备份关键字
  15. BK_KEYWORD="开始备份！"
  16. #！！以上参数请根据实际情况修改！！#
  17. #================================#
  18. #服务器根路径（脚本所在目录）
  19. ROOTPATH=$(dirname $(readlink -f $0))
  20. #准备目录
  21. WDPATH="${ROOTPATH}/worlds"
  22. BKPATH="${ROOTPATH}/bak"
  23. [ ! -d "${WDPATH}" ] && exit 1
  24. [ ! -d "${BKPATH}" ] && mkdir -p "${BKPATH}"
  25. BKWORLD=$(grep ^level-name= "${ROOTPATH}/server.properties" | cut -d = -f 2)
  26. bklevel=""

  28. #读取参数
  29. while getopts 'm:S' OPT; do
  30.     case $OPT in
  31.         m)
  32.             MAX_BACKUPS="$OPTARG";;
  33.         S)
  34.             LOG_TO_FILE=true;;
  35.         *)
  36.             tishi
  37.     esac
  38. done
  39. shift $(($OPTIND - 1))

  41. tishi() {
  42.     echo "Script to backup Minecraft BDS worlds"
  43.     echo ""
  44.     echo "usage: beifen.sh [OPTION] MODE"
  45.     echo "   - hot mode: hold and resume server"
  46.     echo "   - cold mode: restart server"
  47.     echo "   - watch mode: waiting 'backup' command"
  48.     echo "   - init mode: create BDlauncher process with pm2"
  49.     echo "-S Save log to path instead of stdout"
  50.     echo "-m Set max number of backups, default 4"
  51.     exit 0
  52. }

  54. logtofile() {
  55.     [ -n "$1" ] && IN="$1" || read IN
  56.     if [ "$LOG_TO_FILE" = true ]; then
  57.         echo "[`date '+%Y-%m-%d %H:%M:%S'`] $IN" >> ${BKPATH}/worldbak.log
  58.     else
  59.         echo "$IN"
  60.     fi
  61. }
  62. sendtosvr() {
  63.     pm2 send $PID "$*" 2>&1 | logtofile
  64. }

  66. [ $# -eq 0 ] && tishi
  67. logtofile "开始备份，使用$1模式.."

  69. case $1 in
  70. ##cold mode##
  71. cold)
  72. cd ${BKPATH}
  73. #检查需要备份的level
  74. for x in $WDLEVEL; do
  75.     #世界不存在跳过
  76.     [ ! -d "${WDPATH}/${x}" ] && { logtofile "${x} 存档不存在!! 跳过.."; continue; }
  77.     #存档数据未更新跳过
  78.     currtime=`date -r ${WDPATH}/${x} +%Y-%m-%d_%H%M%S`
  79.     [ -n "`find -maxdepth 1 -name "${x}*.tar.gz"`" ] && lastime=`ls -t ${x}*.tar.gz | head -1` || lastime=''
  80.     [ "${lastime}" == "${x}${currtime}.tar.gz" ] && { logtofile "${x} 存档未更新!! 跳过.."; continue; }
  81.     bklevel="$bklevel $x"
  82. done
  83. [ -z "$bklevel" ] && { logtofile "存档无需备份!! 中止.."; exit 0; }
  84. #停止服务器
  85. sendtosvr stop
  86. [ $? -ne 0 ] && { logtofile "服务器通信异常!! 中止.."; exit 1; }
  87. #检查服务器是否关闭
  88. sleep 3
  89. NEXT_WAIT_TIME=0
  90. until pm2 show $PID | grep -q "status.*restart" || [ $NEXT_WAIT_TIME -eq 4 ]; do
  91.     sleep $(( NEXT_WAIT_TIME++ ))
  92. done
  93. [ $NEXT_WAIT_TIME -eq 4 ] && { logtofile "服务器停止失败!! 中止.."; exit 1; }
  94. #正式备份
  95. for x in $WDLEVEL; do
  96.     #压缩文件夹
  97.     tar fczP ${x}latest.tar.gz ${WDPATH}/${x} 2>&1 | logtofile
  98.     logtofile "世界 ${x} 备份$([ $? -eq 0 ] && echo "成功" || echo "失败")!!"
  99.     #只保存前n个备份
  100.     ls -A1t $(find -name "${x}*.tar.gz" -type f) | tail -n +$(( MAX_BACKUPS+1 )) | xargs rm 2>&1 | logtofile
  101. done
  102. #检查服务器是否重启
  103. sleep 5
  104. NEXT_WAIT_TIME=0
  105. until pm2 show $PID | grep -q "status.*online" || [ $NEXT_WAIT_TIME -eq 4 ]; do
  106.     sleep $(( NEXT_WAIT_TIME++ ))
  107. done
  108. [ $NEXT_WAIT_TIME -eq 4 ] && { logtofile "服务器重启失败!! 中止.."; exit 1; }
  109. #备份文件名修改为当前存档时间戳
  110. for x in $WDLEVEL; do
  111.     [ ! -f "${x}latest.tar.gz" ] && continue
  112.     currtime=`date -r ${WDPATH}/${x} +%Y-%m-%d_%H%M%S`
  113.     mv -f ${x}latest.tar.gz ${x}${currtime}.tar.gz
  114. done
  115. ;;

  117. ##watch mode##
  118. watch)
  119. #tail -Fn0 test.log | while read s; do echo $s | awk '/beifen/{print $1,$2} /left/{print $2}'; done
  120. tail -Fn0 player.log | { grep -q ${BK_KEYWORD} && logtofile "收到备份指令！" ; pkill -P $ '^tail
  121. [/spoiler]
  122. 欢迎大佬指正{:710:}

  124. ; }
  125. ;&
  126. ##hot mode##
  127. hot)
  128. #检查存档是否存在
  129. [ ! -d "${WDPATH}/${BKWORLD}" ] && { logtofile "${BKWORLD} 存档不存在!! 中止.."; exit 1; }

  131. sendtosvr "say 存档即将开始备份.."
  132. #挂起服务器
  133. trap 'sendtosvr "save resume"' ERR
  134. sendtosvr "save hold"
  135. #检查是否挂起
  136. NEXT_WAIT_TIME=1
  137. until echo `pm2 logs --out --nostream --lines 2 $PID` | grep -q 'Data saved' || [ $NEXT_WAIT_TIME -eq 4 ]; do
  138.     sendtosvr "save query"
  139.     sleep $(( NEXT_WAIT_TIME++ ))
  140. done
  141. [ $NEXT_WAIT_TIME -eq 4 ] && { logtofile "服务器挂起失败!! 中止.."; sendtosvr "save resume"; exit 1; }
  142. #开始备份
  143. files=$(pm2 logs --out --nostream --lines 2 $PID | grep -Eo "$BKWORLD[^:]+:[0-9]+")
  144. cd ${BKPATH}
  145. rm -rf "$BKWORLD"
  146. #trap 'sendtosvr "save resume"' ERR
  147. echo "$files" | while read -r line; do
  148.     file=${line%:*}
  149.     # https://bugs.mojang.com/browse/BDS-1085
  150.     # save query no longer gives path
  151.     if [ ! -f "$WDPATH/$file" ]; then
  152.         # Trim off $line before first $BKWORLD/
  153.         file=${file#$BKWORLD/}
  154.         # There might be more than one $file in $WDPATH
  155.         file=$(find "${WDPATH}/${BKWORLD}" -name "$file" | head -n 1)
  156.         file=${file#$WDPATH/}
  157.     fi
  158.     dir=$(dirname "$file")
  159.     # Trim off $line before last :
  160.     length=${line##*:}
  161.     mkdir -p "$dir"
  162.     cp "${WDPATH}/${file}" "$dir/"
  163.     truncate --size="$length" "$file"
  164. done
  165. tar fczP ${BKWORLD}$(date +%Y-%m-%d_%H%M%S).tar.gz "$BKWORLD" 2>&1 | logtofile
  166. rm -r "$BKWORLD"
  167. #只保存前n个备份
  168. ls -A1t $(find -name "${BKWORLD}*.tar.gz" -type f) | tail -n +$(( MAX_BACKUPS+1 )) | xargs rm 2>&1 | logtofile
  169. sendtosvr "save resume"
  170. sendtosvr "say 存档备份成功！"
  171. logtofile "${BKWORLD} 备份成功！"
  172. sh $0 $*
  173. ;;
  174. init)
  175. pm2 start -x "./bdlauncher" --restart-delay=10000 -n mc
  176. ;;
  177. *)
  178. tishi
  179. esac
  180. exit 0

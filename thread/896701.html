  1. package cn.ccll.lotrforge.common.item;

  3. import cn.ccll.lotrforge.CreativeTabs.CreativeTabsLoader;
  4. import net.minecraft.item.Item;
  5. import net.minecraft.item.ItemStack;
  6. import net.minecraft.nbt.NBTTagCompound;
  7. import net.minecraft.nbt.NBTTagList;
  8. import net.minecraft.nbt.NBTTagString;

  10. public class ItemSoftBronzeIngot extends Item {

  12.     public ItemSoftBronzeIngot() {
  13.         super();
  14.         this.setUnlocalizedName("softBronzeIngot");
  15.         this.setCreativeTab(CreativeTabsLoader.tabItem);
  16.         this.setTextureName("lotrforge" + ":" + "SoftBronzeIngot");

  18.         ItemStack item = new ItemStack(ItemLoader.softBronzeIngot);
  19.         NBTTagCompound tag = new NBTTagCompound();
  20.         NBTTagCompound display = new NBTTagCompound();
  21.         NBTTagList lore = new NBTTagList();
  22.         lore.appendTag(new NBTTagString("this is a lore"));

  24.         display.setTag("lore",lore);
  25.         tag.setTag("display",display);
  26.         item.setTagCompound(tag);
  27.         }
  28. }

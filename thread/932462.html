方块注册（BlockLoader.java）

  1. public class BlockLoader
  2. {
  3.         public static Block blockChromite=new BlockChromite();
  4.         public static Block fluidMercury=new BlockFluidMercury();
  5.         
  6.         public BlockLoader(FMLPreInitializationEvent event)
  7.         {
  8.                 register(blockChromite,"block_chromite");//设置方块ID。
  9.                 register(fluidMercury,"fluid_mercury");
  10.         }
  11.         
  12.          @SideOnly(Side.CLIENT)
  13.             public static void registerRenders()
  14.             {
  15.                 registerRender(blockChromite);
  16.             }
  17.          
  18.          private static void register(Block block,String name)
  19.          {
  20.                  ForgeRegistries.BLOCKS.register(block.setRegistryName(name));
  21.                  //注册方块。
  22.                  ForgeRegistries.ITEMS.register(new ItemBlock(block).setRegistryName(block.getRegistryName()));
  23.                  //注册该方块对应的模型。
  24.          }
  25.          
  26.          @SideOnly(Side.CLIENT)
  27.             private static void registerRender(Block block)
  28.             {
  29.                  ModelResourceLocation model = new ModelResourceLocation(block.getRegistryName(), "inventory");
  30.                  ModelLoader.setCustomModelResourceLocation(Item.getItemFromBlock(block), 0, model);
  31.             }
  32. }

--------------问-----题---------------
要求编出一个程序，用CP Pascal 或者 FPC编程软件；
输入一个<=32767的数，把它分解质因数并且每个质因数用“*”分隔开；
不要求从小到大排列；
可以用任何功能；
----------------------------------------

回答水贴/跟问题无关直接取消资格；

答案：
我的方法是这样的：
  1. var g,p,k,t,u,b:integer;
  2.           a:array[1..32767] of integer;
  3.           l:boolean;
  4. procedure fj(n:integer);
  5. var i,j,h:integer;
  6.         f:boolean;
  7. begin
  8.   p:=0;
  9.   h:=n;
  10.   repeat
  11.     for i:=2 to h do begin
  12.       f:=true;
  13.       for j:=2 to i-1 do if i mod j=0 then f:=false;
  14.       if (f) and (h mod i=0) then begin h:=h div i; a[p+1]:=i; p:=p+1; end;
  15.     end;
  16.   until h=1;
  17.   if p=1 then l:=false;
  18.   b:=p;
  19. end;
  20. begin
  21.   l:=true;
  22.   readln(g);
  23.   fj(g);
  24.   for t:=1 to b-1 do
  25.     for u:=b downto t+1 do
  26.       if a[u-1]>a[u] then
  27.         begin
  28.           a[0]:=a[u-1];
  29.           a[u-1]:=a[u];
  30.           a[u]:=a[0];
  31.         end;
  32.   if l then begin
  33.     for k:=1 to p-1 do write(a[k],'*');
  34.     write(a[k+1]);
  35.   end else write(g);
  36. end.

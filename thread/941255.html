4. 修改配置
矿区配置文件:

  1. level: "world"
  2. # 矿区等级
  3. areaLevel: 1
  4. # 矿区范围
  5. pos:
  6.   startX: 0
  7.   startY: 0
  8.   startZ: 0
  9.   endX: 0
  10.   endY: 0
  11.   endZ: 0
  12. # 解锁矿区需要的金钱
  13. price: 1000.0
  14. # 矿区是否允许PVP
  15. can-pvp: false
  16. # 矿区是否死亡掉落
  17. can-drop: false
  18. # 矿区重置时间 (-1为不重置)
  19. resetTime(s): 300

  21. #玩家使用矿区时间(-1为无限制)
  22. useTime(s): 600

  24. # 矿区使用次数上限(玩家解锁获得默认次数)
  25. joinCount: 2
  26. # 矿区进入次数重置
  27. reset:
  28. # 上一次刷新时间.. 格式年-月-日
  29. lastUpdataTime: ~
  30. # 时间间隔
  31. day: 7
  32. # 矿区出生点坐标
  33. transfer:
  34.   x: 0
  35.   y: 0
  36.   z: 0
  37. # 传送矿区需要的金钱
  38. transferMoney: 100.0
  39. # 矿区是否开启
  40. key: true
  41. #解锁次矿区前需要解锁的前置矿区
  42. last-area: ""
  43. message: "§a新手矿区 一切发家致富的开始?\n 主要物资:>> 石头>> 木头"
  44. subMessage: "一切的开始..."
  45. buttonImage: "textures/items/diamond_pickaxe"
  46. # 格式为方块id:特殊值 后面的20是生成几率
  47. spawnBlock:
  48.   1: 20
  49.   17: 20

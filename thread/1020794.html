那就生成实体取代假名。
summon armor_stand ~ ~ ~ {Tags:["marker"],Marker:1b}

并创建另一个记分板dummy2

#每次开始前重置marker
#把玩家的dummy2等同于dummy，这样玩家原分数保留在dummy以便后续分数显示的需要，dummy2则用于处理
#处理后的玩家将在后面标上scored
scoreboard players reset @e[tag=marker] dummy2
execute @a[tag=!scored] ~ ~ ~ scoreboard players operation @s dummy2 = @s dummy

#获取最高分，并存入marker中
execute @a[tag=!scored] ~ ~ ~ scoreboard players operation @e[tag=marker,c=1] dummy2 > @s dummy2


#玩家dummy2 - 最高分。0分玩家即为"第一名"
#标记0分玩家，排除此次的"第一名"玩家，分数重置
execute @a[tag=!scored] ~ ~ ~ scoreboard players operation @s dummy2 -= @e[tag=marker,c=1] dummy2
scoreboard players tag @a[tag=!scored,score_dummy2_min=0,score_dummy2=0] add no1
execute @a[tag=!scored] ~ ~ ~ scoreboard players tag @s[tag=no1] add scored
scoreboard players reset @e[tag=marker] dummy2



之后的每次处理和上面一样，差别是标签改为no2, no3...

#获取玩家dummy板中的最高分，记录在假名内
execute as @a run scoreboard players operation $highest dummy > @s dummy

#标记与假名分数相等的玩家
#移除第一名的分数
execute as @a if score @s dummy = $highest dummy run tag @s add no1
scoreboard players reset @a[tag=no1] dummy


#假名分数清空
#第二次记录玩家的最高分
scoreboard players reset $highest dummy
execute as @a run scoreboard players operation $highest dummy > @s dummy

#第二名
execute as @a if score @s dummy = $highest dummy run tag @s add no2
scoreboard players reset @a[tag=no2] dummy



#清空，第三次记录最高分
scoreboard players reset $highest dummy
execute as @a run scoreboard players operation $highest dummy > @s dummy


#第三名
execute as @a if score @s dummy = $highest dummy run tag @s add no3
scoreboard players reset @a[tag=no3] dummy

tellraw @a ["NO1:  ",{"selector":"@p[tag=no1]"},"      NO2:   ",{"selector":"@p[tag=no2]"},"      NO3:   ",{"selector":"@p[tag=no3]"}]

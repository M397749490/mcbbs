  2.     private boolean hasRegion(Location location, String name) {
  3.         FileConfiguration config = Revive.getReviveFile();
  4.         String worldName = config.getString(name + ".point.loc1.world");
  5.         int x1 = config.getInt(name + ".point.loc1.x");
  6.         int x2 = config.getInt(name + ".point.loc2.x");
  7.         int y1 = config.getInt(name + ".point.loc1.y");
  8.         int y2 = config.getInt(name + ".point.loc2.y");
  9.         int z1 = config.getInt(name + ".point.loc1.z");
  10.         int z2 = config.getInt(name + ".point.loc2.z");
  11.         int minY = Math.min(y1, y2) - 1;
  12.         int maxY = Math.max(y1, y2) + 1;
  13.         int minZ = Math.min(z1, z2) - 1;
  14.         int maxZ = Math.max(z1, z2) + 1;
  15.         int minX = Math.min(x1, x2) - 1;
  16.         int maxX = Math.max(x1, x2) + 1;
  17.         if (location.getWorld().getName().equals(worldName)) {
  18.             if (location.getX() > minX && location.getX() < maxX) {
  19.                 if (location.getY() > minY && location.getY() < maxY) {
  20.                     return location.getZ() > minZ && location.getZ() < maxZ;
  21.                 }
  22.             }
  23.         }
  24.         return false;
  25.     }

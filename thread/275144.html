我又學會了一個東西
在此也分享給大家

我要教大家的是
方塊多材質

首先一樣創好一個方塊
這裡方塊名稱以NewBlock為例
再輸入以下代碼

public class BlockNewBlock extends Block{
private Icon[] textures = new Icon[6];
//private Icon[] textures = new Icon[材質數量];
//一般我們會設定6個，因為一個方塊有6面

public BlockNewBlock(int par1) {
  super(par1, Material.rock);
  this.setCreativeTab(MoreArticlesMod.MyTab);

}
public void registerIcons(IconRegister par1IconRegister){
  textures[0] = par1IconRegister.registerIcon("模組名:材質檔名");
  textures[1] = par1IconRegister.registerIcon("模組名:材質檔名");
  textures[2] = par1IconRegister.registerIcon("模組名:材質檔名");
  textures[3] = par1IconRegister.registerIcon("模組名:材質檔名");
  textures[4] = par1IconRegister.registerIcon("模組名:材質檔名");
  textures[5] = par1IconRegister.registerIcon("模組名:材質檔名");
  }
//設定材質
//0是下面,1是上面,2是前面,3是後面,4是右邊,5是左邊
@Override
public Icon getIcon(int i,int j){
  if(i == 0){
   return textures[0];
  }
  if(i == 1){
   return textures[1];
  }
  if(i == 2){
   return textures[2];
  }
  if(i == 3){
   return textures[3];
  }
  if(i == 4){
   return textures[4];
  }
  return textures[5];
}
}
//一樣是設定材質的東西

是0開始
0、1、2、3、4、5
而不是
1、2、3、4、5、6
一定要特別注意

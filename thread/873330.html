在1.12版本如何吧头颅设置成一个指定玩家的名字
作者: 星    时间: 2019-6-14 22:04
你是要获取一个头颅物品么? 如果是的话操作如下:
  1. public static ItemStack Skull(String skullOwner, String displayName, int quantity){
  2.                 ItemStack skull = new ItemStack(Material.SKULL_ITEM, quantity, (short)SkullType.PLAYER.ordinal());
  3.                 SkullMeta skullMeta = (SkullMeta)skull.getItemMeta();
  4.                 boolean shouldSet = false;
  5.                 if(skullOwner != null && !skullOwner.equals("")){
  6.                         skullMeta.setOwner(skullOwner);
  7.                         shouldSet = true;
  8.                 }
  9.                 if(displayName != null){
  10.                         skullMeta.setDisplayName(displayName.replace("&", "§"));
  11.                         shouldSet = true;
  12.                 }
  13.                 if(shouldSet){
  14.                         skull.setItemMeta(skullMeta);
  15.                 }
  16.                 return skull;
  17.         }
复制代码

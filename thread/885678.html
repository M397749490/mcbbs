监听setslot数据包，修改物品Unbreakable，这样客户端能显示无限耐久。

  1.                 ProtocolLibrary.getProtocolManager()
  2.                                 .addPacketListener(new PacketAdapter(null, ListenerPriority.HIGHEST, PacketType.Play.Server.SET_SLOT) {
  3.                                         public void onPacketSending(PacketEvent e) {
  4.                                                 PacketContainer packet = e.getPacket();
  5.                                                 ItemStack itemStack = packet.getItemModifier().read(0);
  6.                                                 if (itemStack == null || itemStack.getType().equals(Material.AIR)) {
  7.                                                         return;
  8.                                                 }
  9.                                                 itemStack.setDurability((short) 0);
  10.                                                 ItemMeta itemMeta = itemStack.getItemMeta();
  11.                                                 itemMeta.spigot().setUnbreakable(true);
  12.                                                 itemStack.setItemMeta(itemMeta);
  13.                                                 packet.getItemModifier().write(0, itemStack);
  14.                                         }
  15.                                 });

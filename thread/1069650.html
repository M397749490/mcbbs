进入世界自动执行指令

< 支持版本 >
测试1.8.X版本可以使用,理论上1.8以上也可以用

< 主要功能 >
当玩家进入了一个指定世界
就自动以后台或者玩家身份执行
自定义指令
配置文件

#插件代码全部原创,不纯在抄袭!
#作者QQ 1474671879 - 小胖次
#插件配置默认UTF-8编码,如果乱码自行转码

#设置启用的世界名字
EnableWorld:
  - 'world'
  - 'world_nether'
  - 'XXX'

#配置进入世界的命令(都是以后台权限执行,%player%是玩家变量)

#启用世界名字:
#  Type: 1 (1是以玩家身份执行命令,2是以后台身份执行)
#  Command:
#    - '执行的命令 - 1'
#    - '执行的命令 - 2'
#    - '执行的命令 - 3'
#    - 以此类推~

world:
  Type: 1
  Command:
    - 'say &c&l我进入了&b&l&n主世界'
    - 'say &c&l我进入了&b&l&n主世界'
    - 'say &c&l我进入了&b&l&n主世界'

world_nether:
  Type: 2
  Command:
    - 'tell %player% &c&l你进入了&b&l&n地狱'
    - 'tell %player% &c&l你进入了&b&l&n地狱'
    - 'tell %player% &c&l你进入了&b&l&n地狱'

XXX:
  Type: 2
  Command:
    - 'tell %player% &c&l你进入了&b&l&nXXX世界'
    - 'tell %player% &c&l你进入了&b&l&nXXX世界'
    - 'tell %player% &c&l你进入了&b&l&nXXX世界'

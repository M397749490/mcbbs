打开目录中的config.yml
接下来详解需要用到的配置文件的含义
这边的代码是我配置好的一个三个服forge版的仅供参考（注意缩进，如果缩进错误那就很惨）
  1. orge_support: true #forge支持
  2. player_limit: -1
  3. permissions:
  4.   default:
  5.   - bungeecord.command.server
  6.   - bungeecord.command.list
  7.   admin:
  8.   - bungeecord.command.alert
  9.   - bungeecord.command.end
  10.   - bungeecord.command.ip
  11.   - bungeecord.command.reload
  12. timeout: 30000
  13. log_commands: false
  14. online_mode: false #正版验证
  15. disabled_commands: #禁止bungeecord中的指令（这个是配合插件使用的）
  16. - disabledcommandhere
  17. servers:  #服务器
  18.   lobby:  #主服务器
  19.     motd: '&1服务器简介'
  20.     address: 127.0.0.1:25564 #主服务器端口（只用改后面25564）
  21.     restricted: false
  22.   lobby1: #主服务器
  23.     motd: '&1服务器简介'
  24.     address: 127.0.0.1:25563 #子服务器端口（只用改后面25563）
  25.     restricted: false
  26.   lobby: #主服务器
  27.     motd: '&1服务器简介'
  28.     address: 127.0.0.1:25562 #子服务器端口（只用改后面25562）
  29.     restricted: false
  30. listeners:
  31. - query_port: 25577 #连入bungeecord的端口（需上下一致）
  32.   motd: '&1Another Bungee server'
  33.   tab_list: GLOBAL_PING
  34.   query_enabled: false
  35.   proxy_protocol: false
  36.   forced_hosts:
  37.     pvp.md-5.net: pvp
  38.   ping_passthrough: false
  39.   priorities:
  40.   - lobby
  41.   bind_local_address: true
  42.   host: 0.0.0.0:25577 #连入bungeecord的端口（需上下一致）
  43.   max_players: 1
  44.   tab_size: 60
  45.   force_default_server: false
  46. ip_forward: false
  47. network_compression_threshold: 256
  48. prevent_proxy_connections: false
  49. groups:
  50.   md_5:
  51.   - admin
  52. connection_throttle: 4000
  53. stats: f73150cc-2259-4f03-a66a-cbf2b2f5d226
  54. connection_throttle_limit: 3
  55. log_pings: true

private HashMap<String,Long> cds=new HashMap<>();
public boolean hasCD(Player player)
{
    Long cd=cds.get(player.getName());
    return cd!=null && cd>System.currentTimeMillis();
}
public void putCD(Player player,int cd)
{
    cds.put(player.getName(),System.currentTimeMillis()+(cd*1000));
}
public boolean onCommand(CommandSender sender,Command cmd,String label,String[] args)
{
    if(hasCD((Player)sender))
    {
        sender.sendMessage("§aCD中,请稍后重试");
    }
    putCD((Player)sender,30);
}

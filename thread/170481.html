    I:autoWorkbench.id=541
    I:blueprintLibrary.id=542
    I:builder.id=543
    I:drill.id=544
    I:engine.id=545
    I:filler.id=546
    I:filteredBuffer.id=547
    I:floodGate.id=548
    I:frame.id=549
    I:fuel.id=550
    I:hopper.id=551
    I:laser.id=552
    I:marker.id=553
    I:miningWell.id=554
    I:oil.id=555
    I:pathMarker.id=556
    I:pipe.id=557
    I:pump.id=558
    I:quarry.id=559
    I:refinery.id=560
    I:springBlock.id=561
    I:tank.id=562
    S:woodenPipe.item.exclusion <
     >
    S:woodenPipe.liquid.exclusion <
     >
}

block字段全部替换成这样 就解决了- -！ 纠结了2天 找了2天都没找到兼容的mod 学了一会解决mod冲突方法 就这么简单- -！ 蛋疼

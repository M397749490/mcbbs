  1.     @SideOnly(Side.CLIENT)
  2.     public static void registerRenders(){
  3.         registerFluidRenders((BlockFluidBase) ModBlocks.FLUID_MERCURY,"fluid_mercury");
  4.     }

  6.     @SideOnly(Side.CLIENT)
  7.     public static void registerFluidRenders(BlockFluidBase blockFluid, String blockStateName){
  8.         final String location = Reference.MODID + ":" + blockStateName;
  9.         final Item itemFluid = Item.getItemFromBlock(blockFluid);
  10.         ModelLoader.setCustomMeshDefinition(itemFluid, new ItemMeshDefinition() {
  11.             @Override
  12.             public ModelResourceLocation getModelLocation(ItemStack stack) {
  13.                 return new ModelResourceLocation("fluid_mercury","fluid");
  14.             }
  15.         });
  16.         ModelLoader.setCustomStateMapper(blockFluid, new StateMapperBase() {
  17.             @Override
  18.             protected ModelResourceLocation getModelResourceLocation(IBlockState state) {
  19.                 return new ModelResourceLocation(location,"fluid");
  20.             }
  21.         });
  22.     }

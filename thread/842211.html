作者: Eiup_Niow    时间: 2019-2-3 00:03
新建一个后缀为js的文件

function run() {
       
        var job = "%jobs_format%";
        if (job == "") {
                return "&8[&fxxxx&8]&f"
        }
        return job
}

run();

放进去之后 在papi插件的javascript_placeholders.yml文件

放
jobs:
  file: jobs.js

然后/papi reload 就ok了 我服务器也弄了这些

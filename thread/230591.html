--安装：
------扔进去MC服务端的插件文件夹
--运行环境：
------小心哟！缺少一个就会报错或崩溃
----------BungeeCord主体
----------Mysql
----------BungeeSuite
--命令：
  1.     setnewspawn:
  2.         描述: 设置一个新的出/重生点.
  3.         权限: bungeesuite.spawns.command.setnewspawn
  4.         别称: [sns, setnewplayerspawn, setnoobspawn]
  5.         无权提示: §cYou do not have permission for this command
  6.     setworldspawn:
  7.         description: 设置proxys出/重生点.
  8.         permission: bungeesuite.spawns.command.setworldspawn
  9.         aliases: [sws]
  10.     setserverspawn:
  11.         description: 设置服务器的出/重生点.
  12.         permission: bungeesuite.spawns.command.setserverspawn
  13.         aliases: [sss]
  14.     setglobalspawn:
  15.         description: 设置代理global出/重生点.
  16.         permission: bungeesuite.spawns.command.setglobalspawn
  17.         aliases: [sgs, setproxyspawn,sethub,sethubspawn]
  18.     spawn:
  19.         description: 把玩家弄到出/重生点
  20.         permission: bungeesuite.spawns.command.spawn
  21.     worldspawn:
  22.         description: 把玩家弄到世界出/重生点
  23.         permission: bungeesuite.spawns.command.worldspawn
  24.         aliases: [ws]
  25.     serverspawn:
  26.         description: 把玩家弄到服务器出/重生点
  27.         permission: bungeesuite.spawns.command.serverspawn
  28.         aliases: [ss]
  29.     globalspawn:
  30.         description: 把玩家弄到proxys出/重生点
  31.         permission: bungeesuite.spawns.command.globalspawn
  32.         aliases: [gs,hub]
复制代码

--权限：
  1.     bungeesuite.spawns.*:
  2.         description: 给予本插件的所有权限
  3.         default: op
  4.         children:
  5.             bungeesuite.spawns.admin: true
  6.     bungeesuite.spawns.admin:
  7.         description: 给予本插件的所有管理权限
  8.         children:
  9.             bungeesuite.spawns.command.setnewspawn: true
  10.             bungeesuite.spawns.command.setworldspawn: true
  11.             bungeesuite.spawns.command.setserverspawn: true
  12.             bungeesuite.spawns.command.setglobalspawn: true
  13.             bungeesuite.spawns.command.spawn: true
  14.             bungeesuite.spawns.command.worldspawn: true
  15.             bungeesuite.spawns.command.serverspawn: true
  16.             bungeesuite.spawns.command.globalspawn: true
  17.     bungeesuite.spawns.new.world:
  18.         description: 服务器新玩家将其设置在世界出生点出生【感谢@SmartMadio 的翻译
  19.         default: false
  20.     bungeesuite.spawns.new.server:
  21.         description: 服务器新玩家将其设置在服务器出生点出生 【感谢@SmartMadio 的翻译
  22.         default: false
  23.     bungeesuite.spawns.new.global:
  24.         description: 服务器新玩家将其设置在代理出生点（即多服务器的proxys出生点） 【感谢@SmartMadio 的翻译
  25.         default: false
  26.     bungeesuite.spawns.spawn.bed:
  27.         description: 如果该玩家用床设置过重生点，那么将会重生到床重生点
  28.         default: false
  29.     bungeesuite.spawns.spawn.world:
  30.         description: 如果没有使用过床设置重生点，那么将会弄到世界重生点
  31.         default: false
  32.     bungeesuite.spawns.spawn.server:
  33.         description: 如果上面那两个都没有的话，那就弄到该服务器的重生点
  34.         default: false
  35.     bungeesuite.spawns.spawn.global:
  36.         description: 如果上面那两个都没有的话，那就弄到Bungee的全局出生点(即多服务器的proxys出生点)。【感谢@SmartMadio 的翻译
  37.         default: false

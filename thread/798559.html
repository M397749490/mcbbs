做流体，从1.8到1.12不太会弄,就自己搞了个流体实现,但是forge的UniversalBucket.java（src/main/java/net/minecraftforge/fluids/UniversalBucket.java）

        float playerRorationPitch = playerIn.rotationPitch;
        float playerRotation = playerIn.rotationYaw;

        Vec3d playerPos = new Vec3d(playerIn.posX, playerIn.posY + playerIn.getEyeHeight(), playerIn.posZ);

        float rorationLeft = MathHelper.cos(-playerRotation * 0.01745329F - 3.141593F);
        float rorationRight = MathHelper.sin(-playerRotation * 0.01745329F - 3.141593F);
        float rorationUp = -MathHelper.cos(-playerRorationPitch * 0.01745329F);
        float rorationDown = MathHelper.sin(-playerRorationPitch * 0.01745329F);
        float maxReachLeft = rorationLeft * rorationUp;
        float maxReachRight = rorationRight * rorationUp;

        double maxReachDist = playerIn.getEntityAttribute(EntityPlayer.REACH_DISTANCE).getAttributeValue();

        Vec3d maxReachBlockDist = playerPos.addVector(maxReachRight * maxReachDist, rorationDown * maxReachDist, maxReachLeft * maxReachDist);

        return worldIn.rayTraceBlocks(playerPos, maxReachBlockDist, useLiquids, !useLiquids, false);
    }

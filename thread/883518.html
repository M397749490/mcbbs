作者: 粘兽    时间: 2019-7-10 03:00
  1. players = world.getAllServerPlayers();
  2. for(var i in players){
  3.     var player = world.getPlayer(players[i].getName());
  4.     if(player != null && player.isSneaking() && player.getHeldItem() != null){
  5.         var held = player.getHeldItem().getDisplayName();
  6.         if(held.indexOf('233') > -1){
  7.             var cmd = 'tp {name} ~ ~2 ~';
  8.             npc.executeCommand(cmd.replace('{name}',player.getName()));
  9.         }
  10.     }
  11. }

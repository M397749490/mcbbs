Edit:
问题已经解决，代码：
  1. public class TestModTransformer implements IClassTransformer, Opcodes {
  2.     private static final Logger logger = LogManager.getLogger();
  3.     private static final ClassRef CL_NetHandlerPlayClient = new ClassRef("cqs", "net/minecraft/client/network/NetHandlerPlayClient");
  4.     private static final ClassRef CL_SPacketSpawnObject = new ClassRef("iz", "net/minecraft/network/play/server/SPacketSpawnObject");
  5.     private static final FieldRef FD_NetHandlerPlayClient_world = new FieldRef("f", "world", "Lcqv;", "Lnet/minecraft/client/multiplayer/WorldClient;");
  6.     private static final MethodRef MD_NetHandlerPlayClient_handleSpawnObject = new MethodRef("a", "handleSpawnObject", "(Liz;)V", "(Lnet/minecraft/network/play/server/SPacketSpawnObject;)V");
  7.     private static final MethodRef MD_SPacketSpawnObject_getType = new MethodRef("l", "getType", "()I", "()I");

  9.     @Override
  10.     public byte[] transform(String name, String transformedName, byte[] basicClass) {
  11.         if (basicClass == null) return null;
  12.         MatchType matchType;
  13.         if ((matchType = CL_NetHandlerPlayClient.matchType(name)) != null) {
  14.             boolean succeeded = false;
  15.             ClassNode clazz = new ClassNode();
  16.             ClassReader reader = new ClassReader(basicClass);
  17.             reader.accept(clazz, 0);
  18.             for (MethodNode method : clazz.methods)
  19.                 if (MD_NetHandlerPlayClient_handleSpawnObject.matchType(method) != null) for (ListIterator<AbstractInsnNode> it = method.instructions.iterator(); it.hasNext();) {
  20.                     AbstractInsnNode next = it.next();
  21.                     if (next.getOpcode() == ASTORE && ((VarInsnNode) next).var == 8) {
  22.                         LabelNode finishLabel = new LabelNode();
  23.                         it.add(new VarInsnNode(ALOAD, 1));
  24.                         it.add(new MethodInsnNode(INVOKEVIRTUAL, CL_SPacketSpawnObject.name(matchType), MD_SPacketSpawnObject_getType.name(matchType), MD_SPacketSpawnObject_getType.desc(matchType), false));
  25.                         it.add(new LdcInsnNode(9398272 /* Magic number */));
  26.                         it.add(new JumpInsnNode(IF_ICMPNE, finishLabel));
  27.                         it.add(new TypeInsnNode(NEW, "com/sjx233/testmod/entity/EntityTestProjectile"));
  28.                         it.add(new InsnNode(DUP));
  29.                         it.add(new VarInsnNode(ALOAD, 0));
  30.                         it.add(new FieldInsnNode(GETFIELD, name.replace('.', '/'), FD_NetHandlerPlayClient_world.name(matchType), FD_NetHandlerPlayClient_world.desc(matchType)));
  31.                         it.add(new VarInsnNode(DLOAD, 2));
  32.                         it.add(new VarInsnNode(DLOAD, 4));
  33.                         it.add(new VarInsnNode(DLOAD, 6));
  34.                         it.add(new MethodInsnNode(INVOKESPECIAL, "com/sjx233/testmod/entity/EntityTestProjectile", "<init>", matchType == MatchType.UNTRANSFORMED ? "(Laxs;DDD)V" : "(Lnet/minecraft/world/World;DDD)V", false));
  35.                         it.add(new VarInsnNode(ASTORE, 8));
  36.                         for (AbstractInsnNode node = next.getNext(); node != null; node = node.getNext())
  37.                             if (node.getOpcode() == IFNULL) {
  38.                                 for (node = node.getPrevious(); node != null; node = node.getPrevious())
  39.                                     if (node.getType() == AbstractInsnNode.LABEL) {
  40.                                         it.add(new JumpInsnNode(GOTO, (LabelNode) node));
  41.                                         succeeded = true;
  42.                                         break;
  43.                                     }
  44.                                 break;
  45.                             }
  46.                         it.add(finishLabel);
  47.                         break;
  48.                     }
  49.                 }
  50.             if (!succeeded) {
  51.                 logger.error("Failed to transform {}", name);
  52.                 return basicClass;
  53.             }
  54.             ClassWriter writer = new ClassWriter(0);
  55.             clazz.accept(writer);
  56.             return writer.toByteArray();
  57.         }
  58.         return basicClass;
  59.     }
  60. }

  25.     @EventHandler

  27.         AtomicReference<String> msg = new AtomicReference<>(e.getMessage());
  28.         AtomicBoolean returnFlag = new AtomicBoolean(true);
  29.         map.keySet().forEach(ks -> {

  39.     @Override

  41.         if (args.length == 2 && args[0].equals("remove")) {

  43.             if (getConfig().contains(beRemove)) {

  45.                 saveConfig();
  46.                 reloadConfig();

  48.                 return true;
  49.             }
  50.             sender.sendMessage("配置文件不存在节点: " + beRemove);

  52.         } else if (args.length == 3 && args[0].equals("add")) {

  55.             if (a.equals(b)) {

  57.                 return true;
  58.             }
  59.             getConfig().set(a, b);
  60.             saveConfig();
  61.             reload();
  62.             sender.sendMessage("添加成功");
  63.             return true;
  64.         }else if (args.length == 1 && args[0].equals("reload")) {

  71.             return true;
  72.         }else {
  73.             String[] help = new String[]{"/cr add x y: 将x替换为y", "/cr remove x: 移除x的替换", "/cr reload: 重读配置文件(只能op执行)"};
  74.             sender.sendMessage(help);

  77.     }

  87.     }

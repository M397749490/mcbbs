已解决。审核的时候解决了。。
建了个config工具类。
把原来代码的
FileConfiguration c = getConfig();
改成了
FileConfiguration c = Config.getConfig();

特此贴上工具类 供其他小伙伴使用和探讨。
  1. private static FileConfiguration config = null;
  2.         private static File configFile = null;
  3.         private static Plugin plugin = Main.getPlugin(Main.class);
  4.        
  5.         public static void reload() {
  6.     if (configFile == null) {
  7.         configFile = new File(plugin.getDataFolder(), "config.yml");
  8.     }
  9.         config = YamlConfiguration.loadConfiguration(configFile);
  10.     }

  12.         public static FileConfiguration getConfig() {
  13.     if (config == null)
  14.         reload();
  15.     return config;
  16.         }

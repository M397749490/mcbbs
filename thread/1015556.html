  466. class TAG_Compound(TAG, MutableMapping):
  467.     """
  468.     TAG_Compound, comparable to a collections.OrderedDict with an
  469.     intrinsic name
  470.     """
  471.     id = TAG_COMPOUND

  473.     def __init__(self, buffer=None, name=None):
  474.         # TODO: add a value parameter as well
  475.         super(TAG_Compound, self).__init__()
  476.         self.tags = []
  477.         self.name = ""
  478.         if buffer:
  479.             self._parse_buffer(buffer)

  481.     # Parsers and Generators
  482.     def _parse_buffer(self, buffer):
  483.         while True:
  484.             type = TAG_Byte(buffer=buffer)
  485.             if type.value == TAG_END:
  486.                 # print("found tag_end")
  487.                 break
  488.             else:
  489.                 name = TAG_String(buffer=buffer).value
  490.                 try:
  491.                     tag = TAGLIST[type.value]()
  492.                 except KeyError:
  493.                     raise ValueError("Unrecognised tag type %d" % type.value)
  494.                 tag.name = name
  495.                 self.tags.append(tag)
  496.                 tag._parse_buffer(buffer)

  498.     def _render_buffer(self, buffer):
  499.         for tag in self.tags:
  500.             TAG_Byte(tag.id)._render_buffer(buffer)
  501.             TAG_String(tag.name)._render_buffer(buffer)
  502.             tag._render_buffer(buffer)
  503.         buffer.write(b'\x00')  # write TAG_END

  505.     # Mixin methods
  506.     def __len__(self):
  507.         return len(self.tags)

  509.     def __iter__(self):
  510.         for key in self.tags:
  511.             yield key.name

  513.     def __contains__(self, key):
  514.         if isinstance(key, int):
  515.             return key <= len(self.tags)
  516.         elif isinstance(key, basestring):
  517.             for tag in self.tags:
  518.                 if tag.name == key:
  519.                     return True
  520.             return False
  521.         elif isinstance(key, TAG):
  522.             return key in self.tags
  523.         return False

  525.     def __getitem__(self, key):
  526.         if isinstance(key, int):
  527.             return self.tags[key]
  528.         elif isinstance(key, basestring):
  529.             for tag in self.tags:
  530.                 if tag.name == key:
  531.                     return tag
  532.             else:
  533.                 raise KeyError("Tag %s does not exist" % key)
  534.         else:
  535.             raise TypeError(
  536.                 "key needs to be either name of tag, or index of tag, "
  537.                 "not a %s" % type(key).__name__)

  539.     def __setitem__(self, key, value):
  540.         assert isinstance(value, TAG), "value must be an nbt.TAG"
  541.         if isinstance(key, int):
  542.             # Just try it. The proper error will be raised if it doesn't work.
  543.             self.tags[key] = value
  544.         elif isinstance(key, basestring):
  545.             value.name = key
  546.             for i, tag in enumerate(self.tags):
  547.                 if tag.name == key:
  548.                     self.tags[i] = value
  549.                     return
  550.             self.tags.append(value)

  552.     def __delitem__(self, key):
  553.         if isinstance(key, int):
  554.             del (self.tags[key])
  555.         elif isinstance(key, basestring):
  556.             self.tags.remove(self.__getitem__(key))
  557.         else:
  558.             raise ValueError(
  559.                 "key needs to be either name of tag, or index of tag")

  561.     def keys(self):
  562.         return [tag.name for tag in self.tags]

  564.     def iteritems(self):
  565.         for tag in self.tags:
  566.             yield (tag.name, tag)

  568.     # Printing and Formatting of tree
  569.     def __unicode__(self):
  570.         return "{" + ", ".join([tag.tag_info() for tag in self.tags]) + "}"

  572.     def __str__(self):
  573.         return "{" + ", ".join([tag.tag_info() for tag in self.tags]) + "}"

  575.     def valuestr(self):
  576.         return '{%i Entries}' % len(self.tags)

  578.     def pretty_tree(self, indent=0):
  579.         output = [super(TAG_Compound, self).pretty_tree(indent)]
  580.         if len(self.tags):
  581.             output.append(("\t" * indent) + "{")
  582.             output.extend([tag.pretty_tree(indent + 1) for tag in self.tags])
  583.             output.append(("\t" * indent) + "}")
  584.         return '\n'.join(output)

  595. class NBTFile(TAG_Compound):
  596.     """Represent an NBT file object."""

  598.     def __init__(self, filename=None, buffer=None, fileobj=None):
  599.         """
  600.         Create a new NBTFile object.
  601.         Specify either a filename, file object or data buffer.
  602.         If filename of file object is specified, data should be GZip-compressed.
  603.         If a data buffer is specified, it is assumed to be uncompressed.
  604.         If filename is specified, the file is closed after reading and writing.
  605.         If file object is specified, the caller is responsible for closing the
  606.         file.
  607.         """
  608.         super(NBTFile, self).__init__()
  609.         self.filename = filename
  610.         self.type = TAG_Byte(self.id)
  611.         closefile = True
  612.         # make a file object
  613.         if filename:
  614.             self.filename = filename
  615.             self.file = GzipFile(filename, 'rb')
  616.         elif buffer:
  617.             if hasattr(buffer, 'name'):
  618.                 self.filename = buffer.name
  619.             self.file = buffer
  620.             closefile = False
  621.         elif fileobj:
  622.             if hasattr(fileobj, 'name'):
  623.                 self.filename = fileobj.name
  624.             self.file = GzipFile(fileobj=fileobj)
  625.         else:
  626.             self.file = None
  627.             closefile = False
  628.         # parse the file given initially
  629.         if self.file:
  630.             self.parse_file()
  631.             if closefile:
  632.                 # Note: GzipFile().close() does NOT close the fileobj,
  633.                 # So we are still responsible for closing that.
  634.                 try:
  635.                     self.file.close()
  636.                 except (AttributeError, IOError):
  637.                     pass
  638.             self.file = None

  640.     def parse_file(self, filename=None, buffer=None, fileobj=None):
  641.         """Completely parse a file, extracting all tags."""
  642.         if filename:
  643.             self.file = GzipFile(filename, 'rb')
  644.         elif buffer:
  645.             if hasattr(buffer, 'name'):
  646.                 self.filename = buffer.name
  647.             self.file = buffer
  648.         elif fileobj:
  649.             if hasattr(fileobj, 'name'):
  650.                 self.filename = fileobj.name
  651.             self.file = GzipFile(fileobj=fileobj)
  652.         if self.file:
  653.             try:
  654.                 type = TAG_Byte(buffer=self.file)
  655.                 if type.value == self.id:
  656.                     name = TAG_String(buffer=self.file).value
  657.                     self._parse_buffer(self.file)
  658.                     self.name = name
  659.                     self.file.close()
  660.                 else:
  661.                     raise MalformedFileError(
  662.                         "First record is not a Compound Tag")
  663.             except StructError as e:
  664.                 raise MalformedFileError(
  665.                     "Partial File Parse: file possibly truncated.")
  666.         else:
  667.             raise ValueError(
  668.                 "NBTFile.parse_file(): Need to specify either a "
  669.                 "filename or a file object"
  670.             )

  672.     def write_file(self, filename=None, buffer=None, fileobj=None):
  673.         """Write this NBT file to a file."""
  674.         closefile = True
  675.         if buffer:
  676.             self.filename = None
  677.             self.file = buffer
  678.             closefile = False
  679.         elif filename:
  680.             self.filename = filename
  681.             self.file = GzipFile(filename, "wb")
  682.         elif fileobj:
  683.             self.filename = None
  684.             self.file = GzipFile(fileobj=fileobj, mode="wb")
  685.         elif self.filename:
  686.             self.file = GzipFile(self.filename, "wb")
  687.         elif not self.file:
  688.             raise ValueError(
  689.                 "NBTFile.write_file(): Need to specify either a "
  690.                 "filename or a file object"
  691.             )
  692.         # Render tree to file
  693.         TAG_Byte(self.id)._render_buffer(self.file)
  694.         TAG_String(self.name)._render_buffer(self.file)
  695.         self._render_buffer(self.file)
  696.         # make sure the file is complete
  697.         try:
  698.             self.file.flush()

  619.         """

  621.         can be found as chunk(x,z).
  622.         """
  623.         if compression == COMPRESSION_GZIP:
  624.             # Python 3.1 and earlier do not yet support `data = gzip.compress(data)`.
  625.             compressed_file = BytesIO()

  627.             f.write(data)
  628.             f.close()

  630.             data = compressed_file.read()

  632.         elif compression == COMPRESSION_ZLIB:
  633.             data = zlib.compress(data) # use zlib compression, rather than Gzip
  634.         elif compression != COMPRESSION_NONE:
  635.             raise ValueError("Unknown compression type %d" % compression)
  636.         length = len(data)

  638.         # 5 extra bytes are required for the chunk block header
  639.         nsectors = self._bytes_to_sector(length + 5)

  641.         if nsectors >= 256:
  642.             raise ChunkDataError("Chunk is too large (%d sectors exceeds 255 maximum)" % (nsectors))

  644.         # Ensure file has a header
  645.         if self.size < 2*SECTOR_LENGTH:
  646.             self._init_file()

  648.         # search for a place where to write the chunk:
  649.         current = self.metadata[x, z]

  651.         sector = self._find_free_location(free_sectors, nsectors, preferred=current.blockstart)

  654.         if self.size < sector*SECTOR_LENGTH:
  655.             # jump to end of file
  656.             self.file.seek(0, SEEK_END)

  658.             assert self.file.tell() == sector*SECTOR_LENGTH

  660.         # write out chunk to region
  661.         self.file.seek(sector*SECTOR_LENGTH)
  662.         self.file.write(pack(">I", length + 1)) #length field
  663.         self.file.write(pack(">B", compression)) #compression field
  664.         self.file.write(data) #compressed data

  666.         # Write zeros up to the end of the chunk
  667.         remaining_length = SECTOR_LENGTH * nsectors - length - 5
  668.         self.file.write(remaining_length * b"\x00")

  670.         #seek to header record and write offset and length records
  671.         self.file.seek(4 * (x + 32*z))
  672.         self.file.write(pack(">IB", sector, nsectors)[1:])

  674.         #write timestamp
  675.         self.file.seek(SECTOR_LENGTH + 4 * (x + 32*z))
  676.         timestamp = int(time.time())
  677.         self.file.write(pack(">I", timestamp))

  679.         # Update free_sectors with newly written block
  680.         # This is required for calculating file truncation and zeroing freed blocks.
  681.         free_sectors.extend((sector + nsectors - len(free_sectors)) * [True])
  682.         for s in range(sector, sector + nsectors):
  683.             free_sectors[s] = False
  684.         
  685.         # Check if file should be truncated:
  686.         truncate_count = list(reversed(free_sectors)).index(False)

  689.             self.file.truncate(self.size)
  690.             free_sectors = free_sectors[:-truncate_count]
  691.         
  692.         # Calculate freed sectors
  693.         for s in range(current.blockstart, min(current.blockstart + current.blocklength, len(free_sectors))):
  694.             if free_sectors[s]:
  695.                 # zero sector s
  696.                 self.file.seek(SECTOR_LENGTH*s)
  697.                 self.file.write(SECTOR_LENGTH*b'\x00')
  698.         
  699.         # update file size and header information
  700.         self.size = max((sector + nsectors)*SECTOR_LENGTH, self.size)
  701.         assert self.get_size() == self.size
  702.         current.blockstart = sector
  703.         current.blocklength = nsectors
  704.         current.status = STATUS_CHUNK_OK
  705.         current.timestamp = timestamp
  706.         current.length = length + 1
  707.         current.compression = COMPRESSION_ZLIB

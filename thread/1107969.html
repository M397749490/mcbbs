public class Employee{
   String name;
   int age;
   String designation;
   double salary;
   public Employee(String name){
      this.name = name;
   }
   public void age(int empAge){
      age =  empAge;
   }
   public void designation(String empDesig){
      designation = empDesig;
   }
   public void salary(double empSalary){
      salary = empSalary;
   }
   public void printEmployee(){
      System.out.println("名字:"+ name );
      System.out.println("年龄:" + age );
      System.out.println("职位:" + designation );
      System.out.println("收入:" + salary);
   }
   public static void main(String[] args){
       Employee dangan1 = new Employee("Mojang");
       Employee dangan2 = new Employee("微软");
       dangan1.age(6);
       dangan1.designation("子公司");
       dangan1.salary(1E9);
       dangan1.printEmployee();
       dangan2.age(45);
       dangan2.designation("母公司");
       dangan2.salary(1.25843E12);
       dangan2.printEmployee();
   }
}

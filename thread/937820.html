# 异步选项
async:
  # 漏斗处理是否异步
  hopper: false
  # 实体处理是否异步
  entityMove: true
  # 区块生成是否异步(测试功能，请勿使用)
  chunkGen: false
  # mod生物是否异步
  modMob: false
#世界配置
world:
  # 预生成出生点区块和保持加载
  keepSpawnInMemory: false
  # 无人区块减少tick
  enableSkipTick: true
  # 使用抓取器防止刷物品,但可能和某些MOD不兼容
  enableCapture: true
  # 世界地形生成
  worldGen:
    # 禁用了mod地形生成器的世界,此世界禁用后将不会自动生成mod方块
    disableForgeGenWorld:
      - "xxxxxxx"
  # 世界生成限制(可以缓解跑图卡服)
  worldGenMaxTick: 15
# 禁用更新正版信息 可以防止头颅卡服
disableUpdateGameProfile: true
# FakePlayer设置
fakePlayer:
  # FakePlayer不callEvent, 设置为true后, FakePlayer将不会触发事件,会无视权限
  # 非特殊情况 请不要开启
  eventPass: false

import java.io.File;
import java.io.FileNotFoundException;
import java.io.IOException;
import org.bukkit.configuration.file.YamlConfiguration;
import org.bukkit.configuration.file.YamlConstructor;
import org.bukkit.configuration.file.YamlRepresenter;
import org.yaml.snakeyaml.DumperOptions;
import org.yaml.snakeyaml.Yaml;
import org.yaml.snakeyaml.representer.Representer;

代码

public class MyYamlConfiguration extends YamlConfiguration
{

    private final DumperOptions yamlOptions = new DumperOptions();
    private final Representer yamlRepresenter = new YamlRepresenter();
    private final Yaml yaml;
    private final File f;

    public MyYamlConfiguration(File f) throws FileNotFoundException
    {
        if(f == null || !f.exists())
            throw new java.io.FileNotFoundException();
        this.f = f;
        yamlOptions.setAllowUnicode(false);  // 这货默认为true   要自己先改成false
        this.yaml = new Yaml(new YamlConstructor(), this.yamlRepresenter, this.yamlOptions);
    }

    @Override  //覆盖
    public String saveToString()
    {
       yamlOptions.setIndent(options().indent());
       yamlOptions.setDefaultFlowStyle(DumperOptions.FlowStyle.BLOCK);
       yamlOptions.setAllowUnicode(false); // 还用问吗
       yamlRepresenter.setDefaultFlowStyle(DumperOptions.FlowStyle.BLOCK);
        String header = buildHeader();
        String dump = this.yaml.dump(getValues(false));
        if (dump.equals("{}\n"))
            dump = "";
        return header + dump;
    }

    public boolean toSave()  // 这个爱要不要
    {
        try
        {   this.save(f);   }
        catch (IOException none)
        {   return false;   }
        return true;
    }

    public void setUnicode(boolean flag)  // 逗你玩233
    {
        yamlOptions.setAllowUnicode(flag);
    }

}




抄下来，实例化，开始玩耍吧 !

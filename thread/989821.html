public final class ByteReader {
    private byte[] bytes;
    public int index = 0;
    public int length;
    public ByteReader(byte[] bytes) {
        refresh(bytes);
    }

    public final void refresh(byte[] bytes) {
        this.bytes = bytes;
        this.length = bytes.length;
    }

    public final byte getByte() {
        return bytes[index++];
    }

    public final short getUByte() {
        return (short)(bytes[index++] & 0xff);
    }

    public final boolean isFinished() {
        return index >= bytes.length;
    }

    public final String getString() {
        return new String(getBytes(getInt()));
    }

    public final void getBytes(byte[] array, int start, int length) {
        if(index + length > bytes.length) {
            throw new RuntimeException("ByteReader: not enough data to process");
        }
        System.arraycopy(bytes, index, array, start, length);
        index += length;
    }

    public final byte[] getBytes(int length) {
        byte[] result = new byte[length];
        getBytes(result, 0, length);
        return result;
    }

    public final byte[] getBytesPlusAnEmpty(int length) {
        byte[] result = new byte[length + 1];
        getBytes(result, 0, length);
        return result;
    }

    // &0xff将byte值无差异转成int,避免Java自动类型提升后,会保留高位的符号位
    public final int getIntR() {
        return (bytes[index++] & 0xff) << 8 | (bytes[index++] & 0xff) | (bytes[index++] & 0xff) << 24 | (bytes[index++] & 0xff) << 16;
    }

    public final int getInt() {
        return (bytes[index++] & 0xff) << 24 | (bytes[index++] & 0xff) << 16 | (bytes[index++] & 0xff) << 8 | (bytes[index++] & 0xff);
    }

    public final long getLongR() {
        return (long) (bytes[index++] & 0xff) << 24 |
          (long) (bytes[index++] & 0xff) << 16 |
          (long) (bytes[index++] & 0xff) << 8  |
          (long) (bytes[index++] & 0xff) |
          (long) (bytes[index++] & 0xff) << 56 |
          (long) (bytes[index++] & 0xff) << 48 |
          (long) (bytes[index++] & 0xff) << 40 |
          (long) (bytes[index++] & 0xff) << 32;

    }

    public final long getLong() {
        return (long) (bytes[index++] & 0xff) << 56 |
          (long) (bytes[index++] & 0xff) << 48 |
          (long) (bytes[index++] & 0xff) << 40 |
          (long) (bytes[index++] & 0xff) << 32 |
          (long) (bytes[index++] & 0xff) << 24 |
          (long) (bytes[index++] & 0xff) << 16 |
          (long) (bytes[index++] & 0xff) << 8  |
          (long) (bytes[index++] & 0xff);

    }

    public final float getFloat() {
        return Float.intBitsToFloat(getInt());
    }

    public final double getDouble() {
        return Double.longBitsToDouble(getLong());
    }

    public final int getUShort() {
        return (bytes[index++] & 0xff) << 8 | (bytes[index++] & 0xff);
    }

    public final int getUShortR() {
        return (bytes[index++] & 0xff) | (bytes[index++] & 0xff) << 8;
    }

    public final short getShort() {
        return (short)((bytes[index++] & 0xff) << 8 | (bytes[index++] & 0xff));
    }

    public static boolean isBitTrue(int i, int bit) {
        return bit == 0 ? ((i & 1) == 1) : ((i & (1 << bit)) >> bit) == 1;
    }

    public static int getSpecifyBit(int n, int fromInclude, int toInclude) {
        int k = 0;
        for(int i = fromInclude; i <= toInclude; i++) {
            k |= (1 << i);
        }
        return (n & k) >>> fromInclude;
    }
}

public class ByteWriter {
    public final ByteList list;
    public ByteWriter() {
        list = new ByteList();
    }

    public ByteWriter(int len) {
        list = new ByteList(len);
    }

    public ByteWriter writeByte(short b) {
        list.add((byte)b);
        return this;
    }

    public ByteWriter writeString(String s) {
        byte[] b;
        writeInt((b = s.getBytes()).length);
        writeBytes(b);
        return this;
    }

    public ByteWriter writeBytes(byte[] bs) {
        if(list.id + bs.length >= list.length) {
            byte[] newArr = new byte[list.id + bs.length + 512];
            System.arraycopy(list.list, 0, newArr, 0, list.id);
            list.list = newArr;
            list.length = newArr.length;
        }
        System.arraycopy(bs, 0, list.list, list.id, bs.length);
        list.id += bs.length;
        return this;
    }

    public ByteWriter writeIntR(int i) {
        list.add((byte)(i >>> 8));
        list.add((byte)(i));
        list.add((byte)(i >>> 24));
        list.add((byte)(i >>> 16));
        return this;
    }

    public ByteWriter writeInt(int i) {
        list.add((byte)(i >>> 24));
        list.add((byte)(i >>> 16));
        list.add((byte)(i >>> 8 ));
        list.add((byte)(i));
        return this;
    }

    public ByteWriter writeLong(long l) {
        list.add((byte)(l >>> 56));
        list.add((byte)(l >>> 48));
        list.add((byte)(l >>> 40));
        list.add((byte)(l >>> 32));
        list.add((byte)(l >>> 24));
        list.add((byte)(l >>> 16));
        list.add((byte)(l >>> 8 ));
        list.add((byte)(l));
        return this;
    }

    public ByteWriter writeFloat(float f) {
        writeInt(Float.floatToIntBits(f));
        return this;
    }

    public ByteWriter writeDouble(double d) {
        writeLong(Double.doubleToLongBits(d));
        return this;
    }

    public ByteWriter writeShort(int s) {
        list.add((byte)(s >>> 8));
        list.add((byte)(s));
        return this;
    }

    public ByteWriter writeShortR(int s) {
        list.add((byte)(s));
        list.add((byte)(s >>> 8));
        return this;
    }

    public byte[] toByteArray() {
        return list.toByteArray();
    }

}

public class ByteList {
    public byte[] list;
    protected int id = 0;
    protected int length;

    public ByteList() {
        this(1024);
    }

    public ByteList(int len) {
        list = new byte[len];
        length = len;
    }

    public int size() {
        return id;
    }

    public int add(byte e) {
        list[id++] = e; // [1,1,1,2]
        if(id >= length){
            byte[] newList = new byte[length+128];
            System.arraycopy(list,0,newList ,0,length);
            list = newList;
            length += 128;
        }
        return id-1; //[3]
    }

    public void set(int index, byte e) {
        list[index] = e;
    }

    public boolean remove(int index){
        if(index >= 0){
            for(int i=index;i<list.length-1;i++){
                list = list[i+1];
            }
            list = Arrays.copyOf(list, list.length-1);
            return true;
        }
        return false;
    }

    public byte get(int _id){
        if(_id >= length) // 3 < 4
            return 0;
        return list[_id]; // 2
    }

    public byte[] toByteArray() {
        byte[] result = new byte[size()];
        System.arraycopy(list, 0, result , 0, result.length);
        return result;
    }

    public void clear() {
        id = 0;
    }
}

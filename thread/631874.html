  18.         end

  20.         a_Player:SendMessage("登录成功")

  22. end

  90. function HandleRemoveAccountCommand(a_Split, a_Player)
  91.         if (not a_Split[2]) then
  92.                 a_Player:SendMessage("删除用户: /removeacc <用户名>")
  93.                 return true
  94.         end
  95.         
  96.         -- Remove /removeacc from the split table
  97.         table.remove(a_Split, 1)
  98.         
  99.         -- Get the full username
  100.         local PlayerName = table.concat(a_Split, " ")
  101.         local TargetUUID = GetUUIDFromPlayerName(PlayerName)

  104.                 a_Player:SendMessage("找不到这个玩家")
  105.                 return true
  106.         end
  107.         
  108.         local res, Err = g_PassStorage:DeleteUUID(TargetUUID)

  111.                 return true
  112.         end

  13.         local OnAllChunksAvaliable = function()

  16.                         Player:TeleportToCoords( warps[Tag]["x"] + 0.5 , warps[Tag]["y"] , warps[Tag]["z"] + 0.5)
  17.                         Player:SendMessageSuccess('传送到地标"' .. Tag .. '".')
  18.                         if change_gm_when_changing_world == true then
  19.                                 Player:SetGameMode(Player:GetWorld():GetGameMode())
  20.                                 return true

  22.                 else
  23.                         Player:TeleportToCoords( warps[Tag]["x"] + 0.5 , warps[Tag]["y"] , warps[Tag]["z"] + 0.5)

  25.                 end
  26.         end

  31. function HandleSetWarpCommand( Split, Player)
  32.         local Server = cRoot:Get():GetServer()
  33.         local World = Player:GetWorld():GetName()
  34.         local pX = math.floor(Player:GetPosX())
  35.         local pY = math.floor(Player:GetPosY())
  36.         local pZ = math.floor(Player:GetPosZ())

  38.         if #Split < 2 then
  39.                 Player:SendMessageInfo('请使用: '..Split[1]..' <地标>')
  40.                 return true
  41.         end
  42.         local Tag = Split[2]
  43.         
  44.         if warps[Tag] == nil then
  45.                 warps[Tag] = {}
  46.         end
  47.         
  48.         if (WarpsINI:FindKey(Tag)<0) then
  49.                 warps[Tag]["w"] = World
  50.                 warps[Tag]["x"] = pX
  51.                 warps[Tag]["y"] = pY
  52.                 warps[Tag]["z"] = pZ
  53.         end

  55.         if (WarpsINI:FindKey(Tag)<0) then
  56.                 WarpsINI:AddKeyName(Tag);
  57.                 WarpsINI:SetValue( Tag , "w" , World)
  58.                 WarpsINI:SetValue( Tag , "x" , pX)
  59.                 WarpsINI:SetValue( Tag , "y" , pY)
  60.                 WarpsINI:SetValue( Tag , "z" , pZ)
  61.                 WarpsINI:WriteFile("warps.ini");
  62.         
  63.                 Player:SendMessageSuccess("地标"" .. Tag .. ""设置在'" .. World .. "' x:'" .. pX .. "' y:'" .. pY .. "' z:'" .. pZ .. "'")
  64.         else
  65.                 Player:SendMessageFailure('地标"' .. Tag .. '"已经存在 请使用/delwarp <地标>删除它')
  66.         end
  67.         return true
  68. end

  1. public class MyPath {
  2. public static String getProjectPath() {
  3. java.net.URL url = MyPath.class.getProtectionDomain().getCodeSource().getLocation();
  4. String filePath = null;
  5. try {
  6. filePath = java.net.URLDecoder.decode(url.getPath(), "utf-8");
  7. } catch (Exception e) {
  8. e.printStackTrace();
  9. }
  10. if (filePath.endsWith(".jar"))
  11. filePath = filePath.substring(0, filePath.lastIndexOf("/") + 1);
  12. java.io.File file = new java.io.File(filePath);
  13. filePath = file.getAbsolutePath();
  14. return filePath;
  15. }

  17. public static String getRealPath() {
  18. String realPath = MyPath.class.getClassLoader().getResource("").getFile();
  19. java.io.File file = new java.io.File(realPath);
  20. realPath = file.getAbsolutePath();
  21. try {
  22. realPath = java.net.URLDecoder.decode(realPath, "utf-8");
  23. } catch (Exception e) {
  24. e.printStackTrace();
  25. }
  26. return realPath;
  27. }
  28. }

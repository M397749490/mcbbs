标题: Minecraft 服务器数据自动上传百度云 【新增Windows + Linux】
本帖最后由 dfc643 于 2014-7-31 15:57 编辑

想必大家有部分人是在 VPS 上面开的 Minecraft 服务器，但是对于 Minecraft 来说地图数据简直非常重要。但是一旦数据丢了可不得了。所以呢咱就编写了一个自动上传百度云的 PHP 脚本。适合在 Linux 的服务器上运行。可以自动将数据打包上传至百度网盘。

另外值得一提的是，由于太懒就直接调用系统 shell 了，这样也可以方便改写成 shell 脚本，则可以省去 php 这样的东西了。
另外近期作成 Windows 版本请继续往下看

Linux 软件包要求：
php5 zip 以及 curl cron

Linux 版如何使用：
1、将 mcbkp.php 中的相关目录路径更换成您的相关路径。
2、请填写您的 百度API 的 AccessToken 代码。
3、为 Linux 计划任务 Cron 添加一条新计划任务。

crontab -e 范例：
0 */6 * * * php /root/mcbkp.php #每隔 6 小时自动上传 MC 数据到百度网盘

mcbkp.php 程序代码 (Linux版)：
  1. <?php
  2. // MINECRAFT data backup tool
  3. // @ by Norckon

  5. date_default_timezone_set('PRC');

  7. // Backup filename and source path
  8. //   BAKPATH: Where want you storage backup archive.
  9. //   SRCPATH: Where Minecraft server installed.
  10. $bakpath = "/root/mcbackups";
  11. $srcpath = "/root/cauldron";

  13. // Create Zip Archive
  14. //   SRC: Which directories want to compress
  15. //   DST: Where storage Zip Archive
  16. //   MAPNAME: Which map want you backup in Minecraft
  17. //   MODE: How to backup
  18. //      1: Current map only
  19. //      2: Current map and mods
  20. //      3: All the server and data
  21. function CreateArchive($src, $dst, $mapname, $mode) {
  22.         // Create new zip archive file
  23.         switch($mode) {
  24.                 case 1:
  25.                         system("zip -qr $dst $src/$mapname/");
  26.                         break;
  27.                 case 2:
  28.                         system("zip -qr $dst $src/$mapname/ $src/mods/");
  29.                         break;
  30.                 case 3:
  31.                         system("zip -qr $dst $src/");
  32.                         break;
  33.                 default:
  34.                         echo "I don't known which mode you want.";
  35.         }
  36. }

  38. // Remove older backups
  39. function RemoveOld ($path) {
  40.         system("rm -f $path/mcbkp_*.mcb");
  41. }

  43. // Upload to Baidu netdisk
  44. function bdpcsUpload($accesstoken, $path ,$filename) {
  45.         system('curl -k -L -F "file=@'.$path.'" "https://c.pcs.baidu.com/rest/2.0/pcs/file?method=upload&access_token='.$accesstoken.'&path=/apps/fmcbackups/'.$filename.'"');
  46. }

  48. // Main function
  49. function FMain() {
  50.         global $bakpath,$srcpath;
  51.         // Remove all older backups
  52.         RemoveOld($bakpath);
  53.         // Create new backup
  54.         $bkparc = $bakpath."/mcbkp_".date("Y_m_d_H_i_s").".mcb";
  55.         CreateArchive($srcpath, $bkparc, "world", 1);
  56.         // Upload to baidudisk
  57.         // 百度网盘 API 使用方法 ：http://www.fcsys.us/webapp/wordpress/?p=1292
  58.         bdpcsUpload("YOUR_ACCESS_TOKEN", $bkparc, "/mcbkp_".date("Y_m_d_H_i_s").".mcb");
  59. }

  61. FMain();
  62. ?>

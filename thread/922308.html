public final Hashtable<String, ArrayList<Object>> Config = new Hashtable<String, ArrayList<Object>>();
public static byte[] readFile(File file) throws Throwable {
                FileInputStream input = new FileInputStream(file);
                byte[] t_ret = new byte[input.available()];
                input.read(t_ret, 0, input.available());
                input.close();
                return t_ret;
        }

        public static boolean writeFile(File file, byte[] content) throws Throwable {
                FileOutputStream output = new FileOutputStream(file);
                output.write(content, 0, content.length);
                output.flush();
                output.close();
                return true;
        }

        public static <T> T parseJson(String json, Class<T> Jsonclass) throws Throwable {
                Gson parse = new GsonBuilder().setPrettyPrinting().enableComplexMapKeySerialization().create();
                return parse.fromJson(json, Jsonclass);
        }

        public static String toJsonString(Object json) throws Throwable {
                Gson parse = new GsonBuilder().setPrettyPrinting().enableComplexMapKeySerialization().create();
                return parse.toJson(json);
        }

        public boolean SaveConfig() throws Throwable {
                boolean result = true;
                for (String key : Config.keySet()) {
                        result = result & SaveConfig(key);
                }
                return result;
        }

        public boolean SaveConfig(String config) throws Throwable {
                return writeFile(new File((String) Config.get(config).get(0)), toJsonString(config).getBytes("UTF-8"));
        }

        public Object ReadConfig(String config) throws Throwable {
                if (!new File((String) Config.get(config).get(0)).exists()
                                || !new File((String) Config.get(config).get(0)).isFile()) {
                        new File(new File((String) Config.get(config).get(0)).getParent()).mkdirs();
                        new File((String) Config.get(config).get(0)).createNewFile();
                        Object ret = ((Class<?>) Config.get(config).get(1)).newInstance();
                        writeFile(new File((String) Config.get(config).get(0)), toJsonString(ret).getBytes("UTF-8"));
                        return ret;
                } else {
                        Object ret=parseJson(new String(readFile(new File((String) Config.get(config).get(0))), "UTF-8"),
                                        (Class<?>) Config.get(config).get(1));
                        if(ret==null)
                        {
                                ret = ((Class<?>) Config.get(config).get(1)).newInstance();
                                writeFile(new File((String) Config.get(config).get(0)), toJsonString(ret).getBytes("UTF-8"));
                        }
                        return ret;
                }
        }

        public void loadConfig() throws Throwable {
                for (String key : Config.keySet()) {
                        Config.get(key).set(2, ReadConfig(key));
                }

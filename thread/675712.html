作者: 呆萌的龙言    时间: 2017-2-20 19:01
应该可能是PlotSquared插件 方法：
配置文件路径:\PlotSquared\config\worlds文件
worlds:
  世界名字:
    plot:
      height: 64 #高度 一般不用动它
      size: 50 #地皮大小
      filling: #地皮表面以下的方块ID
      - '3:0' #泥土一般不用管
      floor:
      - '2:0' #地皮表面的方块 一般都是草方块 别动
      bedrock: true #是否有基岩
      biome: FOREST #地皮上的自然群系
      auto_merge: true #玩家有多块地皮是否自动合体
      create_signs: true #自动创造牌子
    wall:
      block: '44:0' #没认领前围墙的方块
      block_claimed: '44:1' #认领后围墙的方块
      filling: '1:0' #围墙下面的方块
      height: 64 #高度 一般不用动
    road:
      width: 7 #马路的宽
      height: 64 #马路的高 一般不用动
      block: '155:0' #马路方块
    home:
      default: side #传送到家时在地皮外面 好像是这样翻译 不用改就是了
      allow-nonmembers: false #允许没有地皮建筑权限的人也可以破坏 也不用改就是了
    schematic: #这个属于后期的高级教程 以后慢慢说
      specify_on_claim: false
      on_claim: false
      file: 'null'
      schematics: []
    economy:
      prices:
        merge: 100.0 #合并费用
        sell: 100.0 #出售费用
        claim: 100.0 #认领费用
      use: false #是否开启
    chat:
      enabled: false #地皮是否默认开启聊天模式
    limits:
      max-members: 128
    world:
      max_height: 256
      gamemode: creative
      min_height: 1
      border: false
    event:
      spawn:
        egg: false #允许丢鸡蛋生成动物
        breeding: false #允许动物交配
        custom: true
    natural_mob_spawning: false #怪物生成?
    flags: {}

  5. execute if entity @e[nbt={Item:{tag:{isMark:1b,Mark:2b}}}] run scoreboard players add #temp a 26
  6. execute if entity @e[nbt={Item:{tag:{isMark:1b,Mark:3b}}}] run scoreboard players add #temp a 53

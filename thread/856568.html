name: #插件名字main: #继承了javaPlugin类的主类
version: #版本号
author: #作者
commands:
  demo: #指令名   
  description: "测试指令" #指令的注解

  permission: "demo.use" #指令所需要的权限

在尝试编写插件之前，我建议你先看1个好一个java入门视频，再看一篇好一点的bukkit开发入门，之后开始开发项目，想要实现某个功能时，先去查查doc看有没有这样的api，如果没有可以去看有没有实现了这样功能的插件，反编译看看它的源码看它怎么实现的，实在不行你可以百度看看是否有人问过这样的问题或者写过类似的教程。
事实上你连续发的几个贴中的问题都相当基础，如果你想做开发，最好先提升一下自己的学习能力，和自己的解决问题的能力，来论坛提问，是最没有效率、最浪费事件的方式。

public class Test extends JavaPlugin implements Listener {
boolean flg;
public Test(){
   flg=true;//默认开启
}

    @Override
    public boolean onCommand(CommandSender sender, Command cmd, String label, String[] args){
                if(label.equalsIgnoreCase("demo")) {
                        if(args.length==0) {
                                sender.sendMessage("§a/demo true 开启");
                                sender.sendMessage("§a/demo false 关闭");;
                                return true;
                        }
                        if(args[0].equalsIgnoreCase("false")) {
                                flg=false;//关闭后玩家加入后不会提示
                        }

                        if(args[0].equalsIgnoreCase("true")) {
                                flg=true;//开启
                        }
                }

        @EventHandler//玩家加入事件
        public void onJoinMes(PlayerJoinEvent e) {
                if(flg) {//开关
                                Bukkit.broadcastMessage("§d〓〓〓〓〓〓〓〓〓〓〓〓〓〓〓〓〓〓〓〓〓〓");
                                Bukkit.broadcastMessage(e.getPlayer().getName()+"   §d§l上线了");
                                Bukkit.broadcastMessage("§d〓〓〓〓〓〓〓〓〓〓〓〓〓〓〓〓〓〓〓〓〓〓");
                }
        }
}

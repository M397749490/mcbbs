标题: 【电脑Mod】乌龟种树程序
本帖最后由 fly0025 于 2013-1-20 19:31 编辑

非常简单的程序，主要实现的功能就是在指定的区域内种树，可以指定长宽，区域的位置是乌龟向前一个一格的右前方，每隔一格种一棵，如果树种不够的话会提示的。1-15格摆放的是树苗，最后一格放泥土，泥土的作用是如果哪一格无法种树就把哪一格替换成泥土，所以是可以在天上种树的。不过实际上由于一格只有64个，所以在天上最大只支持16x8的，在地面上的话可以非常大的。
欢迎提出意见和建议

使用说明：

将文本复制到文本文件中，放入.minecraft/mods/computercraft/lua/rom/programs中，然后在TurtleOS上运行就可以了。



程序如下：

local tArgs = { ... }

if #tArgs < 1 or #tArgs >3 then
        print( "Usage: planting <length> <width>" )
        return
end

length = tonumber(tArgs[1]) - 1
width = tonumber(tArgs[2])
turnRight = true
count = 0
loc = 1
SaplingCount = 0

local function tryDig()
        while turtle.detect() do
                if turtle.dig() then
                        sleep(0.5)
                else
                        return false
                end
        end
        return true
end

local function tryForward()
        while not turtle.forward() do
                if turtle.detect() then
                        if not tryDig() then
                                return false
                        end
                elseif turtle.attack() then
                else
                        sleep( 0.5 )
                end
        end
        count = count + 1
        return true
end

local function tryDown()
        while not turtle.down() do
                if turtle.detectDown() then
                        if not tryDigDown() then
                                return false
                        end
                elseif turtle.attackDown() then
                else
                        sleep( 0.5 )
                end
        end
        return true
end

local function getSaplingCount() --得到树苗的总数，从1到15格，第16格用来放泥土
        turtle.select(1)
        for b = 1,15 do
                if turtle.compareTo(b) then
                        SaplingCount = SaplingCount + turtle.getItemCount(b)
                end
        end
        return SaplingCount
end

local function checkSaplingCount() --检查树苗的数量是否满足种树区域所需的数量，不足的话等待放入
        NeedSaplingCount = math.ceil(( length * width ) / 2 ) --计算所需树苗的数量
        print("We get " .. getSaplingCount().. " saplings and we need " .. NeedSaplingCount .. " saplings. ")
        if getSaplingCount() < NeedSaplingCount then
                print("Not enough saplings")
                while getSaplingCount() < NeedSaplingCount do
                        sleep(2)
                end
        end
        print("Now we get enough saplings")
end


local function tryPlaceDown(Num) --PlaceDown的简化版
        turtle.select(Num)
        if not turtle.compareDown() then
                turtle.digDown()
        end
        turtle.placeDown()
        turtle.select(1)
end

local function tryPlant() --种树函数，从第一格开始消耗树苗，如果第一格只剩一个树苗，选择第二格，以此类推
        -- if count % ((length*width)/2) == 0 then
                -- tryPlaceDown(16)
        if count % 2 == 0 then
                if turtle.getItemCount(loc) < 2 then
                        loc = loc + 1
                        if loc == 15 then
                                loc = 1
                        end
                end
                if not tryPlaceDown(loc) then
                        if not turtle.compareDown() then
                                turtle.digDown()
                                turtle.down()
                                tryPlaceDown(16)
                                turtle.up()
                                tryPlaceDown(loc)
                        end
                end
        end
end

local function back() --返回起始点
        if not turnRight then --右转
                turtle.turnRight()
        for d = 1,width do
                tryForward()        
        end
                turtle.turnLeft()
        for d = 1,length do
                tryForward()
        end
                tryForward()
                tryDown()
                turtle.turnLeft()
                turtle.turnLeft()
        else --左转
                turtle.turnLeft()
                for d = 1,width do
                        tryForward()
                end
                turtle.turnRight()                turtle.back()
                tryDown()
        end               
end



local function main()
        checkSaplingCount()
        turtle.up()
        tryForward()
        for w = 1,width do
                for l=1,length do
                        tryPlant()
                        tryForward()
                        end
                if turnRight then
                        tryPlant()
                        turtle.turnRight()
                        tryForward()
                        turtle.turnRight()
                        turnRight = false
                else
                        turtle.turnLeft()
                        tryForward()
                        turtle.turnLeft()
                        turnRight = true
                end
        end
        back()
end

main()

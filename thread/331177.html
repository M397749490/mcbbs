打开插件文件夹，找到worlds.yml，修改
  1. world:
  2.     ==: MVWorld
  3.     hidden: 'false'
  4.     alias: world
  5.     color: WHITE '在此处改，基本的颜色都可以，如red，blue等
  6.     style: NORMAL
  7.     pvp: 'false'
  8.     scale: '1.0'
  9.     respawnWorld: ''
  10.     allowWeather: 'true'
  11.     difficulty: EASY
  12.     spawning:
  13.       ==: MVSpawnSettings
  14.       animals:
  15.         ==: MVSpawnSubSettings
  16.         spawn: 'false'
  17.         spawnrate: '-1'
  18.         exceptions: []
  19.       monsters:
  20.         ==: MVSpawnSubSettings
  21.         spawn: 'false'
  22.         spawnrate: '-1'
  23.         exceptions: []
  24.     entryfee:
  25.       ==: MVEntryFee
  26.       amount: '0.0'
  27.       currency: '-1'
  28.     hunger: 'true'
  29.     autoHeal: 'true'
  30.     adjustSpawn: 'true'
  31.     portalForm: ALL
  32.     gameMode: SURVIVAL  
  33.     keepSpawnInMemory: 'false'
  34.     spawnLocation:
  35.       ==: MVSpawnLocation
  36.       yaw: 0.0
  37.       pitch: 0.0
  38.       z: 158.0
  39.       y: 110.0
  40.       x: -196.0
  41.     autoLoad: 'true'
  42.     bedRespawn: 'true'
  43.     worldBlacklist: []
  44.     environment: NORMAL
  45.     seed: '808660315090941829'

        //在这里输出一个告警信息，告诉大家你在做什么，可选项
        logger.debug("Changing class net/minecraft/client/Minecraft");
        ClassReader cr = new ClassReader(basicClass);
        ClassNode cn = new ClassNode();
        cr.accept(cn, 0);

        for(MethodNode mn : cn.methods) {
            for(int i = 0; i< mn.instructions.size(); i++) {
                AbstractInsnNode node = mn.instructions.get(i);
                if(node instanceof TypeInsnNode) {
                    if(node.getOpcode() == Opcodes.NEW) {
                        // remap, 把notch name转为srg name
                        String d = FMLDeobfuscatingRemapper.INSTANCE.map(((TypeInsnNode)node).desc);
                        if(d.equals("net/minecraft/client/renderer/RenderItem")) {
                            logger.info("Found bzu create node.");
                            ((TypeInsnNode)node).desc = "mi/client/renderer/item/RenderItemMI";
                        } else if(d.equals("net/minecraft/client/renderer/ItemRenderer")) {
                            logger.info("Found bus create node.");
                            ((TypeInsnNode)node).desc = "mi/client/renderer/item/ItemRendererMI";
                        }
                    }
                } else if(node instanceof MethodInsnNode) {
                    if(node.getOpcode() == Opcodes.INVOKESPECIAL) {
                        String d = FMLDeobfuscatingRemapper.INSTANCE.map(((MethodInsnNode)node).owner);
                        //logger.info(d);
                        if(!((MethodInsnNode)node).name.equals("<init>")) {

                        }else
                        if(d.equals("net/minecraft/client/renderer/RenderItem")) {
                            logger.info("Found bzu init node.");
                            ((MethodInsnNode)node).owner = "mi/client/renderer/item/RenderItemMI";
                        } else if(d.equals("net/minecraft/client/renderer/ItemRenderer")) {

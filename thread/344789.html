修改配置plugins\Essentials\config.ym
找到这几行
  1.   # 总体的物理现象或行为的设置
  2.   prevent:                                 # 阻止以下现象(译注:true就是阻止,false是允许,这里需要注意一下)
  3.     lava-flow: true                        # 阻止 岩浆的流动
  4.     water-flow: false                      # 阻止 水的流动
  5.     water-bucket-flow: false               # 阻止 水桶水的流动
  6.     fire-spread: false                     # 阻止 火的传播
  7.     lava-fire-spread: false                # 阻止 岩浆引起的火的传播
  8.     flint-fire: false                      # 阻止 打火石产生火
  9.     lightning-fire-spread: true            # 阻止 闪电引起的火的传播
  10.     portal-creation: false                 # 阻止 传送门的创造
  11.     tnt-explosion: false                   # 阻止 TNT的爆炸
  12.     tnt-playerdamage: false                # 阻止 TNT对玩家造成伤害
  13.     tnt-minecart-explosion: false          # 阻止 TNT矿车 爆炸
  14.     tnt-minecart-playerdamage: false       # 阻止 TNT矿车 对玩家造成伤害
  15.     fireball-explosion: true              # 阻止 火球的爆炸(译注:恶魂发射的火球)
  16.     fireball-fire: false                   # 阻止 火球的燃烧
  17.     fireball-playerdamage: false           # 阻止 火球对玩家造成伤害
  18.     witherskull-explosion: false           # 阻止 凋零攻击的爆炸
  19.     witherskull-playerdamage: false        # 阻止 凋零攻击对玩家照成伤害
  20.     wither-spawnexplosion: false           # 阻止 凋零出生时的爆炸
  21.     wither-blockreplace: true              # 阻止 凋零破坏方块
  22.     creeper-explosion: false               # 阻止 爬行者的爆炸
  23.     creeper-playerdamage: false            # 阻止 爬行者对玩家造成伤害
  24.     creeper-blockdamage: true              # 阻止 爬行者炸毁方块
  25.     enderdragon-blockdamage: true          # 阻止 末影龙破坏方块
  26.     enderman-pickup: true                  # 阻止 末影人拾取方块
  27.     villager-death: true                  # 阻止 村民被杀死

NewHonor Example config
PlayerData
#这个玩家是否应用效果组
enableEffects=true
#这个玩家拥有的头衔表
honors=[
    default,
    admin,
    鱼塘塘主
]
#是否使用头衔
usehonor=true
#正在使用的头衔
using=default
honor (honor.conf)
#头衔id 里面的value为头衔实际效果
admin {
    value="[&1ADMIN&r]"
    getMessage="恭喜{playername}获得{newhonor}"
}
default {
    value="[Player]"
}
#里面的effects为使用的效果组id
鱼塘塘主 {
    effects=鱼塘塘主的光环
    value="[&6鱼塘塘主&f]"
}

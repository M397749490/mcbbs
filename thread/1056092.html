作者: 深拥v    时间: 2020-6-3 00:02
@echo off

set restart=0

:restart

set server_name=[BC]大厅

title %server_name%服 重启次数%restart%次

java -Xms1G -Xmx3G  -jar paperclip.jar
timeout /t 5 /NOBREAK

set/a restart=restart+1

goto restart

作者: 749112221a    时间: 2020-6-3 16:09
@echo OFF
@set a=0
:head
title [20021]青空之巅  重启次数[%a%]
Java -Xincgc -Xms1G -Xmx2G -XX:MaxPermSize=2048M -XX:+AggressiveOpts -XX:+UseCompressedOops -jar PaperSpigot-1.12.2.jar
@echo 服务器已关闭，十秒后会自动重启！
@set /a a=%a%+1
ping -n 10 127.1>nul
goto head

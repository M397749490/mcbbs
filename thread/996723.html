如果TP到x0z0附近你就能看到了
RenderPlayerEvent没猜错的话已经预先定位到了player的脚底位置
所以renderEntity(... 0,0,0,0)没有问题
如果要在头上的话就translate(0, 1.85f, 0)关于定位的具体位置多翻翻forgeBin
可以看一下这段我写好的代码，在头上渲染指定生物
    @SubscribeEvent
    public void doHatRender(RenderPlayerEvent.Post event) {
        EntityPlayer player = event.getEntityPlayer();
        Entity entityToRender = 。。。。//可以缓存没必要每次都new
        
        if(entityToRender == null)
            return;
        
        GlStateManager.pushMatrix();
        GlStateManager.translate(event.getX(), event.getY() + 1.85f, event.getZ());
        
        float spinSpeed = 0.8F; // 转啊转。
        GlStateManager.rotate(-player.rotationYaw * player.world.getTotalWorldTime(), 0, spinSpeed, 0);
        // 仰角
        GlStateManager.rotate(player.rotationPitch, 1, 0, 0);
        
        // 缩小到一半
        GlStateManager.scale(0.5, 0.5, 0.5);
        mc.getRenderManager().renderEntity(entityToRender, 0F, 0F, 0F, 0F, 0, false);
        GlStateManager.popMatrix();
    }

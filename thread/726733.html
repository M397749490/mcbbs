  1. COMPRESSOR:
  2.   enabled: true
  3.   can-be-used-in-workbenches: false
  4.   allow-disenchanting: true
  5.   required-permission: ''
  6.   wiki: https://github.com/TheBusyBiscuit/Slimefun4/wiki/Compressor

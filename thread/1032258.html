介绍
GUI时钟是一个小型Mod，是TheVanillaExperience整合包的附属产物，可以简单显示时间。
只需要将时钟放在身上(背包也可以)，就可以在游戏中显示游戏或者现实(本地电脑)的时间，支持十二小时制和二十四小时制，具体的设置可以参考下面的配置方法的折叠，你或许还对[GUI指南针]感兴趣，可以点击这里赞助作者。

关于配置

12小时制：
电脑时间+游戏时间：

以下内容未1.15.2下config/guiclock-common.toml的默认设置

[General]
#启用后需要身上(包括背包)有携带时钟才会显示游戏时间
mustHaveClockInInventoryForGameTime = true
#启用后需要身上(包括背包)有携带时钟才会显示现实(本地电脑)时间
mustHaveClockInInventoryForRealTime = true
#设置时间为二十四小时制
_24hourformat = true
#显示现实(本地电脑)时间和游戏时间
showBothTimes = false
#显示本地时间而非游戏时间
showRealTime = false
#显示时间中的秒数
showRealTimeSeconds = false
#将时间显示置于在左边
clockPositionIsLeft = false
#将时间显示置于在中间
clockPositionIsCenter = false
#将时间显示置于在右边
clockPositionIsRight = true
#将时间显示的位置移动到垂直位置(Y轴)，即上下移动，可以一定程度上避免和其他显示重叠的问题。
#范围: 0 ~ 3000
clockHeightOffset = 5
#修改坐标显示文字的RGB颜色
#修改RGB颜色的R值(Red)
#范围: 0 ~ 255
RGB_R = 255
#修改RGB颜色的G值(Green)
#范围: 0 ~ 255
RGB_G = 255
#修改RGB颜色的B值(Blue)
#范围: 0 ~ 255
RGB_B = 255

  22. - '&cWelcom&ee &f${viewer name}'
  23. - '&cWelcome &f${viewer name}'

  25. headerAnimationUpdateInterval: 0.2

  27. footer:
  28. - '&f&oPowered by BungeeTabListPlus'

  30. footerAnimationUpdateInterval: 0.5

  32. customPlaceholders:
  33.   other_server_prefix: # makes players on other servers gray
  34.     !conditional
  35.     condition: '${viewer server} == ${player server}'
  36.     true: '&f'
  37.     false: '&7'

  39. playerSets:
  40.   global:
  41.     filter: "true"
  42.     hiddenPlayers: "VISIBLE" # Hidden players show up on the tab list, avoids gliches

  44. playerSet: global

  46. playerOrder: "playerServerFirst,serverAlphabetically,vaultGroupInfo,alphabetically"

  48. playerComponent: {text: "${other_server_prefix}${player name}", icon: "${player skin}", ping: "${player ping}"}

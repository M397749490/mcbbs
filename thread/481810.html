作为一个新人，昨天下午才接触mod开发的

首先推荐一下，这个教程真的很好，很适合新人学习http://blog.hakugyokurou.net/?page_id=126  新人们可以去看看，大神请忽略
就在搭建最初环境时候遇到了一些问题，感觉有点小麻烦。各种出错，不过在自己的耐心寻找方法和重试下终于成功了。最后写了一点自己的东西
  1. @Mod(modid="diracon",name="MyFirstMod",version="1.0.0")
  2. public class MyFirstMod {

  4.         public static BlockIron blockIron;
  5.         public static ItemIron itemIron;
  6.         public static IronTools ironTools;
  7.        
  8.         @EventHandler
  9.         public void preLoad(FMLPreInitializationEvent preInitializationEvent){
  10.                
  11.                 //创建一种矿石
  12.                 blockIron = new BlockIron(Material.rock);   //创建一种基于BlockIron类的材质类型为石材质的砖块
  13.                 blockIron.setBlockName("myIron");                                //语言缺失时的名字
  14.                 blockIron.setBlockTextureName("diracon:myIron");                //纹理的名字
  15.                 blockIron.setHardness(4.0f); //设置砖块的硬度,这个硬度是相对于徒手而言的,泥土是0.5,石头是1.5,大部分矿石是3.0
  16.                 blockIron.setResistance(10.0f); //设置对爆炸的抗性,石头是10.0
  17.                 blockIron.setLightLevel(0.0f); //设置发光亮度,范围是0.0~1.0,南瓜灯,萤石和岩浆是1.0,通往下界的传送门是0.75.采集中的红石是0.625.
  18.                 blockIron.setStepSound(Block.soundTypeStone);  //设置踩在上面的脚步声.默认值是Block.soundTypeStone(石头地的声音.)
  19.                 blockIron.setCreativeTab(CreativeTabs.tabBlock);  //设置在创造模式中它在哪个菜单分类里.
  20.                 blockIron.setHarvestLevel("pickaxe", 4); //是设置开采砖块时需要使用的工具,可以是"pickaxe"(镐), "shovel"(铲子)或"axe"(斧头).后面的数值为工具材质要求,-1(默认值)为可以直接手撕,0是木质和金质,1是石质,2是铁质,3是钻石质.
  21.                 GameRegistry.registerBlock(blockIron,"myIron");  //注册一个砖块,只有在被注册后,砖块才能正式使用
  22.                
  23.                 //创建一个新的物品
  24.                 itemIron = new ItemIron();
  25.                 itemIron.setUnlocalizedName("myItem");                        //设置物品的默认名称,作用和砖块的setBlockName一样
  26.                 itemIron.setTextureName("diracon:myItem");                //设置物品的纹理名,和设置砖块的纹理名类似
  27.                 itemIron.setMaxStackSize(64);                                        //设置最大堆叠数量
  28.                 itemIron.setCreativeTab(CreativeTabs.tabMaterials);                        //设置其在创造模式菜单中的分类位置
  29.                 GameRegistry.registerItem(itemIron, "myItem");
  30.                
  31.                 //创建一种工具
  32.                 ironTools = new IronTools();
  33.                 ironTools.setUnlocalizedName("myIronTool");
  34.                 ironTools.setTextureName("diracon:myIronTool");
  35.                 ironTools.setCreativeTab(CreativeTabs.tabTools);
  36.                 ironTools.setHarvestLevel("pickaxe", 4);
  37.                 GameRegistry.registerItem(ironTools, "myIronTool");
  38.         }
  39.        
  40.         @EventHandler
  41.         public void load(FMLInitializationEvent initializationEvent){
  42.                
  43.         }
  44.        
  45.         @EventHandler
  46.         public void postLoad(FMLPostInitializationEvent postInitializationEvent){
  47.                
  48.         }
  49. }

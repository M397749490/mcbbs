基础配置教程
只简单介绍各位服主可能需要的几个基本配置项，更多功能请自行探索。
config.yml
   Global:
   # 离线模式，若你的服务器未开启正版模式，建议将此项设为true.
     OfflineMode: false
  # 版本检查
     versionCheck: true
  # 语言设置，改为Chinese即可。
     Language: English
  # 领地选取工具，默认是木棍。
     SelectionToolId: 280
     Size:
       # 允许的领地最小值
       # 1000 是 10x10x10 大小的领地
       MinimalSize: 1000
       MinimalX: 10
       MinimalY: 10
       MinimalZ: 10
groups.yml
   Global:
       Default: #组的名字
          Residence:
              # 是否可以创建领地
              CanCreate: true
              # 玩家可以拥有的领地数量最大值
              MaxResidences: 3
              # 领地可包含的最大物理区域
              MaxAreasPerResidence: 2
              # 领地最大格数（东西方向）
              MaxEastWest: 32
              # 领地最大格数（南北方向）
              MaxNorthSouth: 32
             # 领地最大格数（高度）
              MaxUpDown: 256

如果你有多世界插件请看以下方法
在你的插件文件夹plugins 找到Multiverse-Core文件夹里的worlds.yml
  plotworld:
    ==: MVWorld
    hidden: 'false'
    alias:
    color: WHITE
    style: NORMAL
    pvp: 'false'
    scale: '1.0'
    respawnWorld: ''
    allowWeather: 'true'
    difficulty: EASY
    spawning:
      ==: MVSpawnSettings
      animals:
        ==: MVSpawnSubSettings
        spawn: 'true'
        spawnrate: '-1'
        exceptions: []
      monsters:
        ==: MVSpawnSubSettings
        spawn: 'true'         ------------------------------------改为false即可 这样就不会自动刷怪了 注意是不刷怪 刷怪笼和刷怪蛋不受影响
        spawnrate: '-1'
        exceptions: []
    entryfee:
      ==: MVEntryFee
      amount: '0.0'
      currency: '-1'
    hunger: 'true'
    autoHeal: 'true'
    adjustSpawn: 'true'
    portalForm: ALL
    gameMode: SURVIVAL
    keepSpawnInMemory: 'true'
    spawnLocation:
      ==: MVSpawnLocation
      yaw: 0.0
      pitch: 0.0
      z: 2.5

      x: 3.5

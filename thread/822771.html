用相片+NBT数据
列子：
          ItemType = "pixelmon:pixelmon_sprite"
          UnsafeData = {
                   display:{
                      Name:"Arceus Photo"
            },
                        SpriteName:"pixelmon:sprites/pokemon/493-normal"
                }
      UnsafeDamage = 0
          DisplayName = "阿尔宙斯”

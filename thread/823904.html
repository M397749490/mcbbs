public void setItemLore(ItemStack itemStack, FileConfiguration config) {
    List<String> lore = config.getStringList("your_list_path");
    ItemMeta itemMeta = itemStack.getItemMeta();
    itemMeta.setLore(lore);
    itemStack.setItemMeta(itemMeta);
}

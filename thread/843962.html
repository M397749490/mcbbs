  41. function getExpByLevel(level, info) {
  42.         if (level <= 1) {
  43.         return 0;
  44.     }
  45.     if (level > 31) {
  46.         var exp = 887;
  47.         level -= 31;
  48.         for (var i = 1; i < level; i++) {
  49.             exp += 62 + (i * 7);
  50.         }
  51.         return exp;
  52.     }
  53.     if (level > 17) {
  54.         var exp = 272;
  55.         level -= 17;
  56.         for (var i = 1; i <= level; i++) {
  57.             exp += 17 + (i * 3);
  58.         }
  59.         return exp;
  60.     }
  61.     return (level - 1) * 13;
  62. }

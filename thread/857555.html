怪物:
  #怪物类型
  Type: PIG_ZOMBIE
  #怪物名字
  Display: '&7&l怪物'
  #怪物属性
  Health: 100
  Damage: 10
  Armor: 0
  #怪物技能
  Skills:
  #怪物装备
  Equipment:
  - 0:0
  - 0:4
  - 0:3
  - 0:2
  - 0:1
  #怪物装饰
  Disguise:
    Type: PLAYER
    Player: '&7&l怪物'
    Skin:
  BossBar:
    Enabled: true
    Title: '&a<mob.name> &a&l生命&8[&b&l<mob.hp>&f/&b&l<mob.mhp>&8] &c&l目标:&b&l<mob.tt.top>'
    Range: 40
    Color: RED
    Style: SEGMENTED_20   
  #怪物设置
  Options:
    PreventOtherDrops: true
    PreventRandomEquipment: true
    MovementSpeed: 0.3
    MaxCombatDistance: 10
    NoDamageTicks: 10
  Drops:
  - money 1
  - qhsg 1-3 1
  - exp 1-5
  #怪物击杀信息
  KillMessages:
  #不常用设置
  Faction: 敌人
  Mount:
  Modules:
    ThreatTable: true
  AIGoalSelectors:
  AITargetSelectors:
  DropsPerLevel:
  DamageModifiers:
  LevelModifiers:

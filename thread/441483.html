  1. auto-removal:
  2.   broadcast-message: '&6[服务器娘] &5一库哟~！服务器娘吞掉 +RemoveAmount 份遗失精华!'  #清理通知
  3.   broadcast-warning: true
  4.   enable-scheduler: true
  5.   autoremoval-interval: 460  #清理时间间隔
  6.   warnings:
  7.         - time:400 msg:&6[服务器娘] &e服务器娘的欲望已被勾起,预计 &7+remaining &e秒后达到高潮,基佬们请坚持勿交货!  #时间到400S时提示

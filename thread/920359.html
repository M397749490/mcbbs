  2. # 让使用此配置选项时, breakBlocks 也应该被设置为 "true".
  3. breakWhitelist:
  4.   # 一个方块ID并且伴有可用于破坏此方块的工具
  5.   SPONGE: ["IRON_PICKAXE", "STICK"]
  6.   # 这代表可以使用任何工具破坏泥土方块
  7.   DIRT: []

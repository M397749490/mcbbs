KEY COMMAND BIND
（可以让你的玩家按下数字键盘执行命令!）

插件功能
自定义绑定 1~9 按键
自定义选择每个按键是否开启
自定义按键冷却
自定义消耗valut
自定义消耗经验
支持shift+按键执行命令
支持飞行时执行命令
支持控制台命令
支持权限不足时自定义提示消息
配置非常简单！


配置文件(仅供参考)
一共有 数字1~9 的配置，自己根据这个配置文件写即可
  1. {
  2.       #关闭消息提示
  3.   "disableMessages": false,
  4.   "swapModeEnabled": true,
  5.   "keyPressModeEnabled": true,
  6.   "keyPressMode": {
  7.     "key1": {
  8.       #是否启用
  9.       "enabled": true,
  10.       "returnKeyEnabled": false,
  11.       #启用键位
  12.       "returnKeySlot": 1,
  13.       #只在潜行时开启(推荐启用)
  14.       "sneakingOnly": true,
  15.       #潜行时禁用
  16.       "disableOnSneak": false,
  17.       #飞行时禁用
  18.       "disableOnFly": false,
  19.       #有权限才能使用
  20.       "usePermission": false,
  21.       "permission": "kcb.key1",
  22.       #是否开启没有权限返回的提示
  23.       "noPermissionMessageEnabled": false,
  24.       #没有权限返回的提示
  25.       "noPermissionMessage": "Key is still in cooldown",
  26.       #冷却时间(毫秒)
  27.       "cooldownMilliSeconds": 500,
  28.       #是否开启冷却消息
  29.       "cooldownMessageEnabled": false,
  30.       #冷却消息
  31.       "cooldownMessage": "Key is still in cooldown",
  32.       #经验消耗是否开启
  33.       "enabledExpCost": false,
  34.       #经验消耗
  35.       "expCost": 1,
  36.       #经验不足是否提示
  37.       "insufficientExpMessageEnabled": false,
  38.       #经验不足的提示
  39.       "insufficientExpMessage": "Insufficient exp to run command",
  40.       #金币消耗是否开启
  41.       "moneyCostEnabled": false,
  42.       #金币消耗
  43.       "moneyCost": 1,
  44.       #金币不足的提示是否开启
  45.       "insufficientMoneyMessageEnabled": false,
  46.       #金币不足提示
  47.       "insufficientMoneyMessage": "Insufficient exp to run command",
  48.       #执行的指令 (在指令前加上-c- 可以转换为控制台指令)
  49.       "commands": [
  50.         "bs sky",

  52.       ]
  53.     },

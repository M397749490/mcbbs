试图使用这bug时会被清理宝石原理:
使用宝石的时候检测副手物品 如果名字中有 石,或灵
就触发并清空
开源:

package cc.timeland.mc;

public class BSPath extends org.bukkit.plugin.java.JavaPlugin implements org.bukkit.event.Listener{
    public void onEnable(){
    org.bukkit.Bukkit.getServer().getPluginManager().registerEvents(this, this);
    }
      @org.bukkit.event.EventHandler(priority=org.bukkit.event.EventPriority.HIGHEST)
  public void onInventoryClick(org.bukkit.event.inventory.InventoryClickEvent e)
  {
      for(String s : e.getWhoClicked().getInventory().getItemInOffHand().getItemMeta().getLore())
  if(e.getWhoClicked().getInventory().getItemInOffHand().getAmount() !=0 && s.contains("5颗同等级宝石")){
  e.setCancelled(true);
  e.getWhoClicked().getInventory().setItemInOffHand(null);
  e.getWhoClicked().sendMessage("使用bug,清空宝石");
   return;
  }
  }
    @org.bukkit.event.EventHandler(priority=org.bukkit.event.EventPriority.HIGHEST)
  public void onRightClick(org.bukkit.event.player.PlayerInteractEvent e)
  {
      for(String s : e.getPlayer().getInventory().getItemInOffHand().getItemMeta().getLore())
    if(e.getPlayer().getInventory().getItemInOffHand().getAmount() !=0 && s.contains("5颗同等级宝石")){
  e.setCancelled(true);
  e.getPlayer().getInventory().setItemInOffHand(null);
  e.getPlayer().sendMessage("使用bug,清空宝石");
   return;
  }
  }
}

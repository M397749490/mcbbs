示例
  1. import httplib
  2. headers = {"Cookie": "开F12在Network里随便右键一个www.mcbbs.net的请求再点复制为cURL就能拿到了" ,"Accept": "text/plain"}
  3. conn = httplib.HTTPConnection("www.mcbbs.net")
  4. conn.request("GET", "/home.php?mod=space&do=notice&type=reward&page=2", "", headers)
  5. response = conn.getresponse()
  6. print response.status, response.reason
  7. data = response.read()
  8. conn.close()

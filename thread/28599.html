plugins\SmartServerTool\config.yml 使用文本编辑器打开：
  1. world-setting:
  2.   disallowbuild: true
  3.   disablebreak: true
  4.   Blockcreeper: true
  5.   prevent-tnt: true
  6.   Blockender: true
  7.   xpsave: true
  8.   prevent-fire-spread: true
  9.   prevent-lava-spread: true
  10.   general-spread: true
  11.   flint-and-steal-spread: true

  disallowbuild: true        改为false
  disablebreak: true        改为false

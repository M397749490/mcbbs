  1. {
  2.   "type": "minecraft:crafting_shaped",
  3.   "group": "test",
  4.   "output": {
  5.     "item": "minecraft:apple"
  6.   },
  7.   "pattern": [ "###", "#*#", "###" ],
  8.   "keys": {
  9.     "#": { "item": "minecraft:gold_ingot" },
  10.     "*": { "item": "minecraft:egg" }
  11.   }

如果需要精准的冷却时间判断的话，可以用这个评分评分
package cn.mcres.karlatemp.mxlib.test;

import java.util.Map;
import java.util.Optional;
import java.util.UUID;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.TimeUnit;

/**
 * Create at 2019/11/20 22:30
 * Copyright Karlatemp
 * MXLib $ cn.mcres.karlatemp.mxlib.test
 */
public class TimedTask {
    private static final Map<
            UUID, // 玩家UUID
            Map<
                    String, // 辨认用, task唯一ID, 用于多task多冷却
                    Long // 存储冷却结束时间
                    >
            > GLOBAL_TIME_STORE = new ConcurrentHashMap<>(); // 定义全局冷却

    /**
     * 清除存储的全部延迟几率, 在PlayerQuitEvent里调用一次解除内存占用
     *
     * @param uniqueId 玩家UUID
     */
public static void clearAllTimeDelay(UUID uniqueId) {
        GLOBAL_TIME_STORE.remove(uniqueId);
    }

    /**
     * 判断目标Task是否在冷却中
     *
     * @param uniqueId 玩家UUID
     * @param taskId   Task的唯一ID
     * @Return 是否在冷却时间
*/
public static boolean isInDelay(UUID uniqueId, String taskId) {
        return Optional.ofNullable(GLOBAL_TIME_STORE.get(uniqueId))
                .map(tasks -> tasks.get(taskId))
                .map(
                        time -> System.currentTimeMillis() < time // 判断是否冷却没结束, 如果系统当前时间没超过存储时间那么还在冷却期内
                ).orElse(false); // 如果没有存储几率那么不在冷却时间内
    }

    /**
     * 获取冷却时间
*
     * @param uniqueId 玩家UUID
     * @param taskId   Task的唯一ID
     * @return 冷却时间(毫秒), 如果为-1的话表示目标Task不在冷却中
*/
public static long getDelayingTime(UUID uniqueId, String taskId) {
        return Optional.ofNullable(GLOBAL_TIME_STORE.get(uniqueId))
                .map(tasks -> tasks.get(taskId))
                .map(end -> end - System.currentTimeMillis())
                .filter(delay -> delay >= 0).orElse(-1L);
    }

    /**
     * 使目标Task进入冷却期
     *
     * @param uniqueId 玩家UUID
     * @param taskId   唯一 Task ID
     * @param delays   冷却时间(毫秒)
     */
public static void startDelay(UUID uniqueId, String taskId, long delays) {
        Map<String, Long> tasks;
        if (GLOBAL_TIME_STORE.containsKey(uniqueId)) {
            tasks = GLOBAL_TIME_STORE.get(uniqueId);
        } else {
            GLOBAL_TIME_STORE.put(uniqueId, tasks = new ConcurrentHashMap<>());
        }
        tasks.put(taskId, System.currentTimeMillis() + delays);
    }

    /**
     * 强行结束目标Task的冷却
     *
     * @param uniqueId 玩家UUID
     * @param taskId   唯一 Task ID
     */
public static void endDelay(UUID uniqueId, String taskId) {
        if (GLOBAL_TIME_STORE.containsKey(uniqueId)) {
            GLOBAL_TIME_STORE.get(uniqueId).remove(taskId);
        }
    }

    /**
     * 示例
     */
public static void main(String[] args) throws Throwable {
        UUID playerUUID = new UUID(0xCED25DD238D94E00L, 0xA0A6B22946BD0D8FL);
        String taskId = "Example Task";
        System.out.println("0: 目标冷却时间:" + TimeUnit.MILLISECONDS.toSeconds(getDelayingTime(playerUUID, taskId)));
        if (isInDelay(playerUUID, taskId)) {
            System.out.println("1: 还在冷却期中, 拒绝访问");
            return;
        }
        startDelay(playerUUID, taskId, TimeUnit.SECONDS.toMillis(5)); // 5 秒
        if (isInDelay(playerUUID, taskId)) {
            System.out.println("2: 进入了冷却期");
        }
        System.out.println("3: 目标冷却时间:" + TimeUnit.MILLISECONDS.toSeconds(getDelayingTime(playerUUID, taskId)));
        TimeUnit.SECONDS.sleep(7); // 停止6 秒
        System.out.println("4: 是否在冷却期: " + isInDelay(playerUUID, taskId));
        System.out.println("5: 目标冷却时间:" + TimeUnit.MILLISECONDS.toSeconds(getDelayingTime(playerUUID, taskId)));
        startDelay(playerUUID, taskId, TimeUnit.SECONDS.toMillis(50));
        endDelay(playerUUID, taskId);
        System.out.println("6: 目标冷却时间:" + TimeUnit.MILLISECONDS.toSeconds(getDelayingTime(playerUUID, taskId)));
    }
}

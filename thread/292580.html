  安装方法
配置主服务器的config.php文件：
  1. $config['version'] = "3.0"; //版本
  2. $config['title'] = "桉树叶免费图床"; //站点标题
  3. $config['filesize'] = "1024"; //文件最大尺寸，单位：KB
  4. $config['allowExt'] = "jpg,jpeg,png,bmp,gif"; //大小写均可，多个间用“,”隔开
  5. $config['length'] = 6; //保存的文件名长度，请不要设置得过小，以免出现重复！
  6. $config['remote'] = "0"; //0:不启用远程备份，1:启用远程备份
  7. $config['remote_only'] = "0"; //0:使用本地储存，1:仅使用远程储存，不使用本地储存
  8. $config['r_host'] = "http://www.example.com/"; //远程服务端地址，必须填写完整，以“/”结尾
  9. $config['key'] = "connection_key"; //连接密钥，【必须】与远程服务端保持一致

   啦啦啦，我是43444，大家都知道Herobiren吧，今天我就来教大家怎么做出吓人的Herobiren。
   先来几张图图君













指令：

  1. 1./summon ArmorStand ~ ~1 ~ {ActiveEffects:[{Id:14,Duration:9999999,Ambient:1}], Attributes:[{Name:generic.attackDamage,Base:-1},{Name:generic.followRange,Base:350}], Equipment:[{id:267,tag:{AttributeModifiers:[{Operation:0,UUIDLeast:1,UUIDMost:1,Amount:-1.0,AttributeName:generic.attackDamage,Name:No_Attack}]}},{id:313},{id:312},{id:311},{id:397,Damage:3,tag:{SkullOwner:Herobrine}}], Invulnerable:1,CustomName:Herobrine,CustomNameVisible:1，NoBasePlate:1}
  2. 2./tellraw @a {text:"",extra:[{text:"Herobiren ",color:yellow,clickEvent:{action:suggest_command,value:"/msg Herobiren"}, hoverEvent:{action:show_text,value:"Herobiren \n1626ae14-1bca-459f-8b2b-3f7bcfa42926"}},{color:yellow,text:"joined the game"}]}
  3. 3./tellraw @a {text:"<",extra:[{text:"Herobiren",clickEvent:{action:suggest_command,value:"/msg Herobiren"}, hoverEvent:{action:show_text,value:"Herobiren \n1626ae14-1bca-459f-8b2b-3f7bcfa42926"},color:white},{text:"> "},{text:"You die"}]}
  4. 4.tp @e[type=ArmorStand] @p
  5. 5.kill @p
  6. 6./tellraw @a {text:"",extra:[{text:"Herobiren ",color:yellow,clickEvent:{action:suggest_command,value:"/msg Herobiren"}, hoverEvent:{action:show_text,value:"Herobiren \n1626ae14-1bca-459f-8b2b-3f7bcfa42926"}},{color:yellow,text:"left the game"}]}
  7. 7.kill @e[type=ArmorStand]

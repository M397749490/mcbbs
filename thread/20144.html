  2. IF /I "%PROCESSOR_ARCHITECTURE:~-2%"=="64"  "%ProgramFiles (x86)%\Java\JRE6\bin\java.exe" -Xms1024M -Xmx2048M -jar "%~dp0craftbukkit.jar"
  3. IF /I "%PROCESSOR_ARCHITECTURE:~-2%"=="86"  "%ProgramFiles%\Java\JRE6\bin\java.exe" -Xincgc -Xmx1024M -jar "%~dp0craftbukkit.jar"

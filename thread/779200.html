代码:
distance填距离,delta填坐标误差允许的范围
  1. public static Entity getTargetEntity(Player player, double distance, double delta) {
  2.     for (Entity entity : player.getNearbyEntities(distance, distance, distance)) {
  3.         double d = entity.getLocation().distance(player.getLocation());
  4.         if (d > distance) {
  5.             continue;
  6.         }
  7.         Location eye = player.getEyeLocation();
  8.         Vector direction = eye.getDirection();
  9.         double a = direction.getX();
  10.         double b = direction.getY();
  11.         double c = direction.getZ();
  12.         double s = Math.sqrt(a * a + b * b + c * c);
  13.         Location entityLoc = entity.getLocation();
  14.         if (Math.abs((entityLoc.getX() - eye.getX()) - (a / s * d)) < delta
  15.                 && Math.abs((entityLoc.getY() - eye.getY()) - (b / s * d)) < delta
  16.                 && Math.abs((entityLoc.getZ() - eye.getZ()) - (c / s * d)) < delta) {
  17.             if (player.hasLineOfSight(entity)) {
  18.                 return entity;
  19.             }
  20.         }
  21.     }
  22.     return null;
  23. }

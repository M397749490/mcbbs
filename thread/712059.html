多世界的Multiverse-Core\worlds.yml里
worlds:
  sc:                        #选择一个世界
    ==: MVWorld
    hidden: 'false'
    alias: ""
    color: WHITE
    style: NORMAL
    pvp: 'true'
    scale: '1.0'
    respawnWorld: ''
    allowWeather: 'true'
    difficulty: NORMAL
    spawning:
      ==: MVSpawnSettings
      animals:                  
        ==: MVSpawnSubSettings
        spawn: 'true'     #允许刷动物    
        spawnrate: '-1'
        exceptions: []
      monsters:
        ==: MVSpawnSubSettings
        spawn: 'true'    #true改为false就不刷怪了
        spawnrate: '-1'
        exceptions: []

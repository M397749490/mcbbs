maven可以使用本地依赖
  1.       <dependency>
  2.          <groupId>ldapjdk</groupId>
  3.          <artifactId>ldapjdk</artifactId>
  4.          <scope>system</scope>
  5.          <version>1.0</version>
  6.          <systemPath>${basedir}\src\lib\ldapjdk.jar</systemPath>
  7.       </dependency>

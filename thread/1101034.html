  2. ConfigVersion: 13

  5.   UseCache: true
  6. #是否启用混淆矿物

  8. #方块被破坏时还原周围的方块为正常状态
  9.   UpdateOnDamage: true
  10. #当它为true时,在NoObfuscationForMetadataTagName中指定了标签(名字)的玩家
  11. #插件不会在其附近持续检测并生成假矿
  12. #这对服务器NPC很有用（例如,在Citizens插件中的NPC）
  13.   NoObfuscationForMetadata: true
  14. #管理员有权查看正常的矿物分布
  15.   NoObfuscationForOps: false
  16. #有权限的人可以查看正常的矿物分布
  17.   NoObfuscationForPermission: false
  18. #通知OP和具有权限的玩家Orebfuscator已被禁用
  19. #应该是进入游戏的时候通知的
  20.   LoginNotification: true
  21. Integers:
  22. #最大加载的缓存文件数
  23.   MaxLoadedCacheFiles: 64
  24. #缓存文件的存活天数,如果到达指定天数不被读写过将被自动删除
  25. #如果指定为0，则功能被禁用
  26.   DeleteCacheFilesAfterDays: 0
  27. #模式
  28. # 1 用指定方块替换目标方块
  29. # 2 用随机方块替换目标方块
  30.   EngineMode: 2
  31. #发送初始方块后,半径搜索透明方块
  32. #建议将值设置为小于3
  33.   InitialRadius: 1
  34. #当方块被击中，破坏或改变状态时
  35. #将附近半径内的风格还原到正常状态
  36. #建议将值设置为小于4
  37.   UpdateRadius: 2
  38. Strings:
  39. #缓存文件将存储到的文件夹的相对路径

  43.   NoObfuscationForMetadataTagName: NPC
  44. #此部分需要手动增加
  45. Lists:

  48.   TransparentBlocks: []
  49. #不透明
  50.   NonTransparentBlocks: []

  54. #有4种类型的世界
  55. Worlds:

  58. #按主世界NORMAL,地狱NETHER和末地THE_END类型的世界设置始终继承DEFAULT世界类型设置

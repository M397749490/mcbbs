配置文件：
#a,b就是物品代号
#name就是物品名字
#lore就是物品下面的介绍
#fix就是能修多少
#id就是物品ID
#物品可以自己加,没有限制
#由于懒得切换颜色,所以颜色符号请使用§,而不是&
Item:
    a:
      name: "§e修复宝石"
      id: 378
      lore:
        - "§e低级修复宝石"
      Fix: 100
    b:
      name: "§e中级宝石"
      id: 378
      lore:
        - "§e中级修复宝石"
      Fix: 300
    c:
      name: "§e超级无敌大宝石"
      id: 378
      lore:
        - "§e超级超级超级厉害的宝石"
        - "§e这里是第二行"
      Fix: 1000
#背包中ID为以下的物品,才进行修复
ID:
    - 310
    - 311
    - 312
    - 313
    - 276

…… MOD不能直接调用bukkit插件的API 但是可以调用Bukkit的API
  1. VaultClassLoader=Bukkit.getPluginManager().getPlugin("Vault").getClass().getClassLoader();
  2. VaultFindClass=VaultClassLoader.getClass().getDeclaredMethod("findClass",Class.forName("java.lang.String"));
  3. VaultFindClass.setAccessible(true);
  4. Economy=VaultFindClass.invoke(VaultClassLoader,"net.milkbowl.vault.economy.Economy");

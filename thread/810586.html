简单说 JSON启动命令如下

  2. -Xmn"最小内存"
  3. -Xmx"最大内存
  4. -Djava.library.path=.\.minecraft\versions\1.12.2\1.12.2-natives
  5. -Dfml.ignoreInvalidMinecraftCertificates=true
  6. -Dfml.ignorePatchDiscrepancies=true
  7. --cp .\.minecraft\libraries\com\mojang\patchy\1.1\patchy-1.1.jar;
  8. 。。
  9. 。。
  10. 。
  11. 。。

  13. net.minecraft.client.main.Main

  15. --version "随意内容"
  16. --gameDir ".\.minecraft"
  17. --assetsDir ".\.minecraft\assets"
  18. --assetIndex 1.12
  19. --uuid "随意一段UUID"
  20. --accessToken "随意一段32字符长度的字符"

  3.   # 10 = 动态岩浆 :: 11 = 静态岩浆 :: 46 = TNT :: 327 = 岩浆桶

  5.     on-placement: 10,11,46,327
  6.     on-use: 327
  7.     on-break:

  9.   blacklist:

  11.     # 阻止玩家放置以下方块
  12.     placement: 10,11,46,327

  14.     # 阻止玩家使用以下物品
  15.     usage: 327

  17.     # 阻止玩家破坏以下方块
  18.     break:

  20.     # 阻止活塞推动以下方块

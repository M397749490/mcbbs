  17.             world = sender.getWorld();
  18.         }

  20.     }

  22.     // 实现/pluginloader reload [<plugin>]
  23.     @CommandAnnotation(args = "[<plugin>]", des = "重载插件")
  24.     public void reload(@Sender CommandSender sender, @OptionalParam String plugin) {
  25.         if (plugin == null) {
  26.             plugin = "";
  27.         }
  28.         logger.info("开始重载插件{}", plugin);
  29.         sender.sendMessage("开始重载插件" + plugin);

  31.         Stopwatch watch = Stopwatch.createStarted();
  32.         List<ServiceHolder> services = BeanContext.getServices();

  34.         String pluginId = plugin;

  36.         services.stream()
  37.                 .forEach(s -> {
  38.                     if (!StringUtils.isEmpty(pluginId) && !s.getPlugin().getId().equalsIgnoreCase(pluginId)) {
  39.                         return;
  40.                     }
  41.                     try {
  42.                         //重载默认配置文件
  43.                         s.getPlugin().setConfigNode(s.getPlugin().getConfigLoader().load());
  44.                         long count = ConfigUtils.injectConfigure(s.getPlugin().getConfigNode(), s.getHandle());
  45.                         if (count > 0) {
  46.                             logger.info("重载配置:" + s.getId());
  47.                         }
  48.                     } catch (Throwable e) {
  49.                         logger.error("配置注入失败:" + s.getId(), e);
  50.                     }
  51.                 });
  52.         LoaderUtils.invokeMethods(services, Reload.class, "reload");
  53.         this.plugin.saveConfig();
  54.         watch.stop();
  55.         sender.sendMessage("重载完毕,耗时" + watch.elapsed(TimeUnit.MILLISECONDS) + "ms");
  56.     }
  57. }

?    mns的混淆名查找，记不住啊。。。神烦

  1. // 假设服务器是多线，怎么能知道玩家连接的是服务器的哪个ip呢
  2. // 这里使用反射拿到玩家对应的nettyChannel，反射使用的名字全是为混淆的名字，会根据1.7.10的混淆规则自动映射混淆的方法
  3. public String getServerIp(Player p) {
  4.          //常规方式
  5.         WrapClass<? extends Player> wc = ClassUtils.wrap(p.getClass());
  6.         WrapField wentity = wc.getField("entity");
  7.         WrapField wplayerNetServerHandler = ClassUtils.wrap(wentity.getType()).getField("playerNetServerHandler");
  8.         WrapField wnetManager = ClassUtils.wrap(wplayerNetServerHandler.getType()).getField("netManager");
  9.         WrapMethod wchannel = ClassUtils.wrap(wnetManager.getType()).getMethod("channel");
  10.         io.netty.channel.Channel channel0 = (Channel) wchannel.invoke(wnetManager.get(wplayerNetServerHandler.get(wentity.get(p))));

  12.         //快捷方式，链式调用
  13.         io.netty.channel.Channel channel = Invoker.of(p)
  14.                 .get("entity")
  15.                 .get("playerNetServerHandler")
  16.                 .get("netManager")
  17.                 .invoke("channel")
  18.                 .get();
  19.         return channel.localAddress().toString().replaceFirst("/", "").split(":", 2)[0];
  20. }

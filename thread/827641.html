  • 配置文件

配置文件 位于.minecraft\config\mineral_expansion.cfg
  1. # Configuration file

  3. general {
  4.     # 是否允许合成 基岩套
  5.     B:RegBedrockArmorRecipe=true            //true=允许合成 false=不允许合成

  7.     # 是否允许合成 钻石牛排
  8.     B:RegDiamondBeefRecipe=true            //true=允许合成 false=不允许合成

  10.     # 是否允许合成 绿宝石套
  11.     B:RegEmeraldArmorRecipe=true            //true=允许合成 false=不允许合成

  13.     # 是否允许合成 混乱之镐
  14.     B:RegHunluanPickaxeRecipe=true            //true=允许合成 false=不允许合成

  16.     # 是否允许合成 混乱
  17.     B:RegHunluanRecipe=true            //true=允许合成 false=不允许合成

  19.     # 是否允许合成 混乱之剑
  20.     B:RegHunluanSwordRecipe=true            //true=允许合成 false=不允许合成

  22.     # 可挖基岩的最低高度 Y
  23.     I:lowestHeight=1            //设为1表示Y小于1的基岩不能被挖掘 可以防止世界被挖穿
  24. }

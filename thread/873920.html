卧槽 ! 这个报错能修复了!!!!!!!!!!!!!!!

而且非常简单,只需要修改下 luckperms 插件的配置文件 config.yml 即可.

  1. data:

  3.   # Define the address and port for the database.
  4.   # - The standard DB engine port is used by default
  5.   #   (MySQL: 3306, PostgreSQL: 5432, MongoDB: 27017)
  6.   # - Specify as "host:port" if differs
  7.   address: localhost

  9.   # The name of the database to store LuckPerms data in.
  10.   # - This must be created already. Don't worry about this setting if you're using MongoDB.
  11.   database: luckperms

  13.   # Credentials for the database.
  14.   username: root
  15.   password: root

  17.   # These settings apply to the MySQL connection pool.
  18.   # - The default values will be suitable for the majority of users.
  19.   # - Do not change these settings unless you know what you're doing!
  20.   pool-settings:

  22.     # Sets the maximum size of the MySQL connection pool.
  23.     # - Basically this value will determine the maximum number of actual
  24.     #   connections to the database backend.
  25.     # - More information about determining the size of connection pools can be found here:
  26.     #   https://github.com/brettwooldridge/HikariCP/wiki/About-Pool-Sizing
  27.     maximum-pool-size: 10

  29.     # Sets the minimum number of idle connections that the pool will try to maintain.
  30.     # - For maximum performance and responsiveness to spike demands, it is recommended to not set
  31.     #   this value and instead allow the pool to act as a fixed size connection pool.
  32.     #   (set this value to the same as 'maximum-pool-size')
  33.     minimum-idle: 10

  35.     # This setting controls the maximum lifetime of a connection in the pool in milliseconds.
  36.     # - The value should be at least 30 seconds less than any database or infrastructure imposed
  37.     #   connection time limit.
  38.     maximum-lifetime: 1800000 # 30 minutes

  40.     # This setting controls the maximum number of milliseconds that the plugin will wait for a
  41.     # connection from the pool, before timing out.
  42.     connection-timeout: 5000 # 5 seconds

  44.     # This setting allows you to define extra properties for connections.
  45.     properties:
  46.       useUnicode: true
  47.       characterEncoding: utf8
  48.       # 删除下面的 # 注释即可关闭 mysql 的 ssl 连接
  49.       # 并且后台不会再出现那个烦人的报错提示了
  50.       #useSSL: false

  52.   # The prefix for all LuckPerms SQL tables.
  53.   # - Change this is you want to use different tables for different servers.
  54.   table_prefix: ''

  56.   # The prefix to use for all LuckPerms collections. Change this if you want to use different
  57.   # collections for different servers. The default is no prefix.
  58.   mongodb_collection_prefix: ''

  60.   # MongoDB ClientConnectionURI for use with replica sets and custom connection options
  61.   # - See https://docs.mongodb.com/manual/reference/connection-string/
  62.   mongodb_connection_URI: ''

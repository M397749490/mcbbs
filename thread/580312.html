标题: VC++ 实现强制释放内存
代码：
#include <windows.h>
#include <Psapi.h>

int WINAPI WinMain(_In_ HINSTANCE hInstance, _In_opt_ HINSTANCE hPrevInstance, _In_ LPSTR lpCmdLine, _In_ int nShowCmd)
{
//VS2013/VS2015编译通过
    MEMORYSTATUSEX stcMemStatus = { sizeof(MEMORYSTATUSEX) };
    GlobalMemoryStatusEx(&stcMemStatus);
    DWORDLONG preUsedMem = stcMemStatus.ullTotalPhys - stcMemStatus.ullAvailPhys;

    DWORD dwPIDList[1000] = { 0 };
    DWORD bufSize = sizeof(dwPIDList);
    DWORD dwNeedSize = 0;
    EnumProcesses(dwPIDList, bufSize, &dwNeedSize);
    for (DWORD i = 0; i < dwNeedSize / sizeof(DWORD); i++)
    {
        HANDLE hProcess = OpenProcess(PROCESS_SET_QUOTA, false, dwPIDList);
        SetProcessWorkingSetSize(hProcess, -1, -1);
    }
    return 0;
}

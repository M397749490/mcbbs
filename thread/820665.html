这个问题可以被抽象成一个概念: throttle,也就是节流，指一个行为在一定时间内只执行一次，Example:
  1. class Throttle(override var wait: Long) {

  3.     private val storeMap = mutableMapOf<UUID, Long>()

  5.     override fun process(uuid: UUID, success: () -> Unit, not: (surplus: Long) -> Unit) {
  6.         if (storeMap.containsKey(uuid)) {
  7.             val prevTime = storeMap[uuid]!!
  8.             val pastTime = System.currentTimeMillis() - prevTime
  9.             if (pastTime > wait) {
  10.                 success.invoke()
  11.                 storeMap[uuid] = System.currentTimeMillis()
  12.             } else {
  13.                 not.invoke(wait - pastTime)
  14.             }
  15.         } else {
  16.             success.invoke()
  17.             storeMap[uuid] = System.currentTimeMillis()
  18.         }
  19.     }

  21. }

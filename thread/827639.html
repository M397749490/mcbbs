BedwarsItemAddon
起床战争·道具拓展




前言

这是一款起床战争道具拓展插件
需要依赖BedwarsRel运行
让起床战争也拥有有趣的小道具
你的起床战争将会变得更加有趣！

3、降落伞

5、搭桥蛋
6、骑行末影珍珠

10、行走平台


指令、权限

指令:
  /bwia  -  显示插件信息
  /bwia reload  -  重载配置文件

权限:
  bedwarsitemaddon.reload  -  允许重载配置文件


配置文件


#信息
message:

  #冷却提示
  # 变量:
  #   {time}    冷却时间
  cooling: "&e冷却中，请等待 &c{time} &e秒后使用！"

#道具
items:


  #火球
  fireball:
  
    #启用
    enabled: true
   
    #冷却时间(秒)
    cooldown: 0
   
    #伤害
    damage: 3


  #TNT自动点燃
  tnt:
  
    #启用
    enabled: true
   
    #冷却时间(秒)
    cooldown: 0
   
    #伤害
    damage: 1
   
    #弹射模式
    ejection: true
   
    #弹射速度
    velocity: 1


  #降落伞
  parachute:
  
    #启用
    enabled: true
   
    #冷却时间(秒)
    cooldown: 5
   
    #物品
    item: "FEATHER"
   
    #弹射速度(高度)
    velocity: 3
   
    #降落速度
    landing_velocity: 0.2


  #蹦床
  trampoline:
  
    #启用
    enabled: true
   
    #冷却时间(秒)
    cooldown: 5
   
    #物品
    item: "DOUBLE_PLANT"
   
    #弹射速度(高度)
    velocity: 3
   
    #蹦床多少秒后消失
    staytime: 20
   
    #空间不足提示
    lack_space: "没有足够的空间使用此道具！"


  #搭桥蛋
  bridge_egg:
  
    #启用
    enabled: true
   
    #冷却时间(秒)
    cooldown: 5
   
    #每次最大放置方块数量
    maxblock: 50


  #骑行末影珍珠
  ender_pearl_chair:
  
    #启用
    enabled: true
   
    #冷却时间(秒)
    cooldown: 0

  #队伍防守铁傀儡
  team_iron_golem:
  
    #启用
    enabled: true
   
    #冷却时间(秒)
    cooldown: 0
   
    #铁傀儡有效时间
    # -1 禁用此功能
    staytime: 120
   
    #血量
    health: 20
   
    #对玩家造成的伤害
    damage: 3
   
    #铁傀儡名称
    name: "{color}{team} 铁傀儡 {time}秒"

使用notepad++打开essentials/config.yml找到保护部分，将想要阻止的设置为true：
  1. protect:

  3.   # 总体的物理现象或行为的设置
  4.   prevent:                                 # 阻止以下现象(译注:true就是阻止,false是允许,这里需要注意一下)
  5.     lava-flow: false                       # 阻止 岩浆的流动
  6.     water-flow: false                      # 阻止 水的流动
  7.     water-bucket-flow: false               # 阻止 水桶水的流动
  8.     fire-spread: true                      # 阻止 火的传播
  9.     lava-fire-spread: true                 # 阻止 岩浆引起的火的传播
  10.     flint-fire: false                      # 阻止 打火石产生火
  11.     lightning-fire-spread: true            # 阻止 闪电引起的火的传播
  12.     portal-creation: false                 # 阻止 传送门的创造
  13.     tnt-explosion: true                    # 阻止 TNT的爆炸
  14.     tnt-playerdamage: false                # 阻止 TNT对玩家造成伤害
  15.     tnt-minecart-explosion: true           # 阻止 TNT矿车 爆炸
  16.     tnt-minecart-playerdamage: false       # 阻止 TNT矿车 对玩家造成伤害
  17.     fireball-explosion: false              # 阻止 火球的爆炸(译注:恶魂发射的火球)
  18.     fireball-fire: false                   # 阻止 火球的燃烧
  19.     fireball-playerdamage: false           # 阻止 火球对玩家造成伤害
  20.     witherskull-explosion: false           # 阻止 凋零攻击的爆炸
  21.     witherskull-playerdamage: false        # 阻止 凋零攻击对玩家照成伤害
  22.     wither-spawnexplosion: true            # 阻止 凋零出生时的爆炸
  23.     wither-blockreplace: false             # 阻止 凋零破坏方块
  24.     creeper-explosion: false               # 阻止 爬行者的爆炸
  25.     creeper-playerdamage: false            # 阻止 爬行者对玩家造成伤害
  26.     creeper-blockdamage: false             # 阻止 爬行者炸毁方块
  27.     enderdragon-blockdamage: false         # 阻止 末影龙破坏方块
  28.     enderman-pickup: true                  # 阻止 末影人拾取方块
  29.     villager-death: false                  # 阻止 村民被杀死

  1. package com.github.berry64;

  3. import java.io.File;

  5. import org.bukkit.Bukkit;
  6. import org.bukkit.ChatColor;
  7. import org.bukkit.command.Command;
  8. import org.bukkit.command.CommandSender;
  9. import org.bukkit.command.ConsoleCommandSender;
  10. import org.bukkit.entity.Player;
  11. import org.bukkit.plugin.java.JavaPlugin;

  13. public class Warnmain extends JavaPlugin{
  14.         @Override
  15.         public void onEnable(){
  16.                 getLogger().info("警告插件已启动!");
  17.                 CreateConfig();
  18.         }
  19.         public boolean onCommand(CommandSender sender, Command cmd, String label, String[] args){
  20.                 if(cmd.getName().equalsIgnoreCase("warn")){
  21.                         if(sender.hasPermission("warn.warn")){
  22.                                 if (args.length != 2){
  23.                                         sender.sendMessage(ChatColor.RED+ "请使用/warn [玩家] [原因]");
  24.                                 }
  25.                                 else{
  26.                                         Player target = (Bukkit.getServer().getPlayer(args[0]));
  27.                                         if(target == null){
  28.                                                 sender.sendMessage(ChatColor.AQUA+ args[0]+"不存在");
  29.                                         }
  30.                                         else{
  31.                                                 sendmessage(args[0], args[1]);
  32.                                                 sender.sendMessage(ChatColor.YELLOW+"已向玩家"+ args[0] +"发送警告, 原因:"+ ChatColor.GREEN+ args[1]);
  33.                                         }
  34.                                 }
  35.                         }
  36.                 }
  37.                 return true;
  38.         }
  39.         public void CreateConfig(){
  40.                 if (!new File(getDataFolder() + File.separator + "config.yml").exists()) {
  41.                         saveDefaultConfig();
  42.                         say(ChatColor.YELLOW + "无法找到config.yml,正在创建");
  43.                 } try {
  44.                         reloadConfig();
  45.                         say(ChatColor.YELLOW + "成功加载config");
  46.                 } catch (Exception e) {
  47.                         e.printStackTrace();
  48.               getServer().getPluginManager().disablePlugin(this);
  49.               say(ChatColor.RED + "无法读取config");
  50.                 }
  51.         }
  52.         public void say(String s){
  53.                 ConsoleCommandSender sender = Bukkit.getConsoleSender();
  54.                 sender.sendMessage(s);
  55.         }
  56.         public void sendmessage(String p, String m){
  57.                 Player asdf = (Bukkit.getServer().getPlayer(p));
  58.                 asdf.sendMessage(ChatColor.DARK_GRAY+"=========================");
  59.                 asdf.sendMessage((getConfig().getString("Warn_message").replaceAll("<reason>", m).replaceAll("&", "§")));
  60.                 asdf.sendMessage(ChatColor.DARK_GRAY+"=========================");
  61.         }
  62. }

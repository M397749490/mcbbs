  1. @Plugin(id = MultiExp.PLUGIN_ID,name = MultiExp.PLUGIN_NAME,authors = {"GiNYAi"})
  2. public class MultiExp {
  3.     public final static String PLUGIN_ID = "multiexp";
  4.     public final static String PLUGIN_NAME = "MultiExp";

  6.     @Inject
  7.     private Logger logger;

  9.     @Inject
  10.     @ConfigDir(sharedRoot = true)
  11.     private Path defaultConfigDir;
  12.     private Path configPath;

  14.     private ConfigurationLoader<CommentedConfigurationNode> configurationLoader;
  15.     private CommentedConfigurationNode rootNode;
  16.     private LinkedHashMap<String,Double> scaleMap;
  17.     private HashMap<String,String> messageMap;

  19.     @Listener
  20.     public void onPreLoad(GamePreInitializationEvent event){
  21.         MinecraftForge.EVENT_BUS.register(this);
  22.         configPath = defaultConfigDir.resolve(PLUGIN_NAME+".conf");
  23.         try {
  24.             if(!configPath.toFile().exists()){
  25.                 Sponge.getAssetManager().getAsset(this,"default_config.conf").get()
  26.                         .copyToFile(configPath);
  27.             }
  28.             reload();
  29.         } catch (IOException e) {
  30.             logger.error(PLUGIN_NAME+" failed to load config",e);
  31.         }
  32.     }

  34.     @Listener
  35.     public void onGameStart(GameStartingServerEvent event){
  36.         CommandSpec reload = CommandSpec.builder()
  37.                 .permission("MultiExp.Commands.Reload")
  38.                 .executor(((src, args) -> {
  39.                     try {
  40.                         reload();
  41.                     }catch (Exception e){
  42.                         logger.error(PLUGIN_NAME+" failed to load config",e);
  43.                         throw new CommandException(Text.of("Failed to reload."),e);
  44.                     }
  45.                     src.sendMessage(Text.of("Reload complete."));
  46.                     return CommandResult.success();
  47.                 })).build();
  48.         CommandSpec root = CommandSpec.builder()
  49.                 .permission("MultiExp.Commands.Command")
  50.                 .child(reload,"reload")
  51.                 .build();
  52.         Sponge.getCommandManager().register(this,root,PLUGIN_NAME);
  53.     }

  55.     @Listener
  56.     public void onReload(GameReloadEvent event) {
  57.         try {
  58.             reload();
  59.         }catch (Exception e){
  60.             logger.error(PLUGIN_NAME+" failed to load config",e);
  61.         }
  62.     }

  64.     private void reload() throws IOException {
  65.         configurationLoader = HoconConfigurationLoader.builder().setPath(configPath).build();
  66.         rootNode = configurationLoader.load();
  67.         scaleMap = new LinkedHashMap<>();
  68.         messageMap = new HashMap<>();
  69.         for(CommentedConfigurationNode node:rootNode.getNode("Multiples").getChildrenMap().values()){
  70.             String permission = node.getNode("Permission").getString("MultiExp.Multiples."+node.getKey().toString());
  71.             double scale = node.getNode("Scale").getDouble();
  72.             String message = node.getNode("Message").getString(null);
  73.             if(scaleMap.containsKey(permission)){
  74.                 logger.warn("Multiple "+node.getKey().toString()+" use the permission:"+permission+" has been used by other multiple");
  75.             }else {
  76.                 scaleMap.put(permission,scale);
  77.                 messageMap.put(permission,message);
  78.             }
  79.         }
  80.     }

  82.     @SubscribeEvent
  83.     public void onGainExp(PlayerPickupXpEvent event){
  84.         Player player = (Player) event.getEntityPlayer();
  85.         for(String permission: scaleMap.keySet()){
  86.             if(player.hasPermission(permission)){
  87.                 int originExp = event.getOrb().getXpValue();
  88.                 double expExp = originExp * scaleMap.get(permission);
  89.                 int finalExp = (int) Math.round(expExp);
  90.                 event.getOrb().xpValue = finalExp;
  91.                 int gainExp = finalExp-originExp;
  92.                 String message = messageMap.get(permission);
  93.                 if(message!=null){
  94.                     message = message.replaceAll("%player%",player.getName());
  95.                     message = message.replaceAll("%origin_exp%",String.valueOf(originExp));
  96.                     message = message.replaceAll("%final_exp%",String.valueOf(finalExp));
  97.                     message = message.replaceAll("%gain_exp%",String.valueOf(gainExp));
  98.                     Text text;
  99.                     try {
  100.                         text = TextSerializers.JSON.deserialize(message);
  101.                     }catch (Exception e){
  102.                         text = TextSerializers.FORMATTING_CODE.deserializeUnchecked(message);
  103.                     }
  104.                     player.sendMessage(text);
  105.                 }
  106.                 break;
  107.             }
  108.         }
  109.     }

  111. }

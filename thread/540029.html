改：config里面的IC2.ini
  1. [protection]
  2. ; Enable logging of players when they remove a machine using a wrench.
  3. wrenchLogging = true
  4. ; Maximum Explosion power of a nuke, where TNT is 4.  TNT范围为4
  5. nukeExplosionPowerLimit = 1   1为自毁
  6. ; Maximum explosion power of a nuclear reactor, where TNT is 4.   TNT范围为4
  7. reactorExplosionPowerLimit = 1   1为自毁
  8. ; Enable the nuke
  9. enableNuke = true

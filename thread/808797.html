作者: MineCrocodile    时间: 2018-7-25 11:09
generalConfig文件
  1. Limit:
  2.     # Money gain limit
  3.     # With this enabled, players will be limited how much they can make in defined time
  4.     # Time in seconds: 60 = 1min, 3600 = 1 hour, 86400 = 24 hours
  5.     Money:
  6.       Use: true
  7.       # Do you want to stop money gain when exp limit reached?是否在经验上限达到时停止获得金币
  8.       StopWithExp: false
  9.       # Do you want to stop money gain when point limit reached?是否在点数上限达到时停止获得金币
  10.       StopWithPoint: false
  11.       # Equation to calculate max limit. Option to use totallevel to include players total amount levels of current jobs
  12.       # You can always use simple number to set money limit
  13.       # Default equation is: 500+500*(totallevel/100), this will add 1% from 500 for each level player have
  14.       # So player with 2 jobs with level 15 and 22 will have 685 limit
  15.       # 金币上限计算公式
  16.       MoneyLimit: 500+500*(totallevel/100)
  17.       # Time in seconds: 60 = 1min, 3600 = 1 hour, 86400 = 24 hours 设置多长时间内达到上述上限时停止收获金币
  18.       TimeLimit: 86400
  19.       # Delay between announcements about reached money limit 通知玩家已达金币上限的时间间隔
  20.       # Keep this from 30 to 5 min (300), as players can get annoyed of constant message displaying
  21.       AnnouncmentDelay: 30

· /referral help
· 权限： referral.use
· 指令注释： 显示此插件的基本命令


· /referral info
· 权限： referral.info
· 指令注释： 查看您的推荐信息


· /referral list|
· 权限： referral.list
· 指令注释： 查看您的推荐列表


· /referral code
· 权限： referral.code
· 指令注释： 生成您的邀请码


· /referral claim
· 权限： referral.claim
· 指令注释： 打开兑换奖品界面


· /referral <numb>
· 权限： referral.use
· 指令注释： 使用邀请码


· /referral top
· 权限： referral.top
· 指令注释： 邀请排行榜


· /referral addpoints <player> <value>
· 权限： referral.admin
· 指令注释： 为一个玩家添加积分点


· /referral setpoints <player> <value>
· 权限： referral.admin
· 指令注释： 设置一个玩家的积分点


· /referral reload
· 权限： referral.admin
· 指令注释： 重载插件
五、插件配置文件
官方wiki
http://en.wiki.pedrojm96.com/wiki/ReferralSystem


config.yml
  1. #自动更新设置 true[开启] false[关闭]
  2. update-check: true
  3. #语言文件设置
  4. messages: EN
  5. #消息提示设置
  6. prefix: '&b&l邀请系统 &6>>> '
  7. #奖励被邀请玩家积分点数
  8. points-reward-player: 20
  9. #奖励邀请者积分点数
  10. points-reward-referrer: 5
  11. #被邀请玩家邀请几人后
  12. #被邀请玩家的邀请者可获得额外奖励
  13. claim-bonus-referrer: 3
  14. #额外奖励积分点数
  15. points-reward-bonus-referrer: 10
  16. #是否开启IP最大邀请限制 true[开启] false[关闭]
  17. limit-player-ip: true
  18. #一个IP最多可邀请几个人
  19. max-player-ip: 5
  20. #是否开启烟花特效 true[开启] false[关闭]
  21. fireworks-enable: true
  22. #是否开启粒子特效 true[开启] false[关闭]
  23. particles-enable: true
  24. #是否开启声音特效 true[开启] false[关闭]
  25. sounds-enable: true
  26. #是否开启兑奖系统 true[开启] false[关闭]
  27. menu-claim: true
  28. #是否开启玩家在线多久后获得积分点数 true[开启] false[关闭]
  29. requires-playtime: true
  30. #玩家在线时间设置 单位：秒
  31. min-playtime: 480
  32. #是否启用其它经济插件 true[开启] false[关闭]
  33. use-external-points:
  34.   #是否开启
  35.   enable: true
  36.   #官方点券插件
  37.   SuperCredits: false
  38.   #点券插件
  39.   PlayerPoints: true
  40.   #经济插件
  41.   Vault: true
  42. #选择数据储存方式 MySQL SQLite 默认SQLite
  43. #使用MySQL储存时请将 useSSL 设置为 true
  44. #SSL是一种更安全的数据库连接方式
  45. data-storage:
  46.   #储存类型
  47.   type: SQLite
  48.   #数据库地址
  49.   host: localhost
  50.   #数据库端口
  51.   port: 3306
  52.   #数据库名称
  53.   database: mine
  54.   #数据库用户名
  55.   username: root
  56.   #数据库密码
  57.   password: ''

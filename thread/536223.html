  6. function skin_18to17($skin){
  7.   $arr=getimagesize($skin);
  8.   if(!$arr)return false;
  9.   if($arr[0]!=$arr[1]||$arr[0]<64)return false;
  10.   $size=$arr[0];
  11.   $img=imagecreatetruecolor($size,$size/2);
  12.   $color=imagecolorallocate($img,255,255,255);
  13.   imagecolortransparent($img,$color);
  14.   imagefill($img,0,0,$color);
  15.   $pic=imagecreatefrompng($skin);
  16.   if(imagecopy($img,$pic,0,0,0,0,$size,$size/2)){
  17.     imagedestroy($pic);
  18.     return $img;
  19.   }else{
  20.     imagedestroy($pic);
  21.     return false;
  22.   }
  23. }
  24. function skin_17to18($skin){
  25.   $arr=getimagesize($skin);
  26.   if(!$arr)return false;
  27.   if($arr[0]!=$arr[1]*2||$arr[0]<64)return false;
  28.   $size=$arr[0];
  29.   $img=imagecreatetruecolor($size,$size);
  30.   $color=imagecolorallocate($img,255,255,255);
  31.   imagecolortransparent($img,$color);
  32.   imagefill($img,0,0,$color);
  33.   $pic=imagecreatefrompng($skin);
  34.   imagecopy($img,$pic,0,0,0,0,$size,$size/2);
  35.   $size=$size/4;
  36.   imagecopy($img,$pic,$size,$size*3,0,$size,$size,$size);
  37.   imagecopy($img,$pic,$size*2,$size*3,$size/2*5,$size,$size,$size);
  38.   imagedestroy($pic);
  39.   if($img!=null)return $img;else return false;
  40. }

  19. set jarfiles=mc1.8.x

  21. set timea=5

  25. set Start=5

  35. ::  修改 提示时间 请在 set timea 处修改 (当然你也可以换成中文提示或者其他信息，不过要确定转码)
  36. ::}
  37. ::-----------------------------------------------
  38. set Msg-ed=Server Restarting in %timea%s after
  39. ::服务器关闭时提示的信息
  40. :: 修改方式{
  41. ::  修改 提示时间 请在 set timea 处修改 (当然你也可以换成中文提示或者其他信息，不过要确定转码)
  42. ::}
  43. ::-----------------------------------------------
  44. set title=ASkyBlock-Server--Port:%prot%--Mem:%Mem%M
  45. ::开启时窗口的标题
  46. :: 修改方式{
  47. ::  修改 标题上的端口 请在 set prot 处修改
  48. ::  修改 标题上显示内存 请在 set Mem 处修改 (与开启命令处最大内存相匹配)
  49. ::}
  50. ::-----------------------------------------------
  51. set StartCommand=%Java% -Xincgc -Xmx%Mem%M -XX:+AggressiveOpts -XX:MaxPermSize=128M -jar %jarfiles%.jar
  52. ::服务端开启命令
  53. :: 修改方式{
  54. ::  修改 Java 路径设置 请在 set Java 处修改
  55. ::  修改 内存大小 请在 set Mem 处修改
  56. ::  修改 MaxPermSize 请在 set MaxXX 处修改
  57. ::  修改 Jar核心文件 请在 set jarfiles 处修改
  58. ::}
  59. ::-----------------------------------------------
  60. set Command-title=title
  61. ::后台窗口标题
  62. set Command-color=color
  63. ::修改字体颜色命令
  64. set Command-echo=echo
  65. ::提示信息
  66. ::-----------------------------------------------
  67. ::延迟时间执行再命令
  68. ::如果要修改时间{
  69. :: 请在
  70. ::  set timea
  71. ::   处修改 (单位:秒)
  72. ::}

  76. set Messagedo=**********************************************************

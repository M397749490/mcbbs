看了一下代码，可以确定你的数据包实例没有修改正确的字段值。原帖是通过直接调用 NMS 实现的，而是你通过 ProtocolLib 方式，这种方式是创建无参构造实例然后反射修改字段值实现的。而 PacketPlayOutPlayerInfo 有 2 个字段，PacketPlayOutNamedEntitySpawn 有 9 个字段。详情可以通过反编译服务端查看。
  1. Player player = ...;
  2.   Location location = player.getLocation().clone();
  3.   MinecraftServer server = ((CraftServer) Bukkit.getServer()).getServer();
  4.   WorldServer worldServer = ((CraftWorld) player.getWorld()).getHandle();
  5.   GameProfile gameProfile = new GameProfile(UUID.randomUUID(), "FakePlayer");
  6.   PlayerInteractManager interactManager = new PlayerInteractManager(worldServer);
  7.   EntityPlayer dummy = new EntityPlayer(server, worldServer, gameProfile, interactManager);
  8.   dummy.setLocation(location.getX(), location.getY(), location.getZ(), location.getYaw(), location.getPitch());

  10.   List<PlayerInfoData> infoDataList = new ArrayList<>();
  11.   infoDataList.add(new PlayerInfoData(
  12.       new WrappedGameProfile(gameProfile.getId(), gameProfile.getName()),
  13.       0,
  14.       EnumWrappers.NativeGameMode.SURVIVAL,
  15.       WrappedChatComponent.fromText(gameProfile.getName())
  16.   ));

  18.   PacketContainer ppopi = new PacketContainer(PacketType.Play.Server.PLAYER_INFO);
  19.   PacketContainer ppones = new PacketContainer(PacketType.Play.Server.NAMED_ENTITY_SPAWN);

  21.   ppopi.getPlayerInfoAction().write(0, EnumWrappers.PlayerInfoAction.ADD_PLAYER);
  22.   ppopi.getPlayerInfoDataLists().write(0, infoDataList);

  24.   ppones.getIntegers().write(0, dummy.getId());
  25.   ppones.getUUIDs().write(0, gameProfile.getId());
  26.   ppones.getDoubles().write(0, dummy.locX);
  27.   ppones.getDoubles().write(1, dummy.locY);
  28.   ppones.getDoubles().write(2, dummy.locZ);
  29.   ppones.getBytes().write(0, (byte) (int) (dummy.yaw * 256f / 360f));
  30.   ppones.getBytes().write(1, (byte) (int) (dummy.pitch * 256f / 360f));
  31.   ppones.getDataWatcherModifier().write(0, new WrappedDataWatcher(dummy));

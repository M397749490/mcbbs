config文件配置使用  配置文件很简单，看一眼就能懂了吧

  1. version: 110
  2. options:
  3.   #全局开启物品不可破坏
  4.   global-unbreakable: true
  5.   #开启锻造成功领取界面
  6.   result-gui: true
  7.   #强度条颜色代码，依次为强度从低到高
  8.   strength-color: '54c6e93b2a'
  9. items:
  10.   #锻造装备配置
  11.   mhxtj:
  12.     id: 268
  13.     data: 0
  14.     #物品名
  15.     display: '&a&l<前缀词>&3魔化玄铁剑'
  16.     #自定义lore
  17.     lores:
  18.     - '&a类型：主武器     品质：稀有'
  19.     - ''
  20.     - '&8&m---------&8&l[&d&l  属性  &8&l]&8&m---------'
  21.     - '&f>> &7暴击几率: &a<暴击几率>%'
  22.     - '&f>> &7生命吸取: &a1%'
  23.     - '&f>> &7致盲: &a1%'
  24.     - '&f>> &7等级: &a10'
  25.     - '&8&m---------&8&l[&b&l  锻造  &8&l]&8&m---------'
  26.     - '&f>> &7攻击力: &c<攻击力1>-<攻击力2>'
  27.     - '&f>> &7暴击伤害: &c178%'
  28.     - '&f>> &7吸血倍率: &c<吸血倍率>%'
  29.     - '&8&m---------&8&l[&e&l  描述  &8&l]&8&m---------'
  30.     - '&7魔族唯一留下来的神器之一'
  31.     - '&7但是还存在了图纸里'
  32.     - '&a二阶装备'
  33.     - ''
  34.     - '&e品质&r:<等级>'
  35.     - '&a强度&r:&3[<strength>]'
  36.     - '&7锻造者&6[&b&l<player>&6]'
  37.     #锻造等级配置
  38.     levels:
  39.       普通:
  40.         #概率权重，必须大于0
  41.         weight: 2
  42.         等级:
  43.           #数值类型，fixed固定内容, random随机整数(value1必须小于value2)，
  44.           #strength强度类型，lores随机文本，float随机小数(start必须小于bound)
  45.           type: fixed
  46.           value: '&6普通'
  47.         暴击几率:
  48.           type: random
  49.           value1: 4
  50.           value2: 6
  51.         攻击力1:
  52.           type: strength
  53.           base: 100
  54.           add: 50
  55.         攻击力2:
  56.           type: strength
  57.           base: 250
  58.           add: 200
  59.         吸血倍率:
  60.           type: float
  61.           #显示的小数位
  62.           decimal: 1
  63.           start: 2.1
  64.           bound: 4.5

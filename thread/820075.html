@echo OFF　
set c=-1　　　　　　　　　　　　　　
:head
set /c c+=1
rem a为设置的内存大小,单位为MByte。

set b=3
title The Wind Rises 已重启%c%
JAVA -Xincgc -Xmx%a%M -jar KCauldron-1.7.10-1614.201.jar  -nojline

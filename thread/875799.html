  3. import string,os

  6. pwd_l = 32 # 密钥长度

  8. pwd_s = string.ascii_letters + string.digits

  10. def g_pwd():
  11.     pwd = []
  12.     while (len(pwd) < pwd_l):
  13.         pwd.append(choice(pwd_s))
  14.     return ''.join(pwd)

  16. if __name__ == '__main__':
  17.     for i in range(0, pwd_c):
  18.         print(g_pwd())
  19.     os.system('pause')

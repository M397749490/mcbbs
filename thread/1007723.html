@echo off

SET WORLD_DIR=world

SET BACKUP_DIR=backup

SET DATE_LOG=date.txt

SET TIME_LOWER=0000

SET TIME_UPPER=0005

SET DAY_OF_WEEK=周六



:head

if exist %DATE_LOG% (goto checkfile) else (goto checkdate)

:checkfile

set /P tmp=<%DATE_LOG%

if %tmp%==%date:~0,10% goto start

:checkdate

if %date:~11,2%==%DAY_OF_WEEK% (goto checktime) else (goto start)

:checktime

for /f %%i in ('time /T') do set tmp=%%i

set tmp=%tmp:~0,2%%tmp:~3,2%

if /I %tmp% LSS %TIME_LOWER% goto start

if /I %tmp% GTR %TIME_UPPER% goto start

REM copy

rd /S /Q %WORLD_DIR%

xcopy %BACKUP_DIR% %WORLD_DIR%\ /E 

echo %date:~0,10%>%DATE_LOG%

:start

echo "Starting server..." 

@echo on

java -version

@echo off

choice /T 10 /D Y /M "Press 'N' in 10 seconds to stop..."

if errorlevel 2 goto end

if errorlevel 1 goto head

:end

  1. const fs = require('fs')
  2. const path = require('path')
  3. let page = []
  4. let line = []
  5. for (let i = 0; i <= parseInt('ffff', 16); i++) {
  6.     const char = String.fromCharCode(i)
  7.     if (i !== 0 && i % 16 === 0) {
  8.         page.push(line.join(''))
  9.         line = []
  10.     }
  11.     if (i !== 0 && i % 256 === 0) {
  12.         let codeHex = (i - 1).toString(16)
  13.         if (codeHex.length === 1) {
  14.             codeHex = `000${codeHex}`
  15.         } else if (codeHex.length === 2) {
  16.             codeHex = `00${codeHex}`
  17.         } else if (codeHex.length === 3) {
  18.             codeHex = `0${codeHex}`
  19.         }
  20.         fs.writeFileSync(path.join(__dirname, `unicode_pages/unicode_page_${codeHex.slice(0, 2)}.txt`), page.join('\r\n'), { encoding: 'utf8' })
  21.         page = []
  22.     }
  23.     line.push(char)
  24. }

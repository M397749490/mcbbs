  47. general {
  48.     # 启用创造模式的魔导手册,在你第一次使用时就授予你所有的研究.
  49.     B:allow_cheat_sheet=true

  51.     # 将值设为false会禁用魔镜的研究及其合成.
  52.     B:allow_mirrors=true

  54.     # 若设为false,铺路石,门和守护玻璃就都会变为装饰品,失去原本的作用(每个人都能轻易破坏它们).
  55.     B:allow_warded_stone=true

  57.     # 腐化是否能由咒波污染产生.
  58.     B:biome_taint_from_flux=true

  60.     # 腐化之地群系传播时每个方块更新的概率(n个中的1个). 设置为0阻止腐化之地传播.
  61.     I:biome_taint_spread=200

  63.     # 设为true将魔法森林变回旧式的蓝色魔法森林.
  64.     B:blue_magical_forest=true

  66.     # 设为true增大某些颜色对比度和明暗度差距以防其对色盲患者来说变为'隐形的'.
  67.     B:color_blind=false

  69.     # 讨厌歪着的标签和这个那个所有炫酷的东西?如果是的话将它设为false.
  70.     B:crooked=true

  72.     # 物品所含的要素默认是隐藏的只有在按住shift时才会显示.将它设置为'true'就会反向设定,只在没有按住shift时才显示物品的要素.
  73.     B:display_aspects=false

  75.     # 设为false会移除腐化纤维生长的紫色结块.这可能会避免一些崩溃并提高性能.
  76.     B:glowing_taint=true

  78.     # 设为true的话傀儡就会在与箱子交互时激活开箱子的动画和音效.
  79.     B:golem_chest_interact=true

  81.     # 傀儡在检测任务时等待时间是多少tick.调高会解放服务器的tick,但会让傀儡反应得更慢.
  82.     I:golem_delay=5

  84.     # 傀儡与标记方块之间连接线的特效质量.设定到4以下完全关闭.
  85.     I:golem_link_quality=16

  87.     # 负面节点如饕餮,腐化,凶险会有附加的,更难对付的效果.
  88.     B:hard_mode_nodes=true

  90.     # 生成的世界中节点多么稀有.数值代表每N个区块会有1个节点(平均).
  91.     I:node_rarity=13

  93.     # 决定了提示信息滚动的速度.
  94.     I:notification_delay=5000

  96.     # 屏幕上所能显示的提示信息的最大数量.
  97.     I:notification_max=15

  99.     # 这是一个以逗号分隔的列表,用于设定便携式洞穴不能穿过的方块名.
  100.     S:portablehole_blacklist=iron_door

  102.     # 单个节点变为特殊节点的概率(纯净,凶险,不稳定等等.).数字代表N个节点中就有1个会变为特殊节点,比如设置为5就代表着每5个节点中就有1个是特殊类型的.
  103.     I:special_node_rarity=16

  105.     # 神秘时代村民id
  106.     I:thaumcraft_villager_id=190

  108.     # 设为true使得法杖要素魔力的显示窗在屏幕左下角而不是左上角.
  109.     B:wand_dial_bottom=false

  111.     # 将它设为true禁用扭曲效果和类似的机制.你这个胆小鬼:P.
  112.     B:wuss_mode=false
  113. }

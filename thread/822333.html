2.配置文件
version: "1.0-beta"
#严禁更改，后果自付
prefix: "[&a&lChatPAPI&f]"
#插件前缀
Message:
#插件消息
  Reload: "%prefix% &c&l插件重载成功!"
  #重载消息
ChatFormat:
                 #聊天格式
#==============================================
#格式：Prefix为前缀 Suffix为后缀 支持PAPI变量
#前缀的格式 其他的文本 玩家 消息 后缀的格式
#       eg Clusters_stars为玩家
#            若开启，则为
# %bw_kills% Clusters_stars 消息 %bw_wins%
#==============================================
  Prefix:
  #前缀格式
    Use: true
    #是否启用
    Format: "%bw_kills%"
    #格式
  Suffix:
  #后缀格式
    Use: true
    #是否启用
    Format: "%bw_wins%"
ChatMessage:
                 #消息格式
#==============================================
#格式：Prefix为前缀 Suffix为后缀 支持PAPI变量
#聊天格式 玩家 前缀的格式 玩家 后缀的格式
#       eg Clusters_stars为玩家
#            若开启，则为
# Clusters_stars %bw_kills% 消息 %bw_wins%
#==============================================
  Prefix:
  #前缀格式
    Use: true
    #是否启用
    Format: "%bw_kills%"
    #格式
  Suffix:
  #后缀格式
    Use: true
    #是否启用
    Format: "%bw_wins%"
    #格式
Chat:
                     #变量替换
#====================================================
#这里仅仅是更改可以自定义但不支持PAPI的聊天格式的插件
#              比如起床战争的某个格式，
#你只需要将变量名称输入，然后再将要替换的文本格式写好
#    就可以替换为你想要的文本了，当然支持PAPI
#               Clusters_stars为玩家
#在起床战争的配置里 {demo} Clusters_stars {fx} 就这样
#   等到玩家发送消息的时候就会将其替换为你要的文本
#   显示的是 %bw_kills% Clusters_stars %bw_wins%
#      当然这些有变量的都会替换为相应的文本
#====================================================
  {demo}:
  #变量名称
    Format: "%bw_kills%"
    #需要替换的文本格式
  {fx}:
  #变量名称
    Format: "%bw_wins%"
    #需要替换的文本格式

按理说是路径的关系，试着把.BAT打成：
  1. @ECHO OFF
  2. SET BINDIR=%~dp0
  3. CD /D "%BINDIR%"
  4. echo.
  5. echo 写出你的JAVA版本 (6 或者 7)???
  6. set /p JR=jre
  7. set JR=jre%JR%
  8. echo Java Version Setted to %JR%
  9. Timeout /t 2 /nobreak >nul
  10. cls
  11. echo.
  12. echo  写出你的craftbukkit文件名。[ex. craftbukkit-1.2.5-R4.0]
  13. set /p CBN=
  14. echo CraftBukkit文件设置到%CBN%
  15. Timeout /t 2 /nobreak >nul
  16. cls
  17. echo.
  18. if %PROCESSOR_ARCHITECTURE%==x86 (
  19. echo 检测出Windows 32位！
  20. echo 程序的设置到 %programfiles%
  21. SET PD=%programfiles%
  22. ) else (
  23. echo 检测出Windows 64位！
  24. SET PD64=%programfiles%
  25. SET PD32=%programfiles(x86)%
  26. echo.
  27. echo 你是在 x32 还是 x64的模式下运行你的Java???
  28. echo x32 = %PD32%
  29. echo x64 = %PD64%
  30. set /p PGF=Run in x
  31. set PGF=x%PGF%
  32. if %PGF%==x64 (
  33.   echo 程序设置到 %PD64%
  34.   set PD=%PD64%
  35. ) else (
  36.   echo 程序设置到 %PD32%
  37.   set PD=%PD32%
  38. )
  39. )
  40. Timeout /t 4 /nobreak >nul
  41. cls
  42. echo.
  43. echo 正在开启服务器...
  44. Timeout /t 2 /nobreak >nul
  45. "%PD%\Java\%JR%\bin\java.exe" -Xmx1024M -Xms1024M -jar %CBN%.jar
  46. echo 服务器正在关闭...
  47. echo.
复制代码

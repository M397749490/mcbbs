  1. int circleElements = 20; //粒子数
  2. double radius = 1.0; //半径
  3. Location centerLoc; //中心
  4. for (int i = 0; i < circleElements; ++i)
  5. {
  6.         double alpha = (360.0 / circleElements) * i;
  7.         double x = radius * Math.sin(Math.toRadians(alpha));
  8.         double z = radius * Math.cos(Math.toRadians(alpha));
  9.         Location particleLoc = new Location(centerLoc.getWorld(), centerLoc.getX() + x, centerLoc.getY(), centerLoc.getZ() + z); //这个就是粒子的坐标
  10.         ParticleEffectUtils.display(ParticleType.FLAME,particleLoc); //这是我自己的粒子效果工具类
  11. }

反作弊自带此功能
#限制使用命令和聊天的频率
ChatCommandRate:
  Enable: true
  #聊天间隔时间（秒）
  ChatDelay: 2
  #使用命令间隔时间（秒）
  CommandDelay: 3
  #命令超频的提示语
  CommandMessage: '&e[服务器娘]&b您输入指令的速度太快了，喝杯茶休息几秒吧:)'
  ChatMessage: '&e[服务器娘]&b人家都要被你刷屏了，速度慢一点啦~'
  #命令白名单，在该白名单的命令无视间隔时间
  #仅需添加命令的开头部分。例如/res tp，仅需添加/res
  #不需要考虑大小写的问题
  CommandWhiteList:
  - '/spawn'
  - '/back'
  - '/res'

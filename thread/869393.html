修改配置文件group.yml
  1. Groups:
  2.     Default: #组名，不要删除这个组
  3.        #删除下面的井号可将此配置应用到其他组
  4.        #Mirror:
  5.          #- 'group1'
  6.          #- 'group2'
  7.        Residence:
  8.            # 该组玩家是否可以创建领地
  9.            CanCreate: true
  10.            # 玩家可以拥有的最大领地数量
  11.            MaxResidences: 3
  12.            # 领地可包含的最大物理区域
  13.            MaxAreasPerResidence: 2
  14.            # 领地最大格数（东西方向）
  15.            MaxEastWest: 16
  16.            # 领地最小格数（东西方向），设为-1则不限制
  17.            MinEastWest: 3
  18.            # 领地最大格数（南北方向）
  19.            MaxNorthSouth: 16
  20.            # 领地最小格数（南北方向）
  21.            MinNorthSouth: 3
  22.            # 领地最大格数（高度）
  23.            MaxUpDown: 256
  24.            # 领地最小格数（高度），设为-1则不限制
  25.            MinUpDown: 3
  26.            # 可创建领地的最小高度
  27.            MinHeight: 0
  28.            # 可创建领地的最大高度
  29.            MaxHeight: 255
  30.            # 所在区域可拥有的最大子领地数量
  31.            MaxSubzonesInArea: 3

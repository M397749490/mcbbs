  10. public class RBQListener implements Listener {

  12.     @EventHandler
  13.     public void onMove(PlayerMoveEvent e) {
  14.         Player p = e.getPlayer();
  15.         Location l = e.getTo();
  16.         World w = l.getWorld();
  17.         if (!p.isOp() && w.getName().equalsIgnoreCase("world") && !p.hasPermission("world.move")) {
  18.             e.setCancelled(true);
  19.             p.sendMessage("我艹你个RBQ，TMD主世界禁止移动");

  21.         }
  22.     }
  23. }

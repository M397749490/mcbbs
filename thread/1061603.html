What is NIO？
NIO全称non-blocking，是Java的新IO API
同时支持非阻塞和阻塞模式

NIO 与 BIO 的区别
NIO与BIO最大的区别是支持非阻塞模式，这使得NIO无须像BIO一样为每个Channel(流)
创建一个Thread阻塞地读，完全可以使用一个Thread完成对多个Channel的读操作

NIO的三大核心组件 ByteBuffer & Channel & Selector 类
What is ByteBuffer?
ByteBuffer 实际上是一块连续的内存块，它是对byte[]的封装,Java提供了几个新的方法使这个byte[]更易于使用
所以ByteBuffer可以看成是一个大小固定的容器
创建ByteBuffer的方法:
  1. public static ByteBuffer allocate(int capacity) //在JVM托管堆上分配一个容量为capacity的新的字节缓冲区
  2. public static ByteBuffer allocateDirect(int capacity) //在非托管堆上分配一个容量为capacity的新的字节缓冲区(Java Zero Copy)
  3. public static ByteBuffer wrap(byte[] array)  //将 byte 数组包装到缓冲区中
复制代码

ByteBuffer的容器操作:
  1. public abstract byte get(int index)  //读取指定索引处的字节
  2. public abstract ByteBuffer put(byte b) //将b插入到buffer的尾部,Buffer必须可写且容量足够
  3. public final byte[] array() //返回实现此缓冲区的 byte 数组，在调用此操作之前必须确保此ByteBuffer有byte[]
  4. public final boolean hasArray() //判断是否可通过一个可访问的 byte 数组实现此缓冲区

What is Channel?
Channel实际上是双向的"流",它同时支持Read和Write并且支持非阻塞模式
SocketChannel的坑:
  • 非阻塞模式下SocketChannel的Write操作不一定会把数据全部写入，如果此时底层socket缓冲区已满或者对端网络状况过于糟糕，写入的数据大小可能小于需要写入的大小
  • 不要调用Socket和ServerSocket的Channel方法
  • SocketChannel在客户端关闭时进行Read和Write会抛出异常
  • 如果使用Selector一定要将Channel设置成非阻塞模式

ServerSocketChannel的用法基本与ServerSocket相同
不同的是需要使用静态的open方法来创建Channel
并且调用bind方法是用户的责任
  1. //新建ServerSocketChannel并监听指定端口
  2. serverSocketChannel = ServerSocketChannel.open();
  3. serverSocketChannel.bind(new InetSocketAddress(port));

accept方法返回的将是SocketChannel
  1. //使用accept获取客户端
  2. SocketChannel client = channel.accept();

What is Selector?
Selector对应底层的IO多路复用，它能让你在一个线程中监听多个Channel,当被监听的Channel有事件时Selector通知select方法的调用者
Selector在底层调用了select/poll/epoll的相关函数
例如在Windows中JVM的Selector底层调用select函数(这将导致大量非活跃连接时效率低下)
在Linux 2.6+中JVM的Selector底层调用epoll的相关函数(LT模式)
尽可能让你的NIO程序运行在Linux 平台上
Selector的坑:
  • epoll的空轮询bug
  • Selector的select和register之间不是线程安全的 如果其他线程调用了select方法 另一线程在这时调用了register 被注册的channel有事件到达时并不会select方法返回

直接使用NIO的Selector略显困难,我们通常使用Reactor模型对Selector进行二次封装
示例的Reactor封装
  1. package cc.sfclub.io.network;

  3. import java.io.IOException;
  4. import java.nio.ByteBuffer;
  5. import java.nio.channels.Channel;
  6. import java.nio.channels.ClosedChannelException;
  7. import java.nio.channels.SelectionKey;
  8. import java.nio.channels.Selector;
  9. import java.nio.channels.ServerSocketChannel;
  10. import java.nio.channels.SocketChannel;
  11. import java.util.Iterator;
  12. import java.util.LinkedList;
  13. import java.util.Set;
  14. import java.util.function.BiConsumer;
  15. import java.util.function.Consumer;

  17. import cc.sfclub.io.SpinLock;
  18. //反应器
  19. //在构造函数中传入回调后调用run方法
  20. //run方法将会导致线程阻塞直到反应器关闭
  21. //注意: 在客户端连接回调中将SocketChannel注册到Reactor是用户的责任
  22. public class Reactor {
  23.     //selector
  24.     private Selector _selector;
  25.     //数据到达回调
  26.     private BiConsumer<byte[],SocketChannel> _readCb;
  27.     //客户端关闭回调
  28.     private Consumer<SocketChannel> _closeCb;
  29.     //客户端连接回调
  30.     private Consumer<SocketChannel> _acceptCb;
  31.     //cancel token
  32.     private volatile boolean _token;
  33.     //tasks
  34.     private LinkedList<Runnable> _tasks;

  36.     //自旋锁
  37.     //用于保护_tasks
  38.     private SpinLock _lock;

  40.     public Reactor(BiConsumer<byte[],SocketChannel> readCb,Consumer<SocketChannel> closeCb,Consumer<SocketChannel> acceptCb) throws IOException
  41.     {
  42.         _selector = Selector.open();
  43.         _readCb = readCb;
  44.         _closeCb = closeCb;
  45.         _token = false;
  46.         _acceptCb = acceptCb;
  47.         _tasks = new LinkedList<>();
  48.         _lock = new SpinLock();
  49.     }

  51.     //用于其他线程向Reactor线程投递任务
  52.     private void runInLoop(Runnable task)
  53.     {
  54.         //将任务放到_tasks里
  55.         _lock.lock();
  56.         try {
  57.             _tasks.add(task);
  58.         }
  59.         finally
  60.         {
  61.             _lock.unlock();
  62.         }
  63.         //唤醒Reactor线程
  64.         _selector.wakeup();
  65.     }

  67.     //获取其他线程投递的任务
  68.     private LinkedList<Runnable> getTasks()
  69.     {
  70.         //交换taks和_tasks
  71.         LinkedList<Runnable> tasks = new LinkedList<>(),tmp = tasks;
  72.         _lock.lock();
  73.         try {
  74.             tasks = _tasks;
  75.             _tasks = tmp;
  76.         }
  77.         finally
  78.         {
  79.             _lock.unlock();
  80.         }
  81.         return tasks;
  82.     }

  84.     //执行其他线程投递的操作
  85.     //返**oolean是为了甄别是否出现了epoll空轮询bug
  86.     private boolean handleTasks()
  87.     {
  88.         //遍历调用run
  89.         LinkedList<Runnable> tasks = getTasks();
  90.         for(Runnable task:tasks)
  91.         {
  92.             task.run();
  93.         }
  94.         return tasks.size() != 0;
  95.     }

  97.     //JVM bug:
  98.     //https://bugs.java.com/bugdatabase/view_bug.do?bug_id=2147719
  99.     //https://bugs.java.com/bugdatabase/view_bug.do?bug_id=6403933
  100.     //重建selector
  101.     private void rebuildSelector() throws IOException
  102.     {
  103.         //创建新selector
  104.         Selector selector = Selector.open();
  105.         //将原来selector的key移动到新建的selector上
  106.         for(SelectionKey key:_selector.keys())
  107.         {
  108.             key.channel().register(selector,key.interestOps());
  109.             key.cancel();
  110.         }
  111.         //交换两个selector
  112.         Selector tmp = _selector;
  113.         _selector = selector;
  114.         //关闭旧的selector
  115.         tmp.close();
  116.     }

  118.     //注册客户端连接
  119.     //注意: selector的register和select都不是线程安全的
  120.     //如果其他线程调用了select方法
  121.     //另一线程在这时调用了register
  122.     //被注册的channel有事件到达时并不会select方法返回
  123.     public void register(SocketChannel channel) throws IOException
  124.     {
  125.         //在Reactor线程中注册
  126.         runInLoop(()->
  127.         {
  128.             try {
  129.                 //设置为非阻塞
  130.                 channel.configureBlocking(false);
  131.                 //将Channel注册到Selector
  132.                  //在生产环境中您可能要监听OP_WRITE来支持非阻塞的写操作
  133.                  //在有数据需要写时监听OP_WRITE
  134.                  //无数据需写入时必须将OP_WRITE移除
  135.                  //否则将导致忙循环
  136.                 channel.register(_selector, SelectionKey.OP_READ);
  137.             } catch (Exception e) {
  138.                 e.printStackTrace();
  139.             }
  140.         });
  141.     }

  143.     //注册服务器监听器
  144.     public void register(ServerSocketChannel channel) throws IOException
  145.     {
  146.         //在Reactor线程中注册
  147.         runInLoop(()->
  148.         {
  149.             try {
  150.                 //设置为非阻塞
  151.                 channel.configureBlocking(false);
  152.                 //将Channel注册到Selector
  153.                 channel.register(_selector, SelectionKey.OP_ACCEPT);
  154.             } catch (Exception e) {
  155.                 e.printStackTrace();
  156.             }
  157.         });
  158.     }

  160.     //处理IO事件
  161.     private void handleIoEvent(SelectionKey key) throws IOException
  162.     {
  163.         //必须先判断key是否有效
  164.         //在处理之前它可能被取消
  165.         if(key.isValid())
  166.         {
  167.              //可读事件
  168.             if(key.isReadable())
  169.             {
  170.                 SocketChannel channel = (SocketChannel) key.channel();
  171.                 //4096是系统Socket读缓冲区的大小
  172.                 //这意味着您一次最多读取4096字节
  173.                 ByteBuffer buf = ByteBuffer.allocate(4096);
  174.                 try
  175.                 {
  176.                     //可能在读的过程中对端关闭
  177.                     //导致read出错
  178.                     int r = channel.read(buf);
  179.                     if(r > 0)
  180.                     {
  181.                         //调用数据到达回调
  182.                         _readCb.accept(buf.array(),channel);
  183.                     }
  184.                     else
  185.                     {
  186.                         //对端关闭了
  187.                         //调用客户端断开回调
  188.                         _closeCb.accept(channel);
  189.                     }
  190.                 }   
  191.                 catch (ClosedChannelException ex)
  192.                 {
  193.                     //对端关闭了
  194.                     //调用客户端断开回调
  195.                     _closeCb.accept(channel);   
  196.                 }
  197.             }
  198.             //客户端连接事件
  199.             else if(key.isAcceptable())
  200.             {
  201.                 //获取ServerSocketChannel
  202.                 ServerSocketChannel channel = (ServerSocketChannel)key.channel();
  203.                 //使用accept获取客户端
  204.                 SocketChannel client = channel.accept();
  205.                 //调用客户端连接回调
  206.                 //注意: 在回调中注册到Reactor是用户的责任
  207.                 _acceptCb.accept(client);
  208.             }
  209.             //可能是客户端关闭
  210.             else
  211.             {
  212.                 Channel ch = key.channel();
  213.                 //客户端关闭
  214.                 if(ch instanceof SocketChannel)
  215.                 {
  216.                     SocketChannel channel = (SocketChannel)ch;
  217.                     _closeCb.accept(channel);
  218.                 }
  219.             }
  220.         }
  221.     }

  223.     //获取并处理事件
  224.     private void run_once() throws IOException
  225.     {
  226.         //获取发生的事件
  227.         int num_ev = _selector.select();
  228.         //判断事件个数大于0
  229.         if(num_ev > 0)
  230.         {
  231.             //获取有事件的Channel
  232.             Set<SelectionKey> keys = _selector.selectedKeys();
  233.             //遍历Channel
  234.             Iterator<SelectionKey> iter = keys.iterator();
  235.             while(iter.hasNext())
  236.             {
  237.                 SelectionKey key = iter.next();
  238.                 iter.remove();
  239.                 //处理IO事件
  240.                 handleIoEvent(key);
  241.             }
  242.         }
  243.         boolean r = handleTasks();
  244.         //如果发生的事件为0
  245.         //且handleTasks返回false
  246.         //则可能发生了空轮询bug
  247.         //也可能是其他线程调用了close方法来关闭了selector
  248.         if(!r && num_ev == 0 && !_token)
  249.         {
  250.             rebuildSelector();
  251.         }
  252.     }

  254.     //运行直到Reactor关闭
  255.     public void run() throws IOException
  256.     {
  257.         while(!_token)
  258.         {
  259.             run_once();
  260.         }
  261.     }

  263.     //关闭Reactor
  264.     public void stop() throws IOException
  265.     {
  266.         _token = true;
  267.         _selector.close();
  268.     }
  269. }

SimpleWebServer同时提供了4种并发模型
  • Multithread Model Server - one connection per thread - BIO模型 - 每个连接一个线程
  • Reactor Model Server - basic reactor - 最简单的Reactor模型使用单线程处理所有连接
  • Reactor Model Server With Thread Pool - reactor + thread pool - 单线程Reactor+多个Worker线程
  • MultiReactor Model Server - one loop per thread - 多个Reactor多线程模型


下面主要介绍MultiReactor模型
这个模型会创建多个Reactor(个数为CPU核心数)并分配给相应的Worker线程
确保每一个线程都有一个Reactor
第一个Reactor为MainReactor
MainReactor与其他Reactor的区别是它不仅处理连接还接受连接
接受后轮询调度到不同的Reactor上
为什么需要多个Reactor?
这与Reactor的工作方式有关
虽然Selector的select方法是线程安全的,但是对Channel的读写操作并不是,所以需要多其进行加锁保护,与其用锁保护不如用多个Reactor,每个Reactor各自管理自己的Channel避免多个线程对同一个Channel进行读写的同时提供并发处理能力

  2. package cc.sfclub.io.network;

  4. import java.io.IOException;
  5. import java.net.InetSocketAddress;
  6. import java.nio.channels.ServerSocketChannel;
  7. import java.nio.channels.SocketChannel;
  8. import java.util.ArrayList;
  9. import java.util.List;
  10. import java.util.concurrent.ExecutorService;
  11. import java.util.concurrent.Executors;
  12. import java.util.concurrent.atomic.AtomicInteger;
  13. import java.util.function.BiConsumer;
  14. import java.util.function.Consumer;

  16. //MultiReactor Server
  17. public class MultiReactorServer implements IServer {

  19.     //多个Reactor
  20.     private List<Reactor> _reactors;

  22.     //指派位置
  23.     //用于轮询调度
  24.     private AtomicInteger _dispath_pos;

  26.     //ServerChannel
  27.     private ServerSocketChannel _serverSocketChannel;

  29.     //运行Reactor的线程池
  30.     private ExecutorService _threadPool;

  32.     public MultiReactorServer(int port,int num_reactors,BiConsumer<byte[],SocketChannel> readCb,Consumer<SocketChannel> closeCb) throws IOException
  33.     {
  34.         //num_reactors必须大于或等于1
  35.         if(num_reactors < 1)
  36.         {
  37.             return;
  38.         }
  39.         //新建ServerSocketChannel并监听指定端口
  40.         _serverSocketChannel = ServerSocketChannel.open();
  41.         _serverSocketChannel.bind(new InetSocketAddress(port));
  42.         //初始化指派位置
  43.         _dispath_pos = new AtomicInteger(0);
  44.         //初始化Reactor列表
  45.         _reactors = new ArrayList<>();
  46.         //批量创建Reactor
  47.         for(int i = 0;i < num_reactors;i++)
  48.         {
  49.             _reactors.add(new Reactor(readCb, closeCb,(sock)->
  50.             {
  51.                 //原子性地递增指派位置并返回它原来的值
  52.                 int pos = _dispath_pos.getAndIncrement();
  53.                 //对size求模获取Reactor
  54.                 pos = pos %_reactors.size();
  55.                 Reactor reactor = _reactors.get(pos);
  56.                 //将channel指派给指定的Reactor
  57.                 try {
  58.                     reactor.register(sock);
  59.                 } catch (Exception e) {
  60.                     e.printStackTrace();
  61.                 }
  62.             }));
  63.         }
  64.         //第一个Reactor总是用来监听ServerChannel
  65.         _reactors.get(0).register(_serverSocketChannel);
  66.         //大于1则需要创建线程池
  67.         if(num_reactors > 1)
  68.         {
  69.             _threadPool = Executors.newFixedThreadPool(_reactors.size()-1);
  70.         }
  71.     }

  73.     //运行服务器
  74.     //mainReactor总是由调用run的线程运行
  75.     //其他Reactor则是由线程池运行
  76.     @Override
  77.     public void run() throws IOException {
  78.         Reactor mainReactor = _reactors.get(0);
  79.         if(_reactors.size() > 1)
  80.         {
  81.             //将Reactor分配给线程池
  82.             for(int i = 1;i < _reactors.size();++i)
  83.             {
  84.                 final int index = i;
  85.                 _threadPool.submit(()->
  86.                 {
  87.                     System.out.println("Reactor number: "+index+" running");
  88.                     try {
  89.                         _reactors.get(index).run();
  90.                     }
  91.                     catch (Exception e) {
  92.                         e.printStackTrace();
  93.                     }
  94.                 });
  95.             }
  96.         }
  97.         System.out.println("Main Reactor running");
  98.         mainReactor.run();
  99.     }

  101.     @Override
  102.     public void close() throws IOException {
  103.         //遍历关闭Reactor
  104.         for ( Reactor r : _reactors) {
  105.             r.stop();
  106.         }
  107.         _serverSocketChannel.close();
  108.     }
  109. }

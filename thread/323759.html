这是门口电脑的代码
shell.run("clear")
print("Please insert your card. . .")
repeat
  wait(1)
until disk.hasData()
cd disk
local file = io.open
local pass = file:read()
local pass = pass + file:read()
local modem = peripheral.wrap("back")
modem.open("back")
rednet.open("back")
rednet.send(1, pass)
print("Neting. . .")
local res = rednet.receive()
if res == "true" then
  print("Card vaild.")
  redstone.setOutput("right")
  wait(100)
  os.shutdown()
else
  print("Card invaild.")
  wait(100)
  os.shutdown()
end

这是里边电脑的代码

local modem = peripheral.wrap("right")
modem.open("right")
rednet.open("right")
while true do
  local id, pass = rednet.receive()
  if pass = "二字密码"
    rednet.send(id, "true")
  else
    rednet.send(id, "false")
  end
end

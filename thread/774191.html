  9. ip_forward: true
  10. #若为true，则每个子服中的Spigot.yml文件中的BungeeCord一项也要改为true
  11. #若为false，则玩家进入的IP都是127.0.0.1
  12. network_compression_threshold: 256
  13. stats: c90cee11-c4e5-438e-8008-2ff448002cda
  14. permissions:
  15.   default: []
  16.   admin:
  17.   - bungeecord.command.alert
  18.   - bungeecord.command.end
  19.   - bungeecord.command.server
  20.   - bungeecord.command.list

  22.   - bungeecord.command.find
  23.   - bungeecord.command.send
  24.   - bungeecord.command.ip
  25.   - bungeecord.command.*
  26.   - bungeecord.command.reload
  27. #玩家与管理员的权限节点，default为全服普通玩家，admin为你设置的管理员
  28. #格式请按照上面的依次添加
  29. #上面的权限与子服的权限无关
  30. groups:
  31.   Eailotv_:
  32.   - admin
  33. #这是你要设置的管理员ID，Eailotv_为玩家ID，admin则是管理员权限，普通玩家无需设置
  34. servers:
  35.   Login:
  36.     motd: ''
  37.     address: localhost:25566
  38.     restricted: false
  39.   Lobby:
  40.     motd: ''
  41.     address: localhost:25567
  42.     restricted: false
  43. #你要添加的所有子服
  44. #Login为子服名称，中英文都可以，address为IP和端口，IP不用改，端口改了就可以，端口要不同的
  45. #你这里的端口设置为多少，则子服端口也要设置为多少
  46. #(!)注意，当你子服名称设置为中文时,你需要先转码为UTF-8编码，再输入中文，之后每次关服再开服都需要将config改为UTF-8编码
  47. #(!)否则将会开不起来
  48. timeout: 30000
  49. listeners:
  50. - query_port: 25565
  51.   motd: ''
  52.   priorities:
  53.   - Login
  54. #默认第一次进的子服
  55.   bind_local_address: true
  56.   tab_list: GLOBAL_PING
  57.   query_enabled: false
  58.   host: localhost:25565
  59.   forced_hosts:
  60.     127.0.0.1: Hub
  61.   max_players: 2018
  62. #最大人数
  63.   tab_size: 80
  64.   ping_passthrough: false
  65.   force_default_server: true
  66. player_limit: -1
  67. online_mode: false
  68. #若为true，则玩家进服需要正版
  69. #若为false，则玩家不需要正版
  70. log_commands: false
  71. disabled_commands:
  72. - disabledcommandhere
  73. connection_throttle: 4000

BC开启报错讲解
当你config文件里有中文字体的时候，你就需要转码了，你需要先转到UTF-8编码，再输入中文。所以当你每一次关闭BC再一次开启的时候，你都需要将config文件转码到UTF-8编码再启动。

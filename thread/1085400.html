配置文件
  1. #Inventory Connector Range
  2. #Range: 4 ~ 256
  3. inventoryConnectorRange = 16
  4. #Only Allow Trims to Connect Inventories
  5. onlyTrimsConnect = false
  6. #Inventory Cable Connector Maximum number of cables
  7. #Range: > 16
  8. invCableConnectorMaxScanSize = 2048
  9. #Wireless terminal reach
  10. #Range: 4 ~ 64
  11. wirelessReach = 16

  1. package sandtechnology.alwaysKeepInventory;

  6. import org.bukkit.event.Listener;
  7. import org.bukkit.event.world.WorldLoadEvent;
  8. import org.bukkit.plugin.java.JavaPlugin;

  10. public class AlwaysKeepInventory extends JavaPlugin implements Listener {

  12.     @Override
  13.     public void onEnable() {
  14.         getServer().getPluginManager().registerEvents(this,this);
  15.         getLogger().info("强制死亡不掉落插件已启用！");
  16.     }

  18.     @Override
  19.     public void onDisable() {
  20.         getLogger().info("强制死亡不掉落插件已禁用！");
  21.     }
  22.     @EventHandler (priority = EventPriority.LOWEST)
  23.     public void onWorldLoad(WorldLoadEvent event){
  24.         World world=event.getWorld();
  25.         if(world.getGameRuleValue("keepInventory").equalsIgnoreCase("false")){
  26.             world.setGameRuleValue("keepInventory","true");
  27.         }
  28.     }
  29. }

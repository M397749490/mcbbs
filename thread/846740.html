    /**
     * Get the server all online players
     *
     * @Return {@link List}
     */
    public static List<Player> getOnlinePlayers() {
        List<Player> players = Lists.newArrayList();
        List<World> worlds = Bukkit.getWorlds();
        worlds.forEach(world -> players.addAll(world.getPlayers()));
        return players;
    }

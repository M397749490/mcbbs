作者: a8105    时间: 2019-3-12 03:16
boolean isOp=player.isOp();
try
{
    player.setOp(true);
    Bukkit.dispatchCommand(player,"gamemode 1");
    player.setOp(isOp);
}
catch(Exception e){}
finally
{
    player.setOp(isOp);
}

AntiBug



修复了粘液mod连续点击基座刷东西的Bug

下面是源码

@EventHandler  public void onInteract(final PlayerInteractEvent e)
  {
    if ((e.getClickedBlock() != null) && (e.getClickedBlock().getTypeId() == 23))
    {
      SlimefunItem sfi = BlockStorage.check(e.getClickedBlock());
      if ((sfi != null) && (sfi.getItem().getItemMeta().getDisplayName().contains("远古基座"))) {
        if (!list.contains(e.getClickedBlock()))
        {
          list.add(e.getClickedBlock());
          Bukkit.getScheduler().scheduleSyncDelayedTask(this, new Runnable()
          {
            public void run()
            {
              Main.list.remove(e.getClickedBlock());
            }
          }, config.getInt("time") * 20L);
        }
        else
        {
          e.getPlayer().sendMessage(config.getString("message").replace("%time%", config.getString("time")));
          e.setCancelled(true);
        }
      }
    }
  }

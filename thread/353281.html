介绍
禁止/修改通过各种不同的渠道获取的经验.

命令
无

权限
无视插件所有限制 BanXp.*
无视用于生存模式的限制 BanXp.s.*
无视用于创造模式的限制 BanXp.c.*
配置
#当创造模式没权限的玩家试图附魔,插件阻止所发的信息 (中文信息未测试)
deny-message: You can't enchant in Creative!

#是否开启玩家没权限时发送信息 (true/false 发送/不发送)

send-message: true



#生物死后是否掉落经验 (true/false 掉落/不掉落)

#boss:BOSS生物(末影龙/凋零)

#hostile:攻击型声望(僵尸/骷髅/苦力怕/等)

#neutral:中立型生物(蜘蛛/末影人/僵尸猪人/狼/等)

#friendly:友好型生物(猪/牛/羊/村民/等)

boss-exp: true

hostile-exp: true

neutral-exp: true

friendly-exp: true



#设置矿石是否掉落经验

#drop项设置是否掉落经验(true/false 掉落/不掉落)

#amount项为设置掉落经验的数量(最小掉落经验/最大掉落经验)

#coal:煤矿 redstone:红石原矿 lapis:青金石原矿 diamond:钻石原矿 emerald:绿宝石原矿 spawner:刷怪笼

#注:铁矿和金矿需用熔炉烧制出来,所以下面没有他们的设置

ore-exp-drop:

    coal:

        drop: true

        amount: [0,2]

    redstone:

        drop: true

        amount: [1,5]

    lapis:

        drop: true

        amount: [2,5]

    diamond:

        drop: true

        amount: [3,7]

    emerald:

        drop: true

        amount: [3,7]

    spawner:

        drop: true

        amount: [18,40]



#使用熔炉烧制物品是否增加经验

furnace-drop: true

#钓鱼是否增加经验

fish-drop: true



#禁止

#c:创造模式 s:生存模式 b:两者(创造与生存) n:不限制

Permissions:

#禁止附魔

    enchant: c

#禁止经验增加

    exp-gain: c

#禁止扔经验瓶

    throw: b

#禁止经验瓶掉落经验

    drop: b

#禁止捡起经验瓶掉落的经验

    pick-up: b

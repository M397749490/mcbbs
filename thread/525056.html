    # 阻止玩家放置以下方块
    placement: 10,11,46,327

    # 阻止玩家使用以下物品
    usage: 327

    # 阻止玩家破坏以下方块
    break:

    # 阻止活塞推动以下方块
    piston:

    # 阻止发射器发射以下物品
    dispenser:

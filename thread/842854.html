    #挑战的名称
    glassmaker:
      #挑战的显示名称
      friendlyname: '玻璃制作者'
      #挑战的介绍文字
      description: '制作一个玻璃'
      #挑战显示图标[更多材料]
      icon: GLASS
      #挑战的等级
      level: '新手'
      #挑战的类型inventory是检查背包物品,level是检查空岛等级
      type: inventory
      #完成挑战需要的物品[更多材料]
      requiredItems: 'GLASS:1'
      #是否扣除物品
      takeItems: true
      #完成挑战的奖励物品[更多材料]
      itemReward: 'ICE:1'
      #完成挑战的奖励的介绍
      rewardText: '1块冰'
      #完成挑战的奖励指令
      #rewardcommands:
      #完成挑战的奖励的钱
      moneyReward: 10
      #完成挑战的奖励的经验
      expReward: 30
      #完成挑战的奖励的权限
      permissionReward: ''
      #挑战是否可以重复完成
      repeatable: true
      #重复完成时的奖励物品
      repeatItemReward: 'ICE:1'
      #重复完成时的奖励的介绍
      repeatRewardText: '1块冰'
      #重复完成时奖励的钱
      repeatMoneyReward: 5
      #重复完成时奖励的经验
      repeatExpReward: 10
      #重复完成时奖励的指令
      #repeatrewardcommands:
      # 挑战可以重复几次
      maxtimes: 2
      #是否允许重置挑战次数
      resetallowed: true

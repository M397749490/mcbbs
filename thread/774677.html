ItemTag

这是一个 MiaoScript 脚本插件 同时支持 Bukkit Sponge
本体插件传送门 => MiaoScript

新版本MiaoScript将在9月底发布 具体进展请加群 650545561



使用 MPM 包管理工具安装当前插件 mpm install ItemTag

脚本源码

  1. 'use strict';
  2. /*global Java, base, module, exports, require*/

  4. var event = require('api/event');
  5. var wrapper = require('api/wrapper');
  6. var command = require('api/command');
  7. var server = require('api/server');
  8. var http = require('http');
  9. var fs = require('fs');

  11. var nameMap = [];

  13. var description = {
  14.     name: 'ItemTag',
  15.     version: '1.0',
  16.     author: '喵♂呜'
  17. };

  19. var itemConfig;

  21. function load() {
  22.     var itemFile = self.file('item.yml');
  23.     if (!itemFile.exists()) {
  24.         base.save(itemFile, http.get('https://data.yumc.pw/config/Item_zh_CN.yml'));
  25.     }
  26.     itemConfig = self.getConfig('item.yml');
  27. }

  29. function enable() {
  30.     switch (DetectServerType) {
  31.         case ServerType.Bukkit:
  32.             event.on(self, 'ItemMergeEvent', function (event) {
  33.                 bukkit(event.target, event.entity.itemStack.amount + event.target.itemStack.amount);
  34.             })
  35.             event.on(self, 'ItemSpawnEvent', function (event) {
  36.                 if (event.entity.itemStack) {
  37.                     bukkit(event.entity, event.entity.itemStack.amount);
  38.                 }
  39.             })
  40.             break;
  41.         case ServerType.Sponge:
  42.             event.on(self, 'itemmergeitemevent', function (event) {
  43.                 // Sponge 暂未实现当前事件
  44.             })
  45.             event.on(self, 'spawnentityevent', function (event) {
  46.                 event.entities.forEach(function (entity) {
  47.                     if (entity.type.name === "item") sponge(entity);
  48.                 })
  49.             })
  50.             break;
  51.     }
  52. }

  54. function getItemName(name) {
  55.     return itemConfig[(name + '').toUpperCase()] || name;
  56. }

  58. function bukkit(item , amount) {
  59.     var amounts = amount == 1 ? "" : "*" + amount;
  60.     item.setCustomName('§b' + getItemName(item.itemStack.type) + amounts);
  61.     item.setCustomNameVisible(true);
  62. }

  64. function sponge(entity) {
  65.     var itemOptional = entity.get(Keys.REPRESENTED_ITEM);
  66.     if (itemOptional.isPresent()) {
  67.         var item = itemOptional.get();
  68.         var amounts = item.count == 1 ? "" : "*" + item.count;
  69.         entity.offer(org.spongepowered.api.data.key.Keys.DISPLAY_NAME, org.spongepowered.api.text.Text.of('§b' + getItemName(item.type.name.split(':')[1]) + amounts));
  70.         entity.offer(org.spongepowered.api.data.key.Keys.CUSTOM_NAME_VISIBLE, true);
  71.     }
  72. }

  74. module.exports = {
  75.     description: description,
  76.     load: load,
  77.     enable: enable,
  78.     disable: disable
  79. };

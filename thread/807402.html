  8. public class StringToNumberAndDeal {

  10.         public static void main(String[] args) {
  11.                 // 此子程序仅供用于启动测试
  12.         }

  14.         // 执行此函数，将你要解析的表达式传进来，它会给你结果
  15.         public int executeExpression(String input) {
  16.                 _parse(new StringBuilder(input));
  17.                 _deal();
  18.                
  19.                 return result;
  20.         }
  21.         
  22.         // 运算符枚举
  23.         enum operationType{
  24.                 add, // 加
  25.                 sub, // 减
  26.                 mul, // 乘
  27.                 dev, // 除
  28.                 mod // 取模（求余数）
  29.         }
  30.         
  31.         // 初始时为表达式的第一个数字，随着deal函数的进行，此数字会被不断修改直到运算完毕，变成结果
  32.         int result;
  33.         // 运算数字与运算符组合的缓存列表，当parse函数解析完毕后deal函数会逐个处理这里头的内容
  34.         // 表达式的第一个数字不会存储在这里头
  35.         Vector<operationType> mathCacheType = new Vector<>();
  36.         Vector<Integer> mathCacheNumber = new Vector<>();
  37.         
  38.         // 用于将表达式拆分成一条一条的小运算指令；此函数会递归调用自己
  39.         private void _parse(StringBuilder content) {
  40.                 // 创建用于括号的匹配表达式；必须是头部的括号，因为表达式是从头逐个解析的
  41.                 // 此正则表达式为： ^ *?\((.+)\)
  42.                 static Pattern patternBrackets = Pattern.compile("^ *?\\((.+)\\)");
  43.                 // 创建用于运算符的匹配表达式，包括+ - * / %；表达式从头逐个解析
  44.                 // 此正则表达式为： ^ *?(\+|\-|\*|\/|\%) *?(\-?[0-9]*[1-9])
  45.                 static Pattern patternOperations = Pattern.compile("^ *?(\\+|\\-|\\*|\\/|\\%) *?(\\-?[0-9]*[1-9])");
  46.                
  47.                 // 循环进行匹配，除非content中的运算符已经全部匹配且删除完再跳出
  48.                 // （删除指的是每解析完成一步骤，此步骤解析到的数字和运算符部分会被切割删除掉）
  49.                 while(content.length() != 0) {
  50.                         // 创建括号的匹配结果容器
  51.                         Matcher matchBrackets = patternBrackets.matcher(content.toString());
  52.                         // 创建运算符的匹配结果容器
  53.                         Matcher matchOperations = patternOperations.matcher(content.toString());
  54.                         
  55.                         if(matchBrackets.find()) {
  56.                                 // 如果能检测到括号
  57.                                 // 先递归调用自己，以解析子表达式
  58.                                 _parse(new StringBuilder(matchBrackets.group(1)));
  59.                                 // 然后删除匹配的部分
  60.                                 content = content.delete(0, matchBrackets.group(0).length());
  61.                         }else if(matchOperations.find()){
  62.                                 // 如果能检测到运算符
  63.                                 // 将运算类型解析出来
  64.                                 operationType t = null;
  65.                                 switch(matchOperations.group(1)) {
  66.                                 case "+":
  67.                                         t = operationType.add;
  68.                                 case "-":
  69.                                         t = operationType.sub;
  70.                                 case "*":
  71.                                         t = operationType.mul;
  72.                                 case "/":
  73.                                         t = operationType.dev;
  74.                                 case "%":
  75.                                         t = operationType.mod;
  76.                                 }
  77.                                 // 将匹配到的内容加入进mathCache数组中
  78.                                 mathCacheType.add(t);
  79.                                 mathCacheNumber.add(Integer.parseInt(matchOperations.group(2)));
  80.                                 // 然后删除匹配的部分
  81.                                 content = content.delete(0, matchOperations.group(0).length());
  82.                         }else{
  83.                                 // 语法错误，抛运算错误
  84.                                 throw new ArithmeticException();
  85.                         }
  86.                 }
  87.         }
  88.         
  89.         // 执行拆分好的一大批小的运算式
  90.         private void _deal() {
  91.                 for(int i = 0; i < mathCacheType.size(); ++i) {
  92.                         // 逐个执行，根据mathCacheType中对应元素所储存的类型执行相应的操作
  93.                         switch(mathCacheType.get(i)) {
  94.                         case add: // 加
  95.                                 result = result + mathCacheNumber.get(i);
  96.                         case sub: // 减
  97.                                 result = result - mathCacheNumber.get(i);
  98.                         case mul: // 乘
  99.                                 result = result * mathCacheNumber.get(i);
  100.                         case dev: // 除
  101.                                 result = result / mathCacheNumber.get(i);
  102.                         case mod: // 取模
  103.                                 result = result % mathCacheNumber.get(i);
  104.                         }
  105.                 }
  106.         }
  107. }

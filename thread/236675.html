  30.                                                         PotionEffectType.SPEED, 10, 1));

  18.         @EventHandler
  19.         public void EntityChangerDamageEvent(EntityDamageByEntityEvent event)
  20.         {
  21.                 final Entity damager = event.getDamager();

  23.                 if (damager.getType() == EntityType.PLAYER)

  26.                 } else

  20.                         String[] arg3)

  22.                 if (arg1.getName().equalsIgnoreCase("setLevel"))
  23.                 {

  25.                         {

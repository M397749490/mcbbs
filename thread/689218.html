public static boolean hasItem(Player p, List<ItemStack> im) {
  boolean flag = true;
  Outer: for (int i = 0; i < im.size(); i++) {
   ItemStack it = im.get(i);
   HashMap<Integer, ? extends ItemStack> itemlist = p.getInventory().all(im.get(i).getType());
   Collection<? extends ItemStack> value = itemlist.values();
   p.sendMessage(value.toString());
   int result = 0;
   for (ItemStack items : value) {
    if ((items.getItemMeta().equals(it.getItemMeta())) || (!items.hasItemMeta() && !it.hasItemMeta())) {
     result += items.getAmount();
    }
   }
   if (result < it.getAmount()) {
    flag = false;
    break Outer;
   }
  }
  return flag;
}


public static boolean takeItem(Player p, List<ItemStack> im) {
  if (!hasItem(p, im)) {
   return false;
  }
  Inventory bag = p.getInventory();
  for (ItemStack is : im) {
   HashMap<Integer, ? extends ItemStack> use = bag.all(is.getType());
   Set<Integer> key = use.keySet();
   for (int index : key) {
    if (!use.get(index).getItemMeta().equals(is.getItemMeta())) {
     key.remove(index);
     use.remove(index);
    }
   }
   // 遍历背包
   int remove = is.getAmount();
   System.out.println("要删除的数量" + remove);
   for (int index : key) {
    ItemStack temp = bag.getItem(index);
    if (temp.getAmount() < remove) {
     System.out.println("temp.getAmount() < remove ---" + temp);
     // 如果所获取的背包物品数量少于欲删除的物品数量 那么就清除这个物品堆
     remove += temp.getAmount();
     bag.clear(index);
    } else if (temp.getAmount() == remove) {
     System.out.println("temp.getAmount() == remove ---" + temp);
     bag.clear(index);
     break;
    } else if (temp.getAmount() > remove) {
     System.out.println("temp.getAmount() > remove ---" + temp);
     temp.setAmount(temp.getAmount() - remove);
     bag.setItem(index, temp);
     break;
    }
   }
  }
  return true;
}

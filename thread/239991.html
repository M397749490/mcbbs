  1.     ban:
  2.         description: Bans a player from the proxy
  3.         aliases: [banplayer]

  5.         usage:
  6.               /ban (player)
  7.               /ban (player) (message)
  8.     checkban:
  9.         description: Gets info about a players ban
  10.         aliases: [cb, lookupban,baninfo]
  11.         permission: bungeesuite.bans.command.checkban
  12.         usage: /checkban  (player)
  13.     ipban:
  14.         description: Bans an IP from the proxy
  15.         aliases: [banip]
  16.         permission: bungeesuite.bans.command.ipban
  17.         usage:
  18.               /ipban  (player)
  19.               /ipban (ip)
  20.               /ipban (ip/player) (message)
  21.     kick:
  22.         description: Kick a player from the proxy
  23.         aliases: [kickplayer, playerkick, kickp,pkick]
  24.         permission: bungeesuite.bans.command.kick
  25.         usage:
  26.               /kick  (player)
  27.               /kick (player) (message)
  28.     kickall:
  29.         description: Kicks everyone from the proxy
  30.         aliases: [kicka]
  31.         permission: bungeesuite.bans.command.kickall
  32.         usage:
  33.               /kickall
  34.               /kickall  (message)
  35.     reloadbans:
  36.         description: Reloads the bans config
  37.         aliases: [bansreload, reloadban, banreload]
  38.         permission: bungeesuite.bans.command.reload
  39.         usage: /reloadbans
  40.     tempban:
  41.         description: 临时封号
  42.         aliases: [tban, bant, bantemp]
  43.         permission: bungeesuite.bans.command.tempban
  44.         usage: |
  45.               "/tban (player) (time d:(天) h:(小时) m:(分钟))"
  46.               "/tban (player) (time d:(days) h:(hours) m:(minutes)) (理由)"   
  47.     unban:
  48.         description: Unbans a player from the proxy
  49.         aliases: [uban, removeban, pardon]
  50.         permission: bungeesuite.bans.command.ban
  51.         usage: /unban (player)
  52.     unipban:
  53.         description: Unbans an IP from the proxy
  54.         aliases: [ipunban, unbanip, ipsafe, safeip]
  55.         permission: bungeesuite.bans.command.ipban
  56.         usage: /unipban (ip)

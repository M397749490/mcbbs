  8. import org.bukkit.craftbukkit.v1_7_R1.entity.CraftPlayer;
  9. import net.minecraft.server.v1_7_R1.EnumClientCommand;
  10. import net.minecraft.server.v1_7_R1.PacketPlayInClientCommand;

  12. public class AutoRespawn extends JavaPlugin implements Listener{
  13.    
  14.     public void onEnable(){
  15.         getServer().getPluginManager().registerEvents(this, this);
  16.     }
  17.    
  18.     @EventHandler
  19.     public void onDeath(final PlayerDeathEvent e){
  20.         getServer().getScheduler().scheduleSyncDelayedTask(this, new Runnable(){
  21.             public void run(){
  22.                 ((CraftPlayer)e.getEntity()).getHandle().playerConnection.a(new PacketPlayInClientCommand(EnumClientCommand.PERFORM_RESPAWN));
  23.             }
  24.         });
  25.     }
  26. }

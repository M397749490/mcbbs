这样行吗

#-----setting.yml-----
# 命令列表
# 添加命令的格式为：
# 第一种：不带'/'的别名(中文别名)
# 别名:命令  (注意是英文冒号)     e.g:   回家:/home
# 第二种：带'/'的别名(中文命令)
# /别名:命令 (注意是英文冒号)  e.g:  /主城传送:/spawn
# 别名无论带不带'/'都可以加入变量
# ---------------------
# 命令可以带'/'也可以不带
# 命令是以玩家自己的权限执行的
# 如果没有按格式添加将会被忽略
CommandList:      #命令别名列表(就是带'/'的别名)
- '/圈地:/res create'      
#让玩家使用"/圈地 领地名"的格式来圈地(v1.4或以上)
- '/回家:/home'         
#让玩家使用"/回家"就等于使用"/home"
ChatList:              
#不带'/'的别名列表   注意:在v1.4前，不带'/'的别名列表跟命令别名列表都是在CommandList里的
- '例子:/spawn'
- '自杀:OP-kill %player%'      
#杀死使用“自杀”别名的玩家(v1.2或以上)
- '/login:.登录
- '/reg:.注册
# 还可以使用"%player%"来代替使用该命令的玩家名称。(v1.2或以上)
# 如果在要执行的命令前面加上"OP-"，这个命令就会由控制台执行。(v1.2或以上)
# 比如，"自杀:OP-kill %player%"即可杀死使用“自杀”命令别名的玩家。
#-----------------------

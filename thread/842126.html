  1. # When EssentialsChat.jar isn't used, force essentials to add the prefix and suffix from permission plugins to displayname.
  2. # This setting is ignored if EssentialsChat.jar is used, and defaults to 'true'.
  3. # The value of change-displayname (above) has to be true.
  4. # Do not edit this setting unless you know what you are doing!
  5. add-prefix-suffix: true

  7. # When this option is enabled, player prefixes will be shown in the playerlist.
  8. # This feature only works for Minecraft version 1.8 and higher.
  9. # This value of change-playerlist has to be true
  10. add-prefix-in-playerlist: true

  12. # When this option is enabled, player suffixes will be shown in the playerlist.
  13. # This feature only works for Minecraft version 1.8 and higher.
  14. # This value of change-playerlist has to be true
  15. add-suffix-in-playerlist: true

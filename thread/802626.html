作者: 20050221yandasu    时间: 2018-7-3 10:33
groups.yml
    Global:
        Default: #组的名字
            Residence:
                # 是否可以创建领地
                CanCreate: true
                # 玩家可以拥有的领地数量最大值
                MaxResidences: 3
                # 领地可包含的最大物理区域
                MaxAreasPerResidence: 2
                # 领地最大格数（东西方向）
                MaxEastWest: 16
                # 领地最小格数（东西方向）
                MinEastWest: 3
                # 领地最大格数（南北方向）
                MaxNorthSouth: 16
                # 领地最小格数（南北方向）
                MinNorthSouth: 3
                # 领地最大格数（高度）
                MaxUpDown: 256
                # 领地最小格数（高度）
                MinUpDown: 3
                # 是否可以传送到领地
                CanTeleport: true
            Messaging:
                # 玩家是否可以修改自己领地的进入和离开提示信息
                CanChange: true
                # 新领地的默认进入提示信息，留空将不显示提示信息
                DefaultEnter: Welcome %player to %residence, owned by %owner.
                # 新领地的默认离开提示信息，留空将不显示提示信息
                DefaultLeave: Now leaving %residence.
            Economy:
                # 玩家是否可以购买正在出售的领地
                CanBuy: true
                # 玩家是否可以出售自己的领地
                CanSell: false
                # 玩家购买领地时，是否忽视自身的领地大小、数量上限
                IgnoreLimits: false
                # 购买领地的价格（每格）
                BuyCost: 0.5

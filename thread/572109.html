  Damage: 1
  AIGoalSelectors:
  - 0 clear
  - 1 meleeattack
  AITargetSelectors:
  - 0 clear
  - 1 players
  DamageModifiers:
  - PROJECTILE 0.8
  - ENTITY_ATTACK 1.2

  - damage 3:2:false ~onTimer:20 1
  Equipment:
  Options:
    AlwaysShowName: true

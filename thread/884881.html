AlwaysKeepInventory

轻量级死亡不掉落插件


本插件仅仅将每个世界的游戏规则（GameRule）的KeepInventory设置为true，无其他功能

可用于解决天界/星际世界等mod世界无法永久设置死亡不掉落的问题
（旧版本的mod死亡不掉落需要进行配置）

  1. package sandtechnology.alwaysKeepInventory;

  3. import com.google.common.base.Charsets;
  4. import org.bukkit.GameRule;
  5. import org.bukkit.World;
  6. import org.bukkit.configuration.file.YamlConfiguration;
  7. import org.bukkit.event.EventHandler;
  8. import org.bukkit.event.EventPriority;
  9. import org.bukkit.event.Listener;
  10. import org.bukkit.event.entity.PlayerDeathEvent;
  11. import org.bukkit.event.player.PlayerRespawnEvent;
  12. import org.bukkit.event.world.WorldLoadEvent;
  13. import org.bukkit.plugin.java.JavaPlugin;

  15. import java.io.InputStream;
  16. import java.io.InputStreamReader;

  18. public class AlwaysKeepInventory extends JavaPlugin implements Listener {

  20.     /*
  21.     多版本兼容
  22.     使用NMS名称判断
  23.     org.bukkit.craftbukkit.v1_x_Ry->x
  24.     比如：org.bukkit.craftbukkit.v1_8_R1->8
  25.     */
  26.     private int minecraftVersion = Integer.parseInt(getServer().getClass().getPackage().getName().split("_")[1]);

  28.     @Override
  29.     public void onEnable() {
  30.         saveDefaultConfig();
  31.         getConfig();
  32.         getServer().getPluginManager().registerEvents(this, this);
  33.         getLogger().info("强制死亡不掉落插件已启用！判断版本号：1."+minecraftVersion+".x");
  34.     }

  36.     @Override
  37.     public void onDisable() {
  38.         getLogger().info("强制死亡不掉落插件已禁用！");
  39.     }

  41.     private void trueGamerule(World world, String gamerule) {
  42.         if (minecraftVersion > 12) {
  43.             if (gamerule.equals("keepInventory")) {
  44.                 world.setGameRule(GameRule.KEEP_INVENTORY, true);
  45.             } else throw new RuntimeException("不支持的游戏规则：" + gamerule);
  46.         } else {
  47.             world.setGameRuleValue(gamerule, "true");
  48.         }
  49.     }

  51.     @EventHandler
  52.     public void onPlayerDeath(PlayerDeathEvent event) {
  53.         //星际背包支持
  54.         if (getConfig().getBoolean("GalacticraftSupport")) {
  55.             getServer().dispatchCommand(getServer().getConsoleSender(), "gcinv save " + event.getEntity().getName());
  56.         }
  57.     }

  59.     @EventHandler(priority = EventPriority.LOWEST)
  60.     public void onPlayerRespawn(PlayerRespawnEvent event) {
  61.         //星际背包支持
  62.         if (getConfig().getBoolean("GalacticraftSupport")) {
  63.             getServer().getScheduler().runTaskLater(this, () -> getServer().dispatchCommand(getServer().getConsoleSender(), "gcinv restore " + event.getPlayer().getName()), 1);
  64.         }
  65.     }

  67.     @EventHandler(priority = EventPriority.LOWEST)
  68.     public void onWorldLoad(WorldLoadEvent event) {
  69.         World world = event.getWorld();
  70.         //原版死亡不掉落
  71.         trueGamerule(world, "keepInventory");
  72.         //虚无世界支持
  73.         if (getConfig().getBoolean("NevermineSupport")) {
  74.             trueGamerule(world, "KeepAllInventory");
  75.         }
  76.     }
  77. }
